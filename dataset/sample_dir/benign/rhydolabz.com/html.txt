function ajax_remove_from_cart(jobj, prodID)
{
	$.ajax({
		'url':"https://www.rhydolabz.com/ajax_addtocart.php",
		'dataType':"json",
		'data': {'cmd':"remove", 'product':prodID, 'qty':0, 'mode':"ajax"},
		'type':"GET",
		'success':function(data) {
			$('#abdatc_overlay').stop(true, true).fadeIn();
			x = ($(window).width()/2)+$(document).scrollLeft()-($('#abdatc_removed').outerWidth()/2);
			y = ($(window).height()/2)+$(document).scrollTop()-($('#abdatc_removed').outerHeight()/2);
			$('#abdatc_removed').stop(true,true).css({"left":x+"px", "top":y+"px"}).fadeIn();
			window.setTimeout(function(){$("#abdatc_removed").fadeOut();$('#abdatc_overlay').stop(true, true).fadeOut(function(){},function(){$('#abdatc_removed').attr("style",""); });}, 4000);
			jobj.slideUp('slow', function() { $(this).remove(); });
			$('.cartTotalsDisplay').html(data.totals);
			$('#cartSubTotal').html(data.subtotal);
			$('#abdatc_overlay').css({"height":$('body').outerHeight(), "width":$('body').outerWidth()}).click(function() {
				$('#abdatc_removed').stop(true, true).fadeOut(function(){},function(){$('#abdatc_removed').attr("style",""); });
			});
			//load minicart dropdown
			load_minicart_ajax();
			//load cart sidebox
			load_sideboxajx(prodID,'0');
		}
	});
}
