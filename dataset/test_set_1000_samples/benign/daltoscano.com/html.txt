$(".SelRicerca").click(function(e) {
    e.preventDefault();
    var Word = $.trim($("input[name='nomeprodotto']").val());
    var Cat =  $.trim($("select[name='categoriaprodotto']").val());
    var Fam =  $.trim($("select[name='codicefamigliaprodotto']").val());

    if(Word == '' && Cat == '' && Fam == ''){
        alert('Attenzione inserire almeno un criterio di ricerca!');
        return false;
    }else{

        if(Word == ''){Word = '0';}
        if(Cat == ''){Cat = '0/0';}
        if(Fam == ''){Fam = '0';}

        var BuildUrl = encodeURI(Cat+'/'+Fam+'/'+Word);
        window.location.href = "https://www.daltoscano.com/ricerca/"+BuildUrl;
    }

});
</script>
