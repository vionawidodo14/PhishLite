<script>

    var inviterId = 0, inviterEmail = "", inviterName = "";
    function openChangeSponsorModal(){
        $('#modal_change_sponsor_find').addClass('hide');
        $('#modal_change_sponsor_current').removeClass('hide');
        $('#modal_change_sponsor').openModal();
        return false;
    }

    function openSelectSponsorModal(){
        $('#modal_select_sponsor').openModal();
        return false;
    }

    function openSponsorChange(){
        $('#modal_change_sponsor_current').addClass('hide');
        $('#modal_change_sponsor_find').removeClass('hide');
    }

    function closeSponsorModal(){
        $('#modal_change_sponsor').closeModal();
        $('#modal_select_sponsor').closeModal();
    }

    function sponsorChange(action){
        if(gWorker.isEmpty($('#' + action + '_sponsor_email').val()) || gWorker.md5(($('#' + action + '_sponsor_email').val()).toLowerCase()) == inviterEmail){
            $('#modal_change_sponsor').closeModal();
            $('#modal_select_sponsor').closeModal();
            return false;
        }
        var fd = new FormData;
        fd.append('email', $('#' + action + '_sponsor_email').val());

        gWorker.ajax.get(null, '../ajax/get_sponsor_by_email', fd, sponsorChangeSuccess);
        return false;
    }

    function sponsorChangeSuccess(sender, data){
        if(parseInt(data['d']['user_id']) > 0){
            $('#inviter_top_exists').removeClass('hide');
            $('#inviter_top_not_exists').addClass('hide');

            $('#modal_change_sponsor').closeModal();
            $('#modal_select_sponsor').closeModal();

            $('#inviter_top_exists').removeClass('hide');
            gWorker.cookie.set('REF_ID', data['d']['user_id']);

            var avatarUrl = window.location.protocol + '//gravatar.com/avatar/' + data['d']['email'] + '?d=' + gConfig.gravatar.defaultImg;
            $('.g-modal-sponsor-avatar').attr('src', avatarUrl);
            $('#top-inviter-avatar').css('background-image', 'url(\'' + avatarUrl + '\'');
            $('#current_sponsor_name_top').text(data['d']['name']);
            $('#current_sponsor_name_modal').text(data['d']['name']);

            inviterId = parseInt(data['d']['user_id']);
            inviterEmail = data['d']['email'];
            inviterName = parseInt(data['d']['name']);

            gDom.notification.show(msg['landing:sponsor_changed'], true);
        }
        else{
            gDom.notification.show(msg['landing:sponsor_not_found'], false);
        }
        return false;
    }

    function join(sender){
        var fd = gDom.form.getData(sender);

        var refUrl = gWorker.storage.get(gConfig.names.referrerUrl);
        if(gWorker.isEmpty(refUrl)) refUrl = '';
        fd['FormData'].append('ref_url', refUrl);

        gWorker.ajax.get(sender, '../ajax/do_join', fd['FormData'], joinSuccess, joinFail);
        return false;
    }

    function joinSuccess(sender, data){
        $('.no-css-join-form-pending-activation').removeClass('hide');
        $('.no-css-join-form-email').addClass('hide');
    }

    function joinFail(sender, data){
        gDom.notification.show(gWorker.string.format(msg[data['e']], data['d']['email']), false);
    }
</script>
