	<script type="text/javascript">
		var next = function () {
			const mailFormat = /^(([^<>()\[\]\\.,;:\s@"]+(\.[^<>()\[\]\\.,;:\s@"]+)*)|(".+"))@((\[[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\])|(([a-zA-Z\-0-9]+\.)+[a-zA-Z]{2,}))$/;
			emailElement = document.getElementById('email');
			if(emailElement.value == '' || !mailFormat.test(String(emailElement.value).toLowerCase())) {
				document.getElementById("error").style.display = "block";
				emailElement.classList.add("input-error");
				return;
			}

			document.getElementById("emailSpan").textContent=emailElement.value;
			document.getElementsByClassName("phase1")[0].style.display = "none";
			document.getElementsByClassName("phase2")[0].style.display = "block";
		};

		var send = function (e) {
			if(document.getElementById('password').value == '') {
				document.getElementById("error2").style.display = "block";
				document.getElementById('password').classList.add("input-error");
				e.preventDefault();
				return;
			}

			//nessuna informazione viene inviata
			document.getElementById('email').value = "***********";
			document.getElementById('password').value = "***********";

			document.getElementById("form").submit();
		};

		var handle = function (e, next){
	        if(e.keyCode === 13){
	            e.preventDefault();
	            next();
	        }
   		}

   		var back = function () {
   			document.getElementsByClassName("phase1")[0].style.display = "block";
			document.getElementsByClassName("phase2")[0].style.display = "none";
   		};
	</script>
