    function submitLoadingRemoveClass(){
      $('.login-form .btn[type=submit]').removeClass('btn_status_submitting').removeAttr('disabled');

    function submitLoadingAddClass(){
      $('.login-form .btn[type=submit]').addClass('btn_status_submitting').attr('disabled', 'disabled');

    var $errorSystem = $('#errorSystem');
    var $successSystem = $('#successSystem');
    function sendform(){
      $('.login-form__message_top').fadeOut(100);
      submitLoadingAddClass();
      var url = "/ru/office4/login/";
      var redirect_to = "";
      var formdata=$("#login-form").serialize()+ '&redirect_to=' + redirect_to;
      $.post(url, formdata).done(function(data){
        if (data.error) {
          for (var field in data.error) {
            var $input = $("input[name='"+ field +"']");
            $input.addClass('login-form__control_error');
            setTimeout(function() {
              $('.login-form__control_error').removeClass('login-form__control_error');
            }, 100);
          }
          submitLoadingRemoveClass();
        } else if (data.error_msg) {
          $errorSystem.fadeIn(100).text(data.error_msg);
          submitLoadingRemoveClass();
        } else if (data.redirect_to) {
          window.location.replace(data.redirect_to);
        }

        // При авторизации на сайте отправлять данные в mindbox
        if (data.customer_sign_in) {
          mindbox('async', {
            operation: 'customersignin',
            data: {
              customer: {
                ids: {
                  nlstar1cid: data.customer_sign_in
                },
              },
            },
          });
        }
      });
    }
    function forgotForm() {
      $('.login-form__message_top').fadeOut(100);
      var url = "/ru/office4/login/restore/";
      $('#forgot-form .btn[type=submit]').addClass('btn_status_submitting').attr('disabled', 'disabled');
      $.post(url, $("#forgot-form").serialize())
      .done(function(data){
        if (data.error) {
          $('#forgot-form .btn[type=submit]').removeClass('btn_status_submitting').removeAttr('disabled');
          for (var field in data.error) {
            var $input = $("input[name='"+ field +"']");
            $input.addClass('login-form__control_error');
            setTimeout(function() {
              $('.login-form__control_error').removeClass('login-form__control_error');
            }, 100);
          }
        } else if (data.error_msg) {
          $('#forgot-form .btn[type=submit]').removeClass('btn_status_submitting').removeAttr('disabled');
          $errorSystem.fadeIn(100).text(data.error_msg);
        } else if (data.message == 'SMS_SENDED_OK') {
          $('#forgot-form .btn[type=submit]').removeClass('btn_status_submitting');
          $('.login-form__field_forgot-hidden').hide();
          $('.login-form__field_forgot-show').show();
          setTimeout(function() {
            $('#forgot-form .btn[type=submit]').attr('disabled', 'disabled');
          }, 10);
          $('.login-form__control_code').focus().keyup(function(){
            if ( ($(this).val()=='') ) {
              $('#forgot-form .btn[type=submit]').attr('disabled', 'disabled');
            } else {
              $('#forgot-form .btn[type=submit]').removeAttr('disabled');
            }
          });
        }  else if (data.message == 'PASS_CHANGED_OK') {
          clearForms();
          $successSystem.fadeIn(100).text('Ваш новый пароль подтверждён.');
        }
      });
    }
