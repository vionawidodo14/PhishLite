        function showPricingTableDiscount(percents)
        {
            $('.js-discount-circle').each(function(index, value) {
                var sale_day = $(this).closest('tr').data('summer-sale');
                var sale_percent = 20;
                var percents_with_sale = percents;
                if (sale_day) {
                    percents_with_sale = percents + sale_percent;
                }

                var before_sale_price_span = $(this).closest('td').find('.js-before-sale-price');

                var real_price_span = $(this).parent().find('.real-price');
                if (typeof real_price_span.data('real-price') == 'undefined') {
                    real_price_span.data('real-price', real_price_span.text());
                }

                var real_price = parseFloat(currency(prices_for_table_global[index], 'number'));

                var before_summer_sale_price = real_price / (100 - sale_percent) * 100;

                $(this).text('-'+percents_with_sale+'%');
                real_price_span.text(twoDigits(real_price - (real_price / 100 * percents)));

                if (percents_with_sale > 0) {
                    $(this).css('display', 'inline-block');

                    $(this).parent().addClass('discounted');
                    $(this).parent().removeClass('not-discounted');

                    before_sale_price_span.show().text(twoDigits(before_summer_sale_price));
                } else {
                    $(this).hide();

                    $(this).parent().removeClass('discounted');
                    $(this).parent().addClass('not-discounted');

                    before_sale_price_span.show().text(twoDigits(before_summer_sale_price));
                }
            });
        }
