<script>
$(document).ready(function() {
  if (window.spResourcesDropdownInitialized) return;
  
  var menu = '.sp-resources-menu';
  var dropdown = '.sp-resources-dropdown';
  var hoverContent = '.sp-resources-hover';
  var mobileContent = '.sp-resources-mobile';
  var anchor = '.sp-resources-anchor';
  var overlay = '.sp-resources-overlay';
  var menuItems = '.semplice-navbar .menu-item';
  var overlayMenu = '#overlay-menu .sp-resources-menu';
  
  var activeClass = 'sp-active';
  var staticClass = 'sp-static';
  
  var isActive = false;
  
  function activate(selector) {
    $(selector).addClass(activeClass);
  }
  
  function deactivate(selector) {
	$(selector).removeClass(activeClass);
  }
  
  $(hoverContent).prependTo(dropdown);
  $(mobileContent).appendTo(overlayMenu);
  
  $('#content-holder').on('hover', menu + ',' + dropdown, function(event) {
	if (event.type === 'mouseenter') {
	  isActive = true;
	  activate(dropdown);
	  activate(menu);
	  activate($(menuItems).last());
	}
	
	if (event.type === 'mouseleave') {
	  isActive = false;
	  window.setTimeout(function() {
		if (!isActive) {
		  $(dropdown).addClass(staticClass);
		  
		  deactivate(dropdown);
		  deactivate(menu);
		  deactivate($(menuItems).last());
		  
		  window.setTimeout(function() {
			$(dropdown).removeClass(staticClass);
		  }, 100);
		}
	  }, 150);
	}
  });
  
  $('#content-holder').on('hover', anchor, function(event) {
	var index = $(anchor).index($(event.target).closest(anchor));

	if (event.type === 'mouseenter') {
	  activate($(overlay).eq(index));
	}
	
	if (event.type === 'mouseleave') {
	  deactivate($(overlay).eq(index));
	}
  });
  
  $('#content-holder').on('click', menu + ' > a', function(event) {
	event.preventDefault();
	return false;
  });
  
  window.spResourcesDropdownInitialized = true;
});
</script>
