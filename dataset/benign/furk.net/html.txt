<script>
	$(document).ready(function() {
		$('#login_form').on('submit', function() {
			$.post('/api/login/login', $(this).serialize(), function(resp){
				$('#login_error').hide();
				if (resp.status == 'error') {
					$('#login_error').show().html(resp.error);
				}
				else if (resp.status == 'ok') {
					document.location = resp.url;
				}
			});
		});

		$('#reg_form').on('submit', function() {
			var form = $(this);
			if (!form.data("validator").checkValidity()) {
				$('#reg_form_fields').show();
				form.data("validator").reflow();
				return false;
			}
			$.post('/api/login/reg', form.serialize(), function(resp){
				$('#reg_error').hide();
				if (resp.status == 'error') {
					$('#reg_form_fields').show();
					form.data("validator").reflow();
					form.data("validator").invalidate(resp.errors)
				}
				else if (resp.status == 'ok') {
					document.location = resp.url;
				}
			});
		});

		$('#reg_form').validator();
        
        $('a.social-popup').on('click', function() {
            window.open($(this).data('url'));
            return true;
        });

        window.social_onLogin = function (resp) {
            if (resp.status == 'error') {
                setTimeout(function() {alert('Error: ' + resp.error)}, 1);
                return;
            }
            else if (resp.status == 'ok' && resp.url) {
                document.location = resp.url;
                return;
            }

            if (resp.state == 'link') {
                $('#social_buttons_cont').hide();
                $('#reg_form').hide();

                $('#social_msg ').show().html('<h3>Your email is already registered with us. Enter your Furk.net password to link your '+resp.user.provider+' ID</h3>');

                $('#login_form [name="social_uid"]').val(resp.social_uid);
                $('#login_form [name="login"]').val(resp.user.email);
                $('#reg_form_cont').hide();
            }
            else if (resp.state == 'create') {
                $('#social_buttons_cont').hide();

    			$('#social_msg').show().html("We've never seen you before! What would you like us to do?");
                $('#reg_form_fields').hide();

                $('#login_h').addClass('alert alert-info').html('1. Link an existing account');
                $('#reg_h').addClass('alert alert-info').html('2. Create a new account');

                $('#login_form [name="social_uid"]').val(resp.social_uid);
                $('#reg_form [name="social_uid"]').val(resp.social_uid);
                $('#reg_form [name="login"]').val(resp.user.login);
                $('#reg_form [name="email"]').val(resp.user.email);
                $('#reg_form [name="pwd"]').val(resp.user.pwd);
            }
            else {
                console.log(resp);
                alert('Internal error. Please report to support.\n' + JSON.stringify(resp) );
            }
        }
   });
</script>
