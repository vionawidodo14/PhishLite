			var self = this,
				$bat = $('<div class="halloweenBat"/>'),
				x,
				y,
				tx,
				ty,
				dx,
				dy,

					imageLength = options.width;
				}

					screenLength = innerHeight;
					imageLength = options.height;

				dTop = options.speed * (top / length);

				ddLeft = (dLeft - dx) / options.flickering;
				ddTop = (dTop - dy) / options.flickering;

				dy += ddTop;

				y += dy;
