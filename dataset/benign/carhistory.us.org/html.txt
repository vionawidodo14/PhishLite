function finalcheckviadecoder() {
  if (xmlHttpvdc.readyState == 4) {
    responsestr = xmlHttpvdc.responseText;
    if (responsestr == 'invalidvin') {
      alert("It appears that you have entered an invalid VIN #.\nPlease ensure that you have entered the proper VIN # and re-enter the 17 digit VIN # below.");
      document.getElementById('vinldimg_org').style.display = "";
      document.getElementById('vinldimg').style.display = "none";
      document.getElementById('vin').style.backgroundColor = "#FFF";
      document.nameSearchcriminal.vin.focus();
      return false;
    }
    if (responsestr == 'invalidrequest') {
      alert("It appears that you have entered an invalid VIN #.\nPlease ensure that you have entered the proper VIN # and re-enter the 17 digit VIN # below.");

      document.getElementById('vinldimg').style.display = "none";

      document.nameSearchcriminal.vin.focus();
      return false;
    }
    if (responsestr == 'validvin') {
      document.nameSearchcriminal.submit();
    }
  }
}
