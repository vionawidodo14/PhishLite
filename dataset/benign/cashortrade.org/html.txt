                if ($('#threads').length) {
                    $("#threads").animate({ scrollTop: $('#threads').prop("scrollHeight")}, 500);
                }

        window.sendBrowserNotification = function (data) {
            if (data == undefined || !data) { return false }
            var title = (data.title === undefined) ? 'Notification' : data.title
            var clickCallback = data.clickCallback
            var message = (data.message === undefined) ? 'null' : data.message
            var icon = (data.icon === undefined) ? 'https://images.cotcdn.org/CashorTrade-Logo-150.png' : data.icon
            var sendNotification = function (){
                var notification = new Notification(title, {
                    icon: icon,
                    body: message
                })
                if (clickCallback !== undefined) {
                    notification.onclick = function () {
                        clickCallback()
                        notification.close()
                    }
                }
            }
            if (!window.Notification) {
                return false
            } else {
                if (Notification.permission === 'default') {
                    Notification.requestPermission(function (p) {
                        if (p !== 'denied') {
                            sendNotification()
                        }
                    })
                } else {
                    sendNotification()
                }
            }
        }

        function socket(attemptArg, retryIntervalArg) {
            var attemptInit = 1;
            var retryIntervalInit = 800;
            var attempt = attemptArg || attemptInit;
            var retryInterval = retryIntervalArg || retryIntervalInit;
            var maxAttempts = 15;
            var successTimeout;
            var pongTimeout;
            var pingInterval;
            var typingTimeout;
            window.initSocketReconnectTimeout();
            var resetTimeoutTimer = function(connection){
                clearTimeout(pongTimeout);
                pongTimeout = setTimeout(function(){
                    connection.close(4001, 'connection timed out');
                }, 30000)
            }
            var resetTypingTimer = function(){
                clearTimeout(typingTimeout);
                typingTimeout = setTimeout(function(){
                    $("#typing").hide();
                }, 6000)
            }
            if (attempt > maxAttempts) {
                window.connectAttemptsExhausted = true;
            } else {
                clearTimeout(pongTimeout);
                window.connectAttemptsExhausted = false;
            }
            if (typeof window.conn === 'undefined' || (window.conn.readyState != 1 && attempt <= maxAttempts)) {
                                    window.conn = new WebSocket('wss://rp1.cashortrade.org:443/?&platform=Desktop&app_version=&ipaddress=185.217.117.52');

                  // refreshInboxNotifications();
                  clearTimeout(successTimeout);
                  clearInterval(pingInterval);
                  clearTimeout(pongTimeout);
                  if (window.conn.readyState !== 1 && e.code !== 4000) {
                    setTimeout(function(){
                        socket(attempt + 1, retryInterval * 1.6);
                    }, retryInterval);
                  }
                }
                window.conn.onerror = function(e) {
                  console.log('ERROR', e);
                }

        Handlebars.registerHelper('ifCond', function (v1, operator, v2, options) {
        var boolFns = {
            fn: function(){return true;},
        inverse: function(){return false;}
        };
        var result = !!options.hash.bool ? boolFns : options;
            switch (operator) {
                case '==':
                    return (v1 == v2) ? result.fn(this) : result.inverse(this);
                case '===':
                    return (v1 === v2) ? result.fn(this) : result.inverse(this);
                case '<':
                    return (v1 < v2) ? result.fn(this) : result.inverse(this);
                case '<=':
                    return (v1 <= v2) ? result.fn(this) : result.inverse(this);
                case '>':
                    return (v1 > v2) ? result.fn(this) : result.inverse(this);
                case '>=':
                    return (v1 >= v2) ? result.fn(this) : result.inverse(this);
                case '&&':
                    return (v1 && v2) ? result.fn(this) : result.inverse(this);
                case '||':
                    return (v1 || v2) ? result.fn(this) : result.inverse(this);
                case '!=':
                    return (v1 != v2) ? result.fn(this) : result.inverse(this);
                default:
                    return result.inverse(this);
            }
        });
