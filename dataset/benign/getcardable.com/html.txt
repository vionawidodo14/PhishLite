                $(this).children('.glyphicon').addClass('glyphicon-chevron-up');
            }
            else
            {
                $(this).children('.glyphicon').removeClass('glyphicon-chevron-up');
                $(this).children('.glyphicon').addClass('glyphicon-chevron-down');
            }
        });
    </script>
