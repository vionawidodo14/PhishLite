     function startCountDown(from, html_els) {        
        var obj = html_els;

        setInterval(function() {
            // Get todays date and time
            var now = new Date().getTime();
            // Find the distance between now and the count down date
            var distance = from - now;

            function n(n){
                return n > 9 ? "" + n: "0" + n;
            }
            // Time calculations for days, hours, minutes and seconds
            var days = n(Math.floor(distance / (1000 * 60 * 60 * 24)));
            var hours = n(Math.floor((distance % (1000 * 60 * 60 * 24)) / (1000 * 60 * 60)));
            var minutes = n(Math.floor((distance % (1000 * 60 * 60)) / (1000 * 60)));
            var seconds = n(Math.floor((distance % (1000 * 60)) / 1000));


            // Display the result in the element with id="demo"              
            if (obj.hasOwnProperty("d"))
                document.getElementById(obj["d"]).innerHTML = days;
            
            if (obj.hasOwnProperty("h"))
                document.getElementById(obj["h"]).innerHTML = hours;

            if (obj.hasOwnProperty("m"))
                document.getElementById(obj["m"]).innerHTML = minutes;

            if (obj.hasOwnProperty("s"))
                document.getElementById(obj["s"]).innerHTML = seconds;            
           
        }, 1000);
        
    }

    function sldVisibleElements(container_selector, item_selector) {
     var container_w = parseInt(window.getComputedStyle(document.querySelectorAll(container_selector)[0]).width.replace("px", ""));
     var item_w = parseInt(window.getComputedStyle(document.querySelectorAll(container_selector + " " + item_selector)[0]).width.replace("px", ""));
     var item_margin = parseInt(window.getComputedStyle(document.querySelectorAll(container_selector + " " + item_selector)[0]).marginRight.replace("px", ""));

     return container_w / (item_w + item_margin);

    }

     function updateCountdown() {
         var dateNow = new Date();
         seconds = dateNow.getSeconds();
         minutes = dateNow.getMinutes();
         hours = dateNow.getHours();

         hours = 23 - hours;
         minutes = 59 - minutes;
         seconds = 59 - seconds;

        if (dateNow.getDate() <= 3)
            hours += 24;

         var countdown = {"hora": document.querySelector("#clock #hours"),
                  "minuto": document.querySelector("#clock #minutes"),
                  "segundo": document.querySelector("#clock #seconds")
         }

         if (countdown['hora']) countdown['hora'].innerHTML = fill(hours);

         if (countdown['minuto']) countdown['minuto'].innerHTML = fill(minutes);

         if (countdown['segundo']) countdown['segundo'].innerHTML = fill(seconds);

         setTimeout(function(){ updateCountdown(); }, 1000);
     }

        function shuffle(array) {
             var currentIndex = array.length, temporaryValue, randomIndex;

             while (0 !== currentIndex) {

             randomIndex = Math.floor(Math.random() * currentIndex);
             currentIndex -= 1;

             temporaryValue = array[currentIndex];
             array[currentIndex] = array[randomIndex];
             array[randomIndex] = temporaryValue;
             }

             return array;
         }

        function transition_manage(transition_name, elements_involved, action){
            var el = null;
            if (action == "add") {
                for (var i = elements_involved.length - 1; i >= 0; i--) {
                    el = document.querySelector(elements_involved[i]);
                    if (el != null){
                        class_manage(el, transition_name, "add");
                    }
                }
            }
            else
            {
                var elements = document.getElementsByClassName(transition_name);
                for (var i = elements.length - 1; i >= 0; i--) {
                    class_manage(elements[i], transition_name, "remove");
                }
            }
        }

        function setAttributes(el, attrs) {
          for(var key in attrs) {
            el.setAttribute(key, attrs[key]);
          }
        }

        function class_manage(el, cls, act) {
            if (act=="add") {
                if (!!!el.className.match(new RegExp('(\\s|^)'+cls+'(\\s|$)')))
                    el.classList.add(cls);
            }
            else{
                if (act=="remove") {
                    if (!!el.className.match(new RegExp('(\\s|^)'+cls+'(\\s|$)')))
                        el.classList.remove(cls);
                }
            }
        }

        function ajaxCall(data) {
            var request = data.req;
            var response = data.res;
            var httpRequest = new XMLHttpRequest();
            httpRequest.open(request.method, request.url, request.async || true);
            request.requestHeader ? httpRequest.setRequestHeader(request.requestHeader[0], request.requestHeader[1]) : httpRequest.setRequestHeader('Content-type' , 'application/x-www-form-urlencoded');
            request.query ? httpRequest.send(request.query) : httpRequest.send();
            httpRequest.onreadystatechange = function () {
                if (response.onsuccess) {
                    if (httpRequest.readyState === XMLHttpRequest.DONE){
                        if (httpRequest.status === 200) {
                            var data = response.onsuccess.params;
                            data.responseText = httpRequest.responseText;
                            response.onsuccess.callback(data);
                        }
                        else {
                            if (response.onerror)
                                response.onerror.callback(response.onerror.params);
                        }
                    }
                }
            }
        }

        function getHistory() {
            var hist = User.get("historial") != null && User.get("historial").length > 0 ? User.get("historial") : JSON.parse(localStorage.getItem("hist-anuncios"));
            var ids = [];
            var top;

            if (hist != null) {
                hist.sort(function(a, b) {
                        return new Date(b.fecha) - new Date(a.fecha);
                });

                if (hist.length > 12) {
                    top = 12;
                }
                else {
                    top = hist.length;
                }

                for (var i = 0; i < top; i++) {
                    ids.push(hist[i].id);
                }
            }

            return ids;
        }
