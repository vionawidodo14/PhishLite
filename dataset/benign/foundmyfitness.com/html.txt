            $('#transcript').show();

          case 'trigger-timeline':
            $('#timeline').show();
            break;
          case 'trigger-glossary':
            $('#glossary').show();
            break;
          case 'trigger-clips':
            $('#clips').show();
            break;
        }
        $(this).addClass('active');

      });
    });

        $('#reference-icon').removeClass("fa-caret-up");
        $('#reference-icon').addClass("fa-caret-down");

        $(this).addClass('expanded');
        $('#reference-label').html("Hide References");
        $('#reference-icon').removeClass("fa-caret-down");
        $('#reference-icon').addClass("fa-caret-up");
        $('.references').find('ol').show();
      }
    });
