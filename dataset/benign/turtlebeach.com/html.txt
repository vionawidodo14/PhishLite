/*==========================================================================================
  Create funcitonality so user can share their compared data
===========================================================================================*/
function shareURL() {
  // Retrieving compare list from local storage
  var retrievedData = localStorage.getItem("compare-list");

  // Parse local storage object to an array
  var parsed_obj = JSON.parse(retrievedData);
  console.log("Parsed List",parsed_obj);

  // Pulling out the product ID's from the parsed_obj
  var productID = '';
  for( i = 0; i < parsed_obj.length; i++ ){
    productID +=  parsed_obj[i].handle + ",";
    // console.log(productID);
  }
  // Make sure the last ID doesn't have a comma
  var productID = productID.substr(0, productID.length-1);

  // Build URL to get ready for copy function
  var baseURL = window.location.hostname + "?compareIDs=" + productID;
  // console.log("updated URL: ",baseURL)

  // paste newly created / updated URL to result section of modal
  $(".display-results").html(baseURL);
  $(".share-url-btn").attr("onclick","copyURL('.display-results')");
  $(".share-url-btn").text("Copy URL");
}

function copyURL(element) {
  // Copy text and change the button styling to let the user know text was copied
  var $temp = $("<input>");
  $("body").append($temp);
  $temp.val($(element).text()).select();
  document.execCommand("copy");
  $(".share-url-btn").text("Copied");
  $(".share-url-btn").css({
    'background-color': '#ffffff',
    'color': '#000000',
    'border': '1px solid rgba(0,0,0,0)'
  });
  // Set a timeout function to change back the copy URL button to it's natural state
  setTimeout(function() {
    $(".share-url-btn").text("Copy URL");
    $(".share-url-btn").css({
      'background-color': '#eb6e26',
      'color': '#ffffff',
      'border': '1px solid rgba(0,0,0,0)'
    });
  }, 3 * 1000);
}
