        function submitExtraInfo(a) {   //a is either "" or "2" depending on submission source
            //if the school wasn't found these fields will show, otherwise text was set elsewhere and will submit normally
            var school = $("#schoolExtra" + a).val();
            var address = $("#addressExtra" + a).val();
            var city = $("#cityExtra" + a).val();
            var state = $("#stateExtra" + a).val();
            var zip = $("#zipExtra" + a).val();
            if (school.length > 0 && address.length > 0 && city.length > 0 && state.length > 0 && zip.length > 0)
                $("#emailOverlayInput" + a).text("School: " + school + ", Address: " + address + ", City: " + city + ", State: " + state + ", Zip: " + zip);
        }

        function parseSchoolInfo(data) {
            var info = data.split("|");
            return { name: info[0], address: info[1], city: info[2], state: info[3], zip: info[4], schoolID: info[5] };
        }

        function chooseSchool(info) {
            if (info.address === "help" || info.address === "refine") {
                $("#txtSchoolSearch").hide();
                $("#txtSchoolSearch2").hide();
                $("#emailOverlayExtra").show();
                $("#emailOverlayExtra2").show();
            }
            else {
                $("#emailOverlayInput").text("School: " + info.name + ", Address: " + info.address + ", City: " + info.city + ", State: " + info.state + ", Zip: " + info.zip);
                $("#emailOverlayInput2").text("School: " + info.name + ", Address: " + info.address + ", City: " + info.city + ", State: " + info.state + ", Zip: " + info.zip);
            }
        }
