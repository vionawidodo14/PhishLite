            function toggleFavorite($favorite) {
                var id = $favorite.data('id');
                var url = "https://www.cartorange.com/area-riservata/toggle/preferiti";

                var action = ($favorite.find('i').hasClass('icon-heart-empty')) ? 'add_favorite' : 'remove_favorite';
                var data = {action:action, id:id};
                console.log(url);
                $.ajax({
                    type: "POST",
                    url: url,
                    data: data,
                    success: function (data) {
                        if (data.success == true && data.code == 200){
                            toggleFavoriteIcon($favorite);
                        }
                    }
                });
            }

            function toggleFavoriteGrowl() {
                $.growl.notice({
                    title: "Preferiti",
                    message: 'La tua lista dei preferiti è stata aggiornata',
                    duration: 2000
                });
            }

            function toggleFavoriteIcon(item) {
                var $favoriteIcon = item.find('i');
                if ($favoriteIcon.hasClass('icon-heart-empty')) {
                    $favoriteIcon.removeClass('icon-heart-empty');
                    $favoriteIcon.addClass('icon-heart');
                    $favoriteIcon.attr('title', 'Rimuovi dai preferiti');
                } else {
                    $favoriteIcon.removeClass('icon-heart');
                    $favoriteIcon.addClass('icon-heart-empty');
                    $favoriteIcon.attr('title', 'Aggiungi ai preferiti');
                }
                if (item.hasClass('remove-favorite')){
                    if ($(window).width() < 768) {
                        location.reload();
                    } else {
                        $("div").find("[data-favorite='" + item.data('id') + "']").remove();
                        $('.item, .item-diaries').matchHeight({
                            byRow: true,
                            property: 'height'
                        });
                        toggleFavoriteGrowl();
                    }
                } else {
                    toggleFavoriteGrowl();
                }
            }
