        function mapCheckoutStepToConverlyticsNumber(step) {
            var map = {
                cart: 1,
                login: null,
                billing: 2,
                shipping_method: 2,
                payment: 2,
                review: 3,
                success: 4
            };
            return map[step];
        }

        function mapDeliveryTimeToConverlyticsGroup(deliveryTime) {
            if (deliveryTime <= 2) return 'a';
            if (deliveryTime <= 5) return 'b';
            if (deliveryTime <= 10) return 'c';
            return 'd';
        }

        function mapEcProductToConverlytics(ecProduct, listType, storeAvailability) {
            var mapped = {
                id: ecProduct.id,
                name: ecProduct.name,
                price: ecProduct.price,
                brand: ecProduct.brand,
                category: ecProduct.category,
                dimension10: ecProduct.in_stock ? 'online-available' : 'online-sold-out',
                dimension11: ecProduct.delivery_time,
                dimension12: mapDeliveryTimeToConverlyticsGroup(ecProduct.deliveryTime),
                dimension13: ecProduct.click_and_reserve ? 'store-available' : 'store-sold-out',
                metric2: ecProduct.original_price,
                metric3: ecProduct.net_price
            };

            if (ecProduct.position) {
                mapped.position = ecProduct.position;
            }

            if (storeAvailability) {
                mapped.dimension14 = storeAvailability;
            }

            if (listType) {
                mapped.dimension16 = listType;
            }

            return mapped;
        }

            $(function () {

                };

                        data['ecommerce']['impressions'].push(impressionData.ecProduct);

                if (detailsElements.length > 0) {

                        if (detailsEventFired) return;

                        var productViewData = [];
                        productViewData['event'] = 'productView';
                        productViewData['eventCategory'] = 'store';
                        productViewData['ecommerce'] = {};
                        productViewData['ecommerce']['detail'] = {
                            'products': []
                        };

                        var storeData = (ev.detail || {}).stores || [];

                        detailsElements.each(function (idx, item) {
                            var detailData = $(item).data();
                            var product = detailData.ecProduct;

                            var storeAvailability = [];

                                // On detail page, index 0 should always be the current product

                            });

                            var trackingList = JSON.parse(localStorage.getItem("productListTracking")) || {};
                            if (trackingList[product.id]) {
                                productViewData['ecommerce']['actionField'] = {list: trackingList[product.id][0]};
                            }

                            productViewData['ecommerce']['detail']['products'].push(mapEcProductToConverlytics(detailData.ecProduct, "", storeAvailability.join("|")));
                        });
                        detailsEventFired = true;
                        dataLayer.push(productViewData);
                    });

                    if (!detailsElements.data('ec-product').click_and_reserve) {
                        window.dispatchEvent(new CustomEvent('Dbl:ClickCollect:Completed'));
                    }


                }

                var promoElements = $('.track-ec-promo');
                if (promoElements.length > 0) {
                    data['ecommerce']['promoView'] = {
                        'promotions': []
                    };
