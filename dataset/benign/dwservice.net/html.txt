        function changeLang(){
            var newlnk="";
            var curlink=window.location.href;
            var ar = curlink.split("?")
            var u = ar[0];
            var q = "";
            var c = "";
            if (ar.length>1){
                q = "?" + ar[1];
            }
            ar = u.split("#")
            u = ar[0];
            if (ar.length>1){
                q = "#" + ar[1];
            }
            if (u.match("\.html$")){
                var ar = u.split("/");
                for (var i=0;i<=ar.length-1;i++){
                    if (i>0){
                        newlnk+="/";
                    }
                    if (i==ar.length-2){
                        newlnk+=dojo.attr("langoption","value");
                    }else{
                        newlnk+=ar[i];
                    }
                }
            }else{
                newlnk=u + dojo.attr("langoption","value") + "/home.html";
            }
            window.location=newlnk + q + c;
        }
