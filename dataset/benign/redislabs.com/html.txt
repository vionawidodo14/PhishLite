(function($) {
    var $nav       = $('#sup-nav');
    var $container = $('#email-first');
    var $form      = $('#emailForm');
    var $email     = $form.find('input[name=wp_email]');
    var $goBtn     = $form.find('button[type=submit]');
    var $submit    = $('#get-started-button');
    var $page      = $(document);
    var $body      = $('body');
    var $mobbutton = $('.top-bar-right button.menu-icon');

    function showForm() {
        //console.log('show form');
        $nav.addClass('hide');
        $container.removeClass('hide');
        $submit.addClass('up2px')
          .text('GO')
          .off('click', showForm)
          .on('click', submitForm);
        $body.addClass('form-visible'); 
        $mobbutton.addClass('up2px'); 
        $email.focus();
        $page.on('click', hideFormOnBlur);
    }
	
    function submitForm(e) {
        // console.log('submit form');
        $email.val($email.val().trim());

        if ($email.val().length) {
            window.open(url(), '_blank');
        }

        hideForm();
        e.preventDefault();
    }

    function url() {
        var base = $form.attr('action'),
             sep = base.indexOf('?') == -1 ? '?' : '&';
        return base + sep + $form.serialize();
    }
	
    function hideForm() {
        // console.log('hide form');
        $nav.removeClass('hide');
        $container.addClass('hide');
        $submit.removeClass('up2px')
          .text('TRY FREE')
          .off('click', submitForm)
          .on('click', showForm);
        $body.removeClass('form-visible');
        $mobbutton.removeClass('up2px'); 
        $page.off('click', hideFormOnBlur);
    }

    function hideFormOnBlur(e) {
        // console.log('click outside');
        var $target = $(e.target);

        if (!$target.closest($container.selector).length && !$target.closest($submit.selector).length) {
            hideForm();
        }
    }

    $submit.on('click', showForm);
    $goBtn.on('click', submitForm)
})(jQuery)
</script>
