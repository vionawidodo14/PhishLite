function getCollection(collectionId, title, outerId)
{

$("#" + outerId).append("<div id='" + id + "'></div>");

    type: 'GET',

    dataType: 'json',

        var products = [];

        {
             var product = {};
             product.id = data.products.hits.hit[k].fields.id;
             product.title = data.products.hits.hit[k].fields.title.text;
             product.description = data.products.hits.hit[k].fields.description;

             product.image = data.products.hits.hit[k].fields.image_url.replace(".png", ".jpg");

    }
});
