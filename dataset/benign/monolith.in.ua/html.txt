<p>Издательство «Монолит» специализируется на разработке и производстве профессиональных руководств по ремонту и эксплуатации автомобилей, предназначенных для автолюбителей и специалистов сервисных станций, а также на выпуске учебной литературы для начинающих водителей. Строгий научный и методический подход, слаженность работы всей команды обеспечивают нашим изданиям статус мастер-класса в области технической и учебной литературы, что подтверждается постоянно растущим спросом на продукцию, а также многочисленными благодарными отзывами как от начинающих автомобилистов, так и от профессиональных специалистов-ремонтников.<br/><br/>
                У всех наших авторов технической литературы имеется личный опыт в эксплуатации, обслуживании и ремонте собственных автомобилей, что обуславливает толковую и доступную подачу материала на страницах выпускаемых руководств. Таким образом, книги по обслуживанию автомобилей одновременно являются профессиональными справочниками, содержащими подробнейшие инструкции по капитальному ремонту, со всеми необходимыми спецификациями, и великолепными пособиями для начинающих владельцев авто, которых заядлые автолюбители иногда с доброй иронией называют «чайниками». Данная особенность выгодно отличает «монолитовские» руководства на профильном рынке, обеспечивая стабильно растущий интерес к ним со стороны книготорговцев, профессиональных ремонтников и автомобилистов различных уровней подготовки.<br/><br/>
                Над созданием учебной литературы работают высококвалифицированные методисты, преподаватели автошкол, специалисты в области права. На этапе подготовки к печати издания проходят несколько этапов рецензирования, один из них — в Центре безопасности дорожного движения и автоматизированных систем ГАИ МВД Украины. Книги данной серии содержат массу полезной информации по Правилам дорожного движения, вопросам правовой защиты автомобилистов, оказания медицинской помощи в дороге и многое другое. Учебные книги издательства «Монолит» отличаются высоким качеством печати, иллюстративных материалов – большая часть рисунков выполнена с использованием 3D-технологий. Доступность подачи материала, выделения по тексту и специальные вставки облегчают процесс усвоения материала. Мы приложили максимум усилий, чтобы обучение приносило будущим водителям максимум удовольствия и пользы.<br/><br/>
                Важнейшую роль в развитии издательства играет кропотливая работа отдела маркетинга, благодаря ей мы можем своевременно реагировать на обновления автомобильного рынка и обеспечивать актуальность выпускаемых изданий.<br/><br/>
                Наша компания всегда готова к сотрудничеству как в плане поставки продукции, так и в плане обмена информацией. С нетерпением и энтузиазмом ждем ваших заказов, предложений и замечаний, уважаемые книготорговцы, водители и специалисты по ремонту.</p>

	var slideIndex = -1;
	var timeoutId;
	var paused = false;
	var from = 3;
	var clicked = 0;
	var resultArticles = 1 * 12.333333333333;

	$('.mpSlide').eq(0).addClass('visibleSlide');
	$(document).ready(function() {
		setPager();
		carousel();
	});
	function setPager(){
		var slidesCount = $('.mpSlide').length;
		if (slidesCount == 0) return;
		var pagerContent = '<ul>';
		for (var i = 0; i < slidesCount; i++){
			pagerContent += '<li id="page_' + i + '" onclick="setSlide(' + i + ')"></li>';
		}
		pagerContent += '</ul>';
		$('#pager').html($('#pager').html() + pagerContent);
		$('#page_1').addClass('activePagerItem');
	}
	function carousel() {
		if (!paused) {
			swipeSlide('next');
		}
		else {
			paused = false;
		}
		timeoutId = setTimeout(carousel, 4000);
	}
	function setSlide(i){
		clearTimeout(timeoutId);
		paused = false;
		slideIndex = i - 1;
		carousel();
		clearTimeout(timeoutId);
		paused = true;
	}
	function swipeSlide(direction){
		$('.activePagerItem').css('background', 'white');
		var slider = $('.mpSlide');
		var currentSlide = $('.visibleSlide');
		currentSlide.removeClass('visibleSlide');
		$('.activePagerItem').removeClass('activePagerItem');
		if (direction == 'next') {
			slideIndex++;
			if (slideIndex >= slider.length) {
				slideIndex = 0
			}
		} else {
			slideIndex--;
			if (slideIndex <= -1) {
				slideIndex = slider.length - 1;
			}
		}
		slider.eq(slideIndex).addClass('visibleSlide');
		$('#page_' + slideIndex).addClass('activePagerItem');
		$('.activePagerItem').css('background',$(".visibleSlide .banner-button").css('background-color'));
		setTimeout(function() {
			$('#slider .mpSlide').removeClass("move");
			$('#slider .moveLeft').removeClass("moveLeft");
			$('#slider .moveRight').removeClass("moveRight");

		}, 500);
	}
	$('#slider').swipe({
		swipeStatus:function(event, phase, direction, distance, duration, fingers)
		{
			if (phase=="move" && direction =="right" && distance > "10") {
				$('#slider .mpSlide').addClass("move");
				$('#slider .visibleSlide').addClass("moveRight");
				swipeSlide('prev');
				paused = true;
				return false;
			}
			if (phase=="move" && direction =="left" && distance > "10") {
				$('#slider .mpSlide').addClass("move");
				$('#slider .visibleSlide').addClass("moveLeft");
				swipeSlide('next');
				paused = true;
				return false;
			}
		}
	});

	var sliderClick = 0;
	var mouseMove = 0;
    $(document).on("touchstart  mousedown", "#slider", function(event) {
		sliderClick = 1;
		mouseMove = 0;
        event.stopPropagation();
	});
    $(document).on("touchmove  mousemove", "#slider", function(event) {
		if (sliderClick == 1) {
			mouseMove++;
		}
        event.stopPropagation();
	});
    $(document).on("touchend  mouseup", "#slider", function(event) {
		if ((sliderClick == 1) && (mouseMove < 3)) {
			sliderClick = 0;
			$('#slider .visibleSlide a')[0].click();
		} else {
            sliderClick = 0;
        }
        event.stopPropagation();
	});

	$('#more_articles').on('click', function () {
		clicked++;

		ajax_load_articles();
		from = from + 3;
	});

	function ajax_load_articles() {
		$.ajax({
			type: 'POST',
			url: 'https://monolith.in.ua/wp-admin/admin-ajax.php',
			data: {
				action: 'get_articles',
				from: from
			},
			success: function (data) {
				dataResult = JSON.parse(data);
				var html = '';
                for (var i = 0; i < dataResult.length; i++) {
                    html += '<div class="col-lg-4 col-md-4 col-sm-4">';
                    if (dataResult[i]['text'] == "") {
                        html += '<a href="/';
                    } else {
                        html += '<a href="/useful/';
                    }
                    html += dataResult[i]['slug'];
                    html += '">';
                    html += dataResult[i]['short_desc'];
                    html += '</a>';
                    html += '</div>';
                }
                $('#interesting-holder .articles .hidden-xs').html($('#interesting-holder .articles .hidden-xs').html() + html);

				// create alt for image
				$('.article-interesting').each(function () {
					var currAttrAlt = $(this).find('p.articles_name').text();
					$(this).find('img').attr('alt', currAttrAlt);
					$(this).find('img').attr('title', currAttrAlt);
				});

				if( clicked == Math.floor(resultArticles) ) {
					$('.more_articles').css('display','none');
				}
			}
		});
	}

        if ($('#advnumb').val() == '' || $('#advnumb').val() == ' ' || $('#advnumb').val().length == 0) {
            showFieldError('send_advertise_phone_need', 'advnumb');
            validationResult = false;
        }

        return validationResult;
    }

    $('#advertisingSend .btn').on('click', function () {
        $('#advertising').modal('hide');
    });

    $('#SendDesign').on('click', function (e) {
        e.preventDefault();
        if(checkInputsDesign()) {
            $.ajax({
                type: 'POST',
                url: "https://monolith.in.ua/wp-admin/admin-ajax.php",
                data: {
                    action: 'orderDesign',
                    design_title: $('title').html(),
                    design_user_message: $('#dmes').val(),
                    design_user_name: $('#dname').val(),
                    design_user_email: $('#demail').val(),
                    design_user_phone: $('#dnumb').val()
                },
                success: function (data) {
                    $('#design').modal('hide');
                    $("#designSend").modal('show');
                }
            });
            setTimeout(function () {
                $("#designSend").modal('hide');
            }, 5000);
        }
    });

    function checkInputsDesign() {
        $('.form_error').css('display', 'none');

        var validationResult = true;

        if ($('#demail').val() == '' || $('#demail').val() == ' ' || $('#demail').val().length == 0) {
            showFieldError('send_design_email_need', 'demail');
            validationResult = false;
        }
        else if ( $('#demail').val().length > 255 ) {
            showFieldError('send_design_email_cause_max', 'demail');
            validationResult = false;
        }
        else if ( !regEmail.test($('#demail').val()) ) {
            showFieldError('send_design_email_correct', 'demail');
            validationResult = false;
        }
        if ($('#dname').val() == '' || $('#dname').val() == ' ' || $('#dname').val().length == 0) {
            showFieldError('send_design_name_need', 'dname');
            validationResult = false;
        }
        else if ($('#dname').val().length > 255) {
            showFieldError('send_design_name_cause_max', 'dname');
            validationResult = false;
        }
        else if (!/^[A-Za-zА-Яа-я0-9ё\s,\.]+$/i.test($('#dname').val())) {
            showFieldError('send_design_name_correct', 'dname');
            validationResult = false;
        }
        if ($('#dmes').val().length > 2000) {
            showFieldError('send_design_text_cause_max', 'dmes');
            validationResult = false;
        }

        if ($('#dnumb').val() == '' || $('#dnumb').val() == ' ' || $('#dnumb').val().length == 0) {
            showFieldError('send_design_phone_need', 'dnumb');
            validationResult = false;
        }

        return validationResult;
    }

        $('#design').modal('hide');
