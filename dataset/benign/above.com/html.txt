    function set_auction_type(search_type){
        document.getElementById('auctype').value        = '';
        document.getElementById('doSearch').value       = '';
        document.getElementById('SortTypeMP').value         = '';
        document.getElementById('SortTypeMP').value       = '';

            document.getElementById('auctype').value        = search_type;
            document.getElementById('doSearch').value       = 'advanced';
        }

        
        if(search_type==='traffic'){
            document.getElementById('SortOnMP').value       = 'reported_sort';
            document.getElementById('SortTypeMP').value     = 'DESC';
        }
        if(search_type==='revenue'){
            document.getElementById('SortOnMP').value       = 'revenue_sort';
            document.getElementById('SortTypeMP').value     = 'DESC';
        }
    }
    
    function validate_search_text(){
        var error = '';
        search_text = trim(document.getElementById('frmSearchDomain').value);
        
        if(search_text===''){
            error += "Please Enter Search Text.\n";
        }
        else if(search_text.match(/^\s+$/i)){
            error += "Invalid characters used. Please use letters and numbers.\n";
        }
        else if(search_text!=='' && !search_text.match(/^([a-zA-Z0-9\-\.],?)+$/i)){
            error += "Invalid characters used. Please use letters and numbers.\n";
        }
        
        if(error !== ''){
            alert(error);
            return false;
        }
		
        return true;
    }
    
    function trim(str) {
        return str.replace(/^\s\s*/, '').replace(/\s\s*$/, '');
    }
</script>
