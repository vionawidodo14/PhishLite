function printPdf(pdf_filename, url, custom, addParam) {
  printPdfButton = $(this);
  printPdfButton.hide();
  $('.icon-pdf-circle').hide();
  ajaxLoader.show();
  try{
  if (custom && typeof printPdfCustom != 'undefined') {
    printPdfCustom(pdf_filename);
    return;
  }
  if (url == '') {
    url = window.location.href;
  }
  var hash = url.indexOf("#");
  if(hash != -1) {
    url = url.substring(0, hash);
  }
  if(addParam !== null && typeof addParam == 'object'){
    for(var par in addParam) {
      if(url.indexOf(par + "=") == -1) {
        url += (url.indexOf("?")==-1 ? "?" : "&") + par + "=" + addParam[par]; 
      }
    }
  }
  printPdfUrl(pdf_filename, url);
  }catch(pdferror){
    ajaxLoader.hide();
    printPdfButton.show();
    $('.icon-pdf-circle').show();
    alert('error');
    }
  console.log('download');
}

function printPdfUrl(pdf_filename, url) {
  url = url.replace(/&/g, ';');
  var token = new Date().getTime();
  $('#iframePopup').attr('src', '/en/pdf' + '?name=' + pdf_filename + '&token=' + token + '&url=' + url);
  checkPdfDownload(token);
}

function printPdfHtml(pdf_filename, html_el) {
  var token = new Date().getTime();
  $('#iframePopup').attr('src', '/en/pdf-html-poster' + '?name=' + pdf_filename + '&token=' + token);
  $('#iframePopup').on('load', function(){
    $('#iframePopup').off('load');
    var content = $('#iframePopup').contents();
    content.find("#html").val(html_el.html());
    content.find("#download-form").submit();
  });
  checkPdfDownload(token);
}
  
  
var fileDownloadCheckTimer;
var intervalcount = 0;

function checkPdfDownload(token) {
  var fileDownloadSetTimout = window.setTimeout(function(){
    ajaxLoader.hide();
    printPdfButton.show();
    $('.icon-pdf-circle').show();
  console.log('Timeout timer');
  clearInterval(fileDownloadCheckTimer);
  }, 30000);
  fileDownloadCheckTimer = window.setInterval(function () {checkit(token);}, 1000);
}

function checkit(token){
  var cookieValue = $.cookie('fileDownloadToken');
  var errorCookieValue = $.cookie('fileDownloadErrorToken');
  intervalcount++;
  if(intervalcount > 19){
    clearInterval(fileDownloadCheckTimer);
    console.log('End timer:'+fileDownloadCheckTimer);
  }
  if (cookieValue == token) {
    ajaxLoader.hide();
    printPdfButton.show();
    $('.icon-pdf-circle').show();
    console.log('Log download:'+cookieValue+' token:'+token);
    clearInterval(fileDownloadCheckTimer);
  } else if (errorCookieValue == token) {
    ajaxLoader.hide();
    printPdfButton.show();
    $('.icon-pdf-circle').show();
    console.log('Log download:'+errorCookieValue+' token:'+token);
    clearInterval(fileDownloadCheckTimer);
  }
  console.log('Token tick:'+token);
}

function initOptionForms() {
  $(document).on('click', '.toggle-option-form', function () {
    var form = $(this).nextAll(".option-form:eq(0)");
     if ($(this).is(":checked")) {
      if (form.length) {
        form.show().find(":input:eq(0):not(.hasDatepick)").focus();
      }
      if ($(this).is(":radio")) {
        var oname = $(this).attr("name");
        $('input[name="' + oname + '"]').not(this).each(function (index) {
          $(this).nextAll(".option-form:eq(0)").hide();
        });
      }
    } else {
      form.hide();
    }
  });
}

function setOptionForms() {
  $(".toggle-option-form").each(function(index) {
    if ($(this).is(":checked")) {
      $(this).nextAll(".option-form:eq(0)").show();
    } else {
      $(this).nextAll(".option-form:eq(0)").hide();
    }
  });
}
