        window.addEventListener("load", function () {
            if (typeof countdowns === 'undefined') {
                const sections = Array.from(document.querySelectorAll('.tlh-block'));
                sections.forEach(function(block) {
                    const countdowns = Array.from(block.querySelectorAll('.tlh-block__content'));
                    const nbBlocks = countdowns.length;
                    var countWidth = 120;

                    countdowns.forEach(function(item) {
                        const trads = new Map();
                        trads.set('Starts in', 'Débute dans');
                        trads.set('Ends in', 'Termine dans');
                        trads.set('day', 'jour');
                        trads.set('second', 'seconde');
                        trads.set('Expired', 'Expiré');
                        trads.set('Coming soon', 'À venir');
                    
                        //Translate countdown in Hydra
                        var hydra_lang  = document.querySelector('.tlh-block').getAttribute('sectionid');
                        if ( hydra_lang.indexOf('fr') !== -1 ) {
                            document.documentElement.lang = 'fr';
                        }

                        function getTrads(keystring) {
                            return document.documentElement.lang === 'fr' ? trads.get(keystring) : keystring;
                        }
                        //Verify if Eastern Standard Time or Eastern Daylight Saving Time
                        Date.prototype.stdTimezoneOffset = function () {
                            var jan = new Date(this.getFullYear(), 0, 1);
                            var jul = new Date(this.getFullYear(), 6, 1);
                            return Math.max(jan.getTimezoneOffset(), jul.getTimezoneOffset());
                        }
                        Date.prototype.isDstObserved = function () {
                            return this.getTimezoneOffset() < this.stdTimezoneOffset() ? '-4' : '-5';
                        }

                        var days, hours, minutes, seconds;
                        var startDate = new Date(item.querySelector('.tlh-block__start').innerHTML).getTime();
                        var endDate = new Date(item.querySelector('.tlh-block__end').innerHTML).getTime();
                        var countText = getTrads('Ends in');
                        var output = '';
                        var offset = new Date().isDstObserved();

                        if ( nbBlocks % 3 === 0 ) {
                            item.classList.add('tlh-block__3-col');
                        }

                        if (isNaN(endDate)) {
                            return;
                        }

                        setInterval(calculate, 1000);
                    
                        function calculate() {
                            function calcTime(date, offset) {
                                utc = date.getTime() + (date.getTimezoneOffset() * 60000);
                                return new Date(utc + (3600000*offset));
                            }
                            
                            var now = new Date();
                            now = calcTime(now, offset);
        
                            var timeRemaining = parseInt((endDate - now) / 1000);
                            var timeBeforeStart = parseInt((startDate - now) / 1000);
                            var countdown = timeRemaining;

                            if (timeBeforeStart > 0) {
                                item.classList.add('upcoming');
                                countText = getTrads('Starts in');
                                countdown = timeBeforeStart;
                                item.querySelector('.tlh-block__msg').innerHTML = getTrads('Coming soon');
                            } else {
                                item.classList.add('current');
                            }
                            if (countdown >= 0) {
                                days = parseInt(parseInt(countdown / 86400), 10);
                                countdown = countdown % 86400;
                    
                                hours = parseInt(countdown / 3600);
                                countdown = countdown % 3600;
                    
                                minutes = ('0' + parseInt(countdown / 60)).slice(-2);
                                countdown = countdown % 60;
                    
                                seconds = ('0' + parseInt(countdown)).slice(-2);countText
                    
                                const daysText = days == 1 ? getTrads('day') : getTrads('day') + 's';
                                const secondText = seconds <= 1 ? getTrads('second'): getTrads('second') + 's';
                    
                                if ((timeBeforeStart > 0 && timeBeforeStart < 60) || (timeRemaining >= 0 && timeRemaining < 60)) {
                                    //Less than a minute remaining
                                    output = seconds + ' ' + secondText;
                                } else {
                                    output = (days ? days + ' ' + daysText + ' ' : '') + (hours + ':') + minutes 
                                }
                            } else {
                                countText = getTrads('Expired');
                                output = '';
                                item.classList.add('expired');
                            }
                            item.querySelector('.tlh-block__timer').style.background = '#fff';
                            item.querySelector('.tlh-block__ends-in').innerHTML = countText;
                            item.querySelector('.tlh-block__countdown').innerHTML = output;
                        }
                        setTimeout(function() {
                            var tempWidth = item.querySelector('.tlh-block__timer').clientWidth;
                            if (tempWidth > countWidth) {
                                countWidth = tempWidth;
                            }
                         }, 1000);
                    }); 
                    //Set timer width
                    setTimeout(function() {
                        countdowns.forEach(function(item) {
                            item.querySelector('.tlh-block__timer').style.width = countWidth + 'px';
                        });
                    }, 1000);
                }); 
            }    
        });
