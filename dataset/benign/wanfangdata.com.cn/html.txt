         function suggestionTest(showFlag){
             var textReg = /^.{20,300}$/;
             var suggestionTextVal = suggestionText.val();
             var textLength= suggestionText.val().length;
             var counter = $("#counter");
             var countText = $(".countText");
             var countEmpty =  $(".countEmpty");
             if( counter.text()<0){
                 counter.text(0);
             }else{
                 counter.text(300-textLength);
             }
             if(!textReg.test(suggestionTextVal)){
                 countText.show();
                 countEmpty.hide();
                 showFlag++;
             }else{
                 countText.hide();
             }
             if(suggestionTextVal == ""){
                 countText.hide();
                 countEmpty.show();
                 showFlag++;
             }else{
                 countEmpty.hide();
             }
             return showFlag;
         }
         function userNameTest(showFlag){
             var textReg = /^.{2,10}$/;
             var userNameError = $(".userNameError");
             var userNameEmpty = $(".userNameEmpty");
             var userNameVal = userName.val();
             if(!textReg.test(userNameVal)){
                 userNameError.show();
                 userNameEmpty.hide();
                 showFlag++;
             }else{
                 userNameError.hide();
             }
             if(userNameVal==""){
                 userNameEmpty.show();
                 userNameError.hide();
                 showFlag++;
             }else{
                 userNameEmpty.hide();
             }
             return showFlag;

                 emailEmpty.hide();
             }
             return showFlag;

             var cookieText = "";
             cookieText += encodeURIComponent(name) + "=" + encodeURIComponent(value);

             }
             if (domain) {
                 cookieText += "; domain=" + domain;
             }
             if (secure) {
                 cookieText += "; secure";
             }

    function GetPageScroll() {
        var x, y; if (window.pageYOffset) {    // all except IE
            y = window.pageYOffset;
            x = window.pageXOffset;
        } else if (document.documentElement && document.documentElement.scrollTop) {    // IE 6 Strict
            y = document.documentElement.scrollTop;
            x = document.documentElement.scrollLeft;
        } else if (document.body) {    // all other IE
            y = document.body.scrollTop;
            x = document.body.scrollLeft;
        }
        /*return {X:x, Y:y};*/
        return y;
    }
    setInterval(function () {
