    for (i = 1; i < QUOTE_TABLE.length; i++) {
      if (QUOTE_TABLE[i].waiting == true) {
        return;
      }
    }
    // find the max sessions
    maxSessions = 0;
    for (i = 1; i < QUOTE_TABLE.length; i++) {
      if (QUOTE_TABLE[i].sessions > maxSessions) {
        maxSessions = QUOTE_TABLE[i].sessions;
      }
    }
    // find the plan with the highest sessions
    bestPlan = 'starter';
    for (var plan in AVAILABLE_PLANS) {
      if (AVAILABLE_PLANS.hasOwnProperty(plan)) {
        if (AVAILABLE_PLANS[plan].basesessions < maxSessions) {
          bestPlan = plan;
        }
      }
    }
    document.getElementById("stratusType").value = bestPlan;
    SELECTED_PRODUCT = bestPlan;
    recalibrateQuoteTable();
  }

  function initQuoteTable() {
    for (i = 1; i <= 12; i++) {
      QUOTE_TABLE[i] = {
        dateStart: '',
        dateEnd: '',
        sessions: null,
        storage: 0,
        amount: 0,
        waiting: true
      }
    }
    date = new Date();
    dateEnd = new Date();
    var j = 1;
    for (i = 0; i < 12; i++) {
      dateEnd.setMonth(date.getMonth() + 1);
      QUOTE_TABLE[j].dateStart = date.getFullYear() + '-' +
        MONTH_NUMBERS[date.getMonth()] + '-01';
      QUOTE_TABLE[j].dateEnd = dateEnd.getFullYear() + '-' +
        MONTH_NUMBERS[dateEnd.getMonth()] + '-01';
      date.setMonth(date.getMonth() - 1);
      dateEnd.setMonth(date.getMonth());
      dateEnd.setFullYear(date.getFullYear());
      j++;
    }
  }

  function recalibrateStorage(storage) {
    SELECTED_STORAGE = storage.value;
    for (i = 1; i < QUOTE_TABLE.length; i++) {
      QUOTE_TABLE[i].storage = storage.value;
    }

  function recalibrateQuoteTable() {
    for (i = 1; i < QUOTE_TABLE.length; i++) {
      gasessions = QUOTE_TABLE[i].sessions - AVAILABLE_PLANS[SELECTED_PRODUCT].basesessions;
      gbstorage = QUOTE_TABLE[i].storage - AVAILABLE_PLANS[SELECTED_PRODUCT].basestorage;
      if (gasessions < 0) {
        gasessions = 0;
      }
      if (gbstorage < 0) {
        gbstorage = 0;
      }
      sessionscost = gasessions * AVAILABLE_PLANS[SELECTED_PRODUCT].sessionrate;
      storagecost = gbstorage * AVAILABLE_PLANS[SELECTED_PRODUCT].storagerate;
      QUOTE_TABLE[i].amount = (AVAILABLE_PLANS[SELECTED_PRODUCT].basecost + sessionscost + storagecost) *
        AVAILABLE_REGIONS[
          SELECTED_REGION].ratefactor;
    }
  }

  function drawQuoteTable() {
    for (i = 1; i < QUOTE_TABLE.length; i++) {
      document.getElementById('tdDate' + i).innerHTML = formatDateQuoteTable(QUOTE_TABLE[i].dateStart);
      document.getElementById('tdSessions' + i).innerHTML = formatSessionsQuoteTable(QUOTE_TABLE[i].sessions);
      document.getElementById('tdAmount' + i).innerHTML = formatPricingQuoteTable(QUOTE_TABLE[i].amount);
    }
  }

  function drawSummaryTable() {
    for (var plan in AVAILABLE_PLANS) {
      if (AVAILABLE_PLANS.hasOwnProperty(plan)) {
        document.getElementById('price' + plan.charAt(0).toUpperCase() + plan.substring(1) + 'Summary').innerHTML =
          formatPricingSummary(AVAILABLE_PLANS[plan].basecost * AVAILABLE_REGIONS[SELECTED_REGION].ratefactor);
      }
    }
  }

  function drawDetailTable() {
    for (var plan in AVAILABLE_PLANS) {
      if (AVAILABLE_PLANS.hasOwnProperty(plan)) {
        document.getElementById('price' + plan.charAt(0).toUpperCase() + plan.substring(1) + 'Detail').innerHTML =
          formatPricingDetail(AVAILABLE_PLANS[plan]
            .basecost *
            AVAILABLE_REGIONS[
              SELECTED_REGION]
            .ratefactor);
        document.getElementById('sessionrate' + plan.charAt(0).toUpperCase() + plan.substring(1) + 'Detail').innerHTML =
          formatSessionrateDetail(AVAILABLE_PLANS[plan]
            .sessionrate *
            AVAILABLE_REGIONS[
              SELECTED_REGION]
            .ratefactor);
        document.getElementById('storagerate' + plan.charAt(0).toUpperCase() + plan.substring(1) + 'Detail').innerHTML =
          formatStoragerateDetail(AVAILABLE_PLANS[plan]
            .storagerate *
            AVAILABLE_REGIONS[
              SELECTED_REGION]
            .ratefactor);
      }
    }
  }

  function formatPricingSummary(price) {
    return "$" + price.toFixed(0).replace(/\d(?=(\d{3})+\.)/g, '$&,') + "/mo";
  }

  function formatPricingDetail(price) {
    return "$" + price.toFixed(0).replace(/\d(?=(\d{3})+\.)/g, '$&,') + "/mo";
  }

  function formatPricingQuoteTable(price) {
    return "$" + price.toFixed(0).replace(/\d(?=(\d{3})+\.)/g, '$&,');
  }

  function formatSessionrateDetail(price) {
    return "$" + price.toFixed(5) + "/ea";
  }

  function formatStoragerateDetail(price) {
    return "$" + price.toFixed(2) + "/gb";
  }

  function formatDateQuoteTable(date) {
    datearr = date.split('-');
    return "" + MONTH_NAMES[datearr[1] - 1] + "<br />" + datearr[0] + "";
  }

  function formatSessionsQuoteTable(sessions) {
    if (sessions == null) {
      return '<font size=1>Connect GA To See</font>';
    } else {
      return sessions.replace(/\B(?=(\d{3})+(?!\d))/g, ",");
    }
  }

  function getCookieValue(a) {
    var b = document.cookie.match('(^|[^;]+)\\s*' + a + '\\s*=\\s*([^;]+)');
    return b ? b.pop() : '';
  }
