                            $("#fca-esma-container").css('margin-top','');
                            $(window).scroll(function () {
                                if ($(this).scrollTop() > 10) {

                                    $('#floating-menu').slideUp(200);
                                    $('.lcg-navigation').slideDown(200);

                    if ($("#mobile-nav").css("display") == "none") {

                    if ($(this).parent().children('.fl-dropdown-menu').is(':visible')) {

                        $(this).removeClass('open-submenu');
                    }
                    else {
                        $(this).parent().children('.fl-dropdown-menu').slideDown();
                        $(this).addClass('open-submenu');
                    }
