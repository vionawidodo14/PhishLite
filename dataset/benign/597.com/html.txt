function stringify(obj){
var JSON = window.JSON || void 0;
if (typeof JSON !== void 0 && JSON instanceof Object && typeof JSON.stringify === 'function') {
return JSON.stringify.apply(null, arguments);
}
var strings = [];
var arr = [];
var isArray = $.isArray(obj);
if (obj === null){
return 'null';
} else if (typeof obj === 'string') {
return '"' + obj.replace(/"/g, '\\"') + '"';
} else if(obj instanceof Object) {
$.each(obj, function(key, item) {
if (item instanceof Object && !$.isPlainObject(item)) return !!1;
// 稀疏数组使用hasOwnProperty会被跳过, 提前处理
if (isArray) {
if (!isNaN(key) && key >= 0 && key < obj.length && key === ~~key) {
strings[key] = stringify(item === void 0 ? null : item);
}
return !!1;
}
if (!obj.hasOwnProperty(key)) return !!1;
strings.push('"' + key + '":' + stringify(item));
});
return isArray ? '[' + strings.join(',') + ']' : '{' + strings.join(',') + '}';
}
// 基本类型
return obj.toString();
}

len = ~~len;

len = len < 0 ? 32 : len;
radix = radix < 2 || radix > 36 ? 36 : radix;
