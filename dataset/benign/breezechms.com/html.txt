    }

        setTimeout(function(){

            demo_email_home_hero_header_options.placement = 'top';

        }, 210);



    }

    function change_demo_modal_view(view) {

        switch(view) {

            case "progress_bar":
                $('div.demo_modal_progress_bar_container').removeClass('hidden');
                $('div.demo_modal_footer').addClass('hidden');
                $('div#demo_inputs_row').addClass('hidden');
                $('div.demo_modal_size').addClass('modal-md');
                $('div.demo_modal_size').removeClass('modal-sm');



                // Disable closing modal when clicking background or clicking escape (need to use this more complex way of disabling modal as the traditional method only works when opening a new modal, not modifying an already-open modal)
                var $modal = $('div#demo.modal');
                var keyboard = false; // Prevent to close by ESC
                var backdrop = 'static'; // Prevent to close on click outside the modal

                $modal.data('bs.modal').options.keyboard = keyboard;
                $modal.data('bs.modal').options.backdrop = backdrop;

                if(keyboard === false) {
                    $modal.off('keydown.dismiss.bs.modal'); // Disable ESC
                } else { //
                    $modal.data('bs.modal').escape(); // Resets ESC
                }

                break;

            case "form":
                $('div.demo_modal_progress_bar_container').addClass('hidden');
                $('div.demo_modal_footer').removeClass('hidden');
                $('div#demo_inputs_row').removeClass('hidden');
                $('div.demo_modal_size').addClass('modal-sm');
                $('div.demo_modal_size').removeClass('modal-md');

                // Enable closing modal when clicking background or clicking escape (need to use this more complex way of enabling modal as the traditional method only works when opening a new modal, not modifying an already-open modal)
                var $modal = $('div#demo.modal');
                var keyboard = true; // Prevent to close by ESC
                var backdrop = true; // Prevent to close on click outside the modal

                $modal.data('bs.modal').options.keyboard = keyboard;
                $modal.data('bs.modal').options.backdrop = backdrop;

                if(keyboard === false) {
                    $modal.off('keydown.dismiss.bs.modal'); // Disable ESC
                } else { //
                    $modal.data('bs.modal').escape(); // Resets ESC
                }

                break;

        }

    }

    // define demo_subdomain outside of function so it can be updated globally
    var demo_subdomain = '0';
    var demo_oid = '';

    function create_and_populate_demo(email) {

        try {

            // create site
            $.ajax({
                url: '../../../../ajax/create_demo',
                dataType: 'json',
                data: {
                    email: email
                },
                cache: false,
                success: function(organization){

                    // mixpanel track
                    breeze_track("Demo Started", null);

                    // capterra track
                    var capterra_vkey = 'e117442959897cb893d8af1fd76a5243',
                    capterra_vid = '2093311',
                    capterra_prefix = (('https:' == document.location.protocol) ? 'https://ct.capterra.com' : 'http://ct.capterra.com');

                    (function() {
                    var ct = document.createElement('script'); ct.type = 'text/javascript'; ct.async = true;
                    ct.src = capterra_prefix + '/capterra_tracker.js?vid=' + capterra_vid + '&vkey=' + capterra_vkey;
                    var s = document.getElementsByTagName('script')[0]; s.parentNode.insertBefore(ct, s);
                    })();

                    demo_subdomain = organization.subdomain;
                    demo_oid = organization.id;

                    // populate site (runs beyond redirect)
                    $.ajax({

                        url: '../../../../ajax/populate_demo',
                        type: 'POST',
                        data: {
                            oid: demo_oid
                        },
                        success: function(result) {
                            // populated successfully
                        },
                        complete: function() {
                        }

                    });

                }
            });

        } catch(err) {

        }

    }

    function redirect_to_demo() {

        // if demo hasn't loaded yet
        if(demo_subdomain == '0') {

            $('span#progress_status_text').text('Almost there...');
            window.setTimeout(redirect_to_demo, 300); // wait x amount of milliseconds and try again

       // if demo has loaded
        } else {

            // go there
            window.location.href = "https://"+demo_subdomain+".breezechms.com";

        }

    }

    function start_demo_progress_bar() {

        var progress_speed = 20; // original was 12

        var $bar = $('.demo_modal_progress_bar_container').find('.bar');

        var progress = setInterval(function() {

            if ($bar.width() > 50) {

            }

            if ($bar.width() > 100) {
                $('span#progress_status_text').text('Creating events...');
            }

            if ($bar.width() > 150) {
                $('span#progress_status_text').text('Attaching turbo thrusters...');
            }

            if ($bar.width() > 180) {
                $('span#progress_status_text').text('Synchronizing particle accelerator...');
            }

            if ($bar.width() > 210) {
                $('span#progress_status_text').text('Analyzing space time continuum...');
            }

            if ($bar.width() > 240) {
                $('span#progress_status_text').text('Calibrating flux capacitor...');
            }

            if ($bar.width() > 270) {
                $('span#progress_status_text').text('Consulting Jedi council...');
            }

            if ($bar.width() > 300) {
                $('span#progress_status_text').text('Prepping fusion reactor...');
            }

            if ($bar.width() > 340) {
                $('span#progress_status_text').text('Entering hyperspace...');
            }

            if ($bar.width() > 380) {

                clearInterval(progress);
                $('.bar').width(400);

            } else {
                $bar.width($bar.width()+progress_speed);
                $bar.text(Math.round($bar.width()/4) + "%");
            }

        }, 120); // 120 was original

    }

    function record_demo_lead(name, email, subscribe_to_blog) {

        $.ajax({
            url: '../../../../ajax/record_lead',
            type: 'POST',
            data: {
                name: name,
                email: email,
                subscribe_to_blog: subscribe_to_blog
            },
            success: function(result){
                ga('send', 'event', 'front', 'demo', 'submit_info', 1);

                
                    if(subscribe_to_blog) {
                          mixpanel.track(
                              "Blog Subscription",
                              {
                                  "source-category": "demo",
                                  "blog": "advice",
                                  "email": email
                              }
                          );
                    }

                
                //window.location.href = "../../front/create?email=" + encodeURIComponent(email); // this was the old method
            }
        });

    }

    $('form#demo').on('submit', function(e) {
        e.preventDefault();

        var name = $('input#name').val();
        var email = $('input#email').val();
        var subscribe_to_blog = false;
        if($('input#subscribe_to_blog:checkbox:checked').length) { subscribe_to_blog = true; }
        var address = $('input#address').val();
        var errors = 0;

        //if(!phone) { $('div#phone_error').removeClass('hidden'); $('input#phone').focus(); errors++; } else { $('div#phone_error').addClass('hidden'); }
        if(!email) { $('div#email_error').removeClass('hidden'); $('input#email').focus(); errors++; } else { $('div#email_error').addClass('hidden'); }
        if(!name) { $('div#name_error').removeClass('hidden'); $('input#name').focus(); errors++; } else { $('div#name_error').addClass('hidden'); }

       if(!address && errors == 0) {

            $('input.launch_demo').addClass('disabled');
            $('input.launch_demo').val('Loading...');

            change_demo_modal_view('progress_bar');
            start_demo_progress_bar();
            create_and_populate_demo(email);
            record_demo_lead(name, email, subscribe_to_blog);

       }

    });

    });

    function complete_purchase() {

        var errors = [];

        $('#signup_errors').addClass('hidden');

        if($('a#complete_purchase').hasClass('disabled')) {
            return false;
        }

        $('#sign_up_page_2_error').addClass('hidden');

        // disable the submit button to prevent repeated clicks
        $('a#complete_purchase').addClass('disabled');
        $('a#complete_purchase').html('<i class="fas fa-spinner fa-spin"></i> Processing...');

        // get data
        var data = get_signup_data();

        var card_extra_data = {};
        card_extra_data.name = data.card_name;
        card_extra_data.address_line1 = data.address_line1;
        card_extra_data.address_city = data.address_city;
        card_extra_data.address_state = data.address_state;
        card_extra_data.address_zip = data.address_zip;

        var country = 7;

        if(card_extra_data.name.length === 0 || !card_extra_data.name.trim()) {
          errors.push('Please provide the name listed on your card');
        }

        if(card_extra_data.address_line1.length === 0 || !card_extra_data.address_line1.trim()) {
          errors.push('Your billing address is missing');
        }
        if(card_extra_data.address_city === 0 || !card_extra_data.address_city.trim()) {
          errors.push('Your billing city is missing');
        }
        if(card_extra_data.address_zip === 0 || !card_extra_data.address_zip.trim()) {
          var postal_zip = country == 2 ? 'zip code' : 'postal code';
          errors.push('Your billing '+postal_zip+' is missing');
        }
        if(data.price_acknowledgement !== 'yes') {
          errors.push('Please check the checkbox acknowledging the pricing');
        }
        if(!card_extra_data.address_state.trim()) {
            var state_territory = country == 2 ? 'state' : 'territory';
            errors.push('Your billing '+state_territory+' is missing');
        }

        if(errors.length) {
            display_signup_error(errors.join('<br />'));
            return false;
        }

        stripe.createToken(card[0], card_extra_data).then(function(result) {
            if (result.error) {
                display_signup_error(result.error.message);
            } else {

                // fetch token (token contains id, last4, and card type)
                var token = result.token.id;

                // get form data
                var data = get_signup_data();

                // attempt sign up
                $.ajax({
                    url: '../../../../../../../ajax/signup',
                    type: 'POST',
                    dataType: 'json',
                    data: {
                        token: token,
                        data: data
                    },
                    success: function(response){

                        if(response.error) {

                            display_signup_error(response.error.message);

                        } else if(response.subdomain) {

                            $('a#go_to_login_page').attr('href', 'https://' + data.subdomain + '.breezechms.com');
                            $('.signup_email_placeholder').text(data.email);
                            go_to_signup_page(3);

                        } else {

                            display_signup_error('Yikes!  Something went wrong.  Please contact Breeze Support via email (support@breezechms.com) or phone (888.320.6030).');

                        }

                    }
                });

                }

            });

        // Prevent the form from submitting with the default action
        return false;

    }

    $('a#signup_page_1_submit').on('click', function(e) {
        e.preventDefault();

        $('a#signup_page_1_submit').html('<i class="fas fa-spinner fa-spin"></i>');

        $('#signup_errors').addClass('hidden');

        var errors = [];

        var data = get_signup_data();

        if(!data.organization_name) { errors.push('Please enter your church or organization\'s name.'); }
        if(!data.subdomain) { errors.push('Please enter an address for your account.'); }
        if(!data.first_name) { errors.push('Please enter your first name.'); }
        if(!data.last_name) { errors.push('Please enter your last name.'); }
        if(!data.email) { errors.push('Please enter your email address.'); }
        if(!data.username) { errors.push('Please enter the username you would like to use.'); }
        if(!data.password) { errors.push('Please enter the password you would like to use.'); }
        if(data.password != data.confirm_password) { errors.push('Password and confirm password fields do not match.'); }

        // ensure valid email (this mostly ensures this - see https://stackoverflow.com/questions/46155/how-to-validate-an-email-address-in-javascript for limitations). Comfortable w/ limitations.
        if(data.email) {

            var re = /^(([^<>()\[\]\\.,;:\s@"]+(\.[^<>()\[\]\\.,;:\s@"]+)*)|(".+"))@((\[[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\])|(([a-zA-Z\-0-9]+\.)+[a-zA-Z]{2,}))$/;

            var is_valid_email = re.test(String(data.email).toLowerCase());

            if(is_valid_email == false) {
                errors.push('Sorry, the email address you entered is not a valid address.');
            }

        }

        // require stronger password
        if(data.password && data.password == data.confirm_password && data.password_strength < 1) {
            errors.push('Sorry, the password you selected is too easy for a hacker to guess.  Please choose a different password. (Typically longer passwords are more secure than shorter passwords.)');
        }

        // if subdomain is invalid, don't allow call
        if (!$('div#subdomain_check_invalid_format').is( ":hidden" )) {
            errors.push('Sorry, this web address is invalid (can only contain numbers, letters, hyphen or underscore).');
        }

        // if JS-detected errors, output errors and stop
        if(errors.length) {
            display_signup_error(errors.join('<br />'));
            return false;
        }

        $.ajax({
            url: '../../../../../../../ajax/is_valid_subdomain',
            type: 'POST',
            dataType: 'text',
            data: {
                subdomain: data.subdomain
            },
            success: function(is_valid){

                if(is_valid == 'no') {
                    errors.push('Sorry, the address you have selected is already taken.');
                }

                // if server-detected errors, output errors and stop
                if(errors.length) {
                    display_signup_error(errors.join('<br />'));
                    return false;
                }

                // proceed to next page
                go_to_signup_page(2);

            }

        });

    });

    var is_valid_subdomain_ajax = '';

    function check_subdomain() {

        var subdomain = $('input#subdomain').val();

        if(is_valid_subdomain_ajax) { is_valid_subdomain_ajax.abort(); } // stop any previous checks

        // check format before ajax call
        if(subdomain.match(/^[a-z0-9-_]+$/i) === null && subdomain.length > 0) {
            $('div#subdomain_check_invalid_format').removeClass('hidden');
            return;
        }
        else {
            $('div#subdomain_check_invalid_format').addClass('hidden');
        }

        if(!subdomain) {
            $('div#subdomain_check').addClass('hidden');
        } else {

            is_valid_subdomain_ajax = $.ajax({
                url: '../../../../../../../ajax/is_valid_subdomain',
                type: 'POST',
                dataType: 'text',
                data: {
                    subdomain: subdomain
                },
                success: function(is_valid){

                    if(is_valid == 'no') {
                        $('div#subdomain_check').removeClass('hidden');
                    } else {
                        $('div#subdomain_check').addClass('hidden');
                    }

                }

            });

        }

    }

        data.password_strength = $('div#sign_up').find('input#password').data('password-score');

        data.card_name = $('div#sign_up').find('input#card_name').val();
        data.address_line1 = $('div#sign_up').find('input#address_line1').val();
        data.address_city = $('div#sign_up').find('input#address_city').val();
        data.address_state = $('div#sign_up').find('select#address_state').val();
        data.address_zip = $('div#sign_up').find('input#address_zip').val();
        data.address_country = $('div#sign_up').find('input#address_country').val();

        var price_acknowledgement = $('div#sign_up').find('input#price_acknowledgement').is(':checked');
        if(price_acknowledgement === false) { data.price_acknowledgement = "no"; } else { data.price_acknowledgement = "yes"; } // passed to a function that is expecting string that is "no" or "yes"

        if(tax_exempt === false) { data.tax_exempt = "no"; } else { data.tax_exempt = "yes"; } // passed to a function that is expecting string that is "no" or "yes"

        return data;

    }

<script>

    $('input.force_sign_in').on('click', function(e) {
        e.preventDefault();

        $('div#subdomain_not_found').addClass('hide');
        var subdomain = $('input#force_subdomain').val();

        $.ajax({
            url: '../../../../ajax/check_subdomain_exists',
            type: 'POST',
            data: {
                subdomain: subdomain
            },
            success: function(result){

                if(result == 'true') {
                    var url = window.location.href;
                    url = url.replace("www.", subdomain+".");
                    window.location.href = url;
                } else {
                    $('div#subdomain_not_found').removeClass('hide');
                }

            }
        });


    });

    // focus on input
    $('div#force_sign_in').on('shown.bs.modal', function () {
        $('input#force_subdomain').focus();
    });

</script>
