  function setSlide($slider, index) {
    $slider.removeClass('slide1 slide2 slide3 slide4 slide5')

  var isVisible = function($e) {
    var l_iScrollX = 0;

    if (document.documentElement && document.documentElement.scrollLeft ) {
      l_iScrollX = document.documentElement.scrollLeft;
    } else if( document.body && document.body.scrollLeft ) {
      l_iScrollX = document.body.scrollLeft;
    } else if( window.pageXOffset ) {
      l_iScrollX = window.pageXOffset;
    } else if( window.scrollX ) {
      l_iScrollX = window.scrollX;
    }


    var l_iScrollY = 0;

    if (document.documentElement && document.documentElement.scrollTop ) {
      l_iScrollY = document.documentElement.scrollTop;
    } else if( document.body && document.body.scrollTop ) {
      l_iScrollY = document.body.scrollTop;
    } else if( window.pageYOffset ) {
      l_iScrollY = window.pageYOffset;
    } else if( window.scrollY ) {
      l_iScrollY = window.scrollY;
    }

    return (
      $e.offset().top + $e.outerHeight() <= l_iScrollY + $(window).height()
      && $e.offset().top >= l_iScrollY
      && $e.offset().left + $e.outerWidth() <= l_iScrollX + $(window).innerWidth()
      && $e.offset().left >= l_iScrollX
    );
  };

  $('.lnk-locale').on('click', function(e){

    var $slider = $(this).closest('.slider');
    setSlide($slider, $(this).index() + 1);
    if ($slider.data('slider-sliding') && undefined != $slider.data('slider-interval')) {
      clearInterval($slider.data('slider-interval'));
      $slider
