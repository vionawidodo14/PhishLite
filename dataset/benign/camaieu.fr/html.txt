	function getSizeResolution() {

			message = "desktop";

			message = "tablet responsive";

			message = "mobile responsive";
		}
		return message;

		var d = new Date();

	function toObject(arr) {
		var arrR = [];
		var rv = {};

	}

			obj3[attrname] = obj2[attrname];
