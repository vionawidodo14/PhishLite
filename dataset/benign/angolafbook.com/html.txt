		toggle: function (value, width, display) {
			this.storage.isResponsive = value;
			this.viewport.attr ('content', 'width=' + width);
			this.handle.html (display);
		},
		showFull:		function ()	{ this.toggle ("false", this.widthFull,	'Versão Otimizada');	},
		showOptimized:	function ()	{ this.toggle ("true",  this.widthDevice, 'Versão de Computador');	},
		run: function ()
		{
			if (this.isResponsive ())	{ this.showFull (); }
			else						{ this.showOptimized (); }
		},
		init: function ()
		{
			var self = this;
			self.setResponsive ();
			self.handle.on ('click', function () {
				self.run ();
				return false;
			})
		}
	};
