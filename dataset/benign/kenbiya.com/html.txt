var needRefresh=false;
function centeringModalSyncer(id){
	var w = $(window).width();
	var h = $(window).height();
	var cw = $("#" + id).outerWidth(true);
	var ch = $("#" + id).outerHeight(true);
	var pxleft = ((w - cw)/2);
	var pxtop = ((h - ch)/2);
	$("#" + id).css({"left": pxleft + "px"});
	$("#" + id).css({"top": pxtop + "px"});
}

function openPopupWindow(url, id, key){
  if (url === "/app/exe/mySchMod1" && $("#overlay-" + id).size() > 0) {
		needRefresh=false;
    closePopupWindow(id, function(){
      $("#" + id).hide();
			needRefresh=true;
      openPopupWindow(url, id, key);
    }, "fast");

    return;
  }
  needRefresh = false;
	//$(this).blur();
	var overlayId = "overlay-" + id;
  if (url === "/app/exe/mySchMod1"
   || url === "/app/exe/mySchDel") {
		var value=$("[name=search_name" + key +"]").val();
    $("#" + id + " [name=sn]").val(value);
    needRefresh=true;
  }

  if (url === "/app/exe/mySchMod1") {
    $("#" + id).addClass("edit");
  }

  if (url === "/app/exe/mySchOrder") {
    needRefresh=true;
  }

  registorMySearch(url, false, id);

	$("body").append('<div id="' + overlayId + '"></div>');
	$("#" + overlayId).fadeIn("fast");

	$("#" + id).fadeIn("fast",function () {
		centeringModalSyncer(id)
	});
}

function closePopupWindow(id, callBack, speed){
  var fadeSpeed = speed?speed:"slow";
	$("#" + id).fadeOut(fadeSpeed,function(){
		$("#overlay-" + id).remove();
    if (callBack) {
      callBack();
    }
	});
  if (needRefresh) {
    //location.href="";
		location.reload();
  }
}

		closePopupWindow("box_edit");
	});

function registorMySearch(url, refresh, formId) {
  var postData = $("#" + formId + " form").serialize();
	if (postData) {
		postData = postData.replace("%C2%A5", "%5C");
	}
	showLoader();
  $.ajax({
    type : "post",
    url : url,
    dataType : "html",
    cache : false,
    data : postData,
    success : function(html) {
      $("#" + formId + " form").replaceWith(html);
      if ($("#" + formId + " form").size() == 0) {
        $("#" + formId).addClass("reg_end");
        needRefresh=refresh;
      }
      Cookies.set("pp_scrolltop", $(window).scrollTop(), { expires: 1 });
			hideLoader();
    },
    error :function(XMLHttpRequest, textStatus, errorThrown){
      $("#" + formId + " p.typo_error").text("処理失敗しました");
      Cookies.set("pp_scrolltop", $(window).scrollTop(), { expires: 1 });
			hideLoader();
    }
  });
}

function showLoader() {
	var loadingSrc = "/images/common/loading.gif";
	var div = $("<div>");

	div.attr("id", "show_loading");
	div.css("position", "fixed");
	div.css("text-align", "center");
	div.width($(window).width());
	div.height($(window).height());
	div.css("line-height", $(window).height() + "px");
	div.css("background-color", "rgba(0,0,0,0)");
	div.css("z-index", 999999);
	div.css("top", 0);
	div.css("left", 0);
	var img = $("<img>");
	img.attr("src", loadingSrc);
	img.css("margin-top", ($(window).height() - 33)/2 + "px");
	div.append(img);
	$("body").append(div);
}

function hideLoader() {
	if ($("#show_loading").size() > 0) {
			$("#show_loading").remove();
	}
}
