function fillRangeAgg(data, aggId, maxPostfix, lessPrefix, morePrefix){
    if (data.from == data.to){
        if (data.to == data.max && morePrefix) {
            upd_agg = morePrefix + ' ' + data.max; }
        else if (data.to == data.max && maxPostfix) {
            upd_agg = data.max + maxPostfix; }
        else if (data.to == data.min && lessPrefix) {
            upd_agg = lessPrefix + ' ' + data.min; }
        else {
            upd_agg = data.from; }
//    } else if (data.from == data.min && data.to == data.max) {
//        upd_agg = defaultValue;
    } else {
        var postfix = "";
        if (data.to == data.max && maxPostfix) {postfix += maxPostfix}
        upd_agg = data.from + ' - ' + data.to + postfix;
    }
    $(aggId).html(upd_agg);
}

function fillRangeInput(data, inputId) {
    var upd_input = data.from + ',' + data.to;
    $(inputId).prop("value", upd_input);
}

function fillDropAgg(visDropId, aggId, defaultValue){
    var result_arr = [];
    var items = $(visDropId + " .catalog-filter__checks .check-item");
    $.each(items, function (key, item) {
        if ($(item).hasClass("active")) {
            result_arr.push($(item).html());
        }
    })
    if (result_arr.length == 0) {
        result_arr.push(defaultValue);
    }
    result_str = result_arr.join(", ")
    $(aggId).html(result_str);
}

function activateDropItems(selectId, visDropId) {
    var items = $(visDropId + " .catalog-filter__checks .check-item");
    $.each(items, function(key, item) {
        $(item).removeClass("active");
    });
    $.each($(selectId + " option"),function (key, opt) {
        if ($(opt).attr("selected") !== undefined) {
            $.each(items, function(key, item) {
                if ($(item).attr("data-value") == $(opt).attr("value")) {
                    $(item).toggleClass("active");
                }
            });
        }
    });
}

function addMultiDropClickHandle(inputId, visDropId, aggId, defaultValue) {
    $(visDropId + " .catalog-filter__checks .check-item").click(function(e) {
        e.preventDefault();
        var x = $(this).attr("data-value");
        var options = $(inputId + " option");
        $.each(options,function (key, opt) {
            if ($(opt).attr("value") == x) {
                if ($(opt).attr("selected")) {
                    $(opt).attr("selected", false);
                } else {
                    $(opt).attr("selected", 1)
                }
            }
        });
        fillDropAgg(visDropId, aggId, defaultValue);
    });
}
