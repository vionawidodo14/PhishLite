<script type="text/javascript">
        $(document).ready(function() {

            $('#ctl00_DefaultContentHolder_btnLogin').click(function() {
                $('#ctl00_DefaultContentHolder_lblMessage').text('');
                $('div.errmsg').remove();
                var isValidtxt = ValidateTextControls("#tabLogin input[type=text]");
                var isValidpswd = ValidateTextControls("#tabLogin input[type=password]");
                var isValidchk = ValidateChkboxControls("#tabLogin input[type=checkbox]");
                if (isValidtxt && isValidpswd && isValidchk) {
                    return true;
                } else { return false; }
            });

            function ValidateTextControls(ctrl) {
                var isValid = true;
                $(ctrl).each(function() {
                    if ($.trim($(this).val()) == '') {
                        $(this).parent().append("<div class='errmsg' style='color:red;'>" + $(this).attr("data-ReqMsg") + "</div>");
                        isValid = false;
                    }
                });
                return isValid;
            }

            function ValidateChkboxControls(ctrl) {
                var isValid = true;
                $(ctrl).each(function() {
                    if (!$(this).prop('checked')) {
                        $(this).parent().append("<div class='errmsg' style='color:red;'>" + $(this).attr("data-ReqMsg") + "</div>");
                        isValid = false;
                    }
                });
                return isValid;
            }
            function ResetControls(ctrl) {
                $(ctrl).each(function() {
                    $(this).val('');
                });
            }
            function ResetControlsChk(ctrl) {
                $(ctrl).each(function() {
                    if ($(this).prop('checked')) {
                        $(this).prop('checked', false);
                    }
                });
            }

            //script to check browser compatibility
            var isOldBrowser = false, isCookiesEnabled = false, isPopupBlockerEnabled = detectPopupBlocker();
            var browser = get_browser();

            if ((browser.name == 'IE' || browser.name == 'MSIE') && browser.version < 11) {
                isOldBrowser = true;
            }

            showElement('js1');
            hideElement('js0');

            isCookiesEnabled = areCookiesEnabled();

            if (isCookiesEnabled)
                showElement('cookies1');
            else
                showElement('cookies0');

            if (isOldBrowser)
                showElement('browser0');
            else
                showElement('browser1');

            if (isPopupBlockerEnabled)
                showElement('popup0');
            else
                showElement('popup1');

            function showElement(e) {
                var x = document.getElementById(e);
                x.style.display = "block";
            }
            function hideElement(e) {
                var x = document.getElementById(e);
                x.style.display = "none";
            }

            function detectPopupBlocker() {
                var myTest = window.open("about:blank", "", "directories=no,height=100,width=100,menubar=no,resizable=no,scrollbars=no,status=no,titlebar=no,top=0,location=no");
                if (!myTest) {
                    return true;
                } else {
                    myTest.close();
                    return false;
                }
            }

            function get_browser() {
                var ua = navigator.userAgent, tem, M = ua.match(/(opera|chrome|safari|firefox|msie|trident(?=\/))\/?\s*(\d+)/i) || [];
                if (/trident/i.test(M[1])) {
                    tem = /\brv[ :]+(\d+)/g.exec(ua) || [];
                    return { name: 'IE', version: (tem[1] || '') };
                }
                if (M[1] === 'Chrome') {
                    tem = ua.match(/\bOPR|Edge\/(\d+)/)
                    if (tem != null) { return { name: 'Opera', version: tem[1] }; }
                }
                M = M[2] ? [M[1], M[2]] : [navigator.appName, navigator.appVersion, '-?'];
                if ((tem = ua.match(/version\/(\d+)/i)) != null) { M.splice(1, 1, tem[1]); }
                return {
                    name: M[0],
                    version: M[1]
                };
            }

            function areCookiesEnabled() {
                try {
                    document.cookie = 'cookietest=1';
                    var cookiesEnabled = document.cookie.indexOf('cookietest=') !== -1;
                    document.cookie = 'cookietest=1; expires=Thu, 01-Jan-1970 00:00:01 GMT';
                    return cookiesEnabled;
                } catch (e) {
                    return false;
                }
            }
        });
    </script>
