    function tsSearch()
    {
      var tyreSize = $('select#tsTyreSizeMenu').val();
      var cPath = "125"; // default depends on current default season
      var winterTyre = "";
      
      if($("select#tsSeasonMenu").length > 0)
      {
          // if($("select#tsSeasonMenu").val().indexOf("33") >= 0 || $("select#tsSeasonMenu").val().indexOf("125") >= 0)
          {
            
            cPath = $("select#tsSeasonMenu").val();
            if(cPath == "125_kitka")
            {
              cPath = "125";
              winterTyre = "kitka";
            }   
            if(cPath == "125_nasta")
            {
              cPath = "125";
              winterTyre = "nasta";
            }              
          }
      }
      var LI = "";
      if(tyreSize.indexOf("_LI") > 0)
      {
        LI = tyreSize.substr((tyreSize.indexOf("_LI") + 4));
        if(parseInt(LI) == 0)
        {
          LI = "";
        }
      }  
      var width = "--";
      var profile = "--";
      var diam = "--";
        
      if(tyreSize.indexOf("/") > 0) // check if tyre is euro format
      {
        var url = "";
        
        
        
        if(tyreSize.indexOf("/") > 0)
        {
          width   = tyreSize.substr(0, tyreSize.indexOf("/"));
          profile = tyreSize.substr( (tyreSize.indexOf("/") + 1), ((tyreSize.indexOf("-")) - (tyreSize.indexOf("/") + 1)) );
          diam    = tyreSize.substr( (tyreSize.indexOf("-") + 1), 2);
          
        }
        
     
        if(width.length > 0 && profile.length > 0 && diam.length > 0)
        {
          if(cPath == "125")
          {
            url += "talvirenkaat-c-125.html";
          }
          else
          {
            url += "kesarenkaat-c-33.html";
          }
          url += "?leveys=" + width + "&korkeus=" + profile + "&tuumaa=" + diam + "";
        }
        
        
        
      }
      else
      {
        // alert("missing data");
        var url = "";
        if(tyreSize.indexOf("WS") >= 0)
        {
          if(cPath == "125")
          {
            url += "talvirenkaat-c-125.html";
          }
          else
          {
            url += "kesarenkaat-c-33.html";
          }
        
        
          diam = tyreSize.substr(3, 2);
          if(parseInt(diam) == 0)
          {
            diam = "--";
          }
          url += "?leveys=" + width + "&korkeus=" + profile + "&tuumaa=" + diam + "";
         
          
          
         
        }
       // 
      }
       
      if(url.length > 0)
      {
        if(winterTyre.length > 0)
        {
            url += "&talvirengas=" + winterTyre;
        }
        if(LI.length > 0)
        {

        }
        url += "&category_depth=search";
        
       // alert(url);
        window.location=url;
      }
    }

function LoadManufacturers(SelectedManufacturer)
{
  $.ajax({
    url:    'getWSData.php',
    data:     {action: 'manufacturers'},
    type:     'GET',
    cache:    false,
    dataType:   'JSON',
    async:    false
  }).done(function(data){
    if(data.length > 0)
    {
      $('#wsManufacturerList').append('<option value="">Valitse automerkki</option>');
      for(var i = 0; i < data.length; i++)
      {
        var optionText = '<option value="' + data[i].ManufacturerID + '"';
        if(SelectedManufacturer != undefined && SelectedManufacturer === parseInt(data[i].ManufacturerID))
        {
          optionText += ' selected ';
        }
        optionText += '>';
        optionText += data[i].ManufacturerName;
        optionText += '</option>';

        $('#wsManufacturerList').append(optionText);
      }
    }
  }).fail(function(jqXHR, textStatus, errorThrown){
    alert(textStatus);
  }); 
}

function CheckFitting(KType, ProductsId)
{
  if(typeof KType != "undefined" && typeof ProductsId != "undefined")
  {
    $.ajax({
      url: 'getWSData.php',
      data:     {action: 'checkfitting', KType: KType, products_id: ProductsId},
      type:     'GET',
      cache:    false,
      dataType:   'JSON',
      timeout:    7000
    }).done(function(data){
      var resulthtml = '';
      

      resulthtml += '<br />';
      // Get car name and model from select list
      resulthtml += '<b>Valittu automerkki ja malli:</b><br />';

      resulthtml += $('[name=wsCarManu] option:selected').text() + ' ' + $('[name=wsCarModel] option:selected').text() + ' ' + $('[name=wsCarType] option:selected').text();
      resulthtml += '';
      resulthtml += '<br /><br />';

      if(data.result === true)
      {
        resulthtml += '<p class="text-success">';
      }
      else
      {
        resulthtml += '<p class="text-danger">';
      }
      
      if(data.text)
      {
        resulthtml += data.text;
      }
      // If pcd and cb is ok
      if(data.pcdAndCbOk === true)
      { 
        resulthtml += '<br />';
        resulthtml += '<div class="text-right">';
        resulthtml += '<a href="#" class="btnSelectCarToCart btn btn-success">Choose</a>';
        resulthtml += '</div>';
      }
      resulthtml += '</p>';
      $('#wsCheckFittingResult').html(resulthtml);
    }).fail(function(jqXHR, textStatus, errorThrown){
      $('#wsCheckFittingResult').html('Error: ' + textStatus);
    });
  }
  
}
