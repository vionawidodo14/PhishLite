			function animateSlide(slider23) {
				slider23.find('.active').find('[data-animate-in]').each(function () {
					var $this = jQuery(this);
					animateClass = $this.data('animate-in');
					$this.addClass(animateClass)
				});

				slider23.find('.active').find('[data-animate-out]').each(function () {
					var $this = jQuery(this);
					animateClass = $this.data('animate-out');
					$this.removeClass(animateClass)
				});
			}

			function animateSlideEnd(slider23) {
				slider23.find('.active').find('[data-animate-in]').each(function () {
					var $this = jQuery(this);
					animateClass = $this.data('animate-in');
					$this.removeClass(animateClass)
				});
				slider23.find('.active').find('[data-animate-out]').each(function () {
					var $this = jQuery(this);
					animateClass = $this.data('animate-out');
					$this.addClass(animateClass)
				});
			}

			animateSlide(slider23);

			slider23.on('slid.bs.carousel', function () {
				animateSlide(slider23);
			});
			slider23.on('slide.bs.carousel', function () {
				animateSlideEnd(slider23);
			});
