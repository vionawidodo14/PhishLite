			validateSearchForm = function(form){
				if(!form) return true;
				if(!form.sp) return true;
				
				if(form.sp.value == '' || form.sp.value.toLowerCase() == DEFAULT_SEARCH_TEXT.toLowerCase()){
					alert('Please enter species name');
					return false;
				}
				refineSearchTerm();
				return true;
				
				var name = form.sp.value.match(/(.*)\((.*)\)/);
				if(!name || name.length<2) return true;
				form.sp.value = trim(name[1]);

				return true;
			}

			submitSearch = function(evt, formId){
				var e = evt || window.event;
				var proceed = true;
				if(e && e.key){	
					proceed = (e.keyCode==13);
				}
				if(proceed){
					if(!formId){
						formId = (!jQuery('form#form_search').is(':visible') ? 
										(jQuery('#content_form_search').is(':visible') ? 'content_form_search' : 'content_form_search2') : '');
					}
					
					var form = document.forms[(formId ? formId : 'form_search')];
					if(form){			
						if(validateSearchForm(form)){
							var url = form.getAttribute('action') + '/'+  ''+ form.sp.value;  
							//form.setAttribute('action', url.replace('//', '/'));	
							if(url.indexOf('(')<0)				
								url += '( )';		

							var t = url.match(/\/loc-([^\/]*)/i);
							if(t && t.length>1){
								if(t[1].toLowerCase() != 'california'){
									url = url.replace(/\/loc-([^\/]*)/i, '/loc-California/');
								}
							}
							url = url.replace('//', '/');
							if(url.indexOf('/editors')>=0){
								window.location = url.replace('//', '/');
								return false;
							}
							
							var m = form.sp.value.match(/([^\(]*)\(([^\)]*)/i);
							if(!m){
								m = [form.sp.value];	
							}
							
							if(m.length<2 || jQuery.trim(m[2]).toLowerCase() == 'all'){
								showPageLoading({page:'home-category', category: m[1], url:url, redirect: true});	
							}
							else{
								showPageLoading({page:'list-species', species: m[1], common_name: m[2], url:url, redirect: true});	
							}
							return false;
							//form.submit();
						}
					}
					return false;
				}
			}
			validateInput = function(arrMessage, el, field, regex){
				if(!el)
					return false;
				if(trim(el.value) == ''){
					var msg = field+' is Required';
					if(arrMessage)
						arrMessage[arrMessage.length] =  '  > '+msg;
					else
						alert(msg);
					return false;
				}
				if(regex){
					if(!el.value.match(regex)){
						var msg = field+'\'s value is Invalid';
						if(arrMessage)
							arrMessage[arrMessage.length] =  '  > '+msg;
						else
							alert(msg);
						return false;
					}
				}
				return trim(el.value);
			}
			
			formatFileSize = function(bytes) {
		        if (typeof bytes !== 'number') {
		            return '';
		        }
		        if (bytes >= 1000000000) {
		            return (bytes / 1000000000).toFixed(2) + ' GB';
		        }
		        if (bytes >= 1000000) {
		            return (bytes / 1000000).toFixed(2) + ' MB';
		        }
		        return (bytes / 1000).toFixed(2) + ' KB';
		    }
