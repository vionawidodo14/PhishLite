<script type="text/javascript">
function login(event) {
	var startsWith = function(value, pattern) {
		return (value.substring(0, pattern.length) === pattern);
	};
	var mapper = {
		"default": {
			"code": [],
			"url": "https://lms.wiseman.com.hk/lms/user/login.do"
		},
		"hk-ebbc": {
			"code": ["lions", "sdbnsm", "rac", "nsm", "dse2"],
			"url": "https://hk-ebbc.wiseman.com.hk/lms/user/login.do"
		}
	};
	var username =  document.getElementById("username").value;
	var loginForm = document.getElementById("login-form");
	for (var key in mapper) {
		var domain = mapper[key];
		if (domain.hasOwnProperty("code")) {
			for (var i = 0; i < domain.code.length; i++) {
				var pattern = domain.code[i];
				if (startsWith(username, pattern)) {
					loginForm.action = domain.url;
					return true;
				}
			}
		}
	}

	loginForm.action = mapper.default.url;
	return true;
}
