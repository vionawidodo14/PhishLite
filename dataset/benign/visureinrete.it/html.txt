function handleData(data) {

	if (data.total == 0)
	{
		location.href = '/01_informazioni_sui_servizi/esito-ricerca-motore.asp?q=' + txtSearch;
		
		//$('#res').hide();
		//$('#tot').html('risultati: 0').show();
		//$('nav#menu').data('mmenu').close();
		//$("#trigger").click();
		return;
	}
	
	$.each(data.results, function(k, v) {

		imageLink = ((typeof(v.image_link)  === "undefined" || v.image_link == null) ? '' : '<img src=' + v.image_link + ' />' );
		prais = (typeof(v.price)  === "undefined" ? '' : '€ ' + parseFloat(v.price).toFixed(2));
		prais = (prais == '' ? '' : prais.replace('.', ','));
		
		$('#res').append('<div id=\"box_ricerca\"><a href=' + v.link + '><div class=\"container_image_ricerca\">' + imageLink + '</div><div class=\"container_ricerca_singola\"><h4>' + v.title + '</h4><div class=\"container_txt_ricerca\"><p>' + v.description + '</p></div><div class=\"prezzo_ricerca\">' + prais + '</div></div></a></div>');
	});
	
	$('#res').show();
	$('#tot').html('Risultati trovati: ' + data.total).show();
	
	$('nav#menu').data('mmenu').close();

	$("#trigger").click();
}
