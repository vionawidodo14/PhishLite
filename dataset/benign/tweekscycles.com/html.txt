    initialize: function() {

        if (this.redirectionConfig.redirectionPreferenceEnabled && this.cookieExists(this.redirectionConfig.cookieName)) {
            /**
             * @type {RegExp}
             *
             * Remove trailing slash from Base and Redirection URLs. Without this, the comparison might fail if only

            var trailingSlashTrimRegex = /\/+$/;

            var redirectedUrl = decodeURIComponent(this.cookieExists(this.redirectionConfig.cookieName));

                }

     * Use native javascript so that we don't need to wait for the unnecessary library to load

    detectRedirection: function(ip, continent, country) {
        // Example: ("5.148.137.82", "EU", "GB")

        // Fallback
        var candidates = [
            continent + '_' + country,
            continent
        ];

        var redirectTo = false;

        // If no valid country, then save where they currently are but do not redirect.
        if (!country) {
            this.setLocationCookie(this.redirectionConfig.baseUrl);
            return;
        }

        for (var i in candidates) {
            // Make sure i is either property of object or index of array. This rules out func like each()
            if (!candidates.hasOwnProperty(i)) {
                continue;
            }
            
            var url = this.redirectionConfig.urlByCode[candidates[i]];

            if (url !== undefined) {
                redirectTo = url;
                break;
            }
        }

        // Most of the cases we don't need to redirect
        if (!redirectTo) {
            return;
        }

        // Agent exceptions
        for (var i in this.redirectionConfig.exception.agents) {
            // Make sure i is either property of object or index of array. This rules out func like each()
            if (!this.redirectionConfig.exception.agents.hasOwnProperty(i)) {
                continue;
            }

            var exceptionAgent = new RegExp(this.redirectionConfig.exception.agents[i], 'gi');

            if (navigator.userAgent.match(exceptionAgent)) {
                // Agent matched
                return;
            }
        }

        // Ip exceptions
        for (var i in this.redirectionConfig.exception.ipAddresses) {
            // Make sure i is either property of object or index of array. This rules out func like each()
            if (!this.redirectionConfig.exception.ipAddresses.hasOwnProperty(i)) {
                continue;
            }

            var exceptionIp = this.redirectionConfig.exception.ipAddresses[i];

            if (exceptionIp == ip) {
                // Single ip matched
                return;
            }

            var exceptionParts = exceptionIp.match(/(\d+).(\d+).(\d+).(\d+)\s*-\s*(\d+).(\d+).(\d+).(\d+)/);

            if (exceptionParts == null) {
                // Invalid expression, continue.
                continue;
            }

            var ipParts = ip.split('.');

            // Matching ip with lower parts and higher parts
            if (parseInt(ipParts[0]) >= parseInt(exceptionParts[1]) && parseInt(ipParts[0]) <= parseInt(exceptionParts[1+4])
                && parseInt(ipParts[1]) >= parseInt(exceptionParts[2]) && parseInt(ipParts[1]) <= parseInt(exceptionParts[2+4])
                && parseInt(ipParts[2]) >= parseInt(exceptionParts[3]) && parseInt(ipParts[2]) <= parseInt(exceptionParts[3+4])
                && parseInt(ipParts[3]) >= parseInt(exceptionParts[4]) && parseInt(ipParts[3]) <= parseInt(exceptionParts[4+4])
            ) {
                // IP range matched
                return;
            }
        }
        this.setLocationCookie(url);

        // If you're from UK and are on /uk then you don't need to be redirected
        if (url !== this.redirectionConfig.baseUrl) {
            this.setLocation(this.addCurrentUrlPathToUrl(redirectTo));
        }
    },

    locateUser: function() {
        var geoipScript = document.createElement('script');
        var self = this;
        geoipScript.src = this.redirectionConfig.jsUri;
        geoipScript.onload = function () {
            geoip2.city(function (location) {
                // Example location
                //
                // {
                //   "traits": {
                //     ...
                //     "ip_address": "5.148.137.82"
                //   },
                //   "continent": {
                //     "code": "EU",
                //     ...
                //   },
                //   "country": {
                //     "iso_code": "GB",
                //     ...
                //   },
                //   ...
                // }
                self.detectRedirection(location.traits.ip_address, location.continent.code, location.country.iso_code);
            }, function (errorObject) {
                console.log(errorObject.error);
                self.setLocationCookie(self.redirectionConfig.baseUrl);
            });
        };
        document.head.appendChild(geoipScript);
    },

    /**
     * Adds the current url path to the specified url.
     *
     * Example
     * Current URL: 'https://demon-tweeks.com/uk/motorsport/'
     * Input: 'https://demon-tweeks.com/us/'
     * Output: 'https://demon-tweeks.com/us/motorsport/'
     *
     * @param url
     * @returns {string}
     */
    addCurrentUrlPathToUrl: function (url) {
        var slashTrimRegex = /\/+$/;

        var path = window.location.href.replace(slashTrimRegex, "") + '/';

        // Remove base urls including the store code to get the raw path
        for (var i in this.redirectionConfig.baseUrls) {
            // Make sure i is either property of object or index of array. This rules out func like each()
            if (!this.redirectionConfig.baseUrls.hasOwnProperty(i)) {
                continue;
            }
            
            var baseUrl = this.redirectionConfig.baseUrls[i];
            baseUrl = baseUrl.replace(slashTrimRegex, "") + '/';
            path = path.replace(baseUrl, "");
        }

        return url.replace(slashTrimRegex, "") + '/' + path;
    },

    cookieExists: function(cookieName) {
        var allCookies = document.cookie.split(';');
        var exists;
        allCookies.forEach(function(cookie) {
            var halvedCookie = cookie.split('=');
            if (cookieName === halvedCookie[0].trim()) {
                exists = halvedCookie[1].trim();
            }
        });
        return exists;
    },

    setLocationCookie: function(cookieValue) {
        var days = new Date();
        days.setTime(days.getTime() + (this.redirectionConfig.cookieExpiration * 24 * 60 * 60 * 1000));
        var expires = '; expires=' + days.toGMTString();
        document.cookie = this.redirectionConfig.cookieName + '=' + cookieValue + expires + '; path=/';
    },

    /**
     * Wrapper to set the window location with defensive rules applied.
     *
     * @param locationUrl
     */
    setLocation: function (locationUrl) {
        if (typeof locationUrl === 'string') {
            if (locationUrl.indexOf('undefined') !== -1) {
                //Do not cause any redirect to occur to any "undefined" location, it's likely an error has occurred.
                return;
            }
            window.location = locationUrl;
        }
    }
