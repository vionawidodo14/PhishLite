	function colocar_columnas() {

		var max_cols,
			ancho_col = $('#cts .col-lg-2:eq(0)').width(),
			ancho_body = $('body').width(),
			col_padding = 15,
			alturas = [],
			i = 1,
			fila;

		if (ancho_body >= 1200) {
			max_cols = 6;
		} else if (ancho_body >= 992) {
			max_cols = 3;
		} else if (ancho_body >= 480) {
			max_cols = 2;
		} else {
			max_cols = 1;
		}

		$('#cts .col-lg-2').each(function() {
			
			fila = Math.ceil(i / max_cols);
			var columna = (i % max_cols === 0 ? max_cols : i % max_cols);
			
			if (alturas[fila] === undefined) {
				alturas[fila] = [];
			}
			
			if (i <= max_cols) {
				$(this).css('left', ((ancho_col + col_padding * 2) * (i - 1)) + 'px');
				$(this).css('top', 0);

				alturas[fila][columna] = {alto: $(this).height(), debajo: false, pos: columna};

			} else {
				
				var menor = Infinity;
				var i_menor = 0;
				
				for (var c = 1; c <= max_cols; c++) {
					if (!alturas[fila - 1][c].debajo && alturas[fila - 1][c].alto <= menor) {
						menor = alturas[fila - 1][c].alto;
						i_menor = c;
					}
				}

				alturas[fila - 1][i_menor].debajo = true;

				$(this).css('left', ((ancho_col + col_padding * 2) * (alturas[fila - 1][i_menor].pos - 1)) + 'px');
				$(this).css('top', menor + 'px');
				

				alturas[fila][columna] = {alto: ($(this).height() + menor), debajo: false, pos: alturas[fila - 1][i_menor].pos};
			}
			i++;
		});
		
		var mayor = -Infinity;
		for (var i = 1; i <= alturas[fila].length - 1; i++) {
			if (alturas[fila][i].alto > mayor) {
				mayor = alturas[fila][i].alto;
			}
		}

		$('#cts').css('height', mayor + 'px');
	}
