        function Buscar() {
              var tamanho = $("#FRMconcursoPesquisa").val();
              $("#resultado-filtro-concurso").html("");
              if(tamanho.length >= 2) {
                var formData = new FormData($("#form-pesquisa")[0]);

                $.ajax({
                  url: $("#form-pesquisa").prop("action"),
                  type: 'POST',
                  data: formData,
                  async: true,
                  cache: false,
                  contentType: false,
                  processData: false,
                  success: function (dados) {
                     $("body").animate({scrollTop:0}, '500');
                    var sHTML = "";
                    var Count = 0;
                    var data = $.parseJSON(dados);
                    var host =  $("#FRMhost").val();
                    for(i=0; i < data['total']; i++) {
                      if(Count == 0) {
                        sHTML += "<tr>";
                      }
                      Count++;
                      sHTML += "<tr>";
                      sHTML += "<td>";
                      sHTML += "<a href='"+host+"concurso/"+ data[i]['id_concurso'] +"'><h4>"+ data[i]['descricao'] + "</h4></a>";
                      sHTML += "</td>";''
                     
                      sHTML += "</tr>";
                      if(Count == 15 || i == data['total']) {
                        Count = 0;
                        sHTML += "</tr>";
                      }
                    }
                    var sCampo = $("#FRMconcursoPesquisa").val();
                    $(".pesquisa").css('opacity', '0');
                    $("#banner-topo").hide();
                    $(".cs-loader").show();
                    if(data['total'] != undefined) {
                      $("#resultado-filtro-concurso").html('<tr><td colspan="2"><p class="bg-primary"> Total de <b>'+ data['total'] +'</b> resultado(s) para " <b>'+ sCampo +'</b> " :</p></td></tr>');

                      $("#resultado-filtro-concurso").append(sHTML);
                      setTimeout(function(){
                        $(".cs-loader").hide();
                        $(".pesquisa").show();
                        $(".pesquisa").css('opacity', '1');
                      }, 1000);
                    } else {
                      setTimeout(function(){
                        $(".cs-loader").hide();
                        $(".pesquisa").show();
                        $(".pesquisa").css('opacity', '1');
                        $("#resultado-filtro-concurso").html('<tr><td colspan="2"><p class="bg-primary">Nenhum concurso foi encontrado para <b>"'+ sCampo +'"</b> :(</p></td></tr>');
                      }, 1000);
                    }
                  }
                });
              }
          }
    });

  function AplicarMasks(){
    $('.date').mask('00/00/0000');

    $('.cep').mask('00000-000');

    $('.phone_us').mask('(000) 000-0000');
    $('.mixed').mask('AAA 000-S0S');
    $('.cpf').mask('000.000.000-00', {reverse: true});
    $('.cnpj').mask('00.000.000/0000-00', {reverse: true});
    $('.money').mask('000.000.000.000.000,00', {reverse: true});
    $('.money2').mask("##.##0,00", {reverse: true});
    $('.ip_address').mask('0ZZ.0ZZ.0ZZ.0ZZ', {
      translation: {
        'Z': {
          pattern: /[0-9]/, optional: true
        }
      }
    });
    $('.ip_address').mask('099.099.099.099');
    $('.percent').mask('##0,00%', {reverse: true});
    $('.clear-if-not-match').mask("00/00/0000", {clearIfNotMatch: true});
    $('.placeholder').mask("00/00/0000", {placeholder: "__/__/____"});
    $('.fallback').mask("00r00r0000", {
        translation: {
          'r': {
            pattern: /[\/]/,
            fallback: '/'
          },
          placeholder: "__/__/____"
        }
      });
    $('.selectonfocus').mask("00/00/0000", {selectOnFocus: true});
  }
