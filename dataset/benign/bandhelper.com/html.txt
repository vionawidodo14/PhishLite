<script type="text/javascript">

	banner_showing = 1;
	quote_showing = 1;

	banner_width = parseInt(document.getElementById("banner_1").clientWidth);
	//banner_count = parseInt(document.getElementById("home_banners_inner").clientWidth) / banner_width;
	max_quotes = 10;
	
	function show_first_banner() {
		currentPosition = document.getElementById("home_banners_inner").offsetLeft;
		more = scroll_inner_content("home_banners", "home_banners_inner", "right", currentPosition, true);
		banner_showing = 1;
	}

	function show_next_banner() {
		next_banner = banner_showing + 1;
		if (!document.getElementById("banner_" + next_banner)) {
			show_first_banner();
			return;
		}
		more = scroll_inner_content("home_banners", "home_banners_inner", "left", 0 - banner_width, true);
		banner_showing = next_banner;
	}
	
	function show_previous_banner() {
		previous_banner = banner_showing - 1;
		if (!document.getElementById("banner_" + previous_banner)) {
			// we don't know how many banners there are, so we'll just continue showing the first banner rather than looping to the last
			return;
		}
		more = scroll_inner_content("home_banners", "home_banners_inner", "right", 0 - banner_width, true);
		banner_showing = previous_banner;
	}
	
	function show_next_banner_and_stop() {
		// before calling show_next_banner, find out if that's going to loop back to the start, and don't, to be consistent with show_previous_banner_and_stop
		next_banner = banner_showing + 1;
		if (document.getElementById("banner_" + next_banner)) {
			show_next_banner();
		}
		clearInterval(banner_timer);
	}

	function show_previous_banner_and_stop() {
		show_previous_banner();
		clearInterval(banner_timer);
	}

	function show_quote(i) {
		//document.getElementById("quote_" + quote_showing).className = "hide"; // use this if other classes are present: += " hide"
		//document.getElementById("quote_" + i).className = ""; // use this if other classes are present: = document.getElementById("quote_" + i).className.replace(/ hide/g, "")
		fade("quote_" + quote_showing, 0, .5, "show_quote_finish()");
		quote_showing = i;
	}

	function show_quote_finish() {
		fade("quote_" + quote_showing, 1, .5, "");
		
		// if we leave this running on a hidden tab in Safari, the quotes don't disappear correctly, so make sure all the other quotes are hidden now
		for (var i=1; i<=max_quotes; i++) {
			if (document.getElementById("quote_" + i)) {
				if (i != quote_showing) {
					hide("quote_" + i);
				}
			} else {
				break;
			}
		}
	}
	
	function show_next_quote() {
		next_quote = quote_showing + 1;
		if (!document.getElementById("quote_" + next_quote)) { next_quote = 1; }
		show_quote(next_quote);
	}
	
	function show_quote_and_stop(i) {
		show_quote(i);
		clearInterval(quote_timer);
	}
	
	quote_timer = setInterval("show_next_quote()", 6000);
	banner_timer = setInterval("show_next_banner()", 6000);

</script>
