<script>
  $( function() {
	min_sum = 100; // минимум к пополнению
	max_sum = 100000; // максимум
	  
    var handle = $( "#custom-handle" );
    var slider = $( "#slider" ).slider({
      create: function() {
        handle.text( $( this ).slider( "value" ) );
      },
      slide: function( event, ui ) {
        handle.text( ui.value );
        calc( ui.value );
      },
      min: min_sum,
      max: max_sum,
    });
	
	function calc(v) {
		okup = 90 ;
		speed_1d = v/okup; // в день
		speed_1m = speed_1d *30; // скорость в месяц
		speed_6m = speed_1m * 6; // скорость в 6 месяцев
		speed_3m = speed_1m * 3; // скорость в 3 месяцев
		speed_1sec = speed_1d/86400; // в секунду 
		speed_1sec = speed_1sec.toFixed(8);
		$('#speed_ind').text(speed_1sec);
		$('#speed_3m').html(speed_3m.toFixed(2));
		$('#speed_6m').html(speed_6m.toFixed(2));
		$('#speed_1m').html(speed_1m.toFixed(2));
		$('#speed_1d').html(speed_1d.toFixed(2));
		$('#span_price').text(v);
		$('#form_price').val(v);
		$('#sum_text').val(v);
	}
	calc(min_sum);
	
	$('#sum_text').keyup(function () {
		if ($('#sum_text').val() =='') return;
		s = parseInt($('#sum_text').val());
		if (isNaN(s)) $('#sum_text').val(min_sum);
		if (s<min_sum) {
			$('#error_text').text("Минимальная сумма: "+min_sum+" руб.");
			return;
		} else  $('#error_text').text("");
		if (s>max_sum) $('#sum_text').val(max_sum);
		s = parseInt($('#sum_text').val());
		slider.slider( "value", s );
        handle.text( s );
		calc(s);
	});
  });
