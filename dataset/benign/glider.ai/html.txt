function isScrolledIntoView(elem){
    var $elem = $(elem);
    var $window = $(window);

    var docViewTop = $window.scrollTop();
    var docViewBottom = docViewTop + $window.height();

    var elemTop = $elem.offset().top;
    var elemBottom = elemTop + $elem.height();

    return ( ((elemTop >= docViewTop) && (elemTop <= docViewBottom)) || ((elemBottom >= docViewTop) && (elemBottom <= docViewBottom)) );
}

          $.post("//hire.glider.ai/api/contactUs", $('form.contact-us').serialize(), function(data){
            if(data){
              // actions to be done after getting response from post API
              $('.contact-us input').val('');
              $('.contact-wrapper .form-content').addClass('submitted');
              $('.contact-wrapper .form-content .top-form-content h1').text('Thank you');
              $('.contact-wrapper .form-content .top-form-content .sub').text('Our sales representative will contact you soon');
              $('.contact-wrapper .form-content form.contact-us').hide();
              $('.contact-wrapper .form-content .submit-contact').hide();
              //$('.contact-wrapper .form-content .submit-contact').text('Done');
            }
          });
          $('.contact-us input').val('');
        }else{
          //focus the first invalid element from submitMap
          for(var i in submitMap){
            if(submitMap[i] == false){
              var focusElement = elementMap[i];
              $(focusElement).focus();
              break;
            }
          }
        }
    });
});

function checkLengthOfElement(currentElement, isNotRequired){
  var currentInputElement = $(currentElement),
      currentInputElementName = currentInputElement.attr('name'),
      currentValue = currentInputElement.val(),
      errorElement = currentInputElement.siblings('.error'),
      rolePattern = /^[a-zA-Z ]{2,30}$/,
      organizationPattern = /^[a-zA-Z0-9]|[- @\.#&!]*$/,
      phonePattern = /[0-9\+-\/()\/\s]{10,}$/,
      emailPattern = /^[-a-z0-9~!$%^&*_=+}{\'?]+(\.[-a-z0-9~!$%^&*_=+}{\'?]+)*(\+[A-Za-z0-9-]+)?@([A-Za-z0-9_][-A-Za-z0-9_]*(\.[-A-Za-z0-9_]+)*\.(aero|arpa|biz|com|coop|edu|gov|info|int|mil|museum|name|net|org|pro|travel|mobi|[a-z][a-z])|([0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}))(:[0-9]{1,5})?$/;
  currentValue = currentValue && currentValue.trim();
  if(!isNotRequired && currentValue.length <= 0){
    currentInputElement.addClass('red-border');
    errorElement.text('Required');
    return false;
  }else{
    currentInputElement.removeClass('red-border');
    switch (currentInputElementName){
      case 'fullName':
        if(!(rolePattern.test(currentValue))){
          currentInputElement.addClass('red-border');
          errorElement.text('Enter a valid full name');
          return false;
        }else{
          errorElement.text('');
          return true;
        }
        break;
      case  'phone':
        if( currentValue && !phonePattern.test(currentValue) ){
          currentInputElement.addClass('red-border');
          errorElement.text('Enter a valid phone number');
          return false;
        }else{
          errorElement.text('');
          return true;
        }
        break;
      case 'role':
        if(currentValue && !(rolePattern.test(currentValue))){
          currentInputElement.addClass('red-border');
          errorElement.text('Enter a valid role');
          return false;
        }else{
          errorElement.text('');
          return true;
        }
        break;
      case 'organization':
        if(!(organizationPattern.test(currentValue))){
          currentInputElement.addClass('red-border');
          errorElement.text('Enter a valid Organization');
          return false;
        }else{
          errorElement.text('');
          return true;
        }
        break;
      case 'email':
        if(!(emailPattern.test(currentValue))){
          currentInputElement.addClass('red-border');
          errorElement.text('Enter a valid email address');
          return false;
        }else{
          errorElement.text('');
          return true;
        }
        break;
    }
  }
}
