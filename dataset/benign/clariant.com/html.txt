    var LanguageSwitchNew = {
        getCookie: function (cname) {
            var name = cname + "=";
            var decodedCookie = decodeURIComponent(document.cookie);
            var ca = decodedCookie.split(';');
            for (var i = 0; i < ca.length; i++) {
                var c = ca[i];
                while (c.charAt(0) == ' ') {
                    c = c.substring(1);
                }
                if (c.indexOf(name) == 0) {
                    return c.substring(name.length, c.length);
                }
            }
            return "";
        },

        init: function () {
            $("div#languageswitchoverlay a.textlink").click(function (e) {
                LanguageSwitchNew.redirectToRoot();
            });

            $("div#languageswitchoverlay a.button").click(function (e) {
                LanguageSwitchNew.execute();
            });

            $("div#languageswitchoverlay a.flaglink").click(function (e) {
                LanguageSwitchNew.switchLanguage(this.id);
            });

            $("select.lsRegions").change(function () {
                var regionId = $(this).val();
                LanguageSwitchNew.onRegionChanged(regionId, true);
            });

            $("select.lsCountries").change(function () {
                var countryId = $(this).val();
                LanguageSwitchNew.onCountryChanged(countryId, true);
            });

            var regionId = $("select.lsRegions").val();
            this.onRegionChanged(regionId, false);


            var countryId = $("select.lsCountries").val();
            this.onCountryChanged(countryId, false);
        },

        switchLanguage: function (lang) {
            this.setCookie("CLALang", lang, 365, "/");

            var urlSplit = window.location.href.split('/');
            if (urlSplit.length < 4) {
                window.location.reload();
            }
            else {
                if (urlSplit[3].length == 2 || urlSplit[3].length == 5)
                    urlSplit[3] = lang;
                else {
                    urlSplit.splice(3, 0, lang);
                }

                window.location.href = urlSplit.join('/');
            }
        },

        onRegionChanged: function (regionId, setDefault) {
            if (regionId === "") {
                $("select.lsCountries option").remove();
                var newOptions = $("select.lsCountriesSrc option").clone();
                $("select.lsCountries").append(newOptions);

            } else {
                $("select.lsCountries option").remove();
                var newOptions = $("select.lsCountriesSrc option." + regionId).clone();
                $("select.lsCountries").append(newOptions);
            }
            $('select.lsCountries').trigger('chosen:updated');
        },


        onCountryChanged: function (countryId, setDefault) {
            //TODO: to be filled in when we implement languages.

        },

        execute: function () {
            var region = $("select.lsRegions").val();
            var country = $("select.lsCountries").val();
            var langUrl = $("select.lsLanguages").val();
            langUrl = ((typeof langUrl === typeof undefined) || (langUrl === "undefined") || (langUrl === "")) ? window.location.href : langUrl;
            this.setCookie("CLARegion", region, 365, "/");
            this.setCookie("CLACountry", country, 365, "/");
            this.setCookie("CLALang", "en", 365, "/ ");
            window.location.href = langUrl;
            $("#languageswitchoverlay").hide();
            $(".country-label").html($("select.lsCountries option:selected").html());
            return false;
        },

        redirectToRoot: function () {
            langUrl = "/en/";
            this.setCookie("CLARegion", "", 365, "/");
            this.setCookie("CLACountry", "", 365, "/");
            this.setCookie("CLALang", "en", 365, "/");
            window.location.href = langUrl;
            return false;
        },

        setCookie: function (name, value, expirationDays, path) {
            var d = new Date();
            d.setTime(d.getTime() + (expirationDays * 24 * 60 * 60 * 1000));
            var expires = "expires=" + d.toGMTString();
            document.cookie = name + "=" + value + "; " + expires + "; path=" + path;
        },

        loaded: false
    };

if (width > 1280) return "desktop";
else if (width > 960 && width <= 1280) return "tabletL";
else if (width > 480 && width <= 960) return "tabletP";
else return "phone";
}

        function claGetCookie(key) {
            return decodeURIComponent(document.cookie.replace(new RegExp("(?:(?:^|.*;)\\s*" + encodeURIComponent(key).replace(/[\-\.\+\*]/g, "\\$&") + "\\s*\\=\\s*([^;]*).*$)|^.*$"), "$1")) || null;
        }
        function claRemoveCookie(name) {
            document.cookie = name + '=; path=/;expires=Thu, 01 Jan 1970 00:00:01 GMT;';
        }
        function claSetCookie(key, value, expires) {
            var sExpires = (!expires || expires == false) ? "; expires=Fri, 31 Dec 9999 23:59:59 GMT" : "";
            document.cookie = encodeURIComponent(key) + "=" + encodeURIComponent(value) + sExpires + "; path=/; secure";
            return true;
        }
        function getUTMParams() {
            var keyPairs = [],
                params = window.location.search.substring(1).split('&');
            for (var i = params.length - 1; i >= 0; i--) {
                var pair = params[i].split("=");
                if (pair && pair.length == 2) {
                    var key = pair[0].toLowerCase();
                    var value = decodeURIComponent(pair[1]);
                    if (key.indexOf("utm_") == 0) {
                        keyPairs.push({ key: "cla_"+key, value: value });
                    }
                }
            };
            return keyPairs;
        }
        function removeUTMCookies() {
            claRemoveCookie("cla_utm_campaign");
            claRemoveCookie("cla_utm_source");
            claRemoveCookie("cla_utm_medium");
            claRemoveCookie("cla_utm_term");
            claRemoveCookie("cla_utm_content");
        }

        function saveCampaignInfo() {
            if (window.location.search.toLowerCase().indexOf("utm_") != -1) {
                removeUTMCookies();
                //the UTM parameters are present, which means this is a landing page for a campaign
                var params = getUTMParams();
                if (params && params.length > 0) {
                    //store the UTM info for this session
                    params.forEach(function (pair) {
                        claSetCookie(pair.key, pair.value, true);
                    });
                }
            }
        }

        function saveCurrentPageId() {
            if (window.location.href.toLowerCase().indexOf("/login") == -1) {
                claSetCookie("cla_lastpageid", $("#hidCurrentPageId").val());
            }
        }
