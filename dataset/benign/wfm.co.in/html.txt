$(document).ready(function ($) {

    function doneResizing() {
        if ($(window).width() != winwidth) {
            window.location.reload();
        }
    }
    if (!$('.sidebar-filter').length && console) {
        console.log('".sidebar-filter" element not found, please check Sticky Element plugin settings');
        return;
    }
    if (!$('.cat_description').length && console) {
        console.log('".cat_description" element not found, please check Sticky Element plugin settings');
        return;
    }
    if (winwidth >= 420) {
        var el = $(".sidebar-filter");
        var elwidthpx = el.css('width');
        var elleftmar = el.css('margin-left');
        var eltop;
        var elheight;
        var elleft = el.offset().left;
        $(".sidebar-filter").css("float", 'left');
        $(".right-container").css("float", 'right');
        $('<div/>').attr('id', 'clone').css('width', elwidthpx).insertAfter($('.sidebar-filter'));

        var style = document.createElement('style');
        style.type = 'text/css';
        style.innerHTML = '.sticky-element-fixed { position: fixed; top: 0; width: ' + elwidthpx + ' !important; }';
        document.getElementsByTagName('head')[0].appendChild(style);

        $(window).scroll(function () {

            if (typeof eltop === "undefined") {
                eltop = el.offset().top;
            }
            elheight = el.outerHeight();

            var end = $(".cat_description");
            endtop = end.offset().top - 73;
            var winscroll = $(window).scrollTop();

            if (winscroll > eltop) {
                $(".sidebar-filter").addClass('sticky-element-fixed');
                $(".sidebar-filter").css("left", elleft);
                $(".sidebar-filter").css("margin-left", 0);
                $('#clone').css('height', elheight);
                $('#clone').css('width', elwidthpx);
                $('#clone').css('float', 'left');
            } else {
                $(".sidebar-filter").removeClass('sticky-element-fixed');
                $(".sidebar-filter").css("left", "auto");
                $(".sidebar-filter").css("margin-left", elleftmar);
                $('#clone').css('height', 0);
                $('#clone').css('width', 0);
                $('#clone').css('float', 'left');
            }

            if (winscroll + elheight > endtop) {
                var amount = endtop - (winscroll + elheight);
                $(".sidebar-filter").css("top", amount + "px");
            } else {
                var amount = endtop - (winscroll + elheight);
                $(".sidebar-filter").css("top", "");
            }
        });
    }
});
