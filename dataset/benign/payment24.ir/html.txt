				o[this.name] = [o[this.name]];
			}
			o[this.name].push(this.value || '');
		} else {
			o[this.name] = this.value || '';
		}
	});
	return o;
};

$.fn.isOnScreen = function(topOffset) {
	var win = $(window);
	var viewport = {
		top: win.scrollTop(),
		left: win.scrollLeft()
	};
	viewport.right = viewport.left + win.width();
	viewport.bottom = viewport.top + win.height() - topOffset;
	var bounds = this.offset();
	bounds.right = bounds.left + this.outerWidth();
	bounds.bottom = bounds.top + this.outerHeight();
	return (!(viewport.right < bounds.left || viewport.left > bounds.right || viewport.bottom < bounds.top || viewport.top > bounds.bottom));
};

var bsMSG = function (msg, type, seprator) {
	html = '';
	msg = (msg instanceof Array ?msg:[msg]);

function SelectText(element) {
	var text = element
		, range, selection ;
	if (document.body.createTextRange) {
		range = document.body.createTextRange();
		range.moveToElementText(text);
		range.select();
	} else if (window.getSelection) {
		selection = window.getSelection();
		range = document.createRange();
		range.selectNodeContents(text);
		selection.removeAllRanges();
		selection.addRange(range);
	}
}

<script>function roundTooman(price, by, up)
{
	up=up||true;
	by=by||1;
	if (up)
		price = Math.ceil(price / by);
	else
		price = Math.floor(price / by);
	return parseInt(price * by);
}


function calculatePrice(){
	$('.stotal span').html('0');
	service = $('select[name=service]').val();
	amount = parseFloat( $('input[name=sprice]').val().toString().replace(',',''));
	currency = $('input[name=scurrency]:checked').val();
	//disable other currencies for Webmoney
	if( service=='perfectMoney:charge' || service=='perfectMoney:buy' || service=='btce:charge') {
		$('input[value=GBP]').parent().fadeOut();
		$('input[value=EUR]').parent().fadeOut();
		$('input[value=USD]').trigger('click');
	}else if(service=='WebMoney:buy' || service=='WebMoney:charge'  || service=='Skrill:charge' || service=='Skrill:buy'){
		$('input[value=GBP]').parent().fadeOut();
		$('input[value=EUR]').parent().fadeOut();
		$('input[value=USD]').trigger('click');
	}else if(service=='BtcController:buy' || service=='BtcController:charge' ){
        $('input[value=GBP]').parent().fadeOut();
        $('input[value=EUR]').parent().fadeOut();
        $('input[value=USD]').parent().fadeOut();
        currency ='USD';

    }else{
		$('input[value=GBP]').parent().fadeIn();
		$('input[value=EUR]').parent().fadeIn();
	}

	if(!(amount>0))
		return;
	if(service.length==0)
		return;
	// Currency rates
	ref_rate = 13198;
	rates=[];
	rates['USD'] = 1;
	rates['EUR'] =1.11;
	rates['GBP'] = 1.287;

	if (service == 'PayPal:buy' || service == 'PayPal:charge') {
		rates['USD'] = 1.02;
		rates['EUR'] = 1.02 * 1.15;
		rates['GBP'] = 1.02 * 1.30;
	}

	if (service == 'VisaMaster:buy' ) {
		rates['EUR'] = 1.15;
		rates['GBP'] = 1.31;
	}
    if (service == 'VisaMaster:buy' ) {
        rates['EUR'] = 1.15;
        rates['GBP'] = 1.31;
    }
	service_rates=[];
	service_rates['PayPal:buy']=1;
	service_rates['PayPal:charge']=1;
	service_rates['WebMoney:buy']=1.038;
	service_rates['WebMoney:charge']=1.038;
	service_rates['btce:charge']=1.01;
	service_rates['VisaMaster:buy']=1.05;
	service_rates['Skrill:buy']=1.08;
	service_rates['Skrill:charge']=1.08;
	service_rates['perfectMoney:buy']=1.013;
    service_rates['perfectMoney:charge']=1.013;
    service_rates['BtcController:charge']=1.038;
    service_rates['BtcController:buy']=1.038;

    if(service=='BtcController:buy' || service=='BtcController:charge' ){
        globalPrice = 10014;
        amount=amount*globalPrice;

	}
	currency_price=Math.round(ref_rate * service_rates[service]) * rates[currency];

	result = parseInt(amount *  currency_price);
	result = result.toString().replace(/(\d)(?=(\d{3})+(?!\d))/g, "$1,");
	$('.stotal span').html(result);

}
