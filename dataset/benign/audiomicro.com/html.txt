	$(document).ready(function () {

		/* Global Variables */
		TRACKS_JSON = null;
		PACKS_JSON = null;
		PACKS = {};

	    
	    // Detect is user logged or not
	    var isLogged = false;
	   
	    var balances = {
	    	sfx: 0, 
	    	music: 0	    };

	    // Change to true to inspect jPlayer calling Events && errors to print them as alert()
	    var debug = false;

	    // Player is ready for playing
	    var playerIsReady = false;

	    // HTML5 is not suported & flash will be used
	    var flashFallback = false;

	    // Handler for ajax requests - for example, if user stopped waiting load track, you can call ajaxHandler.abort() - it's browser api
	    var ajaxHandler = null;

	    // Dom Object of track
	    var trackDOM = null;

	    // Finished Playlist
	    var playlist = [];

	    // Track ids from DOM to get for playlist
	    var dataToGet = {};

	    // Currently played track's index
	    var playingIndex = -1;

	    // Current track id
	    var currentTrackId = null;




	    /**
	     * This method initialize jPlayer with tracks ids from dom
	     * It's called on page's first load
	     * @return
	     */
	    var jPlayerInitialize = function () {
	        
	        /**
	          * Init ttwMusicPlayer (based on jPlayer)
	          * TTW: http://www.codebasehero.com/2011/06/html-music-player/
	          * jPlayer: http://jplayer.org/latest/developer-guide/
	          * To use jPlayer Api you must call it near
	          * $('.jPlayer-container')
	          * Example:
	          * $('.jPlayer-container').jPlayer('play')
	          */

	        var jPlayerOptions = {
	            swfPath: '/player/plugin/jquery-jplayer/',
	            preload: "none",
	            solution: 'html,flash',
	            wmode: 'window',
	            errorAlerts: debug,
	            loop: false,
	            timeFormat: {
	                padMin: true
	            },
	            repeat: function(){}
	        };

	        // Browsers manual html5.canPlayMp3 detect: jplayer can't make fallback so we're doing it manually
	        if (canPlayMp3WithHtml5() === false) {
	            jPlayerOptions.solution = "flash,html";
	        }

	        $('body').ttwMusicPlayer(playlist, {
	            autoplay: false,
	            jPlayer: jPlayerOptions
	        });

	        afterPlayerCreated();
	    }


	    /**
	     * Helps us to detect - is player active or not
	     * Active - means visible
	     * @return   true / false
	     */
	    var isPlayerActive = function () {
	        return ($('.ttw-music-player').css('display') == 'none')? false: true;
	    }


	    /**
	     * This function returns us true if player opened & false in another way 
	     * @return {Boolean}
	     */
	    var isPlayerPaused = function () {
	    	return $('.jPlayer-container').data().jPlayer.status.paused;
	    }


	    /**
	     * Detect HTML5 History is available - we use it in navigation with
	     * Ajax player to write states in history stack
	     * @return      true / false
	     */
	    function historyApiAvailable() {
	        return !!(window.history && history.pushState);
	    }


	    /**
	     * Checking browser for compatibility with mp3
	     * Some of FF 20~ versions can't play mp3 with HTML5 because there's problems with browser licensing
	     * Details: http://stackoverflow.com/questions/4923136/why-doesnt-firefox-support-mp3-file-format-in-audio
	     */
	    var canPlayMp3WithHtml5 = function () {
	        var a = document.createElement('audio');
	        return (!!(a.canPlayType && a.canPlayType('audio/mpeg;').replace(/no/, '')))
	    }


	    /**
	     * afterPlayerCreated - handler for additional processing & logic
	     * @return
	     */
	    var afterPlayerCreated = function () {
	        //This mean that flash solution is used
	        if (!$('.ttw-music-player').find('object').length > 0){
	            $('.ttw-music-player').hide();
	        }

	        return;
	    };


	    /**
	     * Global functions: calls in categories list && ajax_search
	     * This function helps us to gather needed track ids. Then
	     * we'll get information with ajax about them & form playlist
	     * @return
	     */
	    getTracksListFromHtml = function () {

	        // Reinit for ajax-reload
	        dataToGet = {};

	        // Getting Track`s ids
	        if ($('.media-item').
	        		has('.media-item-play').length > 0)
	        {
	            $('.media-item').each(function (i, element) {
	                var $element = $(element).find('.media-item-play a:first');
	                var tr_id    = $element.attr('track_id');
	                var pack_id  = $element.attr('pack_id');
	                dataToGet[i] = tr_id;
	                PACKS[tr_id] = pack_id;
	            });
	        }

	        // Carousel Items
	        if($('.grid-featured-play').length > 0){
	            
	            // Main Page & Carousel Items - Featured tracks
	            $('.grid-featured-play').each(function (i, element) {
	                var tr_id      = $(element).attr('track_id');
	                dataToGet[i] = tr_id;
	            });
	        }

	        // console.group("Obtained Media Items Ids From DOM");
	        //     console.table(dataToGet)
	        // console.groupEnd();
	    }


	    /**
	     * W/ this function we send dataToGet with track's ids to 
	     * obtain additional information for current page & player
	     * @param  {string} action       Additional action?
	     * @param  {string} urlToFollow  Where to send?
	     * @return
	     */
	    reloadPlaylist = function (action, urlToFollow) {

	        dataToGet = {tracks: dataToGet, packs: null};

	        if (PACKS !== null) {
	            dataToGet.packs = PACKS;
	        }

	        // console.group("Obtained Media Items From DOM");
	        //     console.table(dataToGet)
	        // console.groupEnd();

	        if(jQuery.isEmptyObject(dataToGet.tracks) && jQuery.isEmptyObject(dataToGet.packs)){
	            return;
	        }

	        // Get Track`s Data to build URLs and frequency progress images after ajax reload
	        ajaxHandler = $.ajax({
	            url: '/tracks/dialog_player/',
	            data: dataToGet,
	            type: 'post',
	            success: function (resp) {

	                // console.group("Media Items Response from backend");
	                //     console.table(dataToGet)
	                // console.groupEnd();

	                response = JSON.parse(resp);
	                TRACKS_JSON = response.tracks;
	                PACKS_JSON = response.packs;

	                if (action !== undefined && action == "social") {
	                    reinitSocialButtons(urlToFollow);
	                }

	                preloadWavesImages(dataToGet, TRACKS_JSON);
	                playlist = [];

	                if ($('.media-player-activate').length > 0) {
	                    $('.media-player-activate').each(function (i, el) {
	                        
	                        var id        = $(el).attr('track_id');
	                        var trackDOM = $('.media-player-activate[track_id=' + id + ']');
	                        
	                        var is_in_playlist = 
	                            (playlist.map(function(e) { return e.track_id; }).indexOf(id) > -1) ?
	                                true:
	                                false;

	                        if (TRACKS_JSON[id] !== undefined && !is_in_playlist) {
	                            var track = TRACKS_JSON[id];

	                            playlist.push({
	                                title: '<a href="' + track.Track.page_url + '">' + track.Track.title + '</a>',
	                                artist: '<a href="/artist/' + track.User.login + '">' + track.User.login + '</a>',
	                                mp3: '/tracks/listen/' + track.Track.storage_id,
	                                oga: '/tracks/listen/' + track.Track.storage_id,
	                                track_id: id
	                            });
	                        }
	                    });
	                } 

	                var track_id = $('.ttw-music-player').attr('current_track_id');
	                
	                trackDOM = $('.media-player-activate[track_id=' + track_id + ']').parents('.media-item');

	                // Highlight current track
	                trackDOM.addClass('playing');
	                // Is player playing? Ok, switch `play` icon to `pause`
	                if(!isPlayerPaused()){
	                	trackDOM.find('.media-player-activate').addClass('pause');
	                }
	                
	            
	                rebuildPlaylist();

	                if (action != undefined && action == "pageAutoload") {
	                    
	                    var loaded_with = $('.jPlayer-container').data('pageAutoload');
	                    
	                    switch (loaded_with) {
	                        case 'prev':
	                            $('.ttw-music-player .jp-previous').click().click();
	                            playingIndex = 9;
	                            $('.jPlayer-container').jPlayer('play');
	                            break;
	                        case 'next':
	                            $('.jp-previous').click();
	                            playingIndex = 0;
	                            //$('.jPlayer-container').jPlayer('setMedia', playlist[0]);
	                            $('.jPlayer-container').jPlayer('play');
	                            break;
	                    }
	                }

	                $('.play-button').show();

	                $('.media-player-activate').show();
	                $('.preloader-adaptive').hide();

	                return true;
	            },
	            beforeSend: function () {
	                $('.play-button').hide();
	                $('.media-player-activate').hide();
	                $('.preloader-adaptive').show();
	            },
	            error: function () {
	                return;
	            }
	        });
	    }

	    /**
	     * Function helps as to fill data in player
	     * For example, title, artist etc.
	     * @param  {array} data  Data to fill
	     * @return
	     */
	    var fillInfo = function (data) {

	        var info = $('.ttw-music-player .track-info-content');

	        //Additional operations for loops pack
	        /*if(data['Track']['is_loop_pack'] != undefined){
	            $('.recent-plays-link').hide();
	        }else{
	            $('.recent-plays-link').show();
	        }*/
	        
	        $('.track-info .title').html('<a title="'+data.Track.title+'" href="'+data.Track.page_url+'">'+data.Track.title+'</a>');
	        $('.track-info .artist').html('<a href="/artist/'+data.User.login +'">' + data.User.login+'</a>');

	        if (data.TracksAdditionalParam.genres_str) {
	            info.find('.genres').html(format(data.TracksAdditionalParam.genres_str))
	                .parents('.g_parent').show();
	        } else {
	            info.find('.genres').parent().hide();
	        }

	        if (data.TracksAdditionalParam.details) {
	            info.find('.description').html(format(data.TracksAdditionalParam.details))
	                .parent('.description_parent').show();
	        }else{
	            info.find('.description').parent().hide();
	        }

	        info.find('.format').html(format(data['Track']['format']));
	        info.find('.length').html(format(data['Track']['length']) + 's');
	        info.find('.bit_rate').html(format(data['TracksAudioParam']['bit_rate']));
	        info.find('.refresh_rate_khz').html(format(data['TracksAudioParam']['refresh_rate_khz']));
	        info.find('.bit_mode').html(format(data['TracksAudioParam']['bit_mode']));
	        info.find('.bpm').html(format(data['TracksAudioParam']['bpm']));
	        info.find('.music_loop').html((data['TracksAudioParam']['music_loop'] == true) ? 'Yes' : 'No');
	        info.find('.bits_per_sample').html(format(data['TracksAudioParam']['bits_per_sample']));

	        var price = data['buying_price'];
	        if (price > 0) {
	            price = '$' + (price / 100);
	            $('.add2cart-player').html('Add To Cart ' + price);
	        }

	        // Native jPlayer code is filling track's duration only after they get it from s3
	        // It's prefill wait duration
	        var time    = new Date(0, 0, 0, 0, 0, data['Track']['length'], 0);
	        var minutes = ('0' + time.getUTCMinutes()).slice(-2);
	        var seconds = time.getUTCSeconds();
	        
	        // Player waveform dimensions
	        var Dims = getPlayerWaveformDims($(document).width());
	        
	        if(!(minutes && seconds)) seconds = 1;
	        seconds = ('0' + seconds).slice(-2);

	        var duration =  minutes + ':' + seconds;
	        $('.ttw-music-player .jp-duration').html(duration);

	        var uid = data['Track']['user_id'];
	        var storage = data['Track']['storage_id'];

	        // Wave image show
	        $('.ttw-music-player .audiolevel-track').css('background', 
	            'url(/player/waveform.php?user_id='+uid+'&storage_id='+storage+'&width='+Dims.width+'&height='+Dims.height+')'
	        );

	        $('.ttw-music-player .audiolevel-track-loaded').css('background', 
	            'url(/player/waveform.php?user_id='+uid+'&storage_id='+storage+'&played=1&width='+Dims.width+'&height='+Dims.height+')'
	        );

	        return;
	    }


	    /**
	     * This is hack to force waveimages for preload -
	     * we're creating hidden element & set as attributes images urls
	     * so browser starts to load them immediately & we can use them
	     * in future - when user'll rewind to our track
	     * @param  {object} dataToGet [description]
	     * @param  {object} json_data   User & track data to build waveimage url
	     * @return
	     */
	    var preloadWavesImages = function (dataToGet, jsonData) {

	        var preloadRule = '';
	        var width = $(document).width();

	        // Player waveform dimensions
	        var DimsPlayer = getPlayerWaveformDims(width);
	        var DimsBanner = getBannerWaveformDims(width);

	        /**
	         * Check is we're on the track view page. If yeah - preload wide banner waveform 
	         */
	        if($('.player-banner').length > 0){
	            var id = $('.player-banner .media-player-activate').attr('track_id');
	            if (jsonData[id] !== undefined)
	                preloadRule += ' url(/player/waveform-banner.php?user_id=' + 
	                    jsonData[id]['Track']['user_id'] + '&storage_id=' + jsonData[id]['Track']['storage_id'] + 
	                    '&width='+DimsBanner.width+'&height='+DimsBanner.height+')';
	        }

	        /**
	         * Other default waveimages for bottom player
	         */

	        if ($('.media-player-activate, .grid-featured-play').length > 0) {
	        	var preloadIds = [];
	            $('.media-player-activate, .grid-featured-play').each(function (i, el) {
	                var id = $(el).attr('track_id');
	                if (jsonData[id] !== undefined && $.inArray(id, preloadIds) == -1){
	                    preloadRule += ' url(/player/waveform.php?user_id=' + 
	                		jsonData[id]['Track']['user_id'] + '&storage_id=' + jsonData[id]['Track']['storage_id'] + 
	                		'&width='+DimsPlayer.width+'&height='+DimsPlayer.height+')';
	                	preloadIds.push(id);
	                }
	            });
	        }

	        /**
	         * Trick to preload media waves images
	         */
	        $('#preload').css('display', 'none')
	                     .css('content', preloadRule);
	    }

	    /**
	     * Helps us to format string for displaying on player
	     * @param  {string} data_str    String to format
	     * @return
	     */
	    var format = function (data_str) {
	        if (data_str == 0 || data_str == '' || data_str == null || data_str == false){
	            return '-';
	        }else{
	            return data_str;
	        }
	    }


	    /**
	     * Method helps us to send data & get tracks information.
	     * This method is used on first time page loading.
	     * @return
	     */
	    var playerDataInit = function () {
	        
	        dataToGet = {tracks: dataToGet, packs: null};

	        if (PACKS !== null) {
	            dataToGet.packs = PACKS;
	        }

	        // console.group("Obtained Media Items From DOM");
	        //     console.table(dataToGet)
	        // console.groupEnd();

	        // No media items
	        if(jQuery.isEmptyObject(dataToGet.tracks) && jQuery.isEmptyObject(dataToGet.packs)){
	            return;
	        }

	        // Get Track`s Data to build urls and frequency progress images
	        ajaxHandler = $.ajax({
	            url: '/tracks/dialog_player/',
	            data: dataToGet,
	            type: 'post',
	            beforeSend: function () {
	                //$('.play-button,.show-popup-player').each(function(){$(this).hide()});
	                $('.media-player-activate').hide();
	                $('.preloader-adaptive').show();
	            },
	            success: function (resp) {

	                // Parse response
	                response    = JSON.parse(resp);
	                TRACKS_JSON = response.tracks;
	                PACKS_JSON  = response.packs;

	                // Empty resp variable indicates that there's no any tracks on page
	                if(!resp){
	                    return;
	                }

	                // Preload wave images
	                preloadWavesImages(dataToGet, TRACKS_JSON);

	                if ($('.media-player-activate, .grid-featured-play').length > 0) {

	                    $('.media-player-activate, .grid-featured-play').each(function (i, el) {
	                        
	                        var id = $(el).attr('track_id');
	                        
	                        var is_in_playlist = 
	                            (playlist.map(function(e) { return e.track_id; }).indexOf(id) > -1) ?
	                                true:
	                                false;

	                        // Check is track id not empty in response & it isn't in playlist
	                        if (TRACKS_JSON[id] !== undefined && !is_in_playlist) {
	                            var tr = TRACKS_JSON[id];
	                            playlist.push({
	                                title: '<a href="' + tr.Track.page_url + '">' + tr.Track.title + '</a>',
	                                artist: '<a href="/artist/' + tr.User.login + '">' + tr.User.login + '</a>',
	                                mp3: '/tracks/listen/' + tr.Track.storage_id,
	                                oga: '/tracks/listen/' + tr.Track.storage_id,
	                                track_id: id
	                            });
	                        }
	                    });
	                } 

	                jPlayerInitialize();

	                $('.media-player-activate').show();
	                $('.preloader-adaptive').hide();
	            },
	            error: function () {
	                return;
	            }
	        });
	    }

	    /**
	     * Helps us to rebuild our playlist
	     * Using after ajax pages loading
	     * @return
	     */
	    var rebuildPlaylist = function () {

	        $.fn.playerPlaylistRebuild(playlist);

	        //if(!$('.jPlayer-container').data('jPlayer').status.media.mp3){
	        
	        $('.tracks .playing').removeClass('playing');
	        //}
	        // Playing index=-5 signals that page was reloaded with ajax
	        playingIndex = -5;
	        playerIsReady = true;
	    }

	    /**
	     * This function helps us manipulate with in-player buttons such as
	     * add2cart, remove, etc. We use this function in $.jPlayer.event.loadstart
	     * && $(window).resize() to show user appropriate buttons that depends on
	     * window width 
	     * @return
	     */
	    var changePlayerButtons = function() {

	        var width            = window.innerWidth;
	        var buttons          = $('.purchase-ajax-buttons a:visible');
	        var current          = $('.ttw-music-player').attr('current_track_id');
	        var preview_button   = $('.ttw-music-player #download-preview-player');
	        var subscribe_button = $('.purchase-ajax-buttons');

	                           /* Tablet dimensions for narrow buttons */
	        var use_narrow  =  window.innerWidth > 767 && window.innerWidth < 950 ||    
	                           /* Desktop dimensions for narrow buttons */
	                           window.innerWidth > 1198 && window.innerWidth < 1525;

	        if(use_narrow) {
	            subscribe_button = subscribe_button.find('.subscribe-narrow');
	        }else {
	            subscribe_button = subscribe_button.find('.subscribe-button');
	        }
	        
	        if (TRACKS_JSON[current]['Track']['top_category_id'] == 2 /*&& TRACKS_JSON[current]['Track']['watermarked_preview'] == 1*/) {
	            if(!use_narrow)
	                subscribe_button.html('Subscribe $10').attr('title', 'Subscribe $10');
	            subscribe_button.attr('href', '/subscribe-music');
	            preview_button.show().attr('href', '/tracks/download_preview/' + TRACKS_JSON[current]['Track']['storage_id']);
	        } else {
	            if(!use_narrow)
	                subscribe_button.html('Subscribe $1').attr('title', 'Subscribe $1');
	            subscribe_button.attr('href', '/subscribe');
	            preview_button.hide();
	        }
	        $('.purchase-ajax-buttons a, .purchase-mobile-buttons a').hide();


	        // Show corresponding buttons for track
	        pack_id = PACKS !== null && PACKS[current];


	        // Show download button for sfx & music if user have needed balance
	        if(pack_id === undefined && !TRACKS_JSON[current]['is_bought'] && (balances.music || balances.sfx)){
	        	if(
	        		balances.sfx && TRACKS_JSON[current]['Track']['top_category_id'] == 3 || 
	        		balances.music && TRACKS_JSON[current]['Track']['top_category_id'] == 2
	        	){
	        		// Yeah? You want it?
	        		// Hide all buttons
	        		$(subscribe_button).hide();
	        		$('#download-preview-player').hide();
	        		$(cart_class('add')+','+cart_mobile_class('add')).hide();

	        		// Handle
	        		$('.download-button-player').show().
	        			attr('onclick', 'open_dialog('+current+'); return false;');

	        		// Check is previuos dialog html left somewhere behind?
	        		if(!$('#confirm-download-'+current)){
	        			$('<div style="display:none;" class="download-subs-messsage" id="confirm-download-'+current+'">Are you sure?<br /><a href="/payments/buy_track/'+current+'" class="buy">Yes</a>&nbsp;<a href="#" onclick="$("#confirm-download-'+current+'").dialog().dialog("close"); $("#page-loader").show(); return false;" class="cancel">No</a></div>').dialog();
	        		}

	        		return;
	        	}
	        }
	        

	        if (PACKS_JSON !== null && pack_id !== undefined && PACKS_JSON[pack_id]['in_cart'] === true) {
	            $(cart_class('remove')+','+cart_mobile_class('remove')).show();
	        } else if (PACKS_JSON !== null && pack_id !== undefined && isLogged != false && PACKS_JSON[pack_id]['is_bought'] === true) {
	            $('.download-track-owner').attr('pack_id', pack_id).show();
	        } else if (TRACKS_JSON[current]['Track']['assigned_as_free'] == 1) {
	            $('.download-track-owner').show();
	        } else if (TRACKS_JSON[current]['is_bought'] == true && isLogged != false) {
	            $('.download-track-owner').show();
	        } else if (pack_id === undefined && TRACKS_JSON[current]['in_cart'] == false && isLogged != false && TRACKS_JSON[current]['is_bought'] == false) {
	            $(subscribe_button).show();
	            $(cart_class('add')+','+cart_mobile_class('add')).show();
	        } else if (TRACKS_JSON[current]['in_cart'] == true && isLogged != false && TRACKS_JSON[current]['is_bought'] != true) {
	            $(subscribe_button).show();
	            $(cart_class('remove')+','+cart_mobile_class('remove')).show();
	        } else if (TRACKS_JSON[current]['in_cart'] == true && isLogged == false) {
	            $(subscribe_button).show();
	            $(cart_class('remove')+','+cart_mobile_class('remove')).show();
	        } else {
	            if (pack_id === undefined) {
	                $(subscribe_button).show();
	                $('#download-preview-player').show();
	            } else {
	                $('#download-preview-player').hide();
	            }       
	            $(cart_class('add')+','+cart_mobile_class('add')).show();
	        }
	        return;
	    }
	    

	    /**
	     * Function calls on window resize with active player & changes waveforsm bg images for banner player & bottom player
	     * @return
	     */
	    changeWaveformsImages = function(){

	    	var width = $(document).width();
	    	
	    	// Player waveform dimensions
	    	var DimsPlayer = getPlayerWaveformDims(width);
	    	var DimsBanner = getBannerWaveformDims(width);

	    	if(isPlayerActive()){

	    // Track inner rewind
	    $('body').delegate('.audiolevel-track', 'click', function (ev) {
	        if (playerIsReady == false) return;
	        var offset_left = null;
	        
	        // Default offset field
	        if (ev.originalEvent.offsetX != undefined){
	            offset_left = ev.originalEvent.offsetX;
	        }
	        // Mozilla offset field
	        else if (ev.originalEvent.layerX != undefined){
	            offset_left = ev.originalEvent.layerX;
	        }else{
	            offset_left = ev.offsetX;
	        }
	        
	        // Total length in px
	        var total_length = $(ev.currentTarget).first().width();
	        
	        // JPlayer rewinds track in percent. 
	        // Calculate px count that depends on waveimage width for 1 percent
	        var rewind_track_1_perc = total_length / 100;
	        
	        // Total percents to rewind?
	        var rewind_track_percents = Math.floor(offset_left / rewind_track_1_perc);
	        $('.jPlayer-container').jPlayer("playHead", rewind_track_percents);
	    });

	     * Hotkeys handlers for jPlayer
	     * Description:
	     *   If you want to add hotkey, simply add ASCII Code of that button &&
	     *   process logic for this button in keyboardHotkeysHandler()
	     *     e.keyCode - ascii code of clicked symbol
	     *     e.ctrlKey - boolean, means that key was pressed with ctrl
	     * @return
	     */
	    $(document).keydown(function (e) {
	        //console.log(e.keyCode);
	        var availableKeys = [
	            37, // `Left` Key - 37 ascii code
	            39, // `Right` Key - 39 ascii code
	            32, // Space - 32 ascii code
	            27  // Esc - 27 ascii code
	        ];
	        var keyMatched = availableKeys.indexOf(e.keyCode);

	        if (keyMatched > 0 || keyMatched === 0)
	            keyboardHotkeysHandler(e);
	    });


	    /**
	     * Hotkeys Handler to set action & logic on button clicked
	     * @param  Event          Clicked event from handler above
	     * @return
	     */
	    var keyboardHotkeysHandler = function (defEv) {

	        // Activate Hotkeys only on player is active
	        // && there's no any focused textfield
	        if ( isPlayerActive() && !isAnyTextfieldActive()) {
	            
	            if ($('.player-controls').data('ui-tooltip')) {
	                $('.player-controls').tooltip('destroy');
	            }

	            // Left Key Clicked without Ctrl
	            if (defEv.keyCode == 37 && !defEv.ctrlKey) {
	                if ($('.tracks .playing').index() == 0 && (window.location.href).indexOf('/carts') === -1) {
	                    $('.jPlayer-container').data('pageAutoload', 'prev');
	                    $('.jPlayer-container').jPlayer('stop');
	                    $('.pagination-numbers .prev a').click();
	                } else {
	                    $('.jp-previous').click();
	                }
	            }

	            // Right Key Clicked without Ctrl
	            if (defEv.keyCode == 39 && !defEv.ctrlKey) {
	                if ($('.tracks .playing').index() == 9 && (window.location.href).indexOf('/carts') === -1) {
	                    $('.jPlayer-container').data('pageAutoload', 'next');
	                    $('.jPlayer-container').jPlayer('stop');
	                    $('.pagination-numbers .next a').click();
	                } else {
	                    $('.jp-next').click();
	                }
	            }

	            // Esc Key
	            if (defEv.keyCode == 27 && !defEv.ctrlKey) {
	                $('.ttw-music-player').hide();
	                $('.jPlayer-container').jPlayer('stop');
	            }

	            // Space Key
	            if (defEv.keyCode == 32 && !defEv.ctrlKey) {
	                if ($('.jPlayer-container').data('jPlayer').status.paused == true){
	                    $('.jPlayer-container').jPlayer('play');
	                }else{
	                    $('.jPlayer-container').jPlayer('pause');
	                }
	            }
	            defEv.preventDefault();
	        }

	    };

	        for (var i = 0; i < ca.length; i++) {
	            var c = ca[i];
	            while (c.charAt(0) == ' ') c = c.substring(1);

	        }

	    }
