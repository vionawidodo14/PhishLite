      jQuery(function($) {
        ajaxCart.init({
          formSelector: '#AddToCartForm',
          cartContainer: '#CartContainer',
          addToCartSelector: '#AddToCart',
          moneyFormat: "${{amount}}"
        });

      jQuery(document.body).on('afterCartLoad.ajaxCart', function(evt, cart) {
        shop.RightDrawer.open();

        if (cart.item_count >= 1) {
          $('.site-header__cart-indicator').removeClass('hide');
        } else {
          $('.site-header__cart-indicator').addClass('hide');
        }

        
          var currentTotal = $('.ajaxcart__total-price').text().replace('$','').replace(',','');
          var freeShippingAmount = '79';
          var currentTotalNumber = parseFloat(currentTotal);
          var freeShippingAmountNumber = parseFloat(freeShippingAmount);

          if (currentTotalNumber > freeShippingAmountNumber) {
            $('.ajaxcart__progress-bar-content').addClass('hide');
            $('.ajaxcart__progress-bar-success').removeClass('hide');
          }

          if (currentTotalNumber < freeShippingAmountNumber) {
            $('.ajaxcart__progress-bar-content').removeClass('hide');
            $('.ajaxcart__progress-bar-success').addClass('hide');
            var shippingTotal = freeShippingAmount - currentTotal;
            var percentageTotal = (currentTotalNumber / freeShippingAmountNumber * 100).toFixed(2);
            $('.ajaxcart__progress-bar-amount').text(shippingTotal.toFixed(2));
            $('.ajaxcart__progress-bar-percentage').css('width',percentageTotal+'%');
          }

          if ($('.ajaxcart__product.exempt').length == $('.ajaxcart__product').length) {
            $('.ajaxcart__progress-bar-content').addClass('hide');
            $('.ajaxcart__progress-bar-success').removeClass('hide');
          } 

          $('.ajaxcart__price span').each(function() {
            if ($(this).text() == '$0.00') {
              $(this).text('Free');
            }
          });
        
      });
    </script>
