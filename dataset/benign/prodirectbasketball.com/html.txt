        // Set hours, minutes, seconds here single digits
        // require no preceeding 0 i.e. 7 not 07
        endDate = new Date();
        endDate.setHours(21);
        endDate.setMinutes(00);
        endDate.setSeconds(00);

        sunEndDate = new Date();
        sunEndDate.setHours(15);
        sunEndDate.setMinutes(00);

        usWeekendEndDate.setMinutes(00);
        usWeekendEndDate.setSeconds(00);

        var day = (currDate.getDay());

        // This finds the difference between date and endDate
        var diff = 0;

        if ($('#delivery-slider').hasClass('us-timer')) {
            //US Timer
            if (day == 6 || day == 7) {
                // When Timer runs out
                clearInterval(x);
                diff = 0;

            } else {
                if (usWeekendEndDate > currDate) {
                    usWeekendEndDate.setDate(usWeekendEndDate.getDate());
                    diff = usWeekendEndDate - currDate;

                    diff = usWeekendEndDate - currDate;
                }

                // When Timer runs out
                if (diff < 0) {
                    clearInterval(x);
                    display.parentElement.innerHTML = 'Next day delivery window missed';
                    diff = 0;
                    //removeSlide();
                } 
            }
        } else {
            //Sunday
            if (day == 7) {
                if (sunEndDate > currDate) {
                    sunEndDate.setDate(sunEndDate.getDate());
                    diff = sunEndDate - currDate;
                } else {
                    diff = sunEndDate - currDate;
                }

                // When Timer runs out
                if (diff < 0) {
                    var sundayBefore='Order by 8pm weekdays for next day delivery';
                    clearInterval(x);
                    display.parentElement.innerHTML = sundayBefore;
                    diff = 0;
                    //removeSlide();
                }

            } else {

                    endDate.setDate(endDate.getDate());
                    diff = endDate - currDate;

                    diff = endDate - currDate;

                // When Timer runs out
                if (diff < 0) {

                    clearInterval(x);
                    display.parentElement.innerHTML = weekdayBefore;
                    diff = 0;
                    //removeSlide();
                    return;
                }
            }

        hours = Math.floor(diff / (1000 * 60 * 60));
        diff -= hours * 1000 * 60 * 60
        minutes = Math.floor(diff / (1000 * 60));
