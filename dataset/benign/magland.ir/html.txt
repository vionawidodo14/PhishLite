                        $(".registerform").removeClass('d-none');

    function validateMobile(mobile) {//اعتبار سنجی شماره موبایل

        var patt = /09(0[1-2]|1[0-9]|3[0-9]|2[0-1])-?[0-9]{3}-?[0-9]{4}/g;
        var result = mobile.match(patt);
        if (result && result.length > 0)
        {
            return true;
        } else
        {
            return false;
        }
    }

            alert('لطفا عبارت موردنظر خود را وارد نمایید');
            return false;
        }

            alert('لطفا حداقل سه حرف جهت جستجو وارد نمایید');
            return false;

                    {
                        timezone += 60;

                    start = start.setHours(hours, minutes, 0, 0);
                    return start / 1000;
                }
