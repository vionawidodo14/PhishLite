				val = (val * -1);

			var hours = Math.floor(seconds % (3600 * 24) / 3600);
			var minutes = Math.floor(seconds % 3600 / 60);
			var seconds = Math.floor(seconds % 3600 % 60);

			}

				minutes = "0" + minutes;
			}
			if (seconds < 10) {
				seconds = "0" + seconds;
			}
			return days + ':' + hours + ':' + minutes + ':' + seconds;
