    //<![CDATA[
    (function($) {

	$.formatCurrency = {};

	$.fn.formatCurrency = function() {


		// initialize defaults
		var defaults = {
			name: "formatCurrency",
			symbol: '$',
			code: 'USD',
			colorize: false,
			region: '',
			global: true,
			roundToDecimalPlace: 2, // roundToDecimalPlace: -1; for no rounding; 0 to round to the dollar; 1 for one digit cents; 2 for two digit cents; 3 for three digit cents; ...
			eventOnDecimalsEntered: false,
			digitGroupSymbol: ',',
			decimalSymbol: '.',
			positiveFormat: '%s%n',
			negativeFormat: '(%s%n)',
			groupDigits: true
		};


		return this.each(function() {
			$this = $(this);

			// get number
			var num = '0';
			num = $this[$this.is('input, select, textarea') ? 'val' : 'html']();

			if (num === '' || (num === '-' && defaults.roundToDecimalPlace === -1)) {
				return;
			}

			
			// evalutate number input
			var numParts = String(num).split('.');
			var isPositive = (num == Math.abs(num));
			var hasDecimals = (numParts.length > 1);
			var decimals = (hasDecimals ? numParts[1].toString() : '0');
			var originalDecimals = decimals;
			
			// format number
			num = Math.abs(numParts[0]);
			num = isNaN(num) ? 0 : num;
			if (defaults.roundToDecimalPlace >= 0) {
				decimals = parseFloat('1.' + decimals); // prepend "0."; (IE does NOT round 0.50.toFixed(0) up, but (1+0.50).toFixed(0)-1
				decimals = decimals.toFixed(defaults.roundToDecimalPlace); // round
				if (decimals.substring(0, 1) == '2') {
					num = Number(num) + 1;
				}
				decimals = decimals.substring(2); // remove "0."
			}
			num = String(num);

			

			if ((hasDecimals && defaults.roundToDecimalPlace == -1) || defaults.roundToDecimalPlace > 0) {
				num += defaults.decimalSymbol + decimals;
			}

			// format symbol/negative
			var format = isPositive ? defaults.positiveFormat : defaults.negativeFormat;
			var money = format.replace(/%s/g, defaults.symbol);
			money = money.replace(/%n/g, num);
			
			money = money + " " + defaults.code;

			// setup destination
			var $destination = $([]);
			$destination = $this;
			
			// set destination
			$destination[$destination.is('input, select, textarea') ? 'val' : 'html'](money);

			if (
				hasDecimals && 
				defaults.eventOnDecimalsEntered && 
				originalDecimals.length > defaults.roundToDecimalPlace
			) {
				$destination.trigger('decimalsEntered', originalDecimals);
			}

			// colorize
			if (defaults.colorize) {
				$destination.css('color', isPositive ? 'black' : 'red');
			}
		});
	};

})(jQuery);
