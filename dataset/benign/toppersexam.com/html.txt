			jQuery(".storeQuantityButtons").attr("disabled", true);
			var quantity=jQuery("#store_quantity_"+param.store_id).val();
			var quantityOld=quantity;
			//alert(quantity);
			
			if(param.action=="plus"){
				quantity++;
			}else if(param.action=="minus"){
				
				quantity--;
			}
			if(quantity>0){
				
				jQuery("#store_quantity_"+param.store_id).val(quantity);
				jQuery(".storeQuantityButtons").attr("disabled", false);	
				if(typeof param.funType  != "undifined" && param.funType=="setStore"){
					jQuery(".storeQuantityButtons").attr("disabled", true);
					var grandTotal=parseFloat(jQuery("#sGrandTotal").html());
					var price=param.price;
					var totalAmountOld=quantityOld*price;
					var totalAmountNew=quantity*price;
					grandTotal=grandTotal-totalAmountOld+totalAmountNew;
					var parameter=[];
					parameter.push({'store_id':param.store_id, 'quantity':quantity, 'proccess':'set'});
					jQuery.ajax({url: "/store_set_unset.php",type: "post",dataType: "json",data: "parameter="+JSON.stringify(parameter),
						success: function(data){
							jQuery("#sTotalAmount_"+param.store_id).html(totalAmountNew);
							jQuery("#sGrandTotal").html(grandTotal);
							jQuery(".storeQuantityButtons").attr("disabled", false);
						}
					});
					
				}
			}else{
				jQuery(".storeQuantityButtons").attr("disabled", false);
			}
			
	}
