//Fonction pour traiter le Xt_Param récuperer sur les pages 
function processXtParam(param){
  var p
  if (param.indexOf('p=',0)!=1){
    p = param.substring(param.indexOf('p=',0)+2,param.length)
	if (p.indexOf('&',0)!=1){
		var tp = p.split('&');
		p = tp[0];
		
	}
    if (p.indexOf('::',0)!=1){
		var tp = p.split('::');
		p = tp[1];
		
	}
	}
  return p;
}
