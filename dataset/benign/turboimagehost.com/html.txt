function mt_rand (min, max) {
    var argc = arguments.length;
    if (argc === 0) {        min = 0;
        max = 2147483647;
    } else if (argc === 1) {
        throw new Error('Warning: mt_rand() expects exactly 2 parameters, 1 given');

var randomstring = '';
function randomString() {
	var chars = "0123456789abcdefghiklmnopqrstuvwxyz";
	var string_length = 20;
	for (var i=0; i<string_length; i++) {
		var rnum = Math.floor(Math.random() * chars.length);
		randomstring += chars.substring(rnum,rnum+1);
	}
	return randomstring;
}

function getCookie(cname) {
    var name = cname + "=";
    var ca = d.cookie.split(';');
    for(var i = 0; i <ca.length; i++) {
        var c = ca[i];
        while (c.charAt(0)==' ') {
            c = c.substring(1);
        }
        if (c.indexOf(name) == 0) {
            return c.substring(name.length,c.length);
        }
    }
    return "";
}

function checkext1(archive) {
	var filename = archive.value;
	if (filename!="") {
		var dot = filename.lastIndexOf("."); 
		var fileType = filename.substr(dot,filename.length).toLowerCase(); 
		var slash = filename.lastIndexOf("\\")+1;
		filename = filename.substr(slash,filename.length);
		
		if (fileType == '.zip' || fileType == '.rar' || fileType == '.tar' || fileType == '.gz' || fileType == '.tgz') {
			return true;
		} else {
			archive.value = "";
			archive.focus();
			alert('The file "'+filename+'" does not have a valid archive extension.\nValid archive extensions are: zip, rar, tar, tar.gz, gz, tgz.\nPlease select some other file.');
			return false;
		}
	return true;
	}
}
