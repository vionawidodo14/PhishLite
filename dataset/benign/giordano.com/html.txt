        function InitFilter() {
            $divFilterList = $(".div-filter");
            if( $divFilterList.length > 0 )
            {
                _curI = 0;
                _curFilterDiv = $divFilterList[_curI];                               
                GetFilterData();
            }

        function GetFilterData() {
            var src = GetWCFURL();
            var gender = "1";
            var brand = "";
            var tab = "";
            brand = _curFilterDiv.getAttribute("brand");
            gender = _curFilterDiv.getAttribute("gender");
            tab = _curFilterDiv.getAttribute("tabid");

            if (brand == null || brand == undefined) {
                ProcceeNextFilterMenu()
            }
            if (gender == null || gender == undefined) {
                ProcceeNextFilterMenu()
            }
            _curFilterDiv.innerHTML = "loading...";
            _curGender = gender;
            _curBrand = brand;
            _curTab = tab;            

            src = src + "/GetProductColorSizeFilter?langid=" + _menu_Lang + "&brandId=" + brand + "&sysCate=&gender=" + gender + "&customCate=&search=&colorfilter=&sizefilter=";
            $.support.cors = true;
            $.ajax({
                url: src,
                type: "get",
                dataType: "jsonp",
                crossDomain: true,
                success: function (data) {
                    $objData = data.GetProductColorSizeFilterResult;
                    RenderFilterDiv();
                    ProcceeNextFilterMenu();
                },
                error: function (XMLHttpRequest, textStatus, errorThrown) {
                    ProcceeNextFilterMenu();
                }
            });
        }

        function ProcceeNextFilterMenu() {
            _curI++;
            if (_curI < $divFilterList.length) {
                _curFilterDiv = $divFilterList[_curI];
                GetFilterData();
            }
        }

        function IsNum(s)
        {
            if (s!=null && s!="")
            {
                return !isNaN(s);
            }
            return false;
        }


        function RenderFilterDiv() {
            var $colorData = $objData[0].Value;
            var $sizeData = $objData[1].Value;

            /////////////////color filter/////////////////////

            
            var colorHtml = "<div class=\"col-md-12\"><h3>Colors</h3><div><table>";
            var COLMAX = 4;
            var colcur = 1;

            for (var i = 0; i < $colorData.length; i++) {
                var color = $colorData[i];
                var id = color.key;
                var name = color.value;
                if (i == 0) {
                    colorHtml += "<tr>";
                }
                var url = "/c1.aspx?tabid=" + _curTab + "&BND=" + _curBrand + "&BrandID=" + _curBrand + "&GenderID=" + _curGender + "&fcolor=" + id + "&fsize=";
                var td = "<td style=\"width:40px;height:40px;\" align=\"center\">" +
                          "<div style=\"margin:4px 4px 4px 4px\">" +
                          "<a href=\"" + url + "\" >" +
                          "<span style=\"margin-top:10px\"><img src=\"/images/color/" + id + ".jpg\" style=\"border-style:none;margin-bottom:5px\" /><br /><span class=\"colorcell\">" + name + "</span></span>" +
                          "</a></div></td>";
                colorHtml += td;

                //last td
                if (colcur == COLMAX) {
                    colorHtml += "</tr>";
                    colcur = 1;
                }
                else {
                    if ((i + 1) < $colorData.length) {
                        colcur++;
                    }
                    else {
                        var less = COLMAX - colcur;
                        for (var j = 0; j < less ; j++) {
                            colorHtml += "<td style=\"width:40px;height:40px;\" align=\"center\">&nbsp;&nbsp;</td>";
                        }
                        colorHtml += "</tr>";
                    }
                }
            }
            colorHtml += "</table></div></div>";

            /////////////////size filter/////////////////////

            
            var sizeHtml = "<div class=\"col-md-12\"><h3>Sizes</h3><div><table>";;
            COLMAX = 5;
            colcur = 1;
            var preSizeName = "";

            for (var i = 0; i < $sizeData.length; i++) {
                var size = $sizeData[i];
                var id = size.key;
                var name = size.value;
                if (i == 0) {
                    sizeHtml += "<tr>";
                }

                if (i != 0) {
                    var curIsNumeric = IsNum(name);
                    var preIsNumeric = IsNum(preSizeName);
                    if (curIsNumeric != preIsNumeric) {
                        var less = COLMAX - colcur;
                        if (less < (COLMAX-1)) {
                            for (var j = 0; j < less ; j++) {
                                sizeHtml += "<td>&nbsp;</td>";
                            }
                            sizeHtml += "</tr>";
                            colcur = 1;
                            sizeHtml += "<tr>";
                        }
                    }
                }

                var url = "/c1.aspx?tabid=" + _curTab + "&BND=" + _curBrand + "&BrandID=" + _curBrand + "&GenderID=" + _curGender + "&fcolor=&fsize=" + id;
                var td = "<td>" +                          
                          "<a href=\"" + url + "\" >" +
                          "<span class=\"sizecell\">"+name+"</span>" +
                          "</a></td>";
                sizeHtml += td;

                //last td
                if (colcur == COLMAX) {
                    sizeHtml += "</tr>";
                    colcur = 1;
                }
                else {
                    if ((i + 1) < $sizeData.length) {
                        colcur++;
                    }
                    else {
                        var less = COLMAX - colcur;
                        for (var j = 0; j < less ; j++) {
                            sizeHtml += "<td>&nbsp;</td>";
                        }
                        sizeHtml += "</tr>";
                    }
                }
                

                preSizeName = name;
            }


            sizeHtml += "</table></div></div>";

            _curFilterDiv.innerHTML = colorHtml + sizeHtml;
        }

        function pagelogoShow() {
            if ($(document).scrollTop() > 300) {
                $('.affix-pagelogo').css({"left":0})
            } else {
                $('.affix-pagelogo').css({"left":"-160px"})
            }
        }

        function headerAnimation() {
            if ('N' == 'N') {
                checkMenuSite();
                $(document).on('scroll',function () {
                    if ($(document).scrollTop() != 0) {
                        $('#header .affix').addClass('affix-on')
                        $('#header').addClass('affix-header-on')
                        pagelogoShow()
                    } else {
                        $('#header .affix').removeClass('affix-on')
                        $('#header').removeClass('affix-header-on')
                    }
                })
            }else{
                $('#header').css({ "height": "57px" })
                $('.affix-pagelogo').css({"left":0})                
            }
        }

        function checkMenuSite(){
            if ($(document).scrollTop() != 0) {
                $('#header .affix').addClass('affix-on')
                $('#header').addClass('affix-header-on')
                pagelogoShow()
            } else {
                $('#header .affix').removeClass('affix-on')
                $('#header').removeClass('affix-header-on')
            }
        }

        function showCookieMsg() {
            var divshow = document.getElementById('divshowaaa');    
            if (divshow != null) {
                divshow.style.display = "block";
                $('.divFrame').css({ "margin-top": "120px" })
                if ('N' == 'N') {
                    $('#header').addClass('affix-cookie')
                    $('.showBndLogo').addClass('affix-on-hide')
                    $(document).on("scroll", function () {
                        if ($(document).scrollTop() != 0) {
                            $('#header .affix').addClass('affix-cookie-on')
                            $('#header').addClass('affix-header-cookie-on').removeClass('affix-cookie')
                            pagelogoShow()
                        } else {
                            $('#header .affix').removeClass('affix-cookie-on')
                            $('#header').removeClass('affix-header-cookie-on').addClass('affix-cookie')
                        }
                    })
                } else {
                     $('.affix-pagelogo').css({"left":0})
                }
            } 
        }

        function addshowaaa() {
            var key = market + "_CookieAccept";
            //$.cookie(key, "Y");
            setCookie(key, "Y", 90);

            var divshow = document.getElementById('divshowaaa');
            if (divshow != null) {

                $('.divFrame').css({ "margin-top": "60px" })

                    if ($(document).scrollTop() != 0) {
                        $('#header').addClass('affix-header-on')
                        $('#header .affix').removeClass('affix-cookie-on').addClass('affix-on')
                        pagelogoShow()
                    } else {
                        $('#header .affix').removeClass('affix-cookie-on')
                    }
                    $(document).off('scroll')
                }
                headerAnimation()
            }
