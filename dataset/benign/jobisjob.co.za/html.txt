        var script = document.createElement('script');
        script.src = url;
        var head = document.getElementsByTagName('head')[0];

		function fit_position_leaflet(bounds, point){
		    if(bounds == undefined || bounds == null){
		        var pad = 0.049;//10km
		        var sw = new L.LatLng(point.lat - pad, point.lng - pad);
		        var ne = new L.LatLng(point.lat + pad, point.lng + pad);
		        bounds = new L.LatLngBounds(sw, ne);
		    }else if(!bounds.contains(point)){	    
		    	bounds.extend(point);
		    }
		    return bounds;
		}
