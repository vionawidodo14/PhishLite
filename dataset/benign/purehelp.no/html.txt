	var recaptcha_used=false;
	head.ready("jQuery",function() {

  			$.ajax({
				type: "POST",
				url: $(this).attr('href'),
				dataType: 'json',
				success: function(data) {
					if(data.logout_others.length > 0){
						$.each(data.logout_others, function(idx, val){
							$.ajax({
								url: val,
								xhrFields: {
									withCredentials: true
								}
							})
						});
					}
				},
				complete: function(){
					window.location.reload();
				}
			});

			e.preventDefault();
			e.stopPropagation();
		});

		$('#ph_login_form').on('submit', function(e) {
		  	e.preventDefault();
			$.ajax({
				type: "POST",
				url: $(this).attr('action'),
				data: $(this).serialize(),
				dataType: 'json',
			}).done(function(data){
				switch(data.login_state){
					case 1:
					case 9:
						$loginBtn= $('.phLoginBtn');
						$loginBtn.addClass('green lighten-4');
						$loginBtn.find('a').attr('data-target', '#ph_goto_panel');
						$loginBtn.find('a').find('i').removeClass('fa-lock');
						$loginBtn.find('a').find('i').addClass('fa-unlock');
						if(data.has_minside){
							$('a#btn_minside').removeClass('d-none');
						}
						if(data.cmsso_connect.length > 0){
							$('footer').append(data.cmsso_connect);
						}

						$('#ph_login_modal').modal('hide');
						$('#ph_goto_panel').modal('show');
						break;
					case 2:
						window.location = '/m/ph_auth/register_org';
						break;
					case 5:
						// Need confirmation.
						window.location = data.my_path;
						break;
					case 0:
					default:
						if(data.msg.length > 0) {
							$('#ph-login-msg').removeClass('d-none').html('<div class="alert alert-danger" role="alert">'+data.msg+'</div>');
						} else if(data.my_path.length>0){
							$(this).off('submit');
							$(this).submit();
						}

						if (data.show_captcha) {
							if (recaptcha_used){
								grecaptcha.reset();
							}

							head.load({recaptcha: "https://www.google.com/recaptcha/api.js?hl=no"});
							recaptcha_used=true;
						}
				}
				return;
			});
		});
