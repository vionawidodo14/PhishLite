    function getTimeRemaining(endtime, time) {
        let Tt = time != null ? time : T;
        var t = endtime * 1000 + Tt - Date.parse(new Date());
        var seconds = Math.floor((t / 1000) % 60);
        var minutes = Math.floor((t / 1000 / 60) % 60);
        var hours = Math.floor((t / (1000 * 60 * 60)) % 24);
        var days = Math.floor(t / (1000 * 60 * 60 * 24));
        return {
            'total': t,
            'days': days,
            'hours': hours,
            'minutes': minutes,
            'seconds': seconds
        };
    }

    function initializeClock(clock, endtime, time) {

        var timeinterval = setInterval(function () {
            var t = getTimeRemaining(endtime, time);

            clock.html(('0' + t.minutes).slice(-2) + ':' + ('0' + t.seconds).slice(-2));
            if (t.total <= 0) {
                clock.html('');
                clock.parent().parent().find("td").eq(5).html("Завершено");
                clearInterval(timeinterval);
            }
        }, 1000);
    }
