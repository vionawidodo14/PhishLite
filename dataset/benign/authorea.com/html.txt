  $(function(){
    var alertsOnPage = $('.alert-success-v2, .alert-error-v2 ')
    
    $('.alert-success-v2, .alert-error-v2 ').click(function () {
      fadeOutAlert($(this));
      window.setTimeout(function () {animateNextAlert()}, 500)  
    })
    
    function animateNextAlert() {
      var nextAlert = [].pop.call(alertsOnPage)
      if (nextAlert){
        var $nextAlert = $(nextAlert)
        $nextAlert.animate({'bottom':0, 'opacity': 1}, 800);
        window.setTimeout(function () {
          if ($nextAlert.is(":visible")){
            fadeOutAlert($nextAlert );
            window.setTimeout(function () {animateNextAlert()}, 500)  
          }
        }, 7000)
      }
    }
    
    function fadeOutAlert($node) {
      $node.fadeOut()
    }
    
    window.setTimeout(function () {
      animateNextAlert()
    }, 500)
      
  })
</script>
