    function ci_startCall(method)
    {
        jQuery('#' + method + ' .button').addClass('disabled');
        jQuery('#' + method + '-data-all').html('<p class="message">Please wait...</p>');
    }

    function ci_endCall(method)
    {
        jQuery('#' + method + ' .button').removeClass('disabled');
    }

    function processRequest(url, method, data) {

        ci_startCall(method);

        var request = jQuery.ajax({
            url:      url,
            type:     'POST',
            dataType: 'json',
            data:     jQuery.param(data)
        });

        request.done(function(response) {
            jQuery('#' + method + '-data-all').html(response['html']);
            jQuery.fancybox.update();
        });
        
        request.fail(function(response) {
            jQuery('#' + method + '-data-all').html('<p class="error">Error. Please try again.</p>');
        });

        request.always(function() {
            ci_endCall(method);
        });
    }
