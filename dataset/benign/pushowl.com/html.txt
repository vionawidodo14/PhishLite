    localStorage["authToken"] = authToken;
    localStorage["userEmail"] = result["email"];

    if (!result.email.includes("pushowl.com")) {
        window.Intercom('update', {email: result.email, name: result.name, user_id: result.id});
        localStorage["debug"] = false;
    }
    else {
        localStorage["debug"] = true;

        var completedSteps = data["result"]["signup_flow"]["completed_steps"];

        // Send the person to dashboard
        console.log("Send to Dashboard");
        window.location = constants.dashboard + "/login?authToken=" + authToken;
    }
}
