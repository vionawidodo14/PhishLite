	$(".sum_count").click(function() {
		var discount = 0;
		var tid = $('#order_no').val();
		var id_off = "id_off="+tid;
		$.ajax({
			type: "POST",
			async:false,
			data: id_off,
			url: '/require/ajax/price.php',
			success: function(result) { 
				discount  = result;
			}
		});	
		$(".discount_td_tr").html(discount);
		var price 	= ajax_price;
		
		
		var count 	= $("#count").val();
		var address = $("#address").val();
		
		var min 	= $("#count").attr("min");
		var max 	= $("#count").attr("max");
		min 	= parseInt(min);
		max 	= parseInt(max);		
		if(price==0)
		{
			$('#show_price').html('لطفا سرویس خود را انتخاب نمایید');
			$('#show_price_total').html('لطفا سرویس خود را انتخاب نمایی');		
		}
		else
		{
			if(count=='')
			{
				$('#show_price').html('تعداد را وارد نمایید');
				$('#show_price_total').html('تعداد را وارد نمایید');			
			}
			else
			{
				if(address=='')
				{
					$('#show_price').html('آیدی پیج را وارد نمایید');
					$('#show_price_total').html('آیدی پیج را وارد نمایید');	
				}
				else
				{
					if(count < min || count > max)
					{
						$('#show_price').html('تعداد باید بین '+min+' و '+max+' باشد');
						$('#show_price_total').html('تعداد باید بین '+min+' و '+max+' باشد');
					}
					else
					{
						var tprice = price*count
						var total_price = (tprice - (tprice * discount)/100)
						$('#show_price').html(tprice+' تومان');
						$('#show_price_total').html(total_price+' تومان');
						
					}	
				}
			}
