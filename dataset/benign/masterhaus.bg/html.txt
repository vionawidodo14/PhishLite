			}).appendTo(li).prepend($('<a href="javascript:;" class="plus" />').html('+'));
		}
		
		this.$owlItems.eq(this.currentItem).find('img').on('load', function() {
			$(this).off('load', arguments.callee);
			owl_instance.options._afterMoveCustom.call(owl_instance);
			owl_instance.$owlItems.find('.points li').hide();
		});
		
		setTimeout(function() {
			owl_instance.options._afterMoveCustom.call(owl_instance);
		}, 300);
	},
	
	startDragging: function() {
		this.$owlItems.find('.points li').hide();
		this.$elem.next().find('li:eq('+this.currentItem+')').hide().find('.opened').removeClass('opened').children('div').hide();
	},
	
	beforeMove: function() {
		this.options.startDragging.call(this);
	},
	
	beforeUpdate: function() {
		this.options.startDragging.call(this);
	},
	
	afterUpdate: function() {
		this.options._afterMoveCustom.call(this);
	},
	
	_afterMoveCustom: function() {
		var owl_instance = this;
		
		this.$owlItems.eq(this.currentItem).find('.points li').show().each(function(idx) {
			var li = this;
			owl_instance.$elem.next().find('li:eq('+owl_instance.currentItem+') > div:eq('+idx+')').css({
				top: $(this).closest('div').position().top + Math.floor(this.offsetTop) + 'px'
			});
		}).hide();
		
		this.$owlItems.find('.points li').hide();
		this.$elem.next().find('li:eq('+this.currentItem+')').fadeIn('fast');
	}
};
</script>
