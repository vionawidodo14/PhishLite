    function getCoordinates($elem, callback) {
      var $icon = $elem.find('i.icon');
      $icon
        .removeClass('icon-map-marker-o')
        .addClass('icon-spinner-o icon-spin');

      var options = {
        enableHighAccuracy: true
      }

      navigator.geolocation.getCurrentPosition(function(position) {
        var coords = {
          lat: position.coords.latitude,
          lng: position.coords.longitude
        };

        callback(coords);

        $icon
          .removeClass('icon-spinner-o icon-spin')
          .addClass('icon-map-marker-o');

      }, null, options);
    }
  }

<script>
  $(function() {
    var modalInstance;
    var stepsManager;
    var $mobileModal = $('#mobile-modal');
    var $categorySearcherDesktop = $('#category-name');
    var $categorySearcherMobile = $('#interactive-category-name');
    var $locationSearcherMobile = $('#interactive-location-name');

    function getSteps() {
      return $('[data-step]')
        .map(function() {
          return $(this).attr('data-step');
        })
        .get();
    }

    function postParentEscapedEvent() {
      window.parent.postMessage("businessSearcherModalEscaped", "*");
    }

    function restoreMobileContent() {
      $('#business-mobile-searcher').appendTo($mobileModal)
    }

    function clearMobileInputsSearchers() {
      $categorySearcherMobile
        .categorySearcher('val', '')
        .val('');

      $locationSearcherMobile
        .typeahead('val', '')
    }

    function postParentOpenedEvent() {
      window.parent.postMessage("businessSearcherModalOpened", "*");
    }

    function displayModal($element) {
      var $modalContent = $('<div id="bootbox-mobile-content"></div>');
      modalInstance = bootbox.dialog({
        message: $mobileModal.children().appendTo($modalContent),
        className: 'mobile-business-search-modal',
        onEscape: function() {
          postParentEscapedEvent();
          restoreMobileContent();
          clearMobileInputsSearchers();

          $mobileModal.trigger('businessSearcherModalClosed');
        },
      });

      modalInstance.init(function() {
        var stepType = $element.data('step-type');

        if (!stepsManager) {
          var steps = getSteps();
          // Come from web components
          stepsManager = Utilities.createStepManager(steps);
        }

        clearMobileInputsSearchers();
        postParentOpenedEvent();
        stepsManager.hideActiveStep();
        if (!$categorySearcherDesktop.val()) {
          stepsManager.showFirstStep();
        } else {
          stepsManager.showStep(stepType, true);
        }
      });
    }

    function successUserSelection() {
      if (stepsManager && !stepsManager.isCurrLatestStep()) {
        stepsManager.goToNextStep();
      }
      $mobileModal.trigger( "mobileSearcher_DoSearch");
    }

    $('[data-interaction="search-modal"]').on('click', function(ev) {
        if (isMobile()) {
          displayModal($(this));
          ev.stopPropagation();

          window.onpopstate = function (e) {
            if (!e.state) {
              postParentEscapedEvent();
              restoreMobileContent();
              clearMobileInputsSearchers();

              $mobileModal.trigger('businessSearcherModalClosed');
              modalInstance.modal('hide');
            } else {
              stepsManager.goToStep(e.state.step);
            }
          };

          if (window.history.state) {
            window.history.replaceState({
              step: stepsManager.getCurrentStepName()
            }, '', '');
          } else {
            window.history.pushState({
              step: stepsManager.getCurrentStepName()
            }, '', '');
          }
        }
    });

    $categorySearcherMobile.on('typeahead:selected typeahead:autocompleted', function(evt, category) {
      $(this).blur();
      $('[name="category-name"]').val($('[name="interactive-category-name"]').val());
      stepsManager.goToNextStep();

      window.history.pushState({
        step: 'location-step'
      }, '', '');
    });

    $locationSearcherMobile.on('typeahead:selected typeahead:autocompleted', function(evt, location) {
      setTimeout(successUserSelection, 0);

      window.history.pushState({
        step: 'category-step'
      }, '', '');
    });

    $locationSearcherMobile.on('change', function(evt, location) {
      successUserSelection();
    });

    $('#loading-step').append(window.ContentPlaceholder.getPlaceholder('geometric-placeholder'));
  });
</script>

    }

    function suggestionTemplateDesktop(value) {
      var template = $('<span>', {text: value.name});
      if (value.parent_name) {
        var gray_text = __('en') + ' ' + value.parent_name;
        template.append(' ', $('<small>', {text: gray_text}).addClass('text--gray'))
      }

      return template
    }

    function initCategorySearcher($input, suggestionTemplate) {
      var selectedCategoryId = null;
      var selectedCategoryName = null;

      $input.categorySearcher({
        alwaysSelectOne: true,
        id_key: 'id',
        suggestionTemplate: suggestionTemplate,
        tree_level: [1,2],
        extraOption: {
          id: -1,
          name: "Vedi tutte le categorie",
          redirectUrl: "https:\/\/aziende.habitissimo.it\/elenco_servizi",
          target: "_self"        },
      });

      if (selectedCategoryId) {
        $input.categorySearcher('val', selectedCategoryName);
        $input.categorySearcher('keyVal', selectedCategoryId);
      }
    }

    function initLocationSearcher($input) {
      $input.LocationSearcher();
    }

    function getValuesFromInputs($input) {
      return $input
        .map(function() {
          return $(this).val();
        })
        .get()
        .filter(function(value) {
          return value;
        })
    }

    $mobileModal.on( "mobileSearcher_DoSearch", function() {
      getCorrectUrl();
    });

    function getCorrectUrl() {
      var params = {};
      var categoryValues = getValuesFromInputs($('[data-typeahead-id="category-name"]'));
      var locationValues = getValuesFromInputs($('[data-typeahead-id="location-name"]'));
      var $searcher = isMobile() ? $locationSearcherMobile : $locationSearcherDesktop;
      var locationValuesInputVal = $searcher.val().trim();

      
      if (categoryValues.length) {
        params['unified-category-id'] = categoryValues[0];
      }

      if (locationValues.length) {
        params['location-id'] = locationValues[0];
      } else if(locationValuesInputVal) {
        params['location-value'] = locationValuesInputVal;
      }

      if (true) {
        params['from_home_searcher'] = true;
      }

      $.ajax({
        url: "\/do_ajax\/autocomplete_url",
        type: 'GET',
        datatype: 'json',
        data: params,
        success: function(res) {
          trackCustomDimensions();

          if (res.url !== "") {
                      window.top.location.href = res.url;
                    }
        }
      });
    }

     function trackCustomDimensions() {
      if (typeof dataLayer != "undefined" && filters.category && filters.location) {
        dataLayer.push({
          'searched_category': filters.category.normalized_name,
          'searched_location': filters.location.normalized_name === undefined ?
            filters.location :
            filters.location.normalized_name,
        });
      }
    }

    initCategorySearcher($categorySearcherDesktop, suggestionTemplateDesktop);
    initLocationSearcher($locationSearcherDesktop);

    $mobileModal.on('businessSearcherModalClosed', function() {
      initCategorySearcher($categorySearcherDesktop, suggestionTemplateDesktop);
      initLocationSearcher($locationSearcherDesktop);
    });

    $categorySearcherDesktop.on('typeahead:selected typeahead:autocompleted', function(evt, category) {
      filters.category = category;
      updateTrackEvent();

      if (!isNotEmpty($locationSearcherDesktop)) {
        $locationSearcherDesktop.focus();
      }
    });

    $categorySearcherDesktop.on('click', function (evt) {
      if (isMobile()) {
        $categorySearcherDesktop.typeahead('destroy');
      }
    });

    $locationSearcherDesktop.on('click', function (evt) {
      if (isMobile()) {
        $locationSearcherDesktop.typeahead('destroy');
      }
    });

    $locationSearcherDesktop.on('typeahead:selected typeahead:autocompleted', function(evt, location) {
      filters.location = location;
      updateTrackEvent();

      if (!isNotEmpty($categorySearcherDesktop)) {
        $categorySearcherDesktop.focus();
      }
    });

    $categorySearcherMobile.on('typeahead:selected typeahead:autocompleted', function(evt, category) {
      filters.category = category;
      updateTrackEvent();
    });

    $locationSearcherMobile.on('typeahead:selected typeahead:autocompleted', function(evt, location) {
      filters.location = location;
      updateTrackEvent();
    });

    $locationSearcherDesktop.on('blur', function(evt) {
      trackBlurLocationInput(evt);
    });

    $locationSearcherMobile.on('blur', function(evt) {
      trackBlurLocationInput(evt);
    });

    $locationSearcherDesktop.on('keyup', function(evt) {
      var enter_code = 13;
      if (evt.keyCode === enter_code) {
        $('[data-find]').focus().click();
      }
    });

    function trackBlurLocationInput(evt) {
      var locationValues = getValuesFromInputs($('[data-typeahead-id="location-name"]'));
      if (locationValues.length) return;

      filters.location = evt.target.value;
      updateTrackEvent();
    }

    function updateTrackEvent() {
      var trackEvent = 'Directory.MainCTASearch.';
      if (filters.category) {
        trackEvent += filters.category.normalized_name;
      }

      if (filters.location && filters.location.normalized_name) {
        trackEvent += '_' + filters.location.normalized_name;
      }

      if (filters.location && !filters.location.normalized_name) {
        trackEvent += '_' + filters.location;
      }

      var dataFindEvents = $('[data-find]');

      dataFindEvents.each(function(index, element) {
        $(element).attr('data-trackevent', trackEvent);
      });
    }

    function isNotEmpty($input) {
      return $input.val().trim() !== '';
    }

    $('[data-find]')
      .on('click', function(evt) {
        evt.preventDefault();
        if (!isMobile()) {
          $(this).addClass('btn-spinner disabled');

          var isValidCategorySearcher = isNotEmpty($categorySearcherDesktop);
          var isValidLocationSearcher = isNotEmpty($locationSearcherDesktop);
          var isValid = isValidCategorySearcher && isValidLocationSearcher;
          if (isValid) {
            getCorrectUrl();
          } else {
            $(this).removeClass('btn-spinner disabled');

            if (!isValidCategorySearcher) {
              $categorySearcherDesktop.focus();
            } else {
              $locationSearcherDesktop.focus();
            }
          }
        } else {
          getCorrectUrl();
        }
      });
  });
</script>
