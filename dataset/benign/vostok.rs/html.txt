    $(document).ready(function () {
        var url = "/index.php?option=btg_anketa&13758";
        var settings = {
            "url": "/index.php?option=btg_anketa&check="+(Math.round(Math.random()*10000)),
            "method": "GET",
            "headers": {
                "crsf-token": "29203"
            }
        };
        $.ajax(settings).done(function (response) {
            $container = $('.poll-container');
            $container.html(response);
            $container.find("button.btn-primary").click(function () {
                var value = $container.find("input[name='poll']:checked");
                var id = $container.find("input[name='id']");
                if(value.length == 0){
                    alert("Није одговор изабран");
                    return;
                }
                var settings = {
                    "url": "/index.php?option=btg_anketa&check="+(Math.round(Math.random()*10000)),
                    "method": "POST",
                    "headers": {
                        "crsf-token": "29203"
                    },
                    "data": {
                        "value": value.val(),
                        "id": id.val()
                    }
                };
                //
                $.ajax(settings).done(function (response) {
                    if(response.voted){
                        $.get( url+"&rand="+Math.floor(Math.random()*1000000), function( data ) {
                            $container = $('.poll-container');
                            var resources = data;
                            $container.hide("fast", function () {
                                $container.html(resources);
                                $container.show("fast");
                                $container.find("button.btn-success").addClass("disabled");
                            });
                            bindListeners($container);
                        });
                    }
                });
            });
        });

        function bindListeners(variable){
            variable.find("button.btn-success").unbind();
            variable.find("button.btn-success").click(function () {
                var value = variable.find("input[name='poll']:checked");
                var id = variable.find("input[name='id']");
                if(value.length == 0){
                    alert("Није одговор изабран");
                    return;
                }
                var settings = {
                    "url": "http://vostok.rs/index.php?option=btg_anketa&",
                    "method": "POST",
                    "headers": {
                        "crsf-token": "29203"
                    },
                    "data": {
                        "value": value.val(),
                        "id": id.val()
                    }
                };
                //
                $.ajax(settings).done(function (response) {
                    if(response.voted){
                        $.get( url, function( data ) {
                            variable = $('.poll-container');
                            variable.html(data);
                            bindListeners(variable);
                        });
                    }
                });
            });
        }
    });
</script>
