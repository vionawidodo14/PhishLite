                function SearchInSite() {
                    var keyword = document.getElementById("search-keyword").value.toString();
                    if (keyword.trim().length > 0) {
                        window.location = '/search/showresult?input=' + keyword;
                    }

                function SearchRealTime() {
                    var keyword = document.getElementById("search-keyword").value.toString();
                    window.scrollTo(0, 0);

                    var searchResult = $('#searchResults');

                    if (keyword.length === 0) {
                        $('#searchResults').empty();
                        $(searchResult).css({ 'display': 'none' });
                        return;
                    }

                    if (keyword.length >= 2) {
                        $.ajax({
                            //url: '/Home/Search',
                            url: '/Search/Search',
                            type: 'GET',
                            traditional: true,
                            data: {
                                //keyword: keyword
                                'input': keyword
                            },
                            contentType: 'application/json; charset=utf-8',
                            success: function (data) {
                                //var searchResults = document.getElementById("searchResults");
                                //searchResults.style.display = "block";
                                //searchResults.innerHTML = data;

                                var keyword = $('#search-keyword').val();
                                if (keyword.length === 0) {
                                    $(searchResult).css({ 'display': 'none' });
                                    return;
                                }

                                $(searchResult).css({ 'display': 'block' });

                                $('#searchResults').empty();
                                for (var i = 0; i < data.data.length; i++) {

                                    if (data.data[i]['Type'] == 1) {
                                        var eachRow =
                                            '<a class="d-block p-2 font-size-medium" href="/' + data.data[i]['Link'] + '">محصول - ' + data.data[i]['Title'] + '</a>';

                                        $('#searchResults').append(eachRow);

                                        $('#searchResults').append(eachRow);

                                //searchResult.style.display = "none";

                                $(searchResult).css({ 'display': 'none' });
                            }

                        });
                    }

                        $('#searchResults').empty();
                        $(searchResult).css({ 'display': 'none' });
                    }
                }

        function showSlides(n) {
            try {
                var i;
                var slides = document.getElementsByClassName("mySlides");
                var dots = document.getElementsByClassName("dot");

                if (n > slides.length) { slideIndex = 1 }
                if (n < 1) { slideIndex = slides.length }
                for (i = 0; i < slides.length; i++) {
                    slides[i].style.display = "none";
                    dots[i].classList.remove("active");
                }
                for (i = 0; i < slides.length; i++) {
                    slides[i].className = slides[i].className.replace(" active", "");
                }
                slides[slideIndex - 1].style.display = "block";
                dots[slideIndex - 1].classList.add("active");
            } catch{}
        }
