<script>
	jQuery(document).ready(function(){
		jQuery("#Header").children("form").attr("onsubmit","return searchSubmitCludo()");

		//modify the search links
		jQuery('a').each(function(){
			var $this = $(this);
			linkHREF = $this.attr('href');
			if(typeof linkHREF !== 'undefined'){
				if(linkHREF.indexOf("gsa.raleighnc.gov") >= 0){
					//do GSA replacements

					//replace GSA with url
					linkHREF = linkHREF.replace("http://gsa.raleighnc.gov/search", "/home/content/ITechWebServices/Articles/search.html");
					linkHREF = linkHREF.replace("https://gsa.raleighnc.gov/search", "/home/content/ITechWebServices/Articles/search.html");
					linkHREF = linkHREF.replace("?q=","?query=");
					linkHREF = linkHREF.replace("&as_q=","&query=");
					linkHREF = linkHREF.replace("inurl:","Url:");
					linkHREF = linkHREF.replace("+inurl:","Url:");
					linkHREF = linkHREF.replace("Url:","Url:\"");
					linkHREF = linkHREF.replace("/&num","/*\"&num");
					linkHREF = linkHREF.split("&num");
					linkHREF = linkHREF[0];

					//add tracking for archive documents at the end of the link
					linkHREF = linkHREF + "&CORArchive=true";

					$this.attr('href',linkHREF);
				}
			}
				
		})
	});
	function searchSubmitCludo(){
		var query = document.getElementById("SearchInput").value;
		window.location.href = "/home/content/ITechWebServices/Articles/search.html?query="+ query;
		return false;
	}

				document.querySelector('.depts').classList.remove('fixed');
			}
  	};
  });
