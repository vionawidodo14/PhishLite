    jQuery(".options label").html(function () { return $(this).html().replace(".00", ""); });  

    (function() {
        if (typeof optionChange === "function") { 
            var existing_optionChange=optionChange;
            optionChange= function(oid, res) { 
                existing_optionChange(oid, res); 
                jQuery(".selling_price").html(function () { return $(this).html().replace(".00", ""); });
                jQuery(".list_price").html(function () { return $(this).html().replace(".00", ""); });

                if (jQuery('#price_list').html() == "<font class=\"list_price\"><s>£0</s></font>") {
                    jQuery('.product_list_price').css('display', 'none');
                    jQuery('.product_price_discount').css('display', 'none');
                } else {
                    jQuery('.product_list_price').css('display', 'block');
                    jQuery('.product_price_discount').css('display', 'inline');
                }

            }
        }
    })();

    /* Sticky Menu */

    var navtop = 0;
    var navpos = 100;
    
    jQuery(window).on('scroll', function()
    {
        if (window.pageYOffset >= navpos)
        {
            if (!navtop)
            {
                jQuery('nav').addClass('sticky');
                if (jQuery(window).width() >= 1024) {
                    jQuery('#values').css('margin-top', '54px');
                }
                navtop = 1;
            }
        }
        else
        {
            if (navtop)
            {
                jQuery('nav').removeClass('sticky');
                if (jQuery(window).width() >= 1024) {
                    jQuery('#values').css('margin-top', '0px');
                }
                navtop = 0;
            }
        }
    }).trigger('scroll');

    /* Currency */

    var currency = jQuery("select[name='currency'] option:selected").val();

    if (currency == 1) { jQuery("#currency_1").addClass("currency"); jQuery("#currency_flag").html("<img src=\"/admin/images/flags/gb.gif\" alt=\"Sterling (GBP)\">"); }
    else if (currency == 2) { jQuery("#currency_2").addClass("currency"); jQuery("#currency_flag").html("<img src=\"/admin/images/flags/eu.gif\" alt=\"Euro (EUR)\">");  }

    });

    var maxHeight = Math.max.apply(null, jQuery(".product_title").map(function ()
    {
        return jQuery(this).height();
    }).get());

    if (maxHeight) {

    }
