        var Player = null;
        var Preview = null;
        var video_photo = null;
        var video_photo_counter = 0;
        var video_photo_interval = null;
        var playlist_ids = [];
        var flash_type = '';
        var flash_message = '';

        $(document).on('ready', function() {
            setFooter();
            setImages();

            setTimeout(function() { setFooter(); }, 2000);

            if (flash_type !== '') {
                showMessage(flash_message, flash_type, 5000);
            }

            $(document).on('click', '.video-photo .icon:not(.play)', function() {
                var icon = $(this);

                playlist(icon.closest('.video-photo').data('id'), icon);
            });

            $(document).on('click', '.video-photo', function(event) {
                if (
                    $(event.target).hasClass('add')
                    || $(event.target).hasClass('remove')
                ) {
                    return false;
                }

                window.location = '/videos/view/' + $(this).closest('.video-photo').data('id');
            });

            $(document).on('mouseover', '.video-photo', function() {
                if ($(this).find('.image img').hasClass('blur')) {
                    return false;
                }

                video_photo = $(this);
                video_photo_interval = setInterval(function() {
                    if (video_photo_counter === 5) {
                        video_photo_counter = 0;
                        clearInterval(video_photo_interval);
                        return false;
                    }

                    video_photo.find('.photos img').css('opacity', 0);
                    video_photo.find('.photos img').eq(video_photo_counter).css('opacity', 1);

                    video_photo_counter++;
                }, 200);
            });

            $(document).on('mouseout', '.video-photo', function() {
                video_photo_counter = 0;
                video_photo.find('.photos img').css('opacity', 0);
                clearInterval(video_photo_interval);
            });
        });

        function playlist(id, icon) {
            var type = icon.hasClass('add') && !icon.hasClass('active') ? 'add' : 'remove';

            $.get('/playlists/' + type + '/' + id, function() {
                id = id.toString();

                if (type === 'add') {
                    icon.removeClass('add').removeClass('active').addClass('remove');

                    playlist_ids.push(id);

                    showMessage('Video bolo pridané do playlistu');
                } else {
                    icon.removeClass('remove').removeClass('active').addClass('add');

                    $.each(playlist_ids, function(key, value) {
                        if (value == id) {
                            delete playlist_ids[key];
                        }
                    });

                    showMessage('Video bolo odstranené z playlistu');
                }
            });
        }
