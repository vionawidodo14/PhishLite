            var emailRegSubmitOverlay = document.getElementById('hg_home_email_overlay');

            function registerEmail(evt, form) {
                evt.preventDefault();
                //disable submit button
                form.submit.disabled = true;

                //Figure out which form we are on to get the right messaging div
                var f = (form.id === "email_signup_overlay_form") ? "overlaymsg" : "tileerrmsg";

                // set data layer for tracking 
                if (f === 'email_signup_overlay_form') {
                    window.tjxData.formInfo.emailSource = 'Email Signup Lightbox';
                }
                else if (f === 'email_signup_form') {
                    window.tjxData.formInfo.emailSource = 'Email Signup Card';
                }

                //Get loading indicator div for called form
                var indicator = (form.id === "email_signup_overlay_form") ? document.getElementById("overlay-loading-indicator") : document.getElementById("tile-loading-indicator");

                //hide error messaging if displayed
                var er = document.getElementById(f);
                er.style.display = 'none';
                if (er.hasChildNodes()) {
                    er.removeChild(er.childNodes[0]);
                }

                //Tile only - clear success message if it is there
                var success = document.getElementById("tilesuccessmsg");
                if (success !== null)
                {
                    success.style.display = 'none';
                    if (success.hasChildNodes()) {
                        success.removeChild(success.childNodes[0]);
                    }
                }            

                //get inputs
                name = form.first_name.value;
                zipcode = form.zip.value;
                email = form.email.value;

                //validate zipcode
                var isValidZip = /(^\d{5}$)/.test(zipcode);

                //validate email
                var isValidEmail = validateEmail(email);

                if (isValidZip && isValidEmail) { //Process the submission
                    //loading indicator ON
                    indicator.style.display = 'block';

                    axios.post('/Handlers/EmailProxy.ashx', {
                        chain: "HOM",
                        firstname: name,
                        email: email,
                        zip: zipcode
                    })
                    .then(function (response) {
                        //loading indicator OFF
                        indicator.style.display = 'none';

                        if (response.data.Status === 0) {
                            if (form.id === "email_signup_overlay_form")
                            { //Overlay success
                                document.getElementById("hg-signup").style.display = "none";
                                document.getElementById("sister-brands").style.display = "block";
                                // set data layer for tracking 
                                window.tjxData.formInfo.tjxBrand = "HomeGoods";
                                _satellite.track('emailSignup');
                            }
                            else
                            { //Tile success 
                                //enable submit button
                                form.submit.disabled = false;

                                var success = document.getElementById("tilesuccessmsg");
                                var content = document.createTextNode("Success.");
                                success.appendChild(content);
                                success.style.display = 'block';
                                //Clear form fields
                                form.first_name.value = "";
                                form.zip.value = "";
                                form.email.value = "";
                            }
                        }
                        else if (response.data.Status === -10)

                            if(attempts <= 2)

                                //enable submit button
                                form.submit.disabled = false;

                                var err = document.getElementById(f);
                                var content = document.createTextNode("Oops! We noticed an error with your email.  Please edit your email address and resubmit.");
                                err.appendChild(content);
                                err.style.display = 'block';
                            }

                                { //Overlay success

                                    document.getElementById("signup-thanks").style.display = "block";
                                }
                                else
                                { //Tile success 
                                    //enable submit button
                                    form.submit.disabled = false;

                                    //Clear form fields
                                    form.first_name.value = "";
                                    form.zip.value = "";

                                }
                            }

                        else

                        }
                    })

                        indicator.style.display = 'none';

            function sisterBrands(evt) {
                // set data layer for tracking
                window.tjxData.formInfo.formId = 'sister_brand_email_overlay';
                window.tjxData.formInfo.emailSource = 'Email Signup Lightbox';

                evt.preventDefault();

                //loading indicator ON
                document.getElementById("sister-loading-indicator").style.display = "block";

                var mar = document.getElementById("sister-marshalls").checked;
                var hsus = document.getElementById("sister-homesense").checked;
                var stp = document.getElementById("sister-sierra").checked;

                var brands = [];

                //Process TJMaxx - push to brands array
                if (tjm) {
                    brands.push(sendBrand("TJM"));
                    // set data layer for tracking

                    _satellite.track('emailSignup');

                //Process Marshalls - push to brands array
                if (mar) {

                }

                    });

            function validateEmail(e) {
                e.trim();
                var emailRegEx = /^[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,4}$/i;

                //Apostrophe email handling
                emailAddress = replaceCharacters("'", "", e);
                if (emailAddress.search(emailRegEx) === -1) {
                    return false;
                } else {
                    return true;
                }
            }

            function replaceCharacters(inChar, replacewithChar, origString) {
                var newString = origString.split(inChar);
                newString = newString.join(replacewithChar);
                return (newString);
            }
