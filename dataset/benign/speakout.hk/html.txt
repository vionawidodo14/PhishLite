    list = $('.infinite-scroll-list-hot');

                var url = url_str.split(" ")[0];
                var _val = url.match(new RegExp("(^|[&?])page=([^&]*)(&|$)"));
                _val = parseInt(_val[2]) + 1;
                url = url.replace(eval('/(page=)([^&]*)/gi'), "page=" + _val);
                load_more_btn.attr('href', url);
                load_more_btn.show();
                load_more.jscroll({
                    debug: false,
                    autoTrigger: false,
                    nextSelector: '.load-more-btn',
                    contentSelector: '.jscroll',
                    loadingHtml: loading_html

          var xMove = event.originalEvent.touches[0].pageX;
          if( Math.floor(xClick - xMove) > 5 ){
              $(".carousel").carousel('next');
          }
          else if( Math.floor(xClick - xMove) < -5 ){
              $(".carousel").carousel('prev');
          }
      });
      $("#headline").on("touchend", function(){
              $(this).off("touchmove");
      });
