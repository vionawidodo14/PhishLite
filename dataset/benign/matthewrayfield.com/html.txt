    var mainElement = document.getElementById('main'),
        items, columnCount, columns = [], column, i;

    function extend(target, source, deep) {
        var key;

        for (key in source) {
            if (deep && typeof source[key] == 'object' && typeof target[key] == 'object') {
                extend(target[key], source[key], deep);
            }
            else {
                target[key] = source[key];
            }
        }

        return target;
    }

    function createElement(tagName, properties, children) {
        var element = document.createElement(tagName),
            key;

        extend(element, properties, true);

        if (children) {
            children.forEach(function (child) {
                element.appendChild(child);
            });
        }

        return element;
    }

    function loop() {
        setTimeout(loop, 1);
    }

    items = document.querySelectorAll('.item');
    main.style.alignItems = 'flex-start';

    function setupColumns() {
        columns.forEach(function (column) {
            column.parentNode.removeChild(column);
        });
        columns = [];

        columnCount = Math.floor(window.innerWidth / 420) || 1;

        for (i = 0; i < columnCount; i ++) {
            column = createElement('div', {className: 'column'});
            var width = 'calc(' + Math.floor(100/columnCount) + '% - ' + ((20*(columnCount-1))/columnCount) + 'px)';
            console.log(width);
            column.style.width = width;
            columns.push(column);
        }

        items.forEach(function (item, index) {
            item.style.display = 'flex';
            item.style.width = 'auto';
            item.parentNode.removeChild(item);
            columns[index % columnCount].appendChild(item);
        });

        columns.forEach(function (column) {
            main.appendChild(column);
        });
    }
    setupColumns();

    window.addEventListener('load', function () {
        //loop();
    });

    window.addEventListener('resize', function () {
        setupColumns();
