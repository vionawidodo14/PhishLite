  bannerPlayersCount = $(".js-rolling-item--movie").length;

  $(".js-banner-main .js-rolling-item").each(function(index){
    var _this = $(this);
    var videoId = _this.data("movie");
    if(videoId != null) {
      var id = "bannerVideo"+index;
      var player = createYtPlayer(id, 1920, 1080, videoId, 0, onBannerPlayerReady);
      bannerPlayers[index] = player;
      player.addEventListener("onStateChange", function(e){
        if(e.data == YT.PlayerState.ENDED) {
          mainBanner.rotate();
        } else if(e.data == YT.PlayerState.PLAYING) {
          $("#"+id).next().css({"background-image":"none"});
        }
      });
    } else {
      bannerPlayers[index] = null;
    }
  });
  resizeVideo();
  
  mainBanner = new rollingBanner(".js-banner-main");
  if(bannerPlayersCount < 1) {
    mainBanner.setPlayers(bannerPlayers);
    mainBanner.play();
  }
  $(".js-banner_next").click(function(){
    mainBanner.stop();
    mainBanner.rotate();
    mainBanner.play();
  });
  
  // character video load
  characterPlayer = createYtPlayer("caracterVideo", 270, 490, "gtA9o1OwJ2w", 1, onCharacterPlayerReady);
