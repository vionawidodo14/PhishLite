        $('.slider_selector').click(function () {
            changeSlide('toSlide', this)
        })
        $('.slider_next img').click(function () {
            changeSlide('next');
        });
        $('.slider_prev img').click(function () {
            changeSlide('prev');
        });
    });
    $(window).load(function () {
        setSliderHeights();
    });
    $(window).resize(function () {
        setSliderHeights();
    });

    function setSliderHeights() {
        $('.slider').each(function () {
            $(this).children('.slider_slide-wrapper').children().css('height', 'auto').setAllToMaxHeight();
        });
    }
    
    function slideshowNext() {
        if (!slideshowPlay) {
            slideshowPlay = true;
            setTimeout(slideshowNext, 5000);
            return;
        }
        
        changeSlide('auto');

        setTimeout(slideshowNext, 5000);
    }
    
    function changeSlide(direction, target) {
        if(direction != 'auto') slideshowPlay = false;

        //Set up changing slides and selector
        slidePrev = $('.slide--holder');
        slideCur = $('.slide--selected');
        slideOnDeck = $('.slide--selected').next('.slide');
        if (slideOnDeck.length == 0) slideOnDeck = $('.slide').first();
        selectorCur = $('.slider_selector--selected');
        selectorOnDeck = $('.slider_selector--selected').next('.slider_selector');
        if (selectorOnDeck.length === 0) selectorOnDeck = $('.slider_selector').first();

        //Only check if direction is prev or a specific slide, the default is next
        if (direction == 'prev') {
            slideOnDeck = $('.slide--selected').prev('.slide');
            if (slideOnDeck.length == 0) slideOnDeck = $('.slide').last();
            selectorOnDeck = $('.slider_selector--selected').prev('.slider_selector');
            if (selectorOnDeck.length === 0) nextSelector = $('.slider_selector').last();
        } else if(direction == 'toSlide' && target != null) {
            slideOnDeck = $($('.slide').get($(target).index()));
            selectorOnDeck = $(target);
        }

        //Change Slides and selector
        slidePrev.removeClass('slide--holder');
        slideCur.removeClass('slide--selected').addClass('slide--holder');
        slideOnDeck.addClass('slide--selected');
        slideCur.fadeOut(1000);
        slideOnDeck.fadeIn(1000);
        selectorCur.removeClass('slider_selector--selected');
        selectorOnDeck.addClass('slider_selector--selected');
    }
