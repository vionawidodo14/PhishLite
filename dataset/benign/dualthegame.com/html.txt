                                    $(document).ready( function () {
                                        var $form = $('#mc-embedded-subscribe-form');
                                        $token = $("#mc-embedded-subscribe-form").find('input[name=csrfmiddlewaretoken]').val();
                                        $("#email-error").hide();

                                        if ( $form.length > 0 ) {
                                            $('form input[type="submit"]').bind('click', function ( event ) {
                                                if ( event ) event.preventDefault();
                                                if (validate($('#mce-EMAIL').val()))
                                                    register($form);
                                            });
                                        }

                                        $('#modal-newsletter').on('hidden.bs.modal', function () {
                                            $('#newsletter-description').removeClass('hidden');
                                            $('#mc_embed_signup').removeClass('hidden');
                                            $('#success-message').addClass('hidden');
                                            $('#error-message').addClass('hidden');
                                        })
                                    });

                                    function validateEmail(email) {
                                        var re = /^(([^<>()[\]\\.,;:\s@\"]+(\.[^<>()[\]\\.,;:\s@\"]+)*)|(\".+\"))@((\[[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\])|(([a-zA-Z\-0-9]+\.)+[a-zA-Z]{2,}))$/;
                                        return re.test(email);
                                    }

                                    function validate(email) {
                                        if (!validateEmail(email)) {
                                            $("#email-error").fadeIn();
                                            return false;
                                        }
                                        return true;
                                    }

                                    function register($form) {
                                        $.ajax({
                                            beforeSend: function(xhr, settings) {
                                                xhr.setRequestHeader("X-CSRFToken", csrftoken);
                                            },
                                            type: 'POST',
                                            url: 'subscribe/',
                                            data: $form.serialize(),
                                            csrfmiddlewaretoken: $token,
                                            error : function(err) {
                                                $('#newsletter-description').addClass('hidden');
                                                $('#mc_embed_signup').addClass('hidden');
                                                $('#error-message').removeClass('hidden');
                                             },
                                            success : function(data) {
                                                if (data.status != "success") {
                                                    if (data.message)
                                                        $('#error-message .description').html(data.message);
                                                    $('#newsletter-description').addClass('hidden');
                                                    $('#mc_embed_signup').addClass('hidden');
                                                    $('#error-message').removeClass('hidden');
                                                } else {
                                                    if (data.message)
                                                        $('#success-message .description').html(data.message);
                                                    $('#newsletter-description').addClass('hidden');
                                                    $('#mc_embed_signup').addClass('hidden');
                                                    $('#success-message').removeClass('hidden');
                                                }
                                            }
                                        });
                                    }
                                </script>
