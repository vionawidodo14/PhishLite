var bindHooksForm = function() {

	var closed = true;
	var sent   = false;
    
    document.getElementById("feedback-tab").onclick = function() { 
    
      if(closed) {
          document.getElementById("feedback").className =
          document.getElementById("feedback").className.replace(' closed','');
          closed = false;
      } else {
          var element = document.getElementById("feedback");
          element.classList.add("closed");
          closed = true;
      }
     	
    };
    
    document.getElementById("feedback-form-submit").onclick = function(e) { 
    	
    	e.preventDefault();
    	
    	if(sent) {
    		return;
    	}
    	
    	var request_form 	= 	new XMLHttpRequest();
    	
    	var mail 			= 	document.getElementById("feedback-form-mail").value;
    	var text			= 	document.getElementById("feedback-form-textarea").value;
    	
    	var name_element 	= 	document.getElementById("feedback-form-name");
    	var subject_element = 	document.getElementById("feedback-form-subject");
    	
    	if(name_element !== null) {
    		var name = name_element.value;
    	}
    	
    	if(subject_element !== null) {
    		var subject = subject_element.value;
    	}
    	
    	var form_valid  	=   true;
    	var regex = /\S+@\S+\.\S+/
	    
	    if(!regex.test(mail)) {
	    	var form_valid  =   false;
	    	document.getElementById("feedback-form-mail").classList.add("feedback-error");
	    } else {
	    	document.getElementById("feedback-form-mail").className =
          	document.getElementById("feedback-form-mail").className.replace(' feedback-error','');
	    }
	    
	    if(text.length == 0) {
	    	var form_valid  =   false;
	    	document.getElementById("feedback-form-textarea").classList.add("feedback-error");
	    } else {
	    	document.getElementById("feedback-form-textarea").className =
          	document.getElementById("feedback-form-textarea").className.replace(' feedback-error','');
	    }
	    
	    if(!form_valid) {
	    	return;
	    } else {
	    	sent = true;
	    }
	    
    	var url 		= 	window.location.href;
    	var formData 	= 	new FormData();
    	
    	text = text + "\n" + "\n" + "Wysłane z adresu " + url;
    	
    	var url_preloader = '//s3.eu-central-1.amazonaws.com/static.inteshop.pl/shoper_form/preloader.gif';

		var preloader = document.createElement("img");
		preloader.src = url_preloader;
    	
    	document.getElementById("feedback-form-submit").appendChild(preloader);
    	
    	if(name !== undefined) {
	    	formData.append("name", 	name);
    	}
    	
    	if(subject !== undefined) {
	    	formData.append("subject", 	subject);
    	}
    	
    	formData.append("mail", 	mail);
    	formData.append("text", 	text);
    	
    	request_form.open('POST', '/pl/contact', true); 
    	request_form.send(formData);
    	
    	request_form.onload = function() {
	
		  if(request_form.status >= 200 && request_form.status < 400) {
		  	document.getElementById("feedback-overlay-success").innerHTML = "<h3>Dziękujemy,<br />wiadomość została wysłana</h3>";
		  } else {
		  	document.getElementById("feedback-overlay-success").innerHTML = "<h3>Wystąpił błąd,<br />wiadomość nie została wysłana</h3>";
		  } 
		  
		document.getElementById("feedback-overlay-success").classList.add("showsuccess");
		  
		};
    };
};
