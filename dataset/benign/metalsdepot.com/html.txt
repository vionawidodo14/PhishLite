    $(document).ready(function() {

        var duration = 500;
        var unlock_position = null;

        jQuery(window).scroll(function() {
            var footer_top = $('.footer-logo-section').offset().top - 10;
            var back_button_height = $('#back').height();
            var back_button_bottom = $('#back').offset().top + back_button_height;

            if (back_button_bottom >= footer_top || ($(document).scrollTop() >= unlock_position && unlock_position !== null)) {

                $('#back').fadeIn(500);
            } else {
                $('#back').fadeOut(500);
            }
        });

            event.preventDefault();
            jQuery('html, body').animate({scrollTop: 0}, duration);
            return false;
        });
    });
