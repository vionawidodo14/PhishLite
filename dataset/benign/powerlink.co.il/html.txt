<script>

    var inaction = false;
    $(document).ready(function () {
        $(window).scroll(function () {
            if (!inaction) {
                var target = $('#myNavbar li.active').data("target")
                if (typeof (target) !== 'undefined') {
                    $('.activebar').removeClass('activebar')
                    $('.' + target).addClass("activebar");
                }
            }
        });

        $(document).ready(function () {
            // Add scrollspy to <body>
            var spyOffset = 190;
            if (window.innerWidth >= 2400) {
                spyOffset = 600;
            }
            $('body').scrollspy({ target: ".navbar", offset: spyOffset });

            // Add smooth scrolling on all links inside the navbar
            $("#myNavbar a").on('click', function (event) {
                // Make sure this.hash has a value before overriding default behavior
                if (this.hash !== "") {
                    // Prevent default anchor click behavior
                    event.preventDefault();

                    // Store hash
                    var hash = this.hash;

                    // Using jQuery's animate() method to add smooth page scroll
                    // The optional number (800) specifies the number of milliseconds it takes to scroll to the specified area
                    $('html, body').animate({
                        scrollTop: $(hash).offset().top - 180
                    }, 800, function () {

                        // Add hash (#) to URL when done scrolling (default click behavior)
                        window.location.hash = hash;
                    });
                }  // End if
            });
        });

    });



    function MoveTO(elem, target) {
        inaction = true;
        $('.product-bar-items').removeClass('activebar');
        $(elem).addClass('activebar');
        $('html,body').animate({
            scrollTop: $("#" + target).offset().top - 180
        },
            'slow');
        setTimeout(function () { inaction = false; }, 2000)
    }
</script>
<script>
    var h;
    $(document).ready(function ($) {
        StartSlidShow();
        var slideCount = $('#slider ul li').length;
        var slideWidth = $('#slider ul li').width();
        var slideTextWidth = $('#slider-text ul li').width();
        var slideHeight = $('#slider ul li').height();
        var sliderUlWidth = slideCount * slideWidth;

        $('#slider').css({ width: slideWidth, height: slideHeight });

        $('#slider ul').css({ width: sliderUlWidth, marginLeft: - window.innerWidth - 200});

        $('#slider ul li:last-child').prependTo('#slider ul');

        $('#slider-text').css({ width: slideWidth, height: slideHeight + 120 });

        $('#slider-text ul').css({ width: sliderUlWidth });

        $('#slider-text ul li:last-child').prependTo('#slider ul-text');

        function moveLeft() {
            $('#slider ul').animate({
                left: + slideWidth
            }, 1000, function () {
                $('#slider ul li:last-child').prependTo('#slider ul');
                $('#slider ul').css('left', '');
            });

            $('#slider-text ul').animate({
                left: + slideTextWidth + 100
            }, 1000, function () {
                $('#slider-text ul li:last-child').prependTo('#slider-text ul');
                $('#slider-text ul').css('left', '');
            });
        };

        function moveRight() {
            $('#slider ul').animate({
                left: - slideWidth
            }, 1000, function () {
                $('#slider ul li:first-child').appendTo('#slider ul');
                $('#slider ul').css('left', '');
            });

            $('#slider-text ul').animate({
                left: - slideTextWidth - 100
            }, 1000, function () {
                $('#slider-text ul li:first-child').appendTo('#slider-text ul');
                $('#slider-text ul').css('left', '');
            });
        };

        $('div.control_prev').click(function () {
            moveRight();
            clearInterval(h);
            StartSlidShow();
        });

        $('div.control_next').click(function () {
            moveLeft();
            clearInterval(h);
            StartSlidShow();
        });
        function StartSlidShow() {
            h = setInterval(function () {
                moveRight();
            }, 5000);
        }

    });

</script>
