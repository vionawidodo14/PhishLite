{

    // not sure if you wanted this, but I thought I'd add it.
    // get an associative array of just the values.
    var values = {};
    $inputs.each(function() {
        values[this.name] = $(this).val();
    });
	
	if(values['firstname'] == '')
	{
		 $('input#firstname').attr('placeholder', 'Enter Your First Name here');
		 $('input#firstname').focus();
		 return false;
	}
	
	if(values['lastname'] == '')
	{
		 $('input#lastname').attr('placeholder', 'Enter Your Last Name here');
		 $('input#lastname').focus();
		 return false;
	}
	
	var re = /^([\w-]+(?:\.[\w-]+)*)@((?:[\w-]+\.)*\w[\w-]{0,66})\.([a-z]{2,6}(?:\.[a-z]{2})?)$/i;
	if(re.test(values['email']))
	{
		//
	}else{
		
		$('input#email').val('');
		 $('input#email').attr('placeholder', 'Enter Your correct email here');
		 $('input#email').focus();
		 return false;	
	}
	
	if(values['password']== '')
	{
		 $('input#password').val('')
		 $('input#password').attr('placeholder', 'Enter Your Password here');
		 $('input#password').focus();
		 return false;
	}
	
	
	if(values['confirm_password'] != values['password'])
	{
		
		 $('input#confirm_password').val('')
		 $('input#confirm_password').attr('placeholder', 'Enter Correct Password here');
		 $('input#confirm_password').focus();
		 return false;
	}
	
	if($('input#address').val() == '')
	{
		 $('input#address').attr('placeholder', 'Enter Your Address here');
		 $('input#address').focus();
		 return false;
	}
	
	if($('input#phone_mobile').val() == '')
	{
		 $('input#phone_mobile').attr('placeholder', 'Enter Your Cell# here');
		 $('input#phone_mobile').focus();
		 return false;
	}
	
	
	
	if($('input#city_id').val() == '')
	{
		 $('input#city_id').attr('placeholder', '--Select City--');

	}
	
	if($('input#region_id').val() == '')
	{
		 $('input#region_id').attr('placeholder', '--Select Province--');

		 return false;
	}

	
}
