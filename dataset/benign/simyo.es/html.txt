      //  Recorre el slider (sliderId) hacia la derecha en busca de un bono NO deshabilitado
  function iterateSliderUp(sliderLabelsMap,sliderId){
     var foundNotDisabled=false;
    for(i=$('#'+sliderId).slider("value");i<=sliderLabelsMap.length-1;i++){
      foundNotDisabled = !$('.'+sliderLabelsMap[i]).hasClass('disabled');
      if(foundNotDisabled){
        $('#'+sliderId).slider('value', i);
        break;
      }
    }
    return foundNotDisabled;
  }
//  Recorre el slider (sliderId) hacia la izquierda en busca de un bono NO deshabilitado
  function iterateSliderDown(sliderLabelsMap,sliderId){
    var foundNotDisabled=false;
    for(i=$('#'+sliderId).slider("value");i>=0;i--){
      foundNotDisabled = !$('.'+sliderLabelsMap[i]).hasClass('disabled');
      if(foundNotDisabled){
        $('#'+sliderId).slider('value', i);
        break;
      }
    }
    return foundNotDisabled;
  }
     function updateCartText(){
        var internet = internetSliderValueSel;
        var llamadas = llamadasSliderValueSel;
        var pack = getPack(internet, llamadas);
        choices = getChoicesBundleName();

        } else {
          $('.shopping-cart .price-total').html(formatPriceHtml(total) + '<sup class="euro">€</sup></strong>');
          $('.selected-rate').html(choices);
    //      $('.shopping-cart .buttonTotalTarifa').show();
        }
      }

function getPack(internet, llamadas){
  if(internet || llamadas){
    var pack = $.grep( packs, function( n, i ) {
        return n.mb==parseInt(internet) && n.min==parseInt(llamadas);
    });
    
    return pack[0];   
  }else{
    return null;
  }
}

function getChoicesBundleName(){
  
  var selection = [];
  
  if(internetSliderValueSel && internetSliderValueSel!=0){
    selection.push(internetSliderLabelSel);
  }
  if(llamadasSliderValueSel && llamadasSliderValueSel!=0){
    selection.push(llamadasSliderLabelSel);
  }
  
  $('.complete-rate input:checkbox:checked').map(function(){
    return $(this).attr('bundle_name');
  }).each(function(){
    selection.push(this);
  });


  if(selection.length > 4){
     $('.cart-wrapper').addClass("muchasOpciones");
                }else{
                  $('.cart-wrapper').removeClass("muchasOpciones");
                }
  

  return selection.join(' + '); 
}


function formatPrice(price){
  return String(Math.round(price * 100) / 100).replace('.',',');
}

function formatPriceHtml(price){
  var parts = formatPrice(price).split(',');
  
  if(parts.length > 1){
    return '<span class="price-int">' + parts[0] + "</span><sup class='price-decim'>'" +  parts[1] + '</sup>'
  }else{
    return '<span class="price-int">' + parts[0] + '</span>'
  }
}

function getChoicesBundlePrice(pack){
  
  if(!pack){

    var price = $.grep([internetSliderPriceSel, llamadasSliderPriceSel], Boolean).reduce(function (a,b){return a+b;},0);

    price += $('.complete-rate input:checkbox:checked').map(function(){
      return parseFloat($(this).attr('price'));
    }).get().reduce(function(prev, cur){
      return prev + cur;
    }, 0);
  }else{
    var price = $('.complete-rate input:checkbox:checked ').not('[bundle_type="internet"]').not('[bundle_type="voz"]').map(function(){
      return parseFloat($(this).attr('price'));
    }).get().reduce(function(prev, cur){
      return prev + cur;
    }, 0);
    
    price = price + pack.price;
  }
  
  return price;
}
