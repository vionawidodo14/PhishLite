  {
    $('#divLoginRequestError').html('');

          return 'Aircraft';
      }
      else if ($('#qsearch_item_parent_category_id').val() && !$('#qsearch_item_category_id').val() && !$('#qsearch_vehicle_make_id').val() && modelIds == ''){
          return $('#qsearch_item_parent_category_id :selected').text();
      }
      else if ($('#qsearch_item_parent_category_id').val() && $('#qsearch_item_category_id').val() && !$('#qsearch_vehicle_make_id').val() && modelIds == ''){

          return $('#qsearch_vehicle_make_id :selected').text();

          var modelText = $(':checkbox[name^="qsearch[vehicle_model_id][]"]:checked').map(function(){
              return $(this).next("label").text();
          }).get();

          return $('#qsearch_vehicle_make_id :selected').text() +' '+ modelText;

      else if ($('#qsearch_item_parent_category_id').val() && !$('#qsearch_item_category_id').val() && $('#qsearch_vehicle_make_id').val() && modelIds == ''){

function setPromotionWindow(id, sliderobj)
{
  slides = getPromotionWindowSlides();
  slide_width = getPromotionWindowSlideWidth();

  sliderobj.reloadSlider({
    captions: false,
    controls: getPromotionWindowControl(),
    pager: false,
    slideWidth: slide_width,
    minSlides: slides,
    maxSlides: slides,
    responsive: true,
    infiniteLoop: false,
    touchEnabled: false,
    preloadImages: 'all',
    oneToOneTouch: false,
    hideControlOnEnd: true
  });
}

function getPromotionWindowControl()
{

  width = $(window).width();

  if (width <= 568)
    control = true;
  else
    control = true;

  return control;
}

function getPromotionWindowSlides()
{
  width = $(window).width();

  if (width > 1280)
    slides = 3;
  else if (width >= 800 && width <= 1280 )
    slides = 3;
  else if (width >= 640)
    slides = 3;
  else if (width >= 480)
    slides = 2;
  else if (width >= 320)
    slides = 1;

  return slides;
}

function setMainWindow(id)
{
  slide_width = getMainWindowSlideWidth();

  slider.reloadSlider({

    controls: true,
    slideWidth: slide_width,
    hideControlOnEnd: true
  });
}

function getMainWindowSlideWidth()
{
  return $(window).width();
}

function getPromotionWindowSlideWidth()
{
  width = $(window).width();

  if (width >= 1361)
	  slide_width = 260;

  else if (width >= 568)
    slide_width = 270;

    slide_width = 233;
  else
    slide_width = 309;

  return slide_width;
}
