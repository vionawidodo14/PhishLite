    function checkArrowPos() {
        var posistion = $('div.active').data('pos');
        if (posistion == "br") {
            $('.icon').css('color', '#232323');
        } else {
            $('.icon').css('color', '#FFFFFF');
        };

    function carouselGoTo(e) {
        var index = $(e).attr("data-slide-to");
        $('.carousel').carousel(parseInt(index));
    }

    function carouselControl(e) {
        var direction = $(e).attr("data-slide");
        $('.carousel').carousel(direction);
    }

    $(".carousel").swipe({
        swipe: function (event, direction, distance, duration, fingerCount, fingerData) {
            if (direction == 'left') $(this).carousel('next');
            if (direction == 'right') $(this).carousel('prev');

        },
        allowPageScroll: "vertical"

    });


    function trackCarousel(e) {
        var title = $(e).attr('data-title');
        var position = $(e).attr('data-position');
        var href = e.href;      

        // Send GA event with custom dimensions
        try {
            ga('send', 'event', 'Carousel', 'action', {
                'dimension1': title,
                'dimension2': parseInt(position) + 1,
                'dimension3': href
            });
        } catch (e) {
            console.log(e);
        }
    }
</script>
