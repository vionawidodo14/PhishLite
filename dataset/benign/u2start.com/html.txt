        require(['jquery', 'knockout', 'knockout-ext', 'env'], function ($, ko, kext, env) {

                };

                var imgMap = {
                    'topic': '/images/quicksearch/forums.gif',

                    'song': '/images/quicksearch/songs.gif',
                    'studioalbum': '/images/quicksearch/songs.gif'
                };

                this.img = function (data) {
                    return data.type in imgMap ? imgMap[data.type] : none;
                };

                this.results = kext.asyncDependentObservable(function () {
                    if (!this.searchTerm()) {
                        return [];
                    }

                    if (this.searchTerm().length < 3) {
                        return [];
                    } else {
                        $('.quicksearch-input').addClass('search-active');
                    }

                    this.searching(true);

                    return $.ajax({
                        type: 'GET',
                        dataType: "json",
                        url: ssl_prefix + "/awesomesearch/",
                        data: {
                            query: this.searchTerm()
                        }
                    }).then(function (data) {
                            vm.searching(false);
                            vm.display('display: block');
                            $('.quicksearch-input').removeClass('search-active');
                            return data
                        });
                }, this);


                $(function(){
                    if($("#message").length)
                        self.hasBb(true);

                !function () {

                    self.results.subscribe(function (value) {

                        if (timer)

                                function () {
                                    $("[type=search]").blur();
                                },
                                1000
                            );

                    });
                }();
            };

            ko.applyBindings(vm, document.getElementById("searchContext"));
        });

    function timeAgo(now, indate){
        var $ago = (now - indate) / 1000.0, $years, $months, $time, $seconds, $days, $hours, $minutes;

        if ($ago >= 31536000) {
            $years = Math.floor($ago/31536000);
            $time = $years + ' ' + (($years == 1) ? "year" : "years");

        } else if ($ago >= 2635200) {
            $months = Math.floor($ago/2635200);
            $time = $months + ' ' + (($months == 1) ? "month" : "months");
        } else if ($ago >= 86400) {
            $days = Math.floor($ago/86400);
            $time = $days + ' ' + (($days == 1) ? "day" : "days");

        } else if ($ago >= 3600) {
            $hours = Math.floor($ago/3600);
            $time = $hours + ' ' + (($hours == 1) ? "hour" : "hours");
        } else if ($ago >= 60) {
            $minutes = Math.floor($ago/60);
            $time = $minutes + ' ' + (($minutes == 1) ? "minute" : "minutes");
        } else {
            $seconds = Math.round($ago);
            $time = $seconds + ' ' + (($seconds == 1) ? "second" : "seconds");
        }

        return $time;

    }

    $(document).ready(function(){
        var now = new Date(1572086342000);
        window.setInterval(function(){
            now = new Date(now .getTime() + 1000*10);
            $("time.ago[datetime]").each(function(i, el){
                var strAgo = timeAgo(now, new Date($(el).attr("datetime")));

                if (strAgo.substring(0,3) != "NaN") {
                    var strTranslate = "%1% ago";
                    $(el).text(strTranslate.replace("%1%", strAgo));
                }
            });
        }, 10000);
    });
