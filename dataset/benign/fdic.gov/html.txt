        function PRLocation(type, fullYear, num) {
            var yearString = fullYear.toString();
            var s = "";
            if (type == "pr") s = "press/";
            if (type == "fil") s = "financial/";
            var baseURL = "/news/news/" + s + fullYear + "/" + type + yearString.substring(yearString.length - 2);
          // alert("/news/news/" + s + fullYear + "/" + type + yearString.substring(yearString.length - 2)); 
            var URLloc = "";
            if (num.length == 3)
                URLloc = baseURL + num + ".html";
            else if (num.length == 2)
                URLloc = baseURL + "0" + num + ".html";
            else if (num.length == 1)
                URLloc = baseURL + "00" + num + ".html";
            else
                URLloc = "Error";
            return URLloc;
        }

        function displayItems(type, xmlDoc, num, fullYear) {
            var lineItems = "";
            var textItems = xmlDoc.getElementsByTagName("title");
            var date_textItems = xmlDoc.getElementsByTagName("date");
            for (i = 0; i < num; i++) {
                //var text = textItems[i].childNodes[0].nodeValue;
                var text = "";
                for (j = 0; j < textItems[i].childNodes.length; j++) {
                    text += textItems[i].childNodes[j].textContent;

                var date_text = date_textItems[i].childNodes[0].nodeValue;
                var PRID = text.substring(0, text.indexOf(" "));
                var title = text.substring(text.indexOf(" ") + 1);
                if (title.length > 100) title = title.substring(0, 90) + "...";

                var URLloc = PRLocation(type, fullYear, PRID.split("-")[1]);
                lineItems += "<li><!--<h4>" + PRID + "</h4>--><a href=\"" + URLloc + "\">" + title + "</a><br/><span class=\"date\">" + date_text + "</span></li>";
            }
            return lineItems;
        }
