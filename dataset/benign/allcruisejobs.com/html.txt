	function equalHeight(group) {
		var tallest = 0;
		group.each(function() {
			var thisHeight = $(this).height();
			if(thisHeight > tallest) {
				tallest = thisHeight;
			}
		});
		if (tallest > 0) {
			group.height(tallest);
		}
	}

		$.ajax({
			type: "GET",
			url: "https://www.allcruisejobs.com/cookie-consent/accept.php",
			data: {c:c, url:'https%3A%2F%2Fwww.allcruisejobs.com%2F'},
