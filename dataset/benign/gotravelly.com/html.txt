$(function() {
    // pasang fragment utk menentukan tab yg aktif menggunakan jquery ketika men-submit form
    // karena kalau pakai js utk pindah tab setelah page load, maka agak lambat
    // sedangkan kalau fragment dipasang di attr "action" di form, tidak jalan di IE7
    // sumber: https://stackoverflow.com/a/14365249/1235167
    // parameter @elem berisi pembungkus tombol submit dan elemen2 input yg akan dipassingkan. biasanya berupa form.
    // NB: coding ini ver.20180517
    function myNavbarSearchSubmit(elem) {
        var qs = $(elem).serialize();
        // tambahkan fragment (kalau ada)
        if ($('#search_tab').val()) {
            qs += '#'+$('#search_tab').val();
        }
        window.location.href = '/search?'+qs;
    }
    // ketika submit form pencarian di navbar maka lakukan proses di function myNavbarSearchSubmit
    $('.my-search_form').submit(function(e) {
		e.preventDefault();
        myNavbarSearchSubmit(this);
    });

    // ketika klik elemen dgn atribut name="btn-search-nearby"
    $('[name="btn-search-nearby"]').click(function(e) {
        e.preventDefault();

        var positionFlag = 0;
        var myLat = 0; myLng = 0;
        var url = '/search?qtype=nearby';
        // kalau browser nya support geolocation
        if (navigator.geolocation) {
            navigator.geolocation.getCurrentPosition(function(position) {
                positionFlag = 1;
                myLat = position.coords.latitude.toString();
                myLng = position.coords.longitude.toString();
                url = url + '&lat=' + myLat + '&lng=' + myLng;

                // koordinat user jgn dikirimkan via url parameter, tapi pake POST
				// window.location.href = url+'&q='+myLat+','+myLng;
				var name = $('#csrf').attr('name');
				var value = $('#csrf').val();
                var form = $('<form id="form_coord" action="' + url + '" method="post">'
                        + '<input type="hidden" id="csrf" name="' + name + '" value="' + value + '" />'
                        + '<input type="text" name="lat" value="' + myLat + '" />'
                        + '<input type="text" name="lng" value="' + myLng + '" />'
                        + '</form>');
                $('body').append(form);
                form.submit();
            });
        } else {
            alert('Please turn on your GPS to use this feature.');
        }
    });
});
</script>
<script>
$(function() {
    // script utk menampilkan timeline feeds (notifications)
    var tlContainer = '#timelinefeeds-container';
    var myself = $('#allow_edit_user').val();
    var obj = {
        user_id: $('#login_user_id').val(),
    };

    // tiap 1 detik, refresh timeline nya
    var intervalID = setInterval(function() {
        // baca data timeline feeds/notifications
        socketIO.emit('get timeline', obj);
        // // lakukan proses update is_seen notifications
        // socketIO.emit('set timeline is_read', obj);
    }, 1000);

    var this_lang = $('#language').val();
    if (this_lang == 'indonesian') {
        moment.locale('id');
    } else {
        moment.locale('en');
    }

    // dapatkan isi variabel lang dari server
    var notifLang;
    $.ajax({
        url: '/component/get_lang/notif',
        type: 'get',
        async: false,
        dataType: 'json',
        success: function(res) {
            // console.log('success loading notif language.');
            notifLang = res.lang;
        },
        error: function(err) {
            console.log('error loading notif language', err.responseText);
        }
    });

    socketIO.on('get timeline ready', function(data) {
        setTimeout(function() {
            // kalau data timeline nya kosong, maka bersihkan DOM utk timeline
            if (typeof(data.id) == 'undefined' || $('[data-notifID]').length == 0) {
                $(tlContainer).html('');
            }

            if (typeof(data.id) != 'undefined') {
                // lalu tambahkan satu per satu timeline item ke dalam timeline container
                var notifData = prep_notif_item(data, notifLang);
                // sepertinya faktor waktu di server berbeda2 waktu di server GoDaddy sudah benar 
                // waktu di rumahweb harus ditambahkan 7jam
                // utk php 5.6 / hosting di rumahweb
                // var newDate = new Date(data.created);
                // newDate.setHours(8);
                // notifData['created'] = moment(moment(data.created).add(7, 'hours'), 'YYYY-MM-DD H:m:s').fromNow();
                notifData['created'] = moment(moment(data.created), 'YYYY-MM-DD H:m:s').fromNow();
                // notifData['created'] = moment(moment(newDate), 'YYYY-MM-DD H:m:s').fromNow();
                // utk php 7++ / hosting selain di rumahweb
                // notifData['created'] = moment(data.created, 'YYYY-MM-DD H:m:s').fromNow();
                var str = '<div class="media navbar-gttimeline" data-notifID="'+data.id+'">'
                        +    '<div class="media-left media-top">'
                        +      '<a href="'+notifData.user_href+'" title="'+notifData.user_text+'" target="_blank">'
                        +        '<div class="my-media-object" style="background-image:url('+notifData.user_img_src+');"></div>'
                        +      '</a>'
                        +    '</div>'
                        +    '<div class="media-body">'
                        +        '<div class="navbar-gttimeline-maintxt">'
                        +          notifData.message
                        +        '</div>'
                        +        '<div class="navbar-gttimeline-date">'
                        +          notifData.created
                        +        '</div>'
                        +    '</div>'
                        +    '<div class="media-right media-top">'
                        +      '<a href="'+notifData.obj_href+'" title="'+notifData.obj_text+'" target="_blank">'
                        +        '<div class="my-media-object" style="background-image:url('+notifData.obj_img_src+');"></div>'
                        +      '</a>'
                        +    '</div>'
                        +  '</div>'
                        ;
                // kalau notif dgn id tsb sudah ada di timeline maka tidak perlu ditampilkan ulang
                if ($('[data-notifID="'+data.id+'"]').length < 1) {
                    if (data.id > $('#cur_notif_id').val()) {
                        $(tlContainer).prepend(str);
                    } else {
                        $(tlContainer).append(str);
                    }

                    // pasang id notif barusan di DOM
                    $('#cur_notif_id').val(data.id);
                }
            }

            check_total_notif_num(data);

            // cek apakah container timeline sudah ada isinya? kalau kosong, maka tampilkan gambar default
            $(tlContainer).trigger('change');
        }, 1000);
    });
    // end socketIO.on('get timeline ready'

    // kalau container timeline kosong, maka tampilkan gambar default empty timeline
    $('body').on('change', tlContainer, function(e) {
        if ($('#login_username').val()) {
            if ($(this).html() == '') {
                var str = '<div class="text-center navbar-gttimeline">'
                        + '  <img src="/assets/img/userblank-activitynew.png"/>'
                        + '</div>'
                        ;
                $(this).html(str);
            }
            // ketika ada lebih dari 10 notif, tampilkan link utk ke halaman View All Notif
            if ($(this).find('[data-notifID]').length > 10) {
                // hilangkan semua link View All notif spy cm ada 1 saja nantinya
                $(this).find('[data-linktype="view-all"]').remove();

                var str = '<li><a class="text-center" href="/user/'+$('#login_username').val()+'/timeline"';
                $(this).closest('.dropdown-menu').append(str);
            }
        }
    });

    // ketika klik tombol lonceng notification
    $('.notification-wrapper').click(function(e) {
        var is_showing = !$(this).closest('.dropdown').find('.dropdown-menu').is(':visible');
        if (is_showing) {
            // lakukan proses update is_seen notifications
            socketIO.emit('set timeline is_read', obj);
        }
    });

    socketIO.on('timeline is_seen ready', function(data) {
        check_total_notif_num(data);
    });
    // end socketIO.on('timeline is_seen ready'

    function check_total_notif_num(data) {
        // pengecekan utk jaga2 kalau data.total_notif_unread tidak diada di object yg dipassingkan
        // maka beri nilai default 0
        if (typeof data.total_notif_unread == 'undefined') {
            data.total_notif_unread = 0;
        }
        // gunakan variabel local utk memisahkan antara variabel utk label string dan angka total_notif_unread
        var notif_total = data.total_notif_unread + 0; // typecast into integer
        if (notif_total > 99) {
            notif_total = '+99';
        }

        $('.notification-wrapper').find('.notif_bell-num').text(notif_total);
        if (data.total_notif_unread > 0) {
            $('.notification-wrapper').addClass('fcgold');
            $('.notification-wrapper').find('.notif_bell-num').show();
        } else {
            $('.notification-wrapper').removeClass('fcgold');
            $('.notification-wrapper').find('.notif_bell-num').hide();
        }
    }
    // end function check_total_notif_num
});
</script>
