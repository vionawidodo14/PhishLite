		$(window).width() >= 1530 ? $("#sticky-footer").animate({bottom: -232}) : $("#sticky-footer").animate({bottom: -360});

		$("#sticky-footer").addClass("sticky-footer-closed");
		sliderOpen = false;
	}
	function openBar() {
		$("#sticky-footer").animate({bottom: 0});
		$("#sticky-footer").removeClass("sticky-footer-closed");
		sliderOpen = true;
	}

	$("#sticky-toggle").click(function() {
		stayClosed = !stayClosed;
		if (sliderOpen == true){
			closeBar();
			// Start Session
	    sessionStorage.setItem("keepClosed", "true");
		}
		else if (sliderOpen == false) {
			openBar();
			// End Session
			sessionStorage.clear();
		}
		else {
			openBar();
		}
	});

	$(window).resize(function() {
		if ($(window).width() >= 1530) {
			$('#sticky-footer').hasClass('sticky-footer-closed') ? $('#sticky-footer').css("bottom","-232px") : $('#sticky-footer').css("bottom","0px");

		if (isscrolling) {
