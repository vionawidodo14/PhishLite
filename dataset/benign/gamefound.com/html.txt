    (function () {
        var Toggle = function (selector, options) {
            this.selector = selector;
            this.options = this._extend(Toggle.DEFAULTS, options || {});
            this.init();
        };
        Toggle.DEFAULTS = {
            attr: "data-toggle",
            stateCssClass: "is-open"
        };
        Toggle.prototype = {
            _extend: function (obj, src) {
                Array.prototype.forEach.call(Object.keys(src),function (key) {
                    obj[key] = src[key];
                });
                return obj;
            },
            _toggleClass: function (key) {
                var nodes = this.$toggleTargets[key];
                Array.prototype.forEach.call(nodes, function (item) {
                    item.classList.toggle(this.options.stateCssClass)
                }.bind(this));
            },
            _getToggleItems: function (selector) {
                return [].slice.call(document.querySelectorAll(selector));
            },
            _getToggleTargets: function (nodeList) {
                var result = {};
                Array.prototype.forEach.call(nodeList, function (item) {
                    var key = item.getAttribute(this.options.attr);
                    var nodes = document.querySelectorAll(key);
                    result[key] = nodes;
                }.bind(this));
                return result;
            },
            _handleClick: function (e) {
                this._toggleClass(e.currentTarget.getAttribute(this.options.attr));
            },
            _addEvents: function () {
                Array.prototype.forEach.call(this.$toggleItems, function (item) {
                    item.addEventListener("click", this._handleClick.bind(this), false);
                }.bind(this));
            },
            init: function () {
                this.$toggleItems = this._getToggleItems(this.selector);
                this.$toggleTargets = this._getToggleTargets(this.$toggleItems);
                this._addEvents();
            }
        };
        var toggle = new Toggle(".js-toggle");
    })();
</script>
