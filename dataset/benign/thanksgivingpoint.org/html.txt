                        withCredentials: true
                    }).then(function(result) {

                    });
            };

            $scope.getHeaderView();
        });

    if (isShow === "True") {
        if (docCookies.hasItem(alertCookie)) {
            var cookie = docCookies.getItem(alertCookie);
            if (cookie !== "enabled") {
                $(".top-message").css("display", "none");
            } else {
                $(".top-message").css("display", "block");
            }
        } else {
            docCookies.setItem(alertCookie, "enabled", tomorrow, "/");
            $(".top-message").css("display", "block");
        }
    }

    $(".top-message .close")

    $(document).ready( function() {
        var subdomain = window.location.host.split('.')[0]
        if (subdomain == 'cart') {
            $("section.subscribe").hide();
        } else {
            $("button#subscribe_submit").on("click", function (e) {
                e.preventDefault();
                var model = {
                    email: $("input#subscribe_email")[0].value,
                    fName: $("input#subscribe_fname")[0].value,
                    lName: $("input#subscribe_lname")[0].value,
                    isValid: function () {
                        return this.email != "" && this.fName != "" && this.lName != "" && $(".subscribe input.input-validation-error").length == 0;
                    }
                };
                if (model.isValid()) {

                    var submit = $(this);
                    var submitChildren = submit.children();
                    var spinner = document.createElement("span");
                    spinner.className = "fa fa-spinner subscribe-spinner";
                    submit.empty();
                    submit.append(spinner);

                    $("#subscribe_validation-error").hide();
                    $.ajax({
                        contentType: "application/json",
                        type: "POST",
                        url: "/Umbraco/ACM/Subscribe/New",
                        data: JSON.stringify(model),
                        success: function (res) {
                            if (res.success) {
                                alert("Thanks for subscribing!");
                            } else {
                                alert(res.message);
                                window.location = "https://cart.thanksgivingpoint.org/account/login.aspx?ReturnUrl=http://thanksgivingpoint.org/";
                            }
                            submit.empty();
                            $(submitChildren).each(function () {
                                submit.append(this);
                            })
                        },
                        error: function (e) {
                            alert("Something went wrong, please try again.");
                            console.log(e);
                            submit.empty();
                            console.log(submitChildren);
                            $(submitChildren).each(function () {
                                submit.append(this);
                            })
                        }
                    })
                } else {
                    $("#subscribe_validation-error").show();
                }
            });
        }
        var mainUrl = "http://thanksgivingpoint.org";
        $(".footer-nav a")

                    var origUrl = $(this).attr('href');
                    if ($(this).is("#phoneNumber")) {

                    } else {
                        $(this).attr('href', mainUrl + origUrl);
                    }
                }
            });
        $(".footer-bottom a")
            .each(function () {
                if ($(this).attr('href').toString().toLowerCase().indexOf("http")) {
                    var origUrl = $(this).attr('href');
                    $(this).attr('href', mainUrl + origUrl);
                }
            });
