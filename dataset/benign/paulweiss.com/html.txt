    function AutoComplete(request, response, keyword) {
        $.ajax({
            //type: 'POST',
            //url: '/umbraco/Api/SiteSearch/AutoComplete',
            url: '/umbraco/Api/SiteSearch/AutoComplete',
            dataType: "json",
            data: { keyword: keyword },
            success: function (data) {
                var liTags = '';
                if (data != null) {
                    response($.map(data, function (v, i) {
                        return {
                            label: v.Text,
                            value: v.Value
                        };
                    }));

                    for (var i = 0; i < data.length; i++) {
                        liTags += "<li class='ui-menu-item' id='ui-id-" + i + "' tabindex='-1' url='" + data[i].Value + "'>" + data[i].Text + "</li>";
                    }
                    $('#search-results-container ul').html(liTags);
                    $('#search-results-mobcontainer ul').html(liTags);

                    //var divViewAll = document.getElementById("divViewAll");
                    //console.log($(".ui-autocomplete").html().index(divViewAll));
                    if ($(window).width() < 768) {
                        $('.ui-autocomplete').append("<div id='divViewAll' class='searchview-all-results'><a href='javascript:void(0);' onclick='MobileSearch()' onkeypress='MobileSearch()'>VIEW ALL RESULTS</a></div>");
                    } else {
                        $('.ui-autocomplete').append("<div id='divViewAll' class='searchview-all-results'><a href='javascript:void(0);' onclick='Search()' onkeypress='Search()'>VIEW ALL RESULTS</a></div>");
                    }

                } else {
                    response($.map(data, function (v, i) {
                        return {
                            label: "",
                            value: ""
                        };
                    }));
                }
            },
            error: function (jqXHR, textStatus, errorThrown) {
                console.log(textStatus, errorThrown);
            }
        });
    }
