  $(document).ready(function() {

    mixpanel.track("Extension download page view");

    $.cookie("time_zone_offset", new Date().getTimezoneOffset(), { path: '/' });


    var detectedExtension = false;
    var backoffTimeouts = [4000, 1000, 500, 200, 50];

    var detectExtension = function() {
      if (backoffTimeouts.length > 0) {

        if(!Dly.isChromeDesktop()) {
            $("body").addClass('non-chrome');
        }

        detectedExtension = Dly.isExtensionInstalled();

        if(detectedExtension) {
          $('body').addClass('extension-downloaded');
          mixpanel.track("Extension sign up page view");
        } else {
          setTimeout(detectExtension, backoffTimeouts.pop());
        }
      }
    }

    detectExtension();

    $("#chrome-extension-install-button")
      .on('click', function(e) {
        mixpanel.track("Extension install click");
        $(this).addClass("loading");
        if (Dly.isChromeDesktop()) {
          $.cookie("extension_flow_step", 0);
          return true;
        } else {
          e.preventDefault();
          $(".btn.cta").trigger($.Event("install_error"));
        }
      })
      .on("install_error", function(){
        $(this).removeClass("loading");
      });

    function installExtension() {
      chrome.webstore.install("https://chrome.google.com/webstore/detail/dijhcpbkalfgkcebgoncjmfpbamihgaf", function() {
        $.cookie("extension_flow_step", 0);

        mixpanel.track("Extension install success");
      }, function(response) {
        mixpanel.track("Extension install error");
        $(".btn.cta").trigger($.Event("install_error"));
      });
    }

    $('.sign-in a.sign-in').click(function() {
      mixpanel.track("Extension sign up page sign in click");
      launchSignInModal();
    });
    mixpanel.track_links('.chrome_section .authentication-buttons a.cta.btn', 'Extension sign up button click', function(ele) {
      var provider = $(ele).data('provider');
      return {provider: provider};
    });

    function carousel() {
      var tick = setInterval(function(){
        var nextTab = $(".browser_tab").filter(".active").next();
        if(!nextTab.length) {
         nextTab = $(".browser_tab").first();
        }
        nextTab.find(".browser_tab_link").trigger("click");
      }, 7500)
      $(".browser_tab, .browser_tab_content").on("mouseover", function(e){
        clearInterval(tick);
      })
    }
    carousel();
  });
