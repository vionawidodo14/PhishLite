function set_description_text(windowsize,max_text,max_text_inv){
	$( ".products_id_description_text" ).each(function( index, element ) {
    var product_id = $( element ).attr('data-products-id');
		var inventory_isset = $('#' + product_id + '_id_inventory_isset').val();
		var text = $( element ).val();
		if(inventory_isset){
			if (text.length >= max_text_inv){
				$('#' + product_id + '_products_description_listing_inventory').html( text.substring(0,max_text_inv-15));
			}else{
				$('#' + product_id + '_products_description_listing_inventory').html( text);
			}
		}else{
			if (text.length >= max_text){
				$('#' + product_id + '_products_description_listing').html( text.substring(0,max_text-15));
			}else{
				$('#' + product_id + '_products_description_listing').html( text);
			}
		}
	});
}

var dropdowntimerOpen;
var dropdowntimer_out = 500;
var dropdownflowtimeOpen = 200;
var dropdownflowtimeClose = 0;
var dodropdown = true;
var dodropdownclose = true;

function closeDropdownMenu(element) {
  clearTimeout(dropdowntimerOpen);
  dropdowntimerClose =  setTimeout(function () {
	  if(dodropdownclose){
    	setTimeout(function () {
			  if(dodropdownclose){
		    	$('#' + element).hide();
					$('#' + element).parent().find('.dropdown-menu').removeClass('selected');
				}
		  }.bind(this), dropdownflowtimeClose);
		}
  }.bind(this), dropdowntimer_out);
}

function openDropdownMenu(element) {
	clearTimeout(dropdowntimerClose);
  dropdowntimerOpen =  setTimeout(function () {
  	if(dodropdown){
	  	setTimeout(function () {
		  	$('#' + element).parent().parent().find('.f-dropdown').hide();
				$('#' + element).parent().find('.dropdown-menu').removeClass('selected');
				$('#' + element).parent().find('.dropdown-menu[data-dropdown*="' + element + '"]').addClass('selected');
				$('#' + element).show();
				$('#' + element).focus();
		  }.bind(this), dropdownflowtimeOpen);
	  	
		}
  }.bind(this), dropdowntimer_out);
}
