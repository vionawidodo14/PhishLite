function fiTimerPoints(timer,reference,threshold){
	return Math.pow(Math.min(reference-timer+threshold,0),2);
}

function fiTimerScore(){
	let maxscore=10000;
	var thresholds = {rs:100,fp:100,ttvr:100,tti:100,pl:100};
	var rs 	= BOOMR.getPerformance().timing.responseStart - BOOMR.getPerformance().timing.navigationStart;
	var fp 	= BOOMR.plugins.PaintTiming.getTimingFor('first-paint');	
	var ttvr = BOOMR.plugins.Continuity.metrics.timeToVisuallyReady();
	var tti = BOOMR.plugins.Continuity.metrics.timeToInteractive();
	var pl 	= BOOMR.getPerformance().timing.loadEventEnd - BOOMR.getPerformance().timing.navigationStart;
	if(!tti){
		tti=pl+1000;//TODO: get from mPulse app settings
	}
	var totalTimerPoints = 
		fiTimerPoints(rs,0,thresholds.rs) +
		fiTimerPoints(fp,rs,thresholds.fp) +
		fiTimerPoints(ttvr,fp,thresholds.ttvr) +
		fiTimerPoints(tti,ttvr,thresholds.tti) +
		fiTimerPoints(pl,ttvr,thresholds.pl);
	return Math.min((Math.sqrt(totalTimerPoints)/maxscore)*100,100);
}

function fiScore(){
	return Math.ceil(fiTimerScore()+fiMetricScore());
}

function getFrustrationScore(){
	var frustrationScore = fiScore();
	if(BOOMR.plugins.Continuity.metrics.timeToInteractive()){
		BOOMR.addVar('cdim.Frustration', getFrustrationDimension(frustrationScore));
	}
	return frustrationScore;
}

function getFrustrationDimension(score){
	var range = '';
	var step = 10;
	if(score){
		if(score==100){
			range=100;
		}
		else{
			range=(Math.floor(score/step))*step+'-'+((1+Math.floor(score/step))*step-1);
		}
	}
	return range;
}
