                // by using hidden rather than display:none/block etc, we can inspect the current display state to
                // see if the responsiveness has actually hidden it
                changeButtonWrapper.removeClass('hidden');
                changeButtonWrapper.find('input').click(function (e) {
                    e.preventDefault();
                    changeButtonWrapper.addClass('hidden');
                    toggle();
                });

                $('#tag').change(function () {
                    if (changeButtonVisible) {
                        $("#tag").closest("form").submit();
                    }
                });

                toggle();
                window.addEventListener('resize', toggle);
            });
