    function initLocationAutocomplete() {
        const
            service = new google.maps.places.AutocompleteService(),
            $form = $('form[data-places-search]'),
            $searchField = $form.find('input[name="term"]'),
            $suggestionsList = $form.find('ul[data-places-search-suggestions]');
        $suggestionsList.hide();

        const displaySuggestions = function(predictions, status) {
            if (status !== google.maps.places.PlacesServiceStatus.OK) {
                return;
            }

            $suggestionsList.html('');
            $suggestionsList.css('min-height', '150px');
            predictions.forEach(function(prediction) {
                const suggestion = prediction.description.replace(', Deutschland', '');

                $suggestionsList.append("<li>" + suggestion + "</li>");
            });
            $suggestionsList.css('min-height', '');
            $suggestionsList.show();
        };

        $searchField.on('keyup', function() {
            if ($searchField.val().length >= 4) {
                const options = {
                    input: $searchField.val(),
                    types: ['geocode'],
                    componentRestrictions: {country: 'de'}
                };
                service.getPlacePredictions(options, displaySuggestions);
            } else {
                $suggestionsList.hide();
            }
        });

        $suggestionsList.on('click','li', function() {
            $searchField.val($(this).text());
            $form.submit();
        });
    }
