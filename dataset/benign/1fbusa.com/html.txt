function exchangeFunds(obj){
   var sessionActive = checkSession();
   if (sessionActive) {
      var showExchangeFundsTerms = checkExchangeFundsTermsReconsentRequired();
      var showExchangeFundsTermsArray = showExchangeFundsTerms.split("|");
      var wallet = obj.attr('href').split("#");
      if (showExchangeFundsTermsArray[0] == 'true'){

         // open exhange funds terms modal
         $('#modalShell').modal('show');
         $("#modalShell .modal-content").empty();
         $("#modalShell .modal-content").load('/fscm/prepaid/managewallets/viewExchangeFundsTerms.mvc' + '?wallet=' + wallet[1]);
         if ($('#torefresh').length > 0) {
            $('#torefresh').val('no');
         }
      } else if (showExchangeFundsTermsArray[0] == 'false') {
         if (showExchangeFundsTermsArray[1] == 'true'){
            acceptExchangeFundsTerms();
         }
         // Navigate to exhange funds step 1 page
         var destinationUrl = '/fscm/prepaid/managewallets/exchangeFundStep1.mvc';
         window.location.href = destinationUrl;
      } else {
         $("#manageWalletError").show();
           $("#manageWalletError").html(showExchangeFundsTerms);
           if($(".alert").find('.attentionIconADA:first').length == 1){
       			$(".alert").find('.attentionIconADA:first').prepend('<p class="iconError"><img src="/fscm/page_rwd/images/icon_alert.gif\" alt="" title="Attention"><span>Attention</span></p>');
       		}
      }
   }
   else {
      //on session time out refresh the same page, which will redirect to index.do page.
	   var reloadExchangeFundsUrl = window.location.href;
       window.location.href = reloadExchangeFundsUrl.replace('#', '');
   }
   return false;
}

function checkBillPayeeTerms(originMethod){
   var combinedTermsValues="";
   jQuery.ajax({
        url: '/fscm/prepaid/services/ajaxCallToCheckLegalTerms.mvc',
        type: "POST",
        data: 'originMethod=' + originMethod,
        async: false,
        success : function (result) {
          var JSONobj = jQuery.parseJSON(result);
          var showTerms ="";
          var minorVersion ="";
          var ppcTermsError ="";
             $.each(JSONobj, function(errorField, errorMessage) {
                if(errorField == 'showPPCBillPayTerms'){
                   showTerms =  errorMessage;
                }else if (errorField == 'billPayMinorVersionChange'){
                   minorVersion =  errorMessage;
                }else{
                   ppcTermsError = errorMessage;
                }
               });
             combinedTermsValues = showTerms + "|" + minorVersion + "|" + ppcTermsError;
        },
        error: function() {
         combinedTermsValues = "error";
        }
        });
    return combinedTermsValues;
}
