    var app = new Vue({
        el: '#enhanced_products_bottom',
        data: {
            type: 'bottom',
            products: [],
            loading: true,
            page: 0,
            per_page: 8        },
        computed: {
            product_list: function() {
                return this.products.slice(this.page * this.per_page, (this.page * this.per_page) + this.per_page)
            }
        },
        methods: {
            load_products: function(callback) {
                var _this = this;

                $.getJSON("/meska/enhanced_products_json?type=top", {
                    type: _this.type
                }, function(resp) {
                    _this.loading = false;
                    for (var i = 0; i < resp['products'].length; i++) {
                        _this.products.push(resp['products'][i]);
                    }
                    if (callback !== undefined) {
                        callback()
                    }
                });
            },
            pageNext: function() {
                var _this = this;
                this.load_products(function() {
                    _this.page += 1;
                })
            },
            pagePrev: function() {
                var _this = this;
                if (_this.page < 1) {
                    return;
                }
                this.load_products(function() {
                    _this.page -= 1;
                })
            },
            productImagePath(image, webp = false) {
                if (image == undefined) {
                    return "";
                }
                if (image.length < 1) {
                    return "";
                }
                
                if (webp){
                    image = image.substring(0, image.indexOf('.'));
                    image = image + '.webp';
                }

                return (
                    "https://www.meska.hu/img/product/small" +
                    "/" +
                    image
                    .split("")
                    .slice(0, 2)
                    .join("/") +
                    "/" +
                    image
                );
            }
        },
        mounted: function() {
            this.load_products();

            var _this = this;

                setInterval(function(){
