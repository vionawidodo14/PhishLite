function toggleDesktopSlides() {
	var elem = getobj('undermenu');
	if(elem.dataset.state == "inactive") {
		elem.style = "top:18vh;"; 
		elem.dataset.state="active";
		getDefaultMenu();
		
	} else { 
		var grabber = getobj('slidecont');
		grabber.innerHTML = '';
		elem.style.top = "-500px";		
		elem.dataset.state="inactive"; 
	}
}
function getDefaultMenu() {
	var grabber = getobj('slidecont');
	var defaultMenu = document.createElement('div');
	defaultMenu.classList.add('menuSlides');
	grabber.appendChild(defaultMenu);

	$.ajax({
		type: "POST",
		url: "https://www.garagegymreviews.com/model/getNavContents.php?mode=dekstop",
		dataType:"json", 
		success: function(result){
			defaultMenu.innerHTML = result;
		}		
	});
}

	var send = elem.dataset.id;

		dataType:"json", 
		success: function(result){
			var newSlide = document.createElement('div');
			defaultMenu.classList.add('menuSlides');
			grabber.appendChild(newSlide);
		}		
	});
}

}
