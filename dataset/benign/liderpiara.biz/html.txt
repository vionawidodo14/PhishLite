	function updateNum(id, type)
	{
		var parse = $(id).text();
		var rez = 0;
		if (type) { //прибавляем
			if (parse === '') {
				$(id).text('1');
			} else {
				if(parse !== '+99'){
					rez = parseInt(parse) + 1;
					var itog = rez>99?'+99':rez;
					$(id).text(itog);
				}
			}
		} else { //вычитаем
			if (parse === '' || parse === '1') {
				$(id).text('');
				return false;
			} else {
				if(parse !== '+99'){
					rez = parseInt(parse) - 1;
					$(id).text(rez);
				}
			}
		}
		return true;
	}
	function getCookie(name) {
		var matches = document.cookie.match(new RegExp(
			"(?:^|; )" + name.replace(/([\.$?*|{}\(\)\[\]\\\/\+^])/g, '\\$1') + "=([^;]*)"
		));
		return matches ? decodeURIComponent(matches[1]) : undefined;
	}
