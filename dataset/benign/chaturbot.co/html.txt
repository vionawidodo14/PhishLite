		function switchSort(sortBy) {
			$("#sort_time").removeClass("activeicon");
			$("#sort_alpha").removeClass("activeicon");
			
			if (sortBy == 'time') {
				$("#sort_time").addClass("activeicon");
			} else {
				$("#sort_alpha").addClass("activeicon");
			}
			
			var ul = $("#usermodels");
			var lis = ul.children("li");
			
			if (sortBy == 'time') {
				lis.detach().sort(function(a, b){
					return $(b).attr("sortTime") - $(a).attr("sortTime");
				});
			} else {
				lis.detach().sort(function(a, b){
					return $(a).attr("sortName").localeCompare($(b).attr("sortName"));
				});
			}
			ul.append(lis);
		}
