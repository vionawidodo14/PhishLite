                    max: max
                },
                start: [selectedMinIndex, selectedMaxIndex],
                format: {
                    to: function(value) {
                        return getLabel(value);
                    },
                    from: function(value) {
                        return parseInt(value);
                    }
                }
            });

            sliderContainer.noUiSlider.getSelectedSalary = function() {
                return {
                    min: steps[selectedMinIndex],
                    max: steps[selectedMaxIndex],
                    hasSelection: selectedMinIndex > min || selectedMaxIndex < max
                };
            };

            sliderContainer.noUiSlider.resetSelectedSalary = function() {
                slider.set([min, max]);
            };

            updateLabels(selectedMinIndex, selectedMaxIndex);

            slider.on('update', function(labels, index, values) {
                updateLabels(parseInt(values[0]), parseInt(values[1]));
            });

            slider.on('change', function(labels, index, values) {
                var lastMin = selectedMinIndex, lastMax = selectedMaxIndex;
                selectedMinIndex = parseInt(values[0]);
                selectedMaxIndex = parseInt(values[1]);

                if (selectedMinIndex !== lastMin || selectedMaxIndex !== lastMax) {
                    triggerGlobalSliderChangedEvent(selectedMinIndex, selectedMaxIndex);
                }
            });
        });

        function updateLabels(min, max)
        {
            jQuery('#salarySliderexample .nouislider-labels > div:last-child').text(getLabel(min));
            jQuery('#salarySliderexample .nouislider-labels > div:first-child').text(getLabel(max));

            if (minSalaryInput && maxSalaryInput) {
                minSalaryInput.value = getAmount(min);
                maxSalaryInput.value = getAmount(max);
            }
        }

        function triggerGlobalSliderChangedEvent(min, max)
        {
            // this triggers a namespaced salarySliderChanged event
            // depending code should attach to this
            jQuery(window).trigger('salarySliderChanged.example', {
                minSalary: getAmount(min),
                maxSalary: getAmount(max)
            });
        }

        function getLabel(index)
        {
            return steps[index] ? steps[index].label : '';
        }

        function getAmount(index)
        {
            return steps[index] ? steps[index].amount : '';
        }
    })(document, window, jQuery);

<script type="text/javascript">
    function submitPasswortReminder()
    {
        var form = jQuery('#lostPasswordForm');

        if (form.data('requestRunning')) {
            return;
        }

        form.data('requestRunning', true);

        form.find('#user_name').popover('dispose').removeClass('form-control-danger').parent().removeClass('has-danger');

        jQuery.ajax({
            type: "POST",
            url: '/get-password',
            data: form.serialize(),
            complete: function() {
                form.data('requestRunning', false);
            },
            success: function( result ) {
                var stat = result.statusCode;

                if (stat==0 || stat==1 || stat==2) {
                        jQuery('#standardText').hide();
                        jQuery('#companyOrTalentscout').hide();
                        jQuery('#successText').hide();
                        jQuery('#errorText').hide();
                }

                if (stat==0) {
                        jQuery('#successText').show();
                }

                if (stat==1) {
                        jQuery('#standardText').show();

                        form.find('#user_name').popover({
                            animation: false,
                            html: true,
                            content: result.msg,
                            placement: 'top',
                            trigger: 'focus',
                            template: '<div class="popover popover-danger" role="tooltip"><div class="popover-arrow"></div><h3 class="popover-title"></h3><div class="popover-content"></div></div>'
                        }).focus().addClass('form-control-danger').parent().addClass('has-danger');
                }

                if (stat==2) {
                        jQuery('#companyOrTalentscout').show();
                }

                if (jQuery.fn.colorbox) {
                    jQuery.colorbox.resize();
                }
            }
        });

        form.find('#user_name').off('keypress');
        form.find('#user_name').on('keypress', function() {
            if (form.find('#user_name').hasClass('form-control-danger')) {
                form.find('#user_name').popover('dispose').removeClass('form-control-danger').parent().removeClass('has-danger');
            }
        });
    }
</script>
