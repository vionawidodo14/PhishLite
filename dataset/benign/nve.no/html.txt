<script>	
function resizeImages() {
$('.news-ticker-entry-body img').each(function() {
	var src = $(this).attr('src');
	
	var sImg = $(this).data('sImg');
	var lImg = $(this).data('lImg');
	
	if(typeof lImg == 'undefined'){
		$(this).data('sImg', src);
		var strLargerImg = src.replace(/width=([0-9]*)/, 'width=300').replace(/height=([0-9]*)/, 'height=200');
		$(this).data('lImg', strLargerImg);
		if ($(window).width() < 480) {
			$(this).attr('src', strLargerImg);
		}
	} else {	
		if ($(window).width() < 480 && src != lImg) {
			$(this).attr('src', lImg);
		} else if ($(window).width() >= 480 && src != sImg){
			$(this).attr('src', sImg);
		}
	}
});	
}; resizeImages();  
var resizeId;
//http://alvarotrigo.com/blog/firing-resize-event-only-once-when-resizing-is-finished/
$(window).resize(function() {
    clearTimeout(resizeId);
    resizeId = setTimeout(resizeImages, 500);
});
