     function load_resault(resault){
         var tab_history='<ul class="nav nav-tabs">',
             content_history='<div class="tab-content">',
             index=1;
             var id='tab-resault-'+index,
                 keywords=$.parseJSON(resault).keywords;
         tab_history+='<li keywords="'+keywords+'"><a href="#'+id+'" data-toggle="tab">'+keywords+'</a></li>';
         content_history+='<div class="tab-pane" id="'+id+'">'+json2tabs(resault)+'</div>';
         tab_history+='</ul>';
         content_history+='</div>'
         return tab_history+content_history;
     };


     function load_history(){
         var search_history = localStorage.getItem("search-history"),
             tab_history='<ul class="nav nav-tabs">',
             content_history='<div class="tab-content">';
         search_history=$.parseJSON(search_history);
         if (search_history){
         $.each(search_history,function(index,resault){
             index+=1;
             var id='tab-resault-'+index,
                 keywords=$.parseJSON(resault).keywords;
             tab_history+='<li keywords="'+keywords+'"><a href="#'+id+'" data-toggle="tab">'+keywords+'</a></li>';
             content_history+='<div class="tab-pane" id="'+id+'">'+json2tabs(resault)+'</div>';
         });
         tab_history+='</ul>';
         content_history+='</div>'
         return tab_history+content_history;
         } else {
             return "";
         }
     };


     function update_history(jsonresponse){
         var search_history = localStorage.getItem("search-history");
         //console.log(search_history);
         if (search_history){
             search_history=$.parseJSON(search_history);
             search_history.unshift(jsonresponse);
             search_history=search_history.slice(0,10);
             //if (search_history.length>5){search_history=search_history.slice(0,5);};
         } else {
             search_history=[jsonresponse]
         };
         localStorage.setItem("search-history",$.toJSON(search_history));
         return search_history;
     };


     function json2tabs(jsonresponse){
         var data=jsonresponse;
         if ($.type(data) === "string"){
             data=$.parseJSON(data);
         };
         var message=data['message'],
             keywords=data.keywords,
             content=data['content'],
             resault='<div class="message" keywords="'+keywords+'">'+message+'</div>';
         if (content){
             $.each(content,function(tname,trows){
                 var keys=trows['keys'],
                     values=trows['values'],
                     tablename=trows['tablename'],
                     divrows=[];
                 $.each(values,function(index,value){
                     var valuex=[];
                     $.each(value,function(index1,value1){
                         if ($.isArray(value1)){
                             value1=value1.join('</li><li>');
                         };
                         if(value1){valuex.push('<div class="'+keys[index1]+'"><ul><li>'+value1.toString()+'</li></ul></div>');};
                     });
                     divrows.push('<div class="header">'+valuex.join('')+'</div>')
                 });
                 resault+='<div class="'+tname+'">'+'<div>Found in the table: '+tablename+'</div>'+divrows.join('')+'</div>';
             });
         };
         return resault;
     };


     function highlight_keywords(active_key){
         $("div.tab-pane").each(function(indx){
             $(this).unhighlight({element: 'span', className: 'hishighlight'});
             var keywords=$(this).find("div.message").attr("keywords");
             if (keywords) $(this).highlight(keywords,{element: 'span', className: 'hishighlight'});
             });

         $('#side-keywords').bind('keyup', function() {
             jQuery('#side-search-resault').unhighlight({element: 'span', className: 'highlight'});
             var v = jQuery(this).val();
             if (v) jQuery('#side-search-resault').highlight(v,{element: 'span', className: 'highlight'});
             if (is_keywords_in_history(v)){
                 $('li[keywords="'+v+'"]:first>a[data-toggle="tab"]').tab('show');
             };
         });
         if (is_keywords_in_history(active_key)){
                 $('li[keywords="'+active_key+'"]:first>a[data-toggle="tab"]').tab('show');
             };
     };


     function is_keywords_in_history(keywords){
         var search_history = localStorage.getItem("search-history");
         if (search_history){
             search_history=$.parseJSON(search_history);
             var rezault=search_history.some(function (element, index, array) {
                 element=$.parseJSON(element);
                 return element.keywords==keywords;
                 });
             return rezault;
         } else {
             return false;
         }
     };


     $(function() {
         $('#side-search-keywords').submit(function(e) {
             var keywords=$(this).find('#side-keywords').val(),
                 $data={'keywords':keywords},
                 $form = $(this);
             if (window.sessionStorage && window.localStorage) {
                  //объекты sessionStorage и localtorage поддерживаются
                 if (is_keywords_in_history(keywords)){
                     $("#side-search-resault").html(load_history());
                     highlight_keywords(keywords);
                     } else {
                          $.ajax({
                             type: $form.attr('method'),
                             url: "/search/resault1",
                             data: $data,
                             dataType:'html',
                             success: function(response) {
                                 update_history(response);
                                 $("#side-search-resault").html(load_history());
                                 highlight_keywords(keywords);
                             },
                             error: function(response) {
                                 $("#side-search-resault").html(response);
                             }
                         });
                     };
                }
                else {
                  //объекты sessionStorage и localtorage не поддерживаются
                    $.ajax({
                             type: $form.attr('method'),
                             url: "/search/resault1",
                             data: $data,
                             dataType:'html',
                             success: function(response) {
                                 $("#side-search-resault").html(load_resault(response));
                                 highlight_keywords(keywords);
                             },
                             error: function(response) {
                                 $("#side-search-resault").html(response);
                             }
                         });
                };
             //отмена действия по умолчанию для кнопки submit
             e.preventDefault();
         });
     });


     $('#searchModalBox').on('show.bs.modal', function() {
         $("#side-search-resault").html(load_history());
         highlight_keywords();
         $('a[data-toggle="tab"]:first').tab('show');
         });
</script>
