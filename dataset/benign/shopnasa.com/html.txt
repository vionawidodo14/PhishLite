          var cNav = $(this).attr('id');
          cNav = '#' + cNav.replace('-nav','');
          $(cNav).addClass('active');
        } 
      });
      
      //clear out after nav drops the first time
      $('.drop-box-container').mouseleave(function(){
       	$('.drop-box-container').removeClass('active');
      });
      
      //First Level Drop Swap
	  $('.first-level-drop .first-level').mouseenter(function() {      
        
      	//drop the other classes
      	$('.first-level-drop .first-level').removeClass('active');
        
        //add class to this element
        $(this).addClass('active');
        
        //find the second-level nav and add drop
        var cFirstLevel = '#' + $(this).data('drop');
        $(this).closest('.drop-box-container').find(cFirstLevel).addClass('active');

        //change the backgroundout 
        var dropBack = 'https://cdn.shopify.com/s/files/1/2120/7043/files/nasa-' + $(this).data('drop') + '.jpg?ver=92929292929';
        dropBack = 'background-image: url( ' + dropBack + ');'        
        $(this).closest('.drop-box-container').find('#background-drop').attr('style', dropBack);
     
        //Swap out the collection boxes
        var cBox = $(this).data('drop');
        $('#nasa-drop .drop-collection').removeClass('active');
        $('#nasa-drop #' + cBox).addClass('active');
      });
      
      
   
    });
