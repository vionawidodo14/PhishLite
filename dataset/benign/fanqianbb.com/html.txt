                        html += '<li class="' + (i == this.currentPage ? 'active' : '') + '" page="' + i + '">' + i + '</li>'
                    }
                }
                html += '<li class="pagenation1-next ' + (this.currentPage == this.totalPageNumber ? "disabled" : "") + '">下一页</li>';
                return html;
            },
            handle: function() {
                if (this.listObjs[this.currentPage]) {
                    this.changePage();
                } else {
                    this.getLists();
                }
            }
        }
        pagenation.init();
    })
