    function calcC(origin, calc) {
	    if (calc == 1) {
	        
			var userin = $('#c' + origin).val();
			
			var equivalent = arrCurrency[origin].factor;

			userin = parseFloatWithCultureFormatting(userin);
			if (isNaN(userin)) {
			    userin = 0;
			}

			for (i = 0; i < arrCurrency.length; i++) {
				if (i != origin) {
					v = (equivalent / arrCurrency[i].factor * userin).formatFloat();
					$('#c' + i).val(v);
				}
			}
			return true;
		}
	}

    function standardizeNumberString(str) {
        /// <summary>Takes a formatted number string and makes it parsable</summary>
        /// <param name="str">The string you wish to parse</param>
        /// <returns>The number string formatted ready for parsing</returns>
        if ("is-IS" == "en-GB") { // no formatting needed this is the standard that js recognises
            str = str.replace(/\,/g, ''); // remove 'period' commas
            return str;
        }
        if ("is-IS" == "is-IS") {
            str = str.replace(/\./g, ''); // remove 'period' commas
        }
        else { // then its Polish, we should remove spaces 
            str = str.replace(/\s/g, ''); // remove 'space' commas
        }
        return str.replace(/\,/, '.'); // replace the comma decimal place with a period
    }
