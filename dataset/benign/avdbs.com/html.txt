function search_log_move(num){
	var	srch_kwd = $("#search_text_"+num).val();
	srch_kwd = srch_kwd.trim();

	if (srch_kwd==""){
		alert("검색어를 입력해주세요.");
		return false;
	}

	if (srch_kwd.length<2){
		alert("검색어를 2글자 이상 입력하세요.");
		return false;
	}

	var url = 
		"/w2017/api/iux_kwd_srch_log.php"
	+	"?op=srch"
	+	"&kwd="+urlencode(srch_kwd);
	av_api.get_text(url, function(raw){
		var j = JSON.parse(raw);
		if (j.cd){
						cookie.set("search_tab", 0, -1);

			var url_srch = 
				"/menu/search.php"
			+	"?kwd="+urlencode(srch_kwd)
			+	"&seq=" + j.seq;

		}
	});
}
