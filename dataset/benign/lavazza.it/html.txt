$(document).ready(function(){
	// track click on item
	$("[data-js-bannervideo], [data-js-videopopup]").each(function() {
		var name = $("[data-js-bannervideo]").data('js-bannervideo');
		if (name == null || name == "") {
			// fallback on videopopup
			name = $("[data-js-videopoup]").attr('href');
		}
		$(this).click(function() {
			try {
				utag.data['video_title'] = name; 
				//utag.data['video_lenght'] = [VIDEO LENGHT] 
				//utag.data['video_category'] = "[VIDEO CATEGORY]" 
				utag.data['event_name'] = 'video_open'; 
				utag.track('video' , utag.data);	
			} catch (e) { console.warn("no video tracking was possible"); }
		});
	});
	// track start/pause events
	$("video").each(function() {
	    try{
            var name = $(this).attr('src');
            $(this).click(function() {
                try {
                    utag.data['video_title'] = name;
                    utag.data['video_lenght'] = $(this).duration;
                    //utag.data['video_category'] = "[VIDEO CATEGORY]"
                    utag.data['event_name'] = 'video_open';
                    utag.track('video' , utag.data);
                } catch (e) { console.warn("no video tracking was possible"); }
            });
            $(this).onplay(function() {
                try {
                    utag.data['video_title'] = name;
                    utag.data['video_time'] = $(this).currentTime;
                    utag.data['event_name'] = 'video_start';
                    utag.track('video' , utag.data);
                } catch (e) { console.warn("no video tracking was possible"); }
            });
            $(this).onpause(function() {
                try {
                    utag.data['video_title'] = name;
                    utag.data['video_time'] = $(this).currentTime;
                    utag.data['event_name'] = 'video_pause';
                    utag.track('video' , utag.data) ;
                } catch (e) { console.warn("no video tracking was possible"); }
            });
        } catch (e) {
          console.warn("error during track video")
        }
	});
});
