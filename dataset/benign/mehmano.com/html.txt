        function getOrders() {
            var req = $.ajax({
                method: "GET",
                url: "ajax/packing/get_orders_light.php"
            });

            req.done(function(res) {
                if(res=="red") {
                    window.location = "logout.php?ref=index.php";
                } else if(res!="nok") {
                    orders = JSON.parse(res);
                    getCategs();
                }

                else
                    toastr.error('', 'خطایی رخ داده است');
            });

            req.fail(function(jqXHR, textStatus) {
                toastr.error('', 'خطایی رخ داده است');
            });
        }

        function addOrder(id, no, pk) {
            var req = $.ajax({
                method: "POST",
                url: "ajax/packing/add_order.php",
                data: {
                    id: id,
                    no: no,
                    pk: pk
                }
            });

            $('#addBtn_'+id+'_'+pk).html("<i aria-hidden='true' class='fa fa-spinner fa-spin fa-lg fa-fw'></i>");
            req.done(function(res) {
                $('#addBtn_'+id+'_'+pk).html('<i class="fa fa-lg fa-shopping-cart"></i>');
                if(res=="red")
                    window.location = "logout.php?ref=index.php";
                else if(res!="nok") {
                    res = JSON.parse(res);
                    if(res['res']=='ok')
                        $('#snum').html(res['sn']);
                    
                    orders.push(id+'_'+pk);
                    toastr.success('', 'مورد با موفقیت افزوده شد');
                }
                else
                    toastr.error('', 'خطایی رخ داده است');
            });

            req.fail(function(jqXHR, textStatus) {
                $('#addBtn_'+id+'_'+pk).html('<i class="fa fa-lg fa-shopping-cart"></i>');
                toastr.error('', 'خطایی رخ داده است');
            });
        }
