					controller = new ScrollMagic.Controller();
					createSceens();

					controller.destroy(true);
					controller = new ScrollMagic.Controller();
					createSceens();
				}
				catch (error) {
					console.log(error);
					controller.destroy(false);
					controller = new ScrollMagic.Controller();
					createSceens();
				}
			});
		});

		$(window).resize(function () {
			console.log("resize");
			//fixes screens if window is resized
			controller.destroy(true);
			controller = new ScrollMagic.Controller();
			createSceens();
		});

		$(document).ready(function () {
			// init controller, makes sure the sceens are created if the browser has the images cached
			console.log("defualt");

		function createSceens() {
			try {
				console.log("creating screens");
				PanelOneScrollDistance = ($(".scroll-panel-one img").height() - $(".scroll-panel-one").height());
				PanelTwoScrollDistance = ($(".scroll-panel-two img").height() - $(".scroll-panel-two").height());
				PanelThreeScrollDistance = ($(".scroll-panel-three img").height() - $(".scroll-panel-three").height());

				var Offset = ($(".scroll-panel-one").offset().top - 40);
				var Panel = $(".scroll-panel-one .sixteen-by-nine-content");
				createSingleSceen(sceneOne, PanelOneScrollDistance, Offset, Panel);

				var Offset = ($(".scroll-panel-one").offset().top + $(".scroll-panel-one").height() + PanelOneScrollDistance);
				var Panel = $(".scroll-panel-two .sixteen-by-nine-content")
				createSingleSceen(sceneTwo, PanelTwoScrollDistance, Offset, Panel);

				var Offset = ($(".scroll-panel-one").offset().top + $(".scroll-panel-one").height() + $(".scroll-panel-two").height() + PanelOneScrollDistance + PanelTwoScrollDistance);
				var Panel = $(".scroll-panel-three .sixteen-by-nine-content")
				createSingleSceen(sceneThree, PanelThreeScrollDistance, Offset, Panel);
			}
			catch (error) {
				console.log(error);
			}

		}

		function createSingleSceen(Scene, Distance, Offset, Panel) {
			try {
				Scene = new ScrollMagic.Scene({
					duration: Distance,	// the scene should last for a scroll distance given
					offset: Offset	// start this scene after scrolling for 50px
				})
					.setPin(".scroll-panels") // pins the element for the the scene's duration
					.addTo(controller) // assign the scene to the controller
					.on("progress", function (e) {
						var topval = -Distance * e.progress;
						Panel.css('top', topval + 'px');
					});
			}
			catch (error) {
				console.log(error);
			}
		}
