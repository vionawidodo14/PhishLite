var cur_Image = 1;
var timer;
var images_q = 6;

function right() {
    cur_Image += 1;
    if(cur_Image > images_q) cur_Image=1;
    image_switch(cur_Image);
}

function left() {
    cur_Image -= 1;
    if(cur_Image < 1) cur_Image=images_q;
    image_switch(cur_Image);
}

function image_switch(number) {
    $('.thumbnail').removeClass('active_t');
    $('#t'+number).addClass('active_t');
    $('.image').fadeOut(1000);
    $('.header-text').fadeOut(1000);
    $('.left-part').fadeOut(1000);
    $('#i'+number).fadeIn(1000);
    $('#ht'+number).fadeIn(1000);
    $('#lt'+number).fadeIn(1000);
    cur_Image = parseInt(number);
}

function scroll() {
    right();
    timer = setTimeout(function () { scroll(); }, 5000)
}

function scroll_pause() {
    clearTimeout(timer);
    timer = setTimeout(function () { scroll(); }, 10000)
}
