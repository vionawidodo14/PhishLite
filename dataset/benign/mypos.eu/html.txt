    $(document).ready(function(){
        updateGetAQuoteFormButton();
    });

    $(document).on('click', '#iAgreeWithPPGetAQuote', function(){
        updateGetAQuoteFormButton();
    });

    function updateGetAQuoteFormButton(){
        if ($('#iAgreeWithPPGetAQuote').is(':checked')) {
            $('#getAQuoteFormButton').attr('disabled', false);
        } else {
            $('#getAQuoteFormButton').attr('disabled', true);
        }
    }

    $(document).on('submit', 'form#js-get-a-quote-form', function (e) {
        if (!$('#iAgreeWithPPGetAQuote').is(':checked')) {
            e.preventDefault();
        }
    });

    (function () {
        const process_url = "https://www.mypos.eu/en/get-a-quote/process/";

        const errorClass = 'alerted';
        const errorContainer = $('div#comp_error');
        const errorText = $('div#err_text');

        let form = document.getElementById('js-get-a-quote-form');
        let sendDataAndFiles;

        let handleSubmitFunction = function (event) {
            event.preventDefault();

            sendDataAndFiles();
        };

        if (form !== null) {
            form.addEventListener('submit', handleSubmitFunction);
        }

        sendDataAndFiles = function () {
            let form = document.getElementById('js-get-a-quote-form');
            let formData = new FormData(form);
            let request = new XMLHttpRequest();

            // clear errors
            clearErrors();

            $('#getAQuoteFormButton').attr('disabled', true);
            showPreloader();

            request.open("POST", process_url);
            request.onreadystatechange = function() {
                if (request.readyState === 4 && request.status === 200) {
                    let response = JSON.parse(request.responseText);

                    if (response.success) {
                        // hide form and show success
                        showSuccess();
                    } else {
                        // map errors and show them
                        showErrors(response.fields);
                        showErrorMessage(response.message);
                        $('#getAQuoteFormButton').attr('disabled', false);
                        hidePreloader();
                    }

                    regenerateCaptcha();
                }
            };
            request.send(formData);
        };

        function showErrors(fields) {
            let inputFields = ['company_name', 'city', 'address', 'name', 'email', 'phone', 'business_type_description', 'captcha'];
            let textareaFields = [];
            let selectFields = ['is_taking_card_payments', 'annual_turnover', 'average_transaction_amount'];
            let selectChosenFields = ['country', 'phone_code'];

            $.each(fields, function(index, fieldName) {
                // if is input
                if (inputFields.indexOf(fieldName) > -1) {
                    $('input[name="' + fieldName + '"]').addClass(errorClass);
                } else if (textareaFields.indexOf(fieldName) > -1) {
                    $('textarea[name="' + fieldName + '"]').addClass(errorClass);
                } else if (selectFields.indexOf(fieldName) > -1) {
                    $('select[name="' + fieldName + '"]').addClass(errorClass);
                } else if (selectChosenFields.indexOf(fieldName) > -1) {
                    $('select[name="' + fieldName + '"]').parent().find('.chosen-container').find('.chosen-single').addClass(errorClass);
                }
            });
        }

        function showSuccess() {
            $('div#js-form-get-a-quote').hide();
            $('div#js-form-get-a-quote-success').show();
        }

        function showErrorMessage(message) {
            errorContainer.show();
            errorText.html('<div>' + message + '</div>');
        }

        function resetErrorMessage() {
            errorContainer.hide();
            errorText.html('');
        }

        function clearErrors() {
            $('input, textarea, label, select, .chosen-single').removeClass(errorClass);
            resetErrorMessage();
        }

        function regenerateCaptcha() {
            var captchaImageId = 'captchaImageGetAQuote';
            var captchaInputId = 'captchaInputGetAQuote';

            var captchaImage = $('#' + captchaImageId);
            var captchaInput = $('#' + captchaInputId);
            var url = captchaImage.attr('src').split('?')[0];

            captchaImage.attr('src', url + '?' + Date.now());
            captchaInput.val('');
        }

        $(document).ready(function() {
            if ($(".takeCardPaymentWrap").length) {
                $(".showIfTakeCardPayment").hide();

                $(".takeCardPaymentSelect").change(function() {
                    showIfTakeCardPayment();
                });

                showIfTakeCardPayment();
            }
        });

        function showIfTakeCardPayment() {
            if ($(".takeCardPaymentSelect").find(":selected").val() == 1) {
                $(".showIfTakeCardPayment").show();
            } else {
                $(".showIfTakeCardPayment").hide();
            }
        }
    })();
