			function validateForm(containerId) {

				var getVisibleRequiredFields = function(containerId) {

					var fieldArr = document.getElementById(containerId).querySelectorAll('.required');
					var field = [];

					for (var i = 0; i < fieldArr.length; i++) {
						if (fieldArr[i].offsetHeight > 0) {
							field.push(document.getElementById(fieldArr[i].id));
						}
					}

					return field;

				};

				var validateRequiredFields = function(fields) {
					if (fields) {
						var valid = true;
						for (var i in fields) {
							if (!validateRequiredField(fields[i])) {
								valid = false;
							}
						}
						return valid;
					}
				};

				var validateRequiredField = function(field) {
					var emailVar = false;

					if (field.id === 'Email_Address') {
						for (var i = 0; i < field.value.length; i++) {
							if (field.value[i] === '@') {
								emailVar = true;
								break;
							}
						}
						if (!emailVar) {
							showInvalidMessageFor(field);
							return false;
						} else {
							removeInvalidMessageFor(field);
							return true;
						}
					} else {

						if (!(field.value && field.getAttribute("placeholder") !== field.value)) {
							showInvalidMessageFor(field);
							return false;
						} else {
							removeInvalidMessageFor(field);
							return true;
						}
					}
				};

				var showInvalidMessageFor = function(field) {
					if (!isAlreadyInvalidMessageFor(field)) {
						var message = field.title;
						var element = document.createElement('p');
						element.className = 'error';
						element.innerText = message;
						field.parentNode.appendChild(element);
					}
				};

				var removeInvalidMessageFor = function(field) {
					if (isAlreadyInvalidMessageFor(field)) {
						var parent = field.parentNode;
						var message = parent.getElementsByTagName('p')[0];
						parent.removeChild(message);
					}
				};

				var isAlreadyInvalidMessageFor = function(field) {
					var parent = field.parentNode;
					var message = parent.getElementsByTagName('p')[0];
					return !!message;
				};

				var visibleFields = getVisibleRequiredFields(containerId);
				return validateRequiredFields(visibleFields);
			}
