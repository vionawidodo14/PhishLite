function gdpr_consent()
	{
	if (typeof(consent_status) == 'undefined' || consent_status == null)
		{
		return;
		}
	
	//DETERMINE CONSENT STATUS
	var answers = {'cookies': false,'data_use': false,'adsense':false,'terms':false};
	for (var sub in answers)
		{
		if (typeof(consent_status[sub]) != 'undefined' && typeof(consent_status[sub]['consent_date']) != 'undefined' && typeof(consent_status[sub]['consent_response']) != 'undefined' && consent_status[sub]['consent_date'] != false && parseFloat(consent_status[sub]['consent_date']) != 0)
			{
			if (parseFloat(consent_status[sub]['consent_date']) >= sb_consent_effective_date)
				{
				answers[sub] = consent_status[sub]['consent_response'];
				}
			}
		}
	
	//SHOW OR HIDE SUBJECTS AS APPLICABLE
	if (answers['cookies'])
		{
		$('#consent_member_cookies').remove();
		}
	
	//SHOW APPLICABLE POPUP
	var cookie_enable_popup = true;
	if (document.location.href.match(/\/system\/profile\.php/) || document.location.href.match(/\?page=privacy-settings/))
		{
		cookie_enable_popup = false;
		}
	
	var popup_type = 'visitor';
	if (registered_user && (answers['cookies'] != 'consent' || answers['data_use'] != 'consent' || answers['terms'] != 'consent'))
		{
		popup_type = 'member';
		$('#consent_blocker').fadeIn(500);
		$('#consent_member').fadeIn(500);	
		}
	else if (answers['cookies'] != 'consent' && cookie_enable_popup)
		{
		$('#consent_visitor').fadeIn(500);
		}
	else if (answers['adsense'] == false && !adsense_exempt && adsense_enable_popup)
		{
		$('#consent_adsense').fadeIn(500);
		}
	
	
	//DO ACTIONS IF CONSENT
	if (answers['cookies'] == 'consent')
		{
		console.log('Enabled: Cookies');
		ga_cookie_expire = 63072000;
		}
	if (answers['adsense'] == 'consent' || answers['adsense'] == 'non-personal' || (answers['cookies'] == 'consent' && !adsense_enable_popup))
		{
		console.log('Enabled: Adsense');
		(adsbygoogle=window.adsbygoogle||[]).pauseAdRequests=0;
		}
	
	
	//PRINT CURRENT CONSENT OBJECT
	console.log('Consent Status:');
	console.log(consent_status);
	
	}

function save_consent(subject,response,elemid,parentid,method)
	{
	if (typeof(method) == 'undefined' || method == null)
		{
		method = 'popup';
		}
	
	var consent_url = '../home/index.consent.php?subject='+subject+'&method='+method+'&action='+response;
	
	$('#'+elemid).css('opacity','0.65');
	$('#'+elemid).css('cursor','progress');
	$('#'+elemid+' .consent_button').css('cursor','progress');
	$('#'+elemid+' a').css('cursor','progress');
	
	$.ajax({ url: consent_url, dataType: "json", timeout: 10000 }) //10 SECOND TIMEOUT
			.done(function(data)
				{
				//console.log('Updated Consent:');
				//console.log(data);
				
				consent_status = data;
				
				if ($('#'+parentid+' .consent_message').length == 1 || subject == 'adsense')
					{
					$('#'+parentid).fadeOut(500);
					$('#consent_blocker').fadeOut(500);
					setTimeout(gdpr_consent,650);
					}
				else
					{
					$('#'+elemid).slideUp(250,function () {$('#'+elemid).remove();});
					}
				})
			.fail(function(data, status)
				{
				$('#'+elemid).css('opacity','1.0');
				$('#'+elemid).css('cursor','default');
				$('#'+elemid+' .consent_button').css('cursor','pointer');
				$('#'+elemid+' a').css('cursor','pointer');
				old_alert('Error: Could not communicate with the server! Your consent has not been saved.');
				});
	}

$(document).ready(function()
	{
	setTimeout(gdpr_consent,500);
	});
