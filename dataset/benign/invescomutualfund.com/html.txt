            $('#btnSubmit').on('click', function () {
                var ctrName = $('[id$=txtName]');
                var ctrMobile = $('[id$=txtMobile]');
                var ctrEmail = $('[id$=txtEmailAd]');
                var ctrAddress = $('[id$=txtAddress]');
                if (!validateEmptyControl(ctrName, 'Incomplete Information!!\nEnter your name'))
                    return false;
                if (!validateEmptyControl(ctrMobile, 'Incomplete Information!!\nEnter Mobile number'))
                    return false;
                if (ctrMobile.val().length < 10) {
                    alert("Incorrect Information!!\nPlease enter valid Mobile number"); ctrMobile.focus(); return false;
                }
                if (!validateEmptyControl(ctrEmail, 'Incomplete Information!!\nEnter Email Id'))
                    return false;
                if (ctrEmail.val().length < 6) {
                    alert("Incorrect Information!!\nPlease enter valid email"); ctrEmail.focus(); return false;
                }
                else {
                    var emailReg = /^([\w-\.]+@([\w-]+\.)+[\w-]{2,6})?$/;
                    if (!emailReg.test(ctrEmail.val())) {
                        alert("Incorrect Information!!\nPlease enter valid email"); ctrEmail.focus();
                        return false;
                    }
                }

                if (!validateEmptyControl(ctrAddress, 'Enter Address'))
                    return false;
                // $('.overlay').show();



                var data = new FormData();
                data.append("Name", ctrName.val().trim());
                data.append("EmailId", ctrEmail.val().trim());
                data.append("MobileNo", ctrMobile.val().trim());
                data.append("Address", ctrAddress.val().trim());

                $.ajax({
                    type: 'POST',
                    url: '/api/Advisor/SaveAdvisorLead',
                    contentType: false,
                    processData: false,
                    data: data,
                    success: function (data) {
                        if (data == true) {
                            $('#myThanksModal').show();
                            //var popup = alert('Thank You for submitting your details.\nOur representative will get in touch with you shortly.');
                            $(".ThankYouMessage").html("Thank You for submitting your details.\nOur representative will get in touch with you shortly.");
                            $('.overlay5').show();
                            $('body').removeClass('modal-open');
                            $('.modal-backdrop.fade').removeClass('in');
                            $('body .fade').removeClass('modal-backdrop');
                            $("#homepopDiv").hide();
                            $('#myThanksModal').hide();
                            //setInterval(function () { popup.close(); }, 2000);
                            popup.close();
                            setTimeout(function () {
                                window.close();
                            }, 200);

                        }
                        else {
                            //alert('User alreay exists!!!');
                        }
                        ctrName.val("");
                        ctrMobile.val("");
                        ctrEmail.val("");
                        ctrMobile.val("");
                        ctrAddress.val('');
                        //$(".ThankYouMessage").html("Thank You for submitting your details. Your request will be processed in next 2-3 working days.");
                        //$('.overlay2').show();
                    },
                    error: function () {
                    },
                    cache: true
                });

            });
