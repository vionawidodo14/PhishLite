<h1 class="start" style="padding-top:25px;">Führerschein in Sicht - Fahrschule aussuchen</h1>
<p>Sie wollen <a class="menu" href="/Verkehrsrecht/Fuehrerschein/Fuehrerschein_Fahrerlaubnisklassen.html" title="Übersicht der Führerscheinklassen in Deutschland">einen Führerschein machen</a> und wollen wissen wann Sie mit der <a class="menu" href="/fahren_lernen/Tipp11index.html" title="Mindestalter zum Erwerb eines Führerscheins">Fahrausbildung beginnen</a> können? Sie wollen den <a class="menu" href="/fahren_lernen/Tipp33index.html" title="Begleitetes Fahren für Klasse B mit 17 Jahren">Pkw-Führerschein mit 17</a> machen und am begleiteten Fahren teilnehmen? Sie <a class="menu" href="http://www.fahrschulen.de" title="Finden Sie die passende Fahrschule in einer Datenbank mit über 10.000 Fahrschulen">suchen eine Fahrschule</a> und wollen wissen, <a class="menu" href="https://www.fahrschulen.de/PREISE/" title="Führerschein Kosten und Preise vergleichen">wieviel die Fahrausbildung kostet</a> und welche <a class="menu" href="/fahren_lernen/Tipp24index.html" title="Lohnt sich eine spezielle Führerschein Finanzierung?">Möglichkeiten der Finanzierung</a> es gibt? Sie suchen <a class="menu" href="/fahren_lernen/" title="Infos und Tipps für Fahrschüler und Fahranfänger">Tipps zur Fahrausbildung</a> oder zur <a class="menu" href="/fahren_lernen/Tipp2index.html" title="Wie finde ich die richtige Fahrschule für mich?">Auswahl der richtigen Fahrschule</a>. Sie wollen endlich die Freiheit auf zwei Rädern erleben und einen <a class="menu" href="/fahren_lernen/Tipp7index.html" title="Tipps zum Motorradführerschein">Motorradführerschein</a> machen? Hier können Sie Ihre <a class="menu" href="https://www.fahrschulforum.de/" title="Führerschein und Fahrschulforum">Fragen zur Führerscheinausbildung</a> stellen.<br/></p>

Diskutieren Sie in unserem <a class="menu" href="https://www.fahrschulforum.de/" title="Diskussionsforum zu den Themen Fahrschule und Führerschein">Fahrschulforum</a> mit anderen Fahrschülern und Fahrlehrern. Hier gibt es bereits Antworten auf die meisten Fragen.<br/></p>
<h2 class="start">Geschwindigkeit, rote Ampel, Alkohol, Drogen, ..., Führerschein weg</h2>
<p>Sie haben sich (mit Ihrem <a class="menu" href="fahren_lernen/Tipp3index.html" title="Nachschulung mit einem Aufbauseminar für Fahranfänger (ASF)">Führerschein auf Probe</a>) nicht an die <a class="menu" href="Verkehrsrecht/StVO/" title="Komplette und laufend aktualisierte Straßenverkehrsordnung (StVO)">Straßenverkehrsordnung (StVO)</a> gehalten und wollen im <a class="menu" href="Verkehrsrecht/Bussgeld/" title="Auszug aus dem Bußgeldkatalog und Punktekatalog">Bußgeldkatalog</a> nachsehen, was jetzt passiert? Sie haben <a class="menu" href="fahren_lernen/Tipp4index.html" title="Fahreignungsseminar (FES) zum Punkteabbau im Fahreignungsregister">zu viele Punkte im Fahreignungsregister</a> und wollen diese abbauen? Ihr Führerschein ist weg und jetzt müssen Sie zur <a class="menu" href="MPU_Beratung/" title="Medizinisch-Psychologische Untersuchung (MPU) nach Entzug der Fahrerlaubnis">MPU</a> und suchen einen auf <a class="menu" href="Verkehrsrecht/Rechtsanwalt/frmSearch.asp" title="Anwaltsdatenbank">Verkehrsrecht spezialisierten Anwalt</a>?<br/></p>

<script language="JavaScript" type="text/javascript">
const ipRemote = "185.217.117.52";
var plz,city,country	
plz = "";
city = "";
country = "";

var bValidAdsPages = true;
var bValidBLPages = false;
var getCurrentPosition = false;



function fetchGeoIPFree( callback ) {
	console.log("Fetching data from GeoIPFree");
	
	// we skip fetchGeoIPFree atm.
	if (callback) callback(plz,city,ipRemote);
	return;
	
	
	$.ajax({
			type: 'GET',
			dataType: "json",
			url: "https://geoipfree.com/json/"+ipRemote,
			data: {},
			success: function(data) {
				console.log(data);
				if ( plz == "" ) {
					plz = data['zip_code'];
				}
				if ( city == "" ) {
					city = data['city'];
				}
				if ( country == "" ) 
					country = data['country_name'];
				
				if (callback) callback(plz,city,ipRemote)
				
			},
			error: function () { console.log('error'); } 
			}
	);
};


function fetchGoogleMap(latitude, longitude, callback ) {
	
	console.log("Fetching data from GoogleMaps");

	$.ajax({
		type: 'GET',
		dataType: "json",
		url: "https://maps.googleapis.com/maps/api/geocode/json?latlng="+latitude+","+longitude+"&sensor=false",
		data: {},
		success: function(data) {

			$.each( data['results'],function(i, val) {
				$.each( val['address_components'],
						function(i, val) {
						
						if (val['types'] == "postal_code")
							if ( plz == "" ) {
								plz = val['long_name'];
							}

						if (val['types'] == "locality,political")
							if ( city == "" ) {
								city = val['long_name'];
							}
							
						if (val['types'] == "country,political")
							if ( country == "" ) 
								country = val['long_name'];

				});
			});
			console.log('Success');
			
			if (plz!='') {
				if (callback) callback(plz,city,ipRemote)
			} else
				fetchGeoIPFree(callback);
		},
		error: function () { console.log('error'); } 
	}); 
};
	
function displayGeoAd(plz,city,ipRemote) {
	if (bValidAdsPages)
		$("#werbung").load("/werb/geoad.ajax.asp?zipcode="+plz+"&city="+city+"&ip="+ipRemote);
}

function handle_geolocationerror(err) {
    fetchGeoIPFree(displayGeoAd);
}


function initGeoLoc() {
    if (bValidBLPages && navigator.geolocation) {
        navigator.geolocation.getCurrentPosition(function (position) {
            console.log("Got position!");
			getCurrentPosition = true;
			fetchGoogleMap(position.coords.latitude,position.coords.longitude, displayGeoAd );
        },handle_geolocationerror, {maximumAge: 75000});
    } else if (!getCurrentPosition)
		fetchGeoIPFree(displayGeoAd);
}

$(document).ready( function () {    
    initGeoLoc();
});
