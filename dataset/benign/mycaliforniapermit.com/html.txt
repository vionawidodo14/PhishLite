			}

					$('#test').addClass('hover');

					$('#drive').addClass('hover');
				 }, 3000 * (3 + factor) - 1500);
			timeouts.push(to);
			
			if(i == iterations - 1) {
				to = setTimeout(function(){
						$('#drive').removeClass('hover');
					 }, 3000 * (4 + factor) - 1500);
				timeouts.push(to);
			}
		} 
	}
