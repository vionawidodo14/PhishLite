$(document).ready(function() {
	$('#search_box .container .tabs .tab').click(function() {
		$('#search_box .container .tabs .tab').attr('data-show',"false");
		$(this).attr('data-show',"true");
		var index =$(this).attr('data-index');
		$("#search_box .container form").attr('data-show',"false");
		$("#search_box .container form[data-index="+ index +"]").attr('data-show',"true");
	});
	$('.select_style .opts').click(function() {
		if($(this).attr('data-show')=="true"){
			$(this).attr('data-show',"false");
		}else{
			$(this).attr('data-show',"true");
		}
	});

	$('.select_style .opts .opt').click(function() {
		var action = $(this).attr('data-action');
		var href = $(this).attr('data-href');
		var title = $(this).attr('data-title');
		var selectVal = $(this).attr('data-val');
		var text = $(this).text();
		$(this).parent().find(".selected").text(text);
		$(this).parent().parent().find(".link_all").attr("href",href);
		$(this).parent().parent().find(".link_all").text(title);
		$(this).parent().parent().parent().attr("action",action);
		$(this).parent().find('input[type=hidden]').val(selectVal);
	});
});
</script>

<a class="title" data-href="http://khu.ac.ir" target="_blank"><span>دانشگاه خوارزمی</span></a>
<div class="desc">
دانشگاه خوارزمی، نخستین مؤسسۀ آموزش عالی کشور با سابقه ای صدساله است. در 15 شهریور 1297 شمسی نخست باعنوان «دارالمعلمین مركزی» تأسیس شد. در 1307 به «دارالمعلمین عالی»، در 1312 به «دانشسرای عالی»، در 1353 به «دانشگاه تربیت معلم» و در 11 اسفند 1390 به «دانشگاه خوارزمی» تغيير نام يافت.  در 24 اسفند 1393 دانشگاه علوم اقتصادی به دانشگاه خوارزمی الحاق شد. هم اکنون اين دانشگاه دارای 5 معاونت، 16 دانشكده، 4 مؤسسۀ تحقيقاتي، 4 پژوهشكده، 2 قطب علمی و برخوردار از حضور حدود 500 استاد و 13هزار دانشجوست. ساختمان مركزی دانشگاه در تهران، واقع در خيابان شهيد مفتح؛ و پردیس بزرگ آن در حصارک کرج است.
</div>

<a class="title" data-href="http://lh.khu.ac.ir" target="_blank"><span> ادبیات و علوم انسانی</span></a>
<div class="desc">
 دانشکده ادبیات و علوم انسانی، گروه‌های آموزشی: زبان وادبیات فارسی- زبان و ادبیات عربی- زبان‌های خارجه(در رشته ادبیات) و تاریخ - جامعه شناسی - علوم قرآن و حدیث فلسفه - معارف اسلام (در رشته علوم انسانی و اجتماعی) را دارد و چشم انداز آتی آن: ارتقاء رتبه دانشکده به سطح 5 دانشکده اول در حوزه علوم انسانی کشور، تبدیل دانشکده از یک سازمان یاد دهنده صرف به یَژَک سازمان یادگیرنده مستمر، تقویت سنت گفتگو در سطوح مختلف مناسبات میان افراد به عنوان مبنای فهم و عمل مشترک، کاربردی کردن دانش علوم انسانی در جهت شناسایی مشکلات کشور و چاره جویی از آن، رشد و پرورش فرهنگ آکادمیک (دانشگاهی) در سطح دانشکده، گام برداشتن در مسیر بین‌المللی شدن، رعایت اصل انتقادپذیری، حقیقت گرایی، احترام به آزادی آکادمیک و صداقت علمی و تلاش علمی و عملی برای گسترش صلح و عدالت فراگیر است.

<div class="desc">
دانشکده اقتصاد یکی از دانشکده های دانشگاه علوم اقتصادی ، با چند دهه سابقه موثر در آموزش عالی کشور ،‌ بود که پس از ادغام آن در دانشگاه خوارزمی ، همچنان به فعالیت آموزشی دیرینه خود ادامه می دهد. اکنون بیش از 500 دانشجو در مقاطع کارشناسی و کارشناسی ارشد دراین دانشکده مشغول تحصیل بوده و اغلب دانش آموختگان آن در سازمانهای اقتصادی و مالی کشور منشا اثر هستند.

پردیس بین الملل دانشگاه خوارزمی با هدف توسعه آموزش عالی در مقاطع تحصیلات تکمیلی بر اساس قانون ایجاد شده است. پذیرش دانشجو در پردیس بین‌الملل از طریق آزمون سراسری سازمان سنجش صورت می‌گیرد. در حال حاضر پردیس بین‌الملل دانشگاه خوارزمی در رشته‌های مختلف علوم انسانی، فنی و مهندسی و علوم پایه در مقاطع کارشناسی ارشد و دکتری دانشجو می‌پذیرد.

<a class="title" data-href="http://pess.khu.ac.ir" target="_blank"><span>تربیت بدنی و علوم ورزشی</span></a>
<div class="desc">
.دانشکده تربیت بدنی در دو ساختمان در تهران و کرج فعال می باشد. دانشکده در کرج شامل ساختمان آموزشی و اداری و چهار سالن ورزشی، یک استخر سرپوشیده و یک زمین چمن می باشد استخر دانشکده تربیت بدنی جزو اولین استخرهای استاندارد کشور است. همچنین دارای چهار آزمایشگاه تخصصی در رشته های بیومکانیک و حرکات اصلاحی، فیزیولوژی ورزشی و آزمایشگاه رفتار حرکتی و همچنین کتابخانه تخصصی می باشد. ساختمان دانشکده در تهران واقع در مجموعه ورزشی و فرهنگی شهید کشوری شامل ساختمان اداری و آموزشی، سالن ورزشی، کتابخانه تخصصی و مرکز کامپیوتر است و مخصوص دانشجویان کارشناسی ارشد و دکتری می باشد

<div class="desc">
دانشکده حقوق و علوم سیاسی با تصویب هیأت امنای دانشگاه، در سال 1394 از دانشکده ادبیات و علوم انسانی تفکیک گردید و دارای 5 گروه آموزشی حقوق جزا و جرم شناسی ، حقوق خصوصی ،حقوق بین الملل ، علوم سیاسی و روابط بین الملل می باشد. این دانشکده دارای حدود 600 نفر دانشجو در هر سه مقطع و تعداد 27 نفر عضو هیئت علمی که شامل 1 نفر استاد تمام، 7 نفر دانشیار و 19 نفر استادیار می ‌باشد. این دانشکده در راستای توسعه ی فعالیتهای میان رشته ای خود در حال راه اندازی رشته های جدید می باشد.

همزمان با تاسیس دارالمعلمین مرکزی، موضوع‌های تربیتی ویژه‌ای مانند فلسفه، منطق و اصول تعلیم به دانشجویان تربیت معلم ارائه می‌شد و در سال ۱۳۱۲ برای اولین بار موضوع‌هایی مانند تعلیمات متوسطه، تاریخ آموزش و پرورش، اصول آموزش و پرورش، فلسفه آموزش و پرورش، روانشناسی پرورشی و جامعه شناسی به داوطلبان دبیری تدریس شد و آزمایشگاه‌های روانشناسی و علوم تربیتی تأسیس گردید. بدین ترتیب هسته اولیه دانشکده علوم تربیتی با بخش‌های آموزش و پرورش ابتدایی، تربیت بدنی و روانشناسی تشکیل شد. پس از تبدیل دانشسرای عالی به دانشگاه تربیت معلم، به طور رسمی دانشکده‌های علوم تربیتی و مؤسسه تحقیقات تربیتی به وجود آمد.

<div class="desc">
دانشکده شیمی در سال 1375 تاسیس شد و فعالیت آموزشی و پژوهشی خود را ضمن تغییر قالب گروه به دانشکده ادامه داد. در حال حاضر این دانشکده 26 عضو هیئت علمی شامل 4 استاد، 7 دانشیار، 14 استادیار و یک می باشد. دانشکده مشتمل بر هفت گروه آموزشی شیمی فیزیک، شیمی معدنی, شیمی تجزیه، نانوشیمی، شیمی کاربردی، شیمی آلی و پلیمر و فیتوشیمی است. دانشکده در پردیس کرج دارای 2500 متر مربع زیربنا، مشتمل برکلاس های درس، آزمایشگاه های آموزشی و تحقیقاتی، اتاق دستگاه های آنالیز، اتاق کار اعضای هیأت علمی و مدیریت دانشکده است. ساختمان جدید دانشکده در پردیس کرج با زیربنای 5000 متر مربع در حال ساخت می باشد. در تهران نیز دارای زیربنای 800 مترمربع مشتمل 5 آزمایشگاه تحقیقاتی، اتاق های اساتید و دفتر دانشکده است.

    function createCookie(name, value, days) {
        if (days) {
            var date = new Date();
            date.setTime(date.getTime() + (days * 24 * 60 * 60 * 1000));
            var expires = "; expires=" + date.toGMTString();
        } else var expires = "";
        document.cookie = name + "=" + value + expires + "; path=/";
    }
    function readCookie(name) {
        var nameEQ = name + "=";
        var ca = document.cookie.split(';');
        for (var i = 0; i < ca.length; i++) {
            var c = ca[i];
            while (c.charAt(0) == ' ') c = c.substring(1, c.length);
            if (c.indexOf(nameEQ) == 0) return c.substring(nameEQ.length, c.length);
        }
        return null;
    }
