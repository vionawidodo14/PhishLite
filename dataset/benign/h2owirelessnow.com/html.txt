<script>
        function addEvent(obj, evt, fn) {
            if (obj.addEventListener) {
                obj.addEventListener(evt, fn, false)
            }
            else if (obj.attachEvent) {
                obj.attachEvent("on" + evt, fn)
            }
        }

        addEvent(document, "mouseout", function(e) {
            e = e ? e : window.event
            var from = e.relatedTarget || e.toElement
            if (!from || from.nodeName == "HTML") {
                var popupShown = localStorage.getItem('promo.newsletter-popup.has-been-shown')
                // show popup because mouse has left the browser window
                if(!popupShown || popupShown == 'false'){
                    $("#modal-overlay").fadeIn()
                    $("#newsletter_desc").fadeIn()
                    localStorage.setItem('promo.newsletter-popup.has-been-shown', true)
                }
            }
        })

        $(function() {
            $('#close_btn,#modal-overlay').click(function() {
                $("#modal-overlay").fadeOut(100)
                $("#newsletter_desc").fadeOut(100)
            });

            $('#promo-email-signup-button').on('click', function(e){
                if(!$('input[name="promo-email"]')[0].checkValidity()){
                    toastr.error('Please enter a valid email!')
                    return
                }

                var email = $('input[name="promo-email"]').val()

                $.post('/newsletter-guest-signup', { email: email}, function(data){
                    var data = JSON.parse(data)
                    if(data.success){
                        $('#newsletter-signup-message').addClass('success').html(data.message)
                    }else{
                        $('#newsletter-signup-message').addClass('error').html(data.message)
                    }
                });
            })
        });
    </script>
