	function myPage() {
		if (window.opener) {
			if (window.opener.name != window.name) {
				window.close();
				window.opener.location = "/snpapp/login/PsnLandingPage.jsp";
				document.loginForm.username.focus();
			}
		} else {
			if (window.frameElement) {
				window.parent.location = "/snpapp/login/PsnLandingPage.jsp";
			}
		}
	}

	function OpenPop(page) {
		var link;
		var context = '/snpapp/login/';
		var systemRequirementLink="https://pearsonnacommunity.force.com/coco/articles/Informational/Pearson-SuccessNet-System-Requirements";
		if (page == "Privacy Statement") {
			/* PSN-33526 Change for privacy statement long term fix. */
			link = "../../privacy_statement/privacy.htm";
		} else if (page == "Terms of Use") {
			link = context + "termsOfUse.jsp";
		} else if (page == "System Requirements") {
			/* Start - PSN-27732, Modified by Sandhya Ande, replaced the existing PDF link with the knowledgebase url */
			/* link = "http://knowledgebase.pearsonschool.com/kmp/article/AA-04713"; */
			/* End - PSN-27732 */
			link=systemRequirementLink;
		} else if (page == "Getting Started") {
			link = "/snp_pdf/GettingStarted.pdf";
		} else if (page == "Check Setting") {
			link = context + "checkSettings.jsp";
		}

		window
				.open(link, '',
						'top=150,left=150,location=no,scrollbars=yes,HEIGHT=480,WIDTH=800');
	}
