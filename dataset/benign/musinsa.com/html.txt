function scrollFloat(){
	var hidTop = $('.main-wrapper').offset().top + $('.main-wrapper').innerHeight();
	var hidBot = $(document).height() - $(window).height() - $('footer').innerHeight();
	var scTop = $(window).scrollTop()
		, btnUp = $('.code.box-btn-leftbottom .btn.jump_up')
		, btnDown = $('.code.box-btn-leftbottom .btn.jump_down')
	if(scTop < hidTop)
		btnUp.replaceWith('<a class="jump_up btn disabled" style="cursor:default">top</a>');
	else
		btnUp.replaceWith('<a href="#top" id="left_top_btn" class="jump_up btn" title="위로가기">top</a>');
	if(scTop > hidBot)
		btnDown.replaceWith('<a class="jump_down btn disabled" style="cursor:default">bottom</a>');
	else
		btnDown.replaceWith('<a href="javascript:scrollDown();" id="left_bot_btn" class="jump_down btn" title="아래로가기">bottom</a>');
}
$(document).ready(function(){
	scrollFloat();
});
$(window).scroll(function(){
	scrollFloat();
});
$(window).resize(function(){
	scrollFloat();
});
$(window).load(function(){
	scrollFloat();
});
