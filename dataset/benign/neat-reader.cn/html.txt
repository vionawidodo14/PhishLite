    function formatDate(time) {
        var date = new Date(time).getTime();
        var nowDate = new Date().getTime();
        var overDate = (date - nowDate) / 1000;
        var day = Math.floor(overDate / 86400);
        var hour = Math.floor(overDate % 86400 / 3600);
        var min = Math.floor(overDate % 86400 % 3600 / 60);

        if (day > 0) {
            return day + '天';
        } else if (day === 0 && hour >= 1) {
            return hour + '小时' + min + '分钟';
        } else {
            return min + '分钟';
        }
    }

    function formatExpireTime(time) {
        let date = new Date(time);
        let year = date.getFullYear();
        let month = date.getMonth() + 1;
        let day = date.getDate() - 1;

        return year + '年'+ month + '月' + day + '日晚12时';
    }

<script>
    $(function () {
        //用户评价切换
        var oLi = $('.assess-btn')
                , oDiv = $('.assess-item')
                ;

        oLi.on('mouseover', function () {
            $(this).addClass('assess-btn-active').siblings().removeClass('assess-btn-active');
            oDiv.eq($(this).index()).addClass('assess-box-active').siblings().removeClass('assess-box-active');
        });

        /* 页面滚动 */
        var scrollTop = 0;

        $(document).on('scroll', function () {

            scrollTop = $(this).scrollTop();

            var width = document.documentElement.clientWidth;

            if (width > 1662) {
                if (scrollTop >= $('.introduce-container').offset().top) {
                    $('.introduce-center > div').addClass('introduce-center-in');
                }
            }

        });

    });

    //设备识别，如果不存在，则为当前浏览器生成一个
    if ($.cookie("fingerprint") !== null) {
        new Fingerprint2().get(function (result, components) {
            //a hash, representing your device fingerprint
            //console.log(result)
            // an array of FP components
            //console.log(components);

            $.cookie('fingerprint', result, {path: '/', expires: 36500});
        });
    }

    //浏览器检测
    var browserCookie = $.cookie('broswer-check');
    if (browserCookie === null || browserCookie === undefined) {
        var parser = new UAParser().getBrowser().name;
        //获取浏览器标示
        var browser = parser.toLowerCase();
        console.log("browser", browser);
        if (browser === 'ie') {
            //显示浏览器提示弹窗
            $('.browser-check').addClass('slide-in');
            //关闭浏览器弹窗，并保存cookie
            $('.close-btn, .tips-btn').on('click', function () {
                $('.browser-check').addClass('slide-out');
                $.cookie('broswer-check', 1, {path: '/', expires: 365000});
            });
        }
    }
