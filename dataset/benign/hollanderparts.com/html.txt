        function numberToCurrency(amount) {
            var thousandsSeparator = ","
            var currencyNum = "";
            var amountString = amount.toFixed(2).toString();
            var digits = amountString.split("");

            var countDigits = digits.length;
            var revDigits = digits.reverse();

            for (var i = 0; i < countDigits; i++) {
                if ((i % 3 == 0) && (i != 0)) {
                    currencyNum += thousandsSeparator + revDigits[i];
                } else {
                    currencyNum += digits[i];
                }
            };

            var revCurrency = currencyNum.split("").reverse().join("");

            var finalCurrency = "$" + revCurrency;
            return finalCurrency.replace(',.', '.');
        }
        function GetShoppingQuote() {
            var json = "";
            $.ajax({

                success: function (result) {
                    //document.getElementById('lblMsg').innerText = result.Msg;
                    if (result.User == "Guest") {
                        $('#lblQuoteUser').html("");
                    } else {
                        userName = result.User.split(' ');
                        fName = userName[0];
                        $('#lblQuoteUser').html(fName);
                    }
                    $('#cartSummaryItemCnt').html(result.QuoteItemsCount == "" ? "0" : result.QuoteItemsCount);
                    $('#lblPrice').html(numberToCurrency(result.Price));
                    //GetShoppingQuoteExecution(result);
                },
                statusCode: {
                    404: function (content) { alert('cannot find resource'); },
                    505: function (content) { alert('internal server error'); }
                },
                error: function (req, status, errorObj) {
                    json = "";
                }
            });
            return json;


        }
