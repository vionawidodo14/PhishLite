        jQuery_3_2_1("#tags").val(jQuery_3_2_1(html).text());
        jQuery_3_2_1("#tags").focus();

    var customRenderFunction = function(document_type, item) {
        var regex = /(<([^>]+)>)/ig
        var body = item.highlight['sections'];
        var result = body.replace(regex, "");

        return '<p class="title">' + result + '</p>';
    };


    var customResultRenderFunction = function(ctx, results) {
        var $list = ctx.list,
                config = ctx.config;
        var arrDistinct = new Array();
        $.each(results, function(document_type, items) {
            $.each(items, function(idx, item) {
                if ($.inArray(Swiftype.htmlEscape(item.highlight['sections']), arrDistinct) == -1) {
                    arrDistinct.push(Swiftype.htmlEscape(item.highlight['sections']));
                    ctx.registerResult($('<li>' + config.renderFunction(document_type, item) + '</li>').addClass("ui-menu-item").appendTo($list), item);
                }
            });
        });
    };
