function setStartEndPoints($elem, startPoint, endPoint){
    $elem.css('left', startPoint.x)
        .css('top', startPoint.y)
        .css('width', endPoint.x - startPoint.x)
        .css('height', endPoint.y - startPoint.y);

    if(Math.abs(startPoint.x - endPoint.x) < 10){
        $elem.hide();
    } else {
        $elem.show();
    }
}
