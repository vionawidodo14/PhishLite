    }
  
  };
  
//   scroll to top on mobile when adding item to cart
  
  var _scrollToTop = function(){
  	var scroll_top_duration = 700;
    
        $('body,html').animate({
          scrollTop: 0 ,
        }, scroll_top_duration
       );
   
  
  };
  

  
  var _setText = function($button, label) {
    if ($button.children().length) {
      $button.children().each(function() {
        if ($.trim($(this).text()) !== '') {
          $(this).text(label);
        }
      });
    }
    else {
      $button.val(label).text(label);
    }
  };
  
  var _optionTitle = function(option){
    if(option == null ){
      	$('.mini-cart .mini-variant > .size-label').hide();
    }else{
    	$('.mini-cart .mini-variant').append(option);
    }
  };
  
  var _miniCartAction = function(productData, cartBtn){
    	 
    var $productTitle = productData.product_title,
        $productPrice = Shopify.formatMoney(productData.price),
        $productImage = productData.image,
        $productOption = productData.variant_title
    	
    
    cartBtn.addClass('inverted');
    _setText(cartBtn, _config.addedToCartBtnLabel);

    _showFeedback('success', '<div class="mini-row"><div class="mini-img"><img src="' + $productImage + '"></div><div class="mini-des"><h5>' + $productTitle + '</h5><p class="mini-variant"><span class="size-label">Size: </span></p><h5 class="mini-price">' + $productPrice + '</h5></div>' + '</div> ');
    _optionTitle($productOption);

    _addItemToCart($productTitle, $productPrice, $productImage, $productOption);

    // set time out for add to cart button message and notification
    window.setTimeout(function(){
      cartBtn.prop('disabled', false).removeClass('disabled').removeClass('inverted');
      _setText(cartBtn,_config.addToCartBtnLabel);
    }, _config.howLongTillBtnReturnsToNormal);

    window.setTimeout(function(){

      $('.mini-cart').hide();
    }, _config.cartHide);

   
               
  };
  
  var _updateCartAction = function(){
  
  		// Update cart count and show cart link.
              $.getJSON(_config.shopifyAjaxCartURL, function(cart) {
                if (_config.cartCountSelector && $(_config.cartCountSelector).size()) {
                  var value = $(_config.cartCountSelector).html() || '0';
                  $(_config.cartCountSelector).html(value.replace(/[0-9]+/,cart.item_count)).removeClass('hidden-count');
                }
                if (_config.cartTotalSelector && $(_config.cartTotalSelector).size()) {
                  if (typeof Currency !== 'undefined' && typeof Currency.moneyFormats !== 'undefined') {
                    var newCurrency = '';
                    if ($('[name="currencies"]').size()) {
                      newCurrency = $('[name="currencies"]').val();
                    }
                    else if ($('#currencies span.selected').size()) {
                      newCurrency = $('#currencies span.selected').attr('data-currency');
                    }
                    if (newCurrency) {
                      $(_config.cartTotalSelector).html('<span class=money>' + Shopify.formatMoney(Currency.convert(cart.total_price, "AUD", newCurrency), Currency.money_format[newCurrency]) + '</span>');
                    } 
                    else {
                      $(_config.cartTotalSelector).html(Shopify.formatMoney(cart.total_price, "£{{amount}}"));
                    }
                  }
                  else {
                    $(_config.cartTotalSelector).html(Shopify.formatMoney(cart.total_price, "£{{amount}}"));
                  }
                };
              });        
         
  
  };
  
  var _init = function() {   
    $(document).ready(function() { 
      $(_config.addToCartFormSelector).submit(function(e) {
        e.preventDefault();
        var $addToCartForm = $(this);
        var $addToCartBtn = $addToCartForm.find(_config.addToCartBtnSelector);
        _setText($addToCartBtn, _config.addingToCartBtnLabel);
        $addToCartBtn.addClass('disabled').prop('disabled', true);
        // Add to cart.
        $.ajax({
          url: _config.shopifyAjaxAddURL,
          dataType: 'json',
          type: 'post',
          data: $addToCartForm.serialize(),
          success: function(itemData) {
            
            _miniCartAction(itemData, $addToCartBtn);
            
//             scroll to top on mobile device
            if ($(window).width() <= 768) {  
                _scrollToTop();

             }     
            
            _updateCartAction();
          }, 
          error: function(XMLHttpRequest) {
            var response = eval('(' + XMLHttpRequest.responseText + ')');
            response = response.description;
            if (response.slice(0,4) === 'All ') {
              _showFeedbackError('error', '<h5 class="error-msg">There is not enough stock</h5><h5 class="error-msg">' + response.replace('All 1 ', 'All ') + '</h5>', $addToCartForm);
              $addToCartBtn.prop('disabled', false);
              _setText($addToCartBtn, _config.soldOutBtnLabel);
              $addToCartBtn.prop('disabled',true);
             
            }
            else {
              _showFeedbackError('error', '<i class="fa fa-warning"></i> ' + response, $addToCartForm);
              $addToCartBtn.prop('disabled', false).removeClass('disabled');
              _setText($addToCartBtn, _config.addToCartBtnLabel);
            }
          }
        });   
        return false;    
      });
    });
  };

                  success: function(itemData) {

                     //scroll to top

                    //update cart json number


                    _updateCartAction();

                  },

                  error: function(XMLHttpRequest) {
                    var response = eval('(' + XMLHttpRequest.responseText + ')');
                    response = response.description;
                    if (response.slice(0,4) === 'All ') {
                      _showFeedbackError('error', '<h5 class="error-msg">There is not enough stock</h5><h5 class="error-msg">' + response.replace('All 1 ', 'All ') + '</h5>', sizesPanel);
                      cartBtn.prop('disabled', false);
                      _setText(cartBtn, _config.soldOutBtnLabel);
                      cartBtn.prop('disabled',true);

                    }
                    else {
                      _showFeedbackError('error', '<i class="fa fa-warning"></i> ' + response, sizesPanel);
                      cartBtn.prop('disabled', false).removeClass('disabled');
                      _setText(cartBtn, _config.addToCartBtnLabel);
                    }
                  }
                });
               
            }  else{
                              

              _errorFeedback2(variantList);
           }
           return false;   
         });

    });
  };
                                              
  return {
    init: function(params) {
        // Configuration
        params = params || {};
        // Merging with defaults.
        $.extend(_config, params);
        // Action
        $(function() {
          _init();
        });
    }, 
    vCart: function(params) {
        // Configuration
        params = params || {};
        // Merging with defaults.
        $.extend(_config, params);
        // Action
        $(function() {
          _vCart();
        });
    },    
    getConfig: function() {
      return _config;
    }
  }  
                      

})(jQuery);
