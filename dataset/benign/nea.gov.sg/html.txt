            twoBands.forEach(function (band, i) {
                var numeral = band.split(" ")[1];
                numeral = '<span class="romanize">' + numeral + '</span>';
                band = band.split(" ");
                band[1] = numeral;
                twoBands[i] = band.join(" ");
            });
            return twoBands.join(" - ");
        } else {
            range = range.split(" ");
            var numeral = '<span class="romanize">' + range[1] + '</span>';
            range[1] = numeral;
            range = range.join(" ");
            return range;
        }
    }
</script>
