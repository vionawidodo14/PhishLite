                                el.val('')
                                return false;
                            }
                            return true;
                        }
                    });
                }
            }
        },
        onValid: function () {
            $('#form_newsletter_btn').hide();
            $('#form_newsletter_btn').parent('span').next('span.loaderSmall').show();

                $('#form_newsletter_btn').show();
                $('#form_newsletter_btn').parent('span').next('span.loaderSmall').hide();
                $('.form_newsletter_success').show().delay(5000).hide(400)

                $('.form_newsletter_error').show().delay(5000).hide(400)
                $('#form_newsletter_btn').show();
                $('#form_newsletter_btn').parent('span').next('span.loaderSmall').hide();
            }
        }
    });
    $('#form_newsletter_btn').click(function (e) {
        e.preventDefault();
        formnewsletter.checkForm('check');
    });

                if (data.response) {
                    $('.captchaImgCnt').html(data.text)
