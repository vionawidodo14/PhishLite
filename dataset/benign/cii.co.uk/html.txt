<script>

    var filtersSearchApp = angular.module('filtersSearchApp', ['ui.bootstrap', 'ngSanitize']);
    filtersSearchApp.config(['$httpProvider', function ($httpProvider) {
        $httpProvider.defaults.headers.common["X-Requested-With"] = 'XMLHttpRequest';
    }]);

    filtersSearchApp.controller('filtersSearchController', function ($scope, $http) {
        $scope.searchText = '';

        $scope.checkSearchText = function () {
            if ($scope.searchText.length > 0) {
                window.location.href = '/search-results?q=' + $scope.searchText;
            }
        }

        $scope.handleEnter = function (e) {
            if (e.keyCode == 13) {
                $scope.checkSearchText();
            }
        }

        $scope.doSearch = function (searchValue) {

                var searchCriteria = {
                    SearchTerms: searchValue,
                    PageNumber: 1,
                    PageSize: 20,
                    SearchFilters: null,
                    ToDate: null,
                    FromDate: null,
                    SortBy: 3,
                    SourceId: 51277
                };


                return $http({
                    method: "POST",
                    url: "/umbraco/surface/SiteSearch/Search",
                    data: JSON.stringify({
                        siteSearchCriteria: searchCriteria
                    })
                })
                    .then(function (response) {
                        var filters = [];
                        if (response.data.Filters != null) {
                            angular.forEach(response.data.Filters, function (item) {
                                filters.push({
                                    name: item.Name,
                                    id: item.Id,
                                    matchCount: item.RTLSeperator + '(' + item.MatchCount + item.RTLSeperator + ')',
                                    href: '/search-results?q=' + $scope.searchText + '&' + item.Id + '=on'
                                });
                            });
                        }

                        return filters;
                    });
        }

    });

    angular.element(document).ready(function () {
        angular.bootstrap($('#filtersSearchApp')[0], ['filtersSearchApp']);
    });
</script>
