				$(element).trigger('change');
			}
		}

		$(element).trigger('change');
							}

				.after($listContainer.append($suggestionList))

				.bind('keyup', function() {
					window.clearTimeout(triggerSearch);
					if ($(this).val() === "") {
						$suggestionList.show();

						return;
					}
					triggerSearch = window.setTimeout(function() {
										document.getElementById('form_location-suggestion').value = '';
				document.getElementById('form_location-id').value = '';
				document.getElementById('form_location-locality').value = '';
