<script>

    (function (_hooks, localStorage) {
        var exitIntent = {
            cacheDom: function () {
                this.template = document.getElementById('js-template-exit-intent-layer');
            },

            bindEvents: function () {
                this.trackMovementAlias = this.trackMovement.bind(this);
                document.addEventListener('mousemove', this.trackMovementAlias);

                this.handleMouseOutAlias = this.handleMouseOut.bind(this);
                document.addEventListener('mouseout', this.handleMouseOutAlias); // no once in IE
            },

            mouseCoordinates: [{}, {}],

            trackMovementAlias: {},
            trackMovement: function (e) {
                //console.log('trackMovement', e.clientX, e.clientY);
                this.mouseCoordinates.pop();
                this.mouseCoordinates.unshift({
                    x: e.clientX,
                    y: e.clientY
                });
            },

            handleMouseOutAlias: {},
            handleMouseOut: function () {
                //console.log('Maus raus!', this.mouseCoordinates);
                document.removeEventListener('mousemove', this.trackMovementAlias);
                document.removeEventListener('mouseout', this.handleMouseOutAlias);

                var hotWire = 50;
                var delta = {
                    x: Math.abs(this.mouseCoordinates[0].x - this.mouseCoordinates[1].x),
                    y: Math.abs(this.mouseCoordinates[0].y - this.mouseCoordinates[1].y)
                };
                //console.log('delta', delta);


                if (this.mouseCoordinates[0].x < hotWire && delta.x >= delta.y) {
                    console.warn('exit LEFT');
                }
                if (this.mouseCoordinates[0].x > (window.innerWidth - hotWire) && delta.x >= delta.y) {
                    console.warn('exit RIGHT');
                }
                if (this.mouseCoordinates[0].y > (window.innerHeight - hotWire) && delta.x < delta.y) {
                    console.warn('exit BOTTOM');
                }

                if (this.mouseCoordinates[0].y < hotWire && delta.x < delta.y) {
                    console.warn('exit TOP');
                    this.injectLayer();
                } else {
                    this.reinit();
                }
            },

            preloadImages: function () {
                var images = Array.prototype.slice.call(
                    (this.template.content || this.template).querySelectorAll('img'), 0
                )
                    .map(function (img) {
                        return img.src
                    })
                    .forEach(function (src) {
                        var temp = new Image();
                        temp.src = src;
                    });
            },

            injectLayer: function () {
                var self = this;

                this.layer = (this.template.content || this.template).children[0].cloneNode(true);

                document.body.appendChild(this.layer); // insert
                void(this.layer.offsetHeight); // layout
                this.layer.style.opacity = '1'; // show

                _hooks.showLayer();
                localStorage.setItem('exitIntentHistory', 'shown');

                this.layer.addEventListener('click', function (e) {
                    if (e.target.classList.contains('js-close')) {
                        self.closeLayer();
                    }
                    if (e.target.classList.contains('js-button')) {
                        _hooks.clickThru();
                        localStorage.setItem('exitIntentHistory', 'clicked');
                    }
                });
            },

            closeLayer: function () {
                this.layer.style.opacity = '0';

                setTimeout(function () {
                    document.body.removeChild(this.layer);
                }.bind(this), 125);

                _hooks.closeLayer();
                localStorage.setItem('exitIntentHistory', 'closed');
            },

            init: function () {
                if (!(localStorage.getItem('exitIntentHistory') === 'clicked'
                    || localStorage.getItem('exitIntentHistory') === 'closed'
                    || localStorage.getItem('raffle_20years_submitted') == '1')) {
                    console.log('initializing exitIntentLayer');

                    this.cacheDom();
                    this.preloadImages();
                    this.bindEvents();
                } else {
                    console.log('not initializing exitIntentLayer (already ' + localStorage.getItem('exitIntentHistory') + ')');
                }
            },

            reinit: function () {
                //console.log('reinit exitIntentLayer');
                if (!(localStorage.getItem('exitIntentHistory') === 'clicked'
                    || localStorage.getItem('exitIntentHistory') === 'closed'
                    || localStorage.getItem('raffle_20years_submitted') == '1')) {
                    this.bindEvents();
                }
            }
        }

        window.onload = function () {
            setTimeout(function () {
                exitIntent.init();
            }, 2000);
        }

    })(
        {

                try {

                        hitType: 'event',

                        eventAction: 'show'
