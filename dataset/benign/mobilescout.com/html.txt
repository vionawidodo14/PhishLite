var notification = (function() {

	var ovk_channels_socket_url     = 'wss://f8o1ebmds7.execute-api.us-east-1.amazonaws.com/mobilescout2';
	var ovk_channels_polling_url    = 'https://1d5qervool.execute-api.us-east-1.amazonaws.com/mobilescout2/';
	var pollingIntervalTime         = 5000;
	var lastTimestamp               = 1571832248000;
	var reconnectOnUserActivity     = false;
	var timeoutTimer                = undefined;
	var lastActivityTime            = Date.now();
	var userActivityTimeoutMinutes  = 5; // minutes
	var maxSocketConnectionAttempts = 5; // times to retry socket connection

	/*
	 * sets a timer to stop all connections when there are no user activity
	 */
	function setUserActivityTimeout () {

		timeoutTimer = setTimeout(function () {

			if (pollingDriver.pollingEnabled) {

				console.log('OvkChannels: disconnecting user from polling');

				pollingDriver.stopPolling();
			}

			if (socketDriver.webSocketReady) {

				console.log('OvkChannels: disconnecting user from socket');

				socketDriver.webSocketReady = false;
				socketDriver.webSocket.close();
			}

			reconnectOnUserActivity = true;

			console.log('OvkChannels: user activity timeout. disconnect');

		}, 1000 * 60 * userActivityTimeoutMinutes);
	}

	/*
	 * clear the user activity timeout settled by setUserActivityTimeout function
	 */
	function clearUserActivityTimeout () {

		if (timeoutTimer !== undefined) {

			clearTimeout(timeoutTimer);

			if (reconnectOnUserActivity) {

				console.log('OvkChannels: reconnecting user...');

				reconnectOnUserActivity = false;

				var xhr = pollingDriver.poll();

				xhr.then(function () {

					socketDriver.setWebSocket();
				});
			}
		}
	}

	// polling driver to perform updates via polling
	var pollingDriver = new OvkChannelsPolling(ovk_channels_polling_url, pollingIntervalTime, lastTimestamp);

	// socket to enable real time communication
	var socketDriver            = new OvkChannels(ovk_channels_socket_url);
	socketDriver.onFailCallback = function (e) {

		console.log('OvkChannels: socket fails... executing onFailCallback');

		if (socketDriver.connectionAttempts < maxSocketConnectionAttempts) {

			console.log('OvkChannels: socket connection attempt', socketDriver.connectionAttempts);
			console.log('OvkChannels: perform a simple poll to get updates while socket setup is running');

			pollingDriver.poll();

			setTimeout(function () {

				console.log('OvkChannels: calling setWebSocket after ', pollingIntervalTime, ' to no merge events from multiple drivers');
				socketDriver.setWebSocket();

			}, pollingIntervalTime);

		} else {

			console.log('OvkChannels: max connection attempts exceeded, set polling definitively');

			socketDriver.connectionAttempts = 0;
			pollingDriver.startPolling();
			setUserActivityTimeout();
		}
	};

	notification = new OvkNotification(socketDriver, pollingDriver);

	$(document).on('click mousemove keydown scroll', function (e) {

		console.log('OvkChannels: user activity!');

		clearUserActivityTimeout();
		setUserActivityTimeout();
	});

	return notification;
})();
