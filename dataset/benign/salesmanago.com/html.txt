  function resizeHeaderOnScroll() {
    const distanceY = window.pageYOffset || document.documentElement.scrollTop,
      shrinkOn = 30,
      headerEl = document.querySelector('.custom-topbar'),
      dropdownContainer = document.querySelectorAll('.dropdown__container'),
      logo = document.querySelector('.custom-lowbar__logo');

    if (distanceY > shrinkOn) {
      headerEl.classList.add("smaller");
      dropdownContainer.forEach(function(e) {
        if (window.innerWidth >= 991) {
          e.classList.add('menuPositionMobile');
        }
        lowbar.classList.add('lowbarPositionMobile');
        logo.classList.remove('custom-lowbar__logo--fixed');
        logo.classList.add('custom-lowbar__logo--moved');
      })
    } else {
      headerEl.classList.remove("smaller");
      dropdownContainer.forEach(function(e) {
        if (window.innerWidth >= 991) {
          e.classList.remove('menuPositionMobile');
        }
        lowbar.classList.remove('lowbarPositionMobile');
        logo.classList.remove('custom-lowbar__logo--moved');
        logo.classList.add('custom-lowbar__logo--fixed');
      })
    }


  }

  function showMenuOnResize() {
    if (window.innerWidth >= 991) {
      lowbar.classList.add('display-flex');
      lowbar.classList.remove('display-flex-up');
      lowbar.classList.remove('display-flex-down');
      navMenu.forEach(function(e) {
        e.classList.add('display-flex');
        e.classList.remove('display-flex-up');
        e.classList.remove('display-flex-down');
      });
      navElement.forEach(function(e) {
        if (e.children[1]) {
          e.children[1].classList.remove('display-block');
        }
      });
    } else {
      lowbar.classList.remove('display-flex');
      navMenu.forEach(function(e) {
        e.classList.remove('display-flex');
      });
    }
  }

  const loupe = document.querySelector('.custom-topbar__link--loupe'),
    loupeWrapper = document.querySelector('.loupe__wrapper'),
    loupeInput = document.querySelector('.loupe__input'),
    navToggler = document.querySelector('#nav-toggler'),
    navMenu = document.querySelectorAll('.custom-lowbar__item'),
    lowbar = document.querySelector('.custom-lowbar'),
    navElement = document.querySelectorAll('.custom-lowbar__item'),
    navLink = document.querySelectorAll('.custom-lowbar__link'),
    topbarLinks = document.querySelectorAll('.custom-topbar__item');

  loupe.addEventListener('click', function() {
    topbarLinks.forEach(function(el) {
      if (el.style.display !== 'none') {
        el.style.display = 'none';
        loupeWrapper.style.display = 'flex';
        loupeInput.style.display = 'flex';
        loupeInput.focus();
      } else {
        el.style.display = 'flex';
        loupeWrapper.style.display = 'none';
        loupeInput.style.display = 'flex';
      }
    })
  });

  navToggler.addEventListener('click', function() {
    if (!navMenu[0].classList.contains('display-flex-down')) {
      navToggler.classList.add('open');
      navMenu.forEach(function(e) {
        e.classList.remove('display-flex-up');
        lowbar.classList.remove('display-flex-up');
        e.classList.add('display-flex-down');
        lowbar.classList.add('display-flex-down');
      })
    } else {
      navToggler.classList.remove('open');
      navMenu.forEach(function(e) {
        e.classList.remove('display-flex-down');
        lowbar.classList.remove('display-flex-down');
        e.classList.add('display-flex-up');
        lowbar.classList.add('display-flex-up');
      })
    }
  });

  window.addEventListener('scroll', resizeHeaderOnScroll);
  window.addEventListener('resize', showMenuOnResize);

  navLink.forEach(function(e) {
    e.addEventListener('click', function(el) {
      if (!e.nextElementSibling.classList.contains('display-block')) {
        navLink.forEach(function(e) {
          if (e.nextElementSibling) {
            e.nextElementSibling.classList.remove('display-block');
            e.nextElementSibling.previousElementSibling.classList.remove('active-menu-item');
          }
        })
        e.nextElementSibling.classList.add('display-block');
        e.nextElementSibling.previousElementSibling.classList.add('active-menu-item');
      } else if (e.nextElementSibling.classList.contains('display-block')) {
        e.nextElementSibling.classList.remove('display-block');
      }
    })
  });
