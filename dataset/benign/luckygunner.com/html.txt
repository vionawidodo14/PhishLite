            var savedQuery = Mage.Cookies.get('lastAutocompleteSearchQuery');
            if(savedQuery != searchQuery) {
                Mage.Cookies.set('lastAutocompleteSearchQuery', searchQuery);
                Event.fire(document, 'searchAutocomplete:change', {action: 'SearchQuery', label: searchQuery, value: null});
            }

            if(searchQuery){
                $('search_autocomplete').select('a').invoke('observe', 'click', function(e){
                    Mage.Cookies.set('lastSearchQuery', searchQuery);
                });
            }
        },
        clearResults: function() {
            setTimeout(function(){
                $('search_autocomplete').hide();
                $('search_autocomplete_categories').update('');
                $('search_autocomplete_products').update('');
                $('search_autocomplete_pages').update('');
                $$('.header-container + .backdrop-algolia').invoke('remove');
            }, 500);
        }
    });
    $('search').observe('keydown', function(e) {
        if (e.keyCode == 13) { // Enter
            var label = $('search').getValue();
            Event.fire(document, 'search:action', {action: 'Submit simple search', label: label, value: null});

            var element = Event.element(e);
            if (element.value != '' && element.tagName == 'INPUT' && $$('.algoliasearch-autocomplete-hit.marked').length == 0) {
                e.stop();
                $('search_mini_form').submit();
            }
        }
    });
