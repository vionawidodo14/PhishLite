    function codeAddress(uselatlng, updateDB, useDefaultAddress) {

        if (uselatlng) {

        latitude = position.coords.latitude;
        longitude = position.coords.longitude;

    function parseAddress(results, updateDB, isDefault) {

        var city = "";
        var state = "";
        var country = "";

        latitude = results[0].geometry.location.lat();
        longitude = results[0].geometry.location.lng();

        //get the city and state/province values from the results
        var result = results[0].address_components;
        for (j = 0 ; j < result.length ; ++j) {
            var resultTypes = result[j].types;
            for (k = 0 ; k < resultTypes.length ; ++k) {
                //find city
                if (resultTypes[k] == "locality") {
                    city = result[j].long_name;
                }

                //if locality is empty, some times the city name is in administrative_area_level_3 field, for example, in case of Mission, BC
                if (!city) {
                    if (resultTypes[k] == "administrative_area_level_3") {
                        city = result[j].long_name;
                    }
                }
                //find State
                if (resultTypes[k] == "administrative_area_level_1") {
                    //put the state abbreviation in the form
                    state = result[j].short_name;
                }
                //find Country
                if (resultTypes[k] == "country") {
                    //put the state abbreviation in the form
                    country = result[j].short_name;
                }
            }
        }
        var localeDisplay = "";

        if (city != '' && state != '') {
            localeDisplay = city + ", " + state
        }
        else if (city != '' && state == '' && country != '') {
            localeDisplay = city + ", " + country
        }
        else if (city == '' && state == '' && country != '') {
            localeDisplay = country;
        }
        else {
            localeDisplay = "Unknown";
            updateDB = false;
        }

        // Set the global variable for selected country
        selectedCountry = country;

        if (!isDefault) {
            $("#currentLocale").text(localeDisplay);
        }
        if (updateDB) {
            SaveSettings(city, state, country, longitude, latitude);
        }
        if (localeDisplay != "Unknown" && localeDisplay!='') {
            $('#txtLocale').val(localeDisplay);
            locale=localeDisplay
        }

      }

                state = place.address_components[i].short_name;
            }
            else if (addressType == "administrative_area_level_3") {
                city2 = place.address_components[i].long_name;
            }

            else if (addressType == "country") {
                country = place.address_components[i].long_name;
            }
        }
        if (!city)
            city = city;

        var localeDisplay = "";

        if (city != '' && state != '') {
            localeDisplay = city + ", " + state
        }
        else if (city != '' && state == '' && country != '') {
            localeDisplay = city + ", " + country
        }
        else if (city == '' && state == '' && country != '') {
            localeDisplay = country;

        // Set the global variable for selected country

            SaveSettings(city, state, country, longitude, latitude);
        }
    }

    function blink(elem, times, speed, css) {

        if (times > 0 || times < 0) {
            if ($(elem).hasClass(css))
                $(elem).removeClass(css);
            else
                $(elem).addClass(css);
        }

        clearTimeout(function () {
            blink(elem, times, speed);
        });

        if (times > 0 || times < 0) {
            setTimeout(function () {
                blink(elem, times, speed, css);
            }, speed);
            times -= .5;
        }
    }

            var src = vid.attr('src');
            // empty the iframe's src attribute
            // (this will kill the video playing)
            vid.attr('src', '');
            // and restore the iframe src url, ready to be played
            // again when the lightbox is displayed
            vid.attr('src', src);
        }
    });
