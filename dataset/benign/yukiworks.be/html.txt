		function pageLoad() {
			var openDomainLink = document.getElementById('OpenDomainLink').value;
			if (openDomainLink.length > 0) {
				var errorMessage = $('ErrorMessage');
				if (errorMessage != null) errorMessage.style.display = "none";
				window.open(openDomainLink, "_blank");
				document.getElementById('OpenDomainLink').value = '';
			}

			document.getElementById("openid_url").setAttribute("onkeydown", "var e = event.keyCode || event.which;if(e==13) { return false; }");
			var e = document.getElementById("Email");
			if (e.value.length > 0) {
				document.getElementById("Password").focus();
			} else { e.focus(); }

			if (document.getElementById('ShowOpenID').value) { ToggleLoginMethod(false) } else { ToggleLoginMethod(true); }
			if (IsIE()) { document.getElementById("MaintenanceMessage").hidden = false; }
		}

		function SwitchPassWordVisiblity() {
			if (document.getElementById("Password").type === "password") {
				document.getElementById("Password").type = "text";
				document.getElementById("passwordVisisble").classList.remove("fa-eye");
				document.getElementById("passwordVisisble").classList.add("fa-eye-slash");
			} else {
				document.getElementById("Password").type = "password";
				document.getElementById("passwordVisisble").classList.add("fa-eye");
				document.getElementById("passwordVisisble").classList.remove("fa-eye-slash");
			}
		}

		function IsIE() {
			var agent = navigator.userAgent.toLowerCase();
			if ((agent.indexOf("msie") > -1 && agent.indexOf("opera") === -1) || (!!window.MSInputMethodContext && !!document.documentMode)) {
				return true;
			} else return false;
		}

		function ChangeLanguage(lang) {
			var url = new URL(window.location.href)
			url.searchParams.set("Language", lang);
			window.location.href = url.href;
		}

		var resizeInfoWindow = function () {
			var ratio = 1440 / 1920;
			var clientHeight = window.innerHeight;
			var windowWidth = clientHeight * ratio;
			document.getElementById("InfoWindow").style.width = windowWidth + "px";
			if (window.innerWidth - windowWidth < 600) {
				if (!document.getElementById("InfoWindow").classList.contains("hide")) document.getElementById("InfoWindow").classList.add("hide")
			} else {
				document.getElementById("InfoWindow").classList.remove("hide")
			}
		}
		resizeInfoWindow();
		window.onresize = resizeInfoWindow;
	</script>
