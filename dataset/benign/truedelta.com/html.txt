		function move_callouts(step)
		{
			// get current parameters
			var first_col = parseInt($("#first-callout").val());
			var total_col = parseInt($("#total-callouts").val());
			var max_col;

			var window_width = $(window).width();
			console.log("window_width = " + window_width);
			var margin_right = "0.8%";
			var use_leftmost = 1;
			if(window_width < 640)
			{
				use_leftmost = -1;
				max_col = 1;
			}
			else
			{
				if(window_width < 960)
				{
					margin_right = "0";
					max_col = 2;
				}
				else
					max_col = 3;
			}
			
			// move the columns
			first_col = first_col + step;
			if(first_col < 1)
				first_col = 1;
			if(first_col > total_col)
				first_col = 1;
				
			var last_col = first_col + (max_col - 1);
			if(last_col > total_col)
				last_col = total_col;

			// remove classes
			// $(".leftmost").find('span').remove();
			$(".leftmost").unbind("click");
			$(".rightmost").unbind("click");
			$(".leftmost").removeClass("leftmost");
			$(".edge").removeClass("edge");
			$(".rightmost").removeClass("rightmost");
						
			// iterate through the columns to toggle visibility
			for(var i = 1; i <= total_col; i++)
			{
				var toggle_callout = "#homepage-callout-" + i;
				var toggle_linklist = "#top-comparison-" + i;
				var toggle_section = $(toggle_linklist).parent();

				if(i < first_col)
				{
					$(toggle_callout).hide();
					toggle_section.hide();
				}
				else
				{
					if(i > last_col)
					{
						$(toggle_callout).hide();
						toggle_section.hide();
					}
					else
					{
						$(toggle_callout).show();
						toggle_section.show();
					}
				}
				
				if((i == first_col) && (use_leftmost == 1))
				{
					$(toggle_callout).addClass("leftmost");
					toggle_section.addClass("leftmost");
					
					if(i == 1)
					{
						$(toggle_callout).addClass("edge");
						toggle_section.addClass("edge");
					}
				}
				else
				{
					if((i == last_col) || (use_leftmost == -1))
					{
						$(toggle_callout).addClass("rightmost");
						toggle_section.addClass("rightmost");
						
						if(window_width >= 640)
						{
							$(toggle_callout).css("float", "right");
							$(toggle_callout).css("margin-right", "0");
							$(toggle_linklist).parent().css("float", "right");
							$(toggle_linklist).parent().css("margin-right", "0");
						}
						
						if((i == total_col) && (use_leftmost == 1))
						{
							$(toggle_callout).addClass("edge");
							toggle_section.addClass("edge");
						}
					}
				}
				
				if(i != last_col)
				{
					$(toggle_callout).css("float", "left");
					$(toggle_callout).css("margin-right", margin_right);
					$(toggle_linklist).parent().css("float", "left");
					$(toggle_linklist).parent().css("margin-right", margin_right);					
				}
			}
			
			if((last_col != total_col) || (window_width < 640))
			{		
				console.log("binding rightmost");
				$(".rightmost").bind("click", function()
				{
					move_callouts(1);
				});
			}
			
			if(first_col != 1)
			{
				console.log("binding leftmost");
				$(".leftmost").bind("click", function()
				{
					move_callouts(-1);
				});				
			}
			
			// reset first column parameter
			$("#first-callout").val(first_col);
		}
