function search_project($form){
    var requestData = $($form).serializeArray();
    requestData.push({name:"go_search_project",value:""});

    if(requestData.length == 1){
        $("header [data-search-loading]").fadeOut(0);
        $(".result_wrapper , [data-search-result-wrap]").fadeOut(0);
        $("[data-search-welcom]").fadeIn(200);
    }else{
        $("[data-search-welcom]").fadeOut(0);
        $("header [data-search-loading]").fadeIn(200);

        $.post("_includes/ajax_process.php",requestData,function(response){
            var data = $.parseJSON(response);
            setTimeout(function(){$("header [data-search-loading]").fadeOut(0)},500);

            $(".result_wrapper , [data-search-result-wrap]").fadeIn(200);
            $("[data-search-result] [data-search-result-wrap]").html(data.html);
            $("[data-search-result-no]").html(data.count);
            ellipsis();
            $(".have_scroll_y").mCustomScrollbar({
                snapAmount:40,
                scrollButtons:{enable:true},
                keyboard:{scrollAmount:40},
                mouseWheel:{deltaFactor:40},
                scrollInertia:400,
                theme:"dark-thin",
                mouseWheel:{preventDefault:true},
            });
        });//post
    }//else
}//adv_search
