        function ChangePassword() {
            var oldPassword = document.getElementById('oldPass').value
            var newPassword = document.getElementById('newPwd').value
            var confirmNewPwd = document.getElementById('confirmPwd').value

            if (oldPassword.length <= 0) {
                swal({
                    title: "",
                    text: "الرجاء إدخال كلمة مرور قديمة صالحة!",
                    type: "error"
                },
                          function () {
                              document.getElementById('oldPass').select();

                          });
                return false;
            }
            if (newPassword.length <= 0) {
                swal({
                    title: "",
                    text: "الرجاء إدخال كلمة مرور الجديد",
                    type: "error"
                },
                          function () {
                              document.getElementById('newPwd').select();

                          });
                return false;
            }
            if (newPassword != confirmNewPwd) {
                swal({
                    title: "",
                    text: "كلمة المرور غير متطابقة",
                    type: "error"
                },
                          function () {
                              document.getElementById('newPwd').select();
                              document.getElementById('newPwd').value = "";
                              document.getElementById('confirmPwd').value = "";
                          });
                return false;
            }

            var dataparam = "{oldPassword:'" + oldPassword + "',newPassword:'" + newPassword + "'}";
            $.ajax({
                type: "POST",
                url: "../WebMethods/WebMethods.aspx/ChangePasswordCareer",
                data: dataparam,
                contentType: "application/json; charset=utf-8",
                dataType: "json",
                success: function (msg) {
                    var result = msg.d;
                    if (result == "InValid") {

                        swal({
                            title: "",
                            text: "الرجاء إدخال كلمة المرور القديمة الصحيحة.",
                            type: "error"
                        });

                    }
                    if (result == "Valid") {
                        swal({
                            title: "",
                            text: "تم تغيير كلمة المرور بنجاح",
                            type: "success"
                        });
                        CloseModal();
                    }
                }

            });

        }
