<script type="text/javascript">
var homeparkingForm = {
    buttonFlagLinks: null,
    bookingParkingInputs: null,
    errorElement: null,

    initialStartDate: "24/10/2019",
    initialEndDate: "31/10/2019",
    initialStartTime: "8:00",
    initialEndTime: "17:30",

    startDate: null,
    endDate: null,
    startTime: null,
    endTime: null,

    error: null,
    emptyDate: true,

    init: function () {
        this.buttonFlagLinks =  jQuery('.button-flag');
        this.bookingParkingInputs = jQuery('.input-booking-parking');
        this.errorElement = jQuery('#homeparkingError');

        this.listenBookingParkingInputs();
        this.listenButtonFlagLinks();
    },

    formatDate: function(dateString) {
        try {
            jQuery.datepicker.parseDate("dd/mm/yy", dateString);
        } catch (e) {
            this.error = "Veuillez saisir des dates valides";
            return false;
        }

        var dateParts = dateString.split("/");

        return dateParts[2] + '-' + dateParts[1] + '-' + dateParts[0];
    },

    createDate: function(strDate, strTime) {
        strDate = strDate.trim();
        strTime = strTime.trim();

        if (strTime.length < 5) {
            strTime = '0' + strTime;
        }
        return new Date(strDate + 'T' + strTime + ':00.000Z');
    },

    listenBookingParkingInputs: function() {
        var self = this;
        this.bookingParkingInputs.on('change', function()  {
            self.error = false;
            self.refreshError();

            self.startDate = self.formatDate(jQuery('.startDate').val());

            self.startTime = jQuery('.startHour').val();
            self.endDate = self.formatDate(jQuery('.endDate').val());
            self.endTime = jQuery('.endHour').val();

            self.emptyDate = false;
            if (!self.startDate
                || !self.startTime
                || !self.endDate
                || !self.endDate) {
                self.emptyDate = true;
            }

            if (self.emptyDate) {
                self.refreshError();
                return;
            }

            if (self.startDate === false || self.endDate === false) {
                self.error = "Veuillez saisir des dates valides";
                self.refreshError();
                return;
            }

            var startDate = self.createDate(self.startDate, self.startTime);
            var endDate = self.createDate(self.endDate, self.endTime);

            if (startDate > endDate) {
                self.error = "La date de sortie est avant la date d'entrée";
                self.refreshError();
                return;
            }
        });
    },

    listenButtonFlagLinks: function() {
        var self = this;
        this.buttonFlagLinks.on('click', function(e) {
            e.preventDefault();

            var url = jQuery(this).attr('href');

            if (self.error || self.emptyDate) {
                url += '?ad=' + self.initialStartDate + '&at=' + self.initialStartTime + '&dd=' + self.initialEndDate + '&dt=' + self.initialEndTime;
            } else {
                url += '?ad=' + self.startDate + '&at=' + self.startTime + '&dd=' + self.endDate + '&dt=' + self.endTime;
            }

            if (!self.error) {
                window.open(url,'_blank');
            }
        });
    },

    refreshError: function() {
        if (this.error) {
            this.errorElement.find('li').text(this.error);
            this.errorElement.removeClass('hidden');
        } else {
            this.errorElement.addClass('hidden');
            this.errorElement.find('li').text("");
        }
    },
};


    jQuery(document).ready(function(){
        homeparkingForm.init();
    });</script>
