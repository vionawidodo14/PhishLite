                                            this.bindUIElements();

                                        },
                                        accept: function(){
                                            this.confirmIsApproved = true;
                                            this.$el.modal('hide');
                                        },
                                        deny: function(){
                                            this.confirmIsApproved = false;
                                            this.$el.modal('hide');
                                        },
                                        onHide: function(){
                                            this.confirmDfd[this.confirmIsApproved ? 'resolve' : 'reject']();
