                _create: function() {
                    this._super();
                    this.widget().menu( "option", "items", "> :not(.ui-autocomplete-category)" );
                },
                _renderMenu: function( ul, items ) {
                    var that = this,
                        currentCategory = "";
                    $.each(items, function (index, item) {
                        var li;
                        if (item.category !== currentCategory) {
                            ul.append("<li class='ui-autocomplete-category'>" + item.category + "</li>");
                            currentCategory = item.category;
                        }

                        li = (function (ctx, ul, item) {
                            return $("<li></li>")
                                .data("ui-autocomplete-item", item)
                                .append("<a>" +
                                    item.label.replace(new RegExp(ctx.term, "i"), '<b class="ui-match">$&</b>') +
                                    "</a>")
                                .appendTo(ul);
                        })(that, ul, item);

                        if (item.category) {
                            li.attr("aria-label", item.category + " : " + item.label);
                        }
                    });
                }
            });


            var input = $('#main-search'),
                term = null,
                form,
                auto = null;

            if (input.length <= 0) {
                return;
            }

            form = input.closest('form');
            input.catcomplete({
                source: function (request, response) {
                    $.getJSON('/search', {q: request.term, src: 'jq'}, function (data) {

                        var result = [];

                        var properties = [
                            "Orte",
                            "Länder",
                            "Sehenswürdigkeiten",
                        ];

                        for(var i = 0;  i < properties.length; i++){
                            for(var j = 0;  j < data.length; j++){
                                if(data[j].category === properties[i]){
                                    result.push(data[j]);
                                }
                            }
                        }

                        response(result);
                    });
                },
                minLength: 2,
                select: function (event, ui) {
                    if (ui.item) {
                        event.preventDefault();

                        if (auto) {
                            window.clearTimeout(auto);
                        }

                        term = input.val();
                        log(term, 'select', ui.item.value);
                        input.val(ui.item.label);

                        window.location.href = ui.item.value;
                    }
                },
                search: function (event, ui) {
                    term = input.val();
                    if (auto) {
                        window.clearTimeout(auto);
                    }
                    auto = window.setTimeout(function () {
                        log(term, 'auto');
                    }, 10000);
                },
                focus: function (event, ui) {
                    var origEvent = event.originalEvent.originalEvent;

                    if (origEvent && /^key/.test(origEvent.type)) {
                        event.preventDefault();

                        if (!term) {
                            term = input.val();
                        }

                        if (ui.item) {
                            input.val(ui.item.label);
                        }
                    }
                },
                autoFocus: true,
            });

            form.submit(function (ev) {
                term = input.val();

                if (term.trim() === "") {
                    ev.preventDefault();

                        duration: 2500,
                        step: function () {

                        },
                        complete: function () {
                            $(input).css("border", "none");
                        }
                    });
                }

                log(term, 'submit', null);
            });

                    "src": src,
                    "redir": redir,
                    "__csrf_key": "17510998455dafa33d7b81e8.99399049"
                };
