        // get all carousel slide color classes
        var slides = $('.carousel-item');
        var carouselItemClasses = slides.map((idx,x) => $(x).attr('data-arrow-color'));
        var updateCarouselArrows = function(){
            let currentSlideArrowColor = $("#carousel-165676").find('.carousel-item.active').attr('data-arrow-color');
            $("#carousel-165676").find('.right-arrow-carousel').removeClass(carouselItemClasses.toArray().join(' '));
            $("#carousel-165676").find('.left-arrow-carousel').removeClass(carouselItemClasses.toArray().join(' '));
            if (currentSlideArrowColor){
                $("#carousel-165676").find('.right-arrow-carousel').addClass(currentSlideArrowColor);
                $("#carousel-165676").find('.left-arrow-carousel').addClass(currentSlideArrowColor);
            }
        }
        $(window).on('slid.bs.carousel', updateCarouselArrows);
        $('.carousel').bcSwipe({ threshold: 50 });
        updateCarouselArrows();
   </script>
