<header><h1>Entertainment and Event Technology Solutions from the World's Leading Provider, PRG</h1></header>
</div>
<div class="clear"></div>
<div class="routing_page_content_inner_text" id="routing_page_content_inner_text">
<p>
Production Resource Group is the global partner of choice for the world’s leading entertainment and event producers, designers, and creative talents.  Our teams stage some of the most unique, challenging, and groundbreaking projects across a broad range of markets. From corporate, association, and automotive events, live music tours and festivals, theatre, television and film, to sports and other major public events, to permanent installations at theme parks, retails stores, and performing arts centers  - PRG is on-site, from concept through execution, with the most creative and cutting edge production solutions available.  No other company can match the depth and breadth of PRG's global resources, technological expertise and experience.<br/>
<br/>
A Passion for Results is at the core of everything we do at PRG. It is this passion for results that has made PRG the world's foremost production solutions provider.</p>
<p><br/>
</p>

	function submitMail(form, route) {
		var recipientEmail = form.find('#share_to_mail');
		var formValid = validateEmail(recipientEmail);
		var sender = form.find("#share_from_mail").val();
		formValid = formValid && sender != "";

		if (formValid) {
			var data = {
				Sender: sender,
				RecipientEmail: recipientEmail.val(),
				ShareUrl: form.find('#share_url').val(),
				Message: form.find('#share_msg').val()
			};

			$.ajax({
				url: BASE_URL + route + '?sc_site=us&sc_lang=en',
				type: 'POST',
				data: data,
				beforeSend: function () {
					$('#window_overlay_inner').spin({ color: '#000' });
				},
				complete: function () {
					$('#window_overlay_inner').spin(false);
				},
				success: function () {
					form.find('.form_success_msg').fadeIn(300);
					form.find('.form_error_msg').hide(300);
				}
			});
		} else {
			form.find('.form_error_msg').fadeIn(300);
		}
	}
</script>
