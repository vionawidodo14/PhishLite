	function getNewPricing(){
		if($(".annually-button").is(":visible")){
		 	isAnnual = true;
		 	$("#pricing-sub-text").empty().text("(per month, paid annually)");
		 	$("#25-percent-off").show();
		} else{
			isAnnual = false;
			$("#pricing-sub-text").empty().text("(per month)");
			$("#25-percent-off").hide();
		}

		var subscriberCount = parseInt($("#subscriber-count-input").val());
		if (!subscriberCount){
			subscriberCount = defaultSubscriberCount;
		}

		var lowerbound = 0;
		var upperbound = 0;
		if (subscriberCount){
			for (var i=0; i<pricingJson.length; i++){
				if (subscriberCount > parseInt(pricingJson[pricingJson.length-1].count)){
					paintLast3TierWithCTA();
				} else if (parseInt(pricingJson[i].count) >= subscriberCount){
					
					if (i === 0){
						lowerbound = 0;
					} else{
						lowerbound = parseInt(pricingJson[i-1].count);
					}

					upperbound = parseInt(pricingJson[i].count);
					currentPrice = "$"+getPricing(i);

					paintNewPricingTable(lowerbound, upperbound);
					
					break;

	function numberWithCommas(x) {
		return x.toString().replace(/\B(?=(\d{3})+(?!\d))/g, ",");
	}

			return pricingJson[index].price;	
		}
		
	}
