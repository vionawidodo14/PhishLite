    function updateResults(event) {
        if (event.newTab.IsDynamic) {
            selectedCategory = event.newTab.TabText;

            if (selectedCategory === "All") {
                GetAll(6, false, true);
            }
            else {
                FilterResults(selectedCategory, 6, false, true);
            }
        }
    }

    function GetAll(rows, showHeaderColumn, isWidget) {

        var url = "all/" + rows + "/" + showHeaderColumn + "/" + isWidget;

        WarcUtitlies.AjaxTools
            .getFromController("calendar", url).done(
                function (result) {
                    compileAngularElement('#active-content-area-table', result);
                    waitLogo.hide();
                });
        $('#calendar-url').attr('href', '/calendar');
        $('#calendar-url').on("click", function () { AddTrackingEvent('View_all'); });
        waitLogo.show();
    }

                $(window).scrollTop(targetDistance);

            //Safari can scroll into a negative position when hitting the top of the page and then bounces itself back which looks like a scroll down

                currentScrollPosition = 0;

            var scrollChange = lastScrollPosition - currentScrollPosition;

                $(".header-items-holder").stop();

                navHidden = true;
            }
            else if (scrollChange > 0 && navHidden) {
                $(".header-items-holder").stop();
                $(".header-items-holder").slideToggle();
                navHidden = false;
            }

            lastScrollPosition = currentScrollPosition;
