      (function() {
        var carousel = {
          interval:    5000,
          $menuItems:  $j('#slides-menu .menu-item'),
          timer:       null,

          /**
           * Initialize the carousel, with events handlers.
           */
          init: function() {
            var self = this;

            this.currentItem = this.$menuItems[0];

            $j('#slides-menu')
              .on('mouseover', '.menu-item', function (event) {
                self.seek(this, false);
                self.pause(this);
              })
              .on('mouseout', function() {
                self.play();
              });;

            // Let's play…
            setTimeout(function() {
              self.play();
            });
          },

          /**
           * Pause the carousel.
           *
           * @param  {Object} element DOM element for an menu item
           */
          pause: function(element) {
            var self = this;
            this.timer = clearInterval(this.timer);
          },

          /**
           * Play the carousel.
           */
          play: function() {
            if (this.timer) { return; }

            var self = this;
                          this.timer = setInterval(function() {
                self.next();
              }, self.interval);
                      },

          /**
           * Display next slide, then loop.
           */
          next: function() {
            var indexCurrentItem = this.$menuItems.toArray().indexOf(this.currentItem);

            if (indexCurrentItem === this.$menuItems.length - 1 || indexCurrentItem < 0) {
              this.seek(this.$menuItems[0], true); // back to the beginning
            } else {
              this.seek(this.$menuItems[indexCurrentItem + 1], true);
            }
          },

          /**
           * Display a slide based on the current active item of the menu.
           *
           * @param  {Object} targetItem DOM Object representing a menu item
           * @param  {String} fade       Should the fade effect be triggered?
           */
          seek: function(targetItem, fade) {
            var $targetItem   = $j(targetItem),
                $currentSlide = $j('#' + $j(this.currentItem).data('slide')),
                $targetSlide  = $j('#' + $targetItem.data('slide'));

            // Don't do anything if the target slide is already displayed.
            if (targetItem === this.currentItem) { return; }

            // Reset all active menu items
            this.$menuItems.filter('.active').removeClass('active');

            // Hide existing element, then show target
            if (fade) {
              $currentSlide.fadeOut(700);
              $targetSlide.fadeIn(700);
            } else {
              $currentSlide.hide();
              $targetSlide.show();
            }

            $targetItem.addClass('active');

            // Set the current active slide
            this.currentItem = targetItem;
          }
        };

        carousel.init();
      })();
