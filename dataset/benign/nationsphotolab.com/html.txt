<script type="text/javascript">
	Dropzone.layoutversion=3;
	Dropzone.zones=[];
    Dropzone.autoDiscover = false;
    Dropzone.galleryid = null;
    Dropzone.totalUploaded = 0;
    Dropzone.processQueues = function () {
        for (var i = 0; i < Dropzone.zones.length; i++) {
            Dropzone.zones[i].dropzone.options.autoProcessQueue = true;
            Dropzone.zones[i].dropzone.processQueue();
        }
    };

    var CMYKError = 0;

    var addCMYKError = function () {
        var url = "/cmykerror";
        jQuery.get(url + "?pagecontentsegment=modal", function (data) {
            jQuery('#CMYKmodal').html(data);
            jQuery('#CMYKmodal').modal();
        });
        return false;
    }

	Dropzone.defaults={
		acceptedFiles:'.jpg,.jpeg,.tif,.tiff',
		parallelUploads: 2,
		batchOnly: true,
        timeout:0,
		maxThumbnailFilesize:4,
		autoProcessQueue:false,
		accept:function (file, done) {
			if (Dropzone.galleryid == null) {
				//initialize galleryid to empty to stop multiple creation
			    Dropzone.galleryid = '';
				//create gallery
				jQuery.ajax({
					url: '/Gallery/AddOrGetGallery'
				}).then(function (data) {
					Dropzone.galleryid = data;
					nextButtonGotoLocation(data);

					Dropzone.processQueues();

				});
			}
			done();
		},
		init:function(){
		    // Set up any event handlers
		    enableDisableNextButton();

			this.options.url = "/GalleryPhoto/UploadImagesModal";
			this.on('sending', function (file, xhr, formData) {

				formData.append('galleryId', Dropzone.galleryid);
			});


			this.on('addedfile', function () {
			    showhideuploadingtext('show');
			    window.onbeforeunload = function (ev) {
			        return 'If you leave this page your uploads will stop';
			    };
			});

			this.on('success', function (file, responseText) {
			    if (responseText[0]) {
                    if (responseText[0].Succeeded) {
                        showhideNoImagesUploaded('hide');
                        Dropzone.totalUploaded++;
                        setTotalUploaded(Dropzone.totalUploaded);
                        addImgToCompleted(file, this, responseText);
                    } else {
                        var message = responseText[0].ThumbPath;
                        this.defaultOptions.error(file, message);
                        if (CMYKError == 0 && message.includes("CMYK")) {
                            addCMYKError();
                            CMYKError++;
                        }
                    }
                } else {
                    this.defaultOptions.error(file, "Please try reuploading this file");
                }
			});

			this.on('complete', function (file) {

				if (this.getUploadingFiles().length === 0 && this.getQueuedFiles().length === 0) {
				    showhideuploadingtext('hide');
				    window.onbeforeunload = null;
				}
				enableDisableNextButton(this.getAcceptedFiles().length);
			});
		}
	};

    jQuery(document).ready(function () {
        jQuery('#addPhotosModal').on('hidden.bs.modal', function () {
            if (Dropzone.totalUploaded > 0) {
                location.reload();
            }
        });
    });

    function initDropZoneAndShowModal(galleryId) {
        var autoprocess = false;
        if (galleryId && galleryId != "") {
            Dropzone.galleryid = galleryId;
            autoprocess = true;
        }
        initDropZone();
        jQuery("#addPhotosModal").modal({
            backdrop: 'static',
            keyboard: false
        });
        if (autoprocess) Dropzone.processQueues();
    }

    function initDropZoneControl(sel,template) {
        var dz=jQuery(sel).dropzone({
            acceptedFiles: Dropzone.defaults.acceptedFiles,
            batchOnly: Dropzone.defaults.batchOnly,
            timeout: Dropzone.defaults.timeout,
            parallelUploads: Dropzone.defaults.parallelUploads,
			autoProcessQueue:Dropzone.defaults.autoProcessQueue,
			maxThumbnailFilesize:Dropzone.defaults.maxThumbnailFilesize,
            dictDefaultMessage: template,
            accept:Dropzone.defaults.accept,
            init: Dropzone.defaults.init,
            timeout: Dropzone.defaults.timeout
        });

		return dz[0];
    }

	function initDropZone(){
		//destroy all dropzones
		for(var i=0;i<Dropzone.zones.length;i++){
			try {
				Dropzone.zones[i].dropzone.destroy();
			} catch(ex){}
		}
		try {
			jQuery("#mobileCarousel").slick('unslick');
		} catch(ex){}

		Dropzone.totalUploaded=0;

		//create dropzones
		Dropzone.zones.push(initDropZoneControl('#galleryImgDropZone','<img src="//d2nysvt2e2u12u.cloudfront.net/img/NPLWeb_PhotoUploader_Uploader_button.jpg"/><h3><span>Drag and Drop </span><span style="font-size:15px; font-weight:normal; font-family:FuturaLT-Light">photos from your computer or</span></h3><div class="text-center"><button type="button" class="btn btn-primary">CHOOSE PHOTOS</button></div>'));
		Dropzone.zones.push(initDropZoneControl('#galleryImgDropZoneMobile','<img style="height:50px" src="//d2nysvt2e2u12u.cloudfront.net/img/NPLWeb_PhotoUploader_Uploader_button.jpg"/><h3><span>Tap here </span><span style="font-size:15px; font-weight:normal; font-family:FuturaLT-Light">photos from your computer or</span></h3><div class="text-center"><button type="button" class="btn btn-primary">CHOOSE PHOTOS</button></div>'));

		initmobileCarouselSlick();
	}

    function initmobileCarouselSlick() {
        jQuery("#mobileCarousel").slick({
            slidesToShow: 4,
            slidesToScroll: 4,
            nextArrow: '<img class="slick-next imgarrowright"  src="//d2nysvt2e2u12u.cloudfront.net/img/500w_NPLMobile_PhotoUploader_right_arrow.png"/>',
            prevArrow: '<img class="slick-prev imgarrowleft"  src="//d2nysvt2e2u12u.cloudfront.net/img/500w_NPLMobile_PhotoUploader_left_arrow.png"/>',
            arrows:false,
            responsive: [
                {
                    breakpoint: 767,
                    settings: {
                        slidesToShow: 2,
                        slidesToScroll: 2
                    }
                },
                {
                    breakpoint: 524,
                    settings: {
                        slidesToShow: 2,
                        slidesToScroll: 2
                    }
                }
            ]
        })
    }

    function showhideNoImagesUploaded(hs) {
        var noimagesdiv = jQuery('#noimages');
        var imagesdiv = jQuery("#imagesUploaded");
        if (hs == 'hide') {
            noimagesdiv.hide();
            imagesdiv.show();
            jQuery("#previews").removeClass('dropzoneactive');
            jQuery("#previewsmobile").show();
        } else {
            noimagesdiv.show();
            imagesdiv.hide();
        }
    }

    function showhideuploadingtext(hs) {
        if (hs == 'hide') {
            jQuery("#uploadingtext").hide();
            jQuery("#uploadingtextmobile").hide();
        } else {
            jQuery("#uploadingtext").show();
            jQuery("#uploadingtextmobile").show();
        }
    }

    function addImgToCompleted(file, dropzone, responsetext) {
		try {
			var dataurl = responsetext[0].ThumbPath.replace('https:', '');
			var name = file.name;
			if (name.length > 16) {
				name = name.substring(0, 16) + '...';
			}

			//append thumbnail to right side for desktop
			var desktopTemplate = '<div class="col-md-6"><div class="row pad2-lr"><img class="col-md-12 no-pad" src="' + dataurl + '" alt="' + name + '"/><div><div class="row pad2-lr"><div class="text-center pad2-tb">' + name + '</div></div></div>';
			var mobileTemplate ='<div class="col-xs-6"><img class="col-xs-12 no-pad" style="max-width:100" alt="' + name + '" src="' + dataurl + '"/></div>';

			//add slide to desktop and mobile
			jQuery(desktopTemplate).prependTo('#previewstemplate');
			jQuery("#mobileCarousel").slick('slickAdd', mobileTemplate,true);
		} catch(ex){
			console.log(ex);
		}
		//remove the file (most of the time)
        dropzone.removeFile(file);
    }

    function setTotalUploaded(numberofuploadedImages) {
        var iup = jQuery("#imagesUploaded");
        var iupmobile = jQuery("#imagesUploadedmobile");
        iup.html("Uploaded Images (" + numberofuploadedImages + "):");
        iupmobile.html("Uploaded Images (" + numberofuploadedImages + "):");
    }

    function setUploaderLocationClass(t) {
        var that = jQuery(t);
        var parent = jQuery("#uploaderLocation");
        var parentmobile = jQuery("#uploaderLocationmobile");
        var children = parent.children(".dropzoneactive");
        var childrenmobile = parentmobile.children(".dropzoneactive");
        children.each(function (ind, el) {
            jQuery(el).removeClass("dropzoneactive");
        })
        childrenmobile.each(function (ind, el) {
            jQuery(el).removeClass("dropzoneactive");
        })
        that.addClass("dropzoneactive");
    }

    function nextButtonRefresh() {
        location.reload();
    }

    function nextButtonGotoLocation(galleryId) {
        jQuery("#nextButton").attr('onclick', "");
        jQuery("#nextbuttonmobile").attr('onclick', "");
        jQuery("#nextButton").attr('onclick', 'location="/Users/ViewPhotos?galleryId=' + galleryId + '"');
        jQuery("#nextbuttonmobile").attr('onclick', 'location="/Users/ViewPhotos?galleryId=' + galleryId + '"');
    }

    function enableDisableNextButton( queuelength) {
        if (Dropzone.totalUploaded > 0 && queuelength == 0) {
            jQuery("#nextButton").attr('disabled', false);
            jQuery("#nextbuttonmobile").attr('disabled', false);
        } else {
            jQuery("#nextButton").attr('disabled', true);
            jQuery("#nextbuttonmobile").attr('disabled', true);
        }
    };

    function changeActionMethodForModal(t) {
        var newActionUrl = jQuery(t).attr('data-value');
        var newLocationUrl = jQuery(t).attr('data-newlocation');
        if (newLocationUrl != null) {
            nextButtonGotoLocation(newLocationUrl);
        }
        initDropZoneAndShowModal(newActionUrl);
    }
