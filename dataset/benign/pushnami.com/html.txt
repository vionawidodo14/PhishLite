<script type="text/javascript">
(function() {
  var pushStatsInitializationInterval = null,
      pushStatsCounterObject = null,
      pushStatsCounterUpdateDurationSeconds = 60,
      lastAPIResult = {
        when: null,
        count: 0
      },
      fallbackStatsProjection = {
        //9037400000 (+413 every second) starting on June 25, 2018 at 19:30 UTC
        when: (new Date('2018-06-25T19:30:00Z')).valueOf(),
        baseCount: 31087348552,
        incrementPerSecond: 413
      };

  //Generates fake data as a fallback when the API cannot be reached
  function getStatsCounterProjectedValue() {
    var lastCount = lastAPIResult.count;
    if (lastCount) {
      return lastCount + (fallbackStatsProjection.incrementPerSecond * (Date.now() - lastAPIResult.when.valueOf()) / 1000);
    } else {
      return fallbackStatsProjection.baseCount + (fallbackStatsProjection.incrementPerSecond * (Date.now() - fallbackStatsProjection.when) / 1000);
    }
  }
  
  //Responsible for periodically updating our stats counter
  function updateStatsCounter() {
    //Make a request to the API and attempt to get the latest counter statistic
    var xhr = new XMLHttpRequest();
    xhr.open('GET', 'https://api.pushnami.com/api/pushnami.com/push');
    xhr.onload = function() {
      var countToUpdateTo = null,
          isFirstAPIUpdate = false;
      if (xhr.status === 200) {
        var responseNumber = parseInt(xhr.responseText);
        if (responseNumber) {
          isFirstAPIUpdate = !lastAPIResult.count;
          countToUpdateTo = responseNumber;
          lastAPIResult.count = responseNumber;
          lastAPIResult.when = Date.now();
        }
      }
      if (!countToUpdateTo) {
        /* This is a special case whereby we cannot accurately get the latest count from the API.
           If/when this happens, we need to FAKE the data. */
        countToUpdateTo = getStatsCounterProjectedValue();
      }
      if (isFirstAPIUpdate || countToUpdateTo < pushStatsCounterObject.startVal) {
        /* If the new value is less than the old one, then decrement it and then immediately start counting up
           to give the illusion that our counter never stopped increasing. */
        var newStartVal = countToUpdateTo - (fallbackStatsProjection.incrementPerSecond * pushStatsCounterUpdateDurationSeconds);
        pushStatsCounterObject.reset(newStartVal >= 0 ? newStartVal : 0);
        pushStatsCounterObject.start();
      }
      pushStatsCounterObject.update(countToUpdateTo);
      setTimeout(updateStatsCounter, 15000);
    };
    xhr.onerror = xhr.onload;
    xhr.send();
  }
  
  //Responsible for initializing the stats counter as soon as practical
  function attemptStatsCounterInitialization() {
    //We will not attempt to initialize everything until our required libraries are available
    if (window.CountUp) {
      var initialStatsNumber = getStatsCounterProjectedValue();
      pushStatsCounterObject = new CountUp(
        'push-stats', //Element that we want to update
        initialStatsNumber,
        initialStatsNumber,
        0, //No decimal places
        pushStatsCounterUpdateDurationSeconds,
        {
          useEasing: true, 
          useGrouping: true, 
          separator: ',', 
          decimal: '.', 
        }
      );
      //Since we have initialized the counter, we can go ahead and wipe our interval and move to the fun stuff
      clearInterval(pushStatsInitializationInterval);
      if (!pushStatsCounterObject.error) {
        pushStatsCounterObject.start();
        updateStatsCounter();
      } else {
        console.error(pushStatsCounterObject.error);
      }
    }
  }
  
  //Get everything started!
  pushStatsInitializationInterval = setInterval(attemptStatsCounterInitialization, 100);
})();
</script>
