    var th = jQuery('th'),
    //li = jQuery('li'),
        inverse = false;

    th.click(function () {

        var isAuthor =false;
         if(($(this).children('a').html()=='Autor') ||($(this).children('a').html()=='Author')){
             isAuthor=true;
         }


        var header = $(this),
            index = header.index();

        header
            .closest('table')
            .find('td')
            .filter(function () {
                return $(this).index() === index;
            })
            .sort(function (a, b) {

                a = $(a).text();
                b = $(b).text();

                if(isAuthor){
                    var reta =a.replace(/\s{2,}/g, ' ').split(' ')
                    var retb =b.replace(/\s{2,}/g, ' ').split(' ')

                    if(reta[1]!='') a =reta[1];
                    if(retb[1]!='') b =retb[1];
                }


                return (
                    isNaN(a) || isNaN(b) ?
                        a > b : +a > +b
                    ) ?
                    inverse ? -1 : 1 :
                    inverse ? 1 : -1;

            }, function () {

                return this.parentNode;

            });

        inverse = !inverse;

    });
