function validate_form()
{

	var forms = document.frm_search;
	var valid_form 			= true;
	if ((forms.job_category_id.value == ""))
	{
		
		$('#message_keyword').html('Category  required');
		$('#message_keyword').show();
		$('#message_keyword').focus();
		valid_form 		= false;
	}
	
	if(valid_form)
	{
		$('#frm_search').submit();
	}

	

}
