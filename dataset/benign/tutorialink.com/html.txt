<script>
	function getNewsletter() {
		var email = $("#email").attr('value');
		var mailformat = /^\w+([\.-]?\w+)*@\w+([\.-]?\w+)*(\.\w{2,3})+$/;
		if (email.length === 0 ) {
			$("#n_letter").html('<input type="text" name="email" id="email" class="form-control newsletter-input" placeholder="Please Enter your Email ID"><span class="input-group-btn"><button class="btn btn-inverse btn-hg" type="submit" name="submit" id="sign-up-btn">Subscribe</button></span>');
		} else if(!email.match(mailformat)) {
			$("#n_letter").html('<input type="text" name="email" id="email" class="form-control newsletter-input" placeholder="Please enter valid Email id. ex. xyz@domain.com"><span class="input-group-btn"><button class="btn btn-inverse btn-hg" type="submit" name="submit" id="sign-up-btn">Subscribe</button></span>');
		} else if (email.length > 0 ) {
		 $.ajax({
				type: "POST",
				url: "newsletter.html",
				data: "email="+email,
				cache: false,
				beforeSend: function () {
					$('#n_letter').html('<input type="text" name="email" id="email" class="form-control newsletter-input" placeholder="Siging-Up..."><span class="input-group-btn"><button class="btn btn-inverse btn-hg" type="submit" name="submit" id="sign-up-btn">Subscribing..&nbsp;&nbsp;<img src="img/loading.gif" alt="" width="20" height="20" /></button></span>');
				},
				success: function(html) {
					$("#n_letter").html( html );
				}
			});
		}
		return false;
	};
</script>
