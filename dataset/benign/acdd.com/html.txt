jQuery(document).ready(function(){

	jQuery('#loginbox ul.links li:nth-child(2)').append('<span id="cart-item-qty"></span>');
	window.qty_added = 0;
	get_miniCartItem_JSON();

	var fadeOut = false;
	jQuery('#loginbox ul.links li:nth-child(2)').on({
		mouseover: function(){
			if(jQuery('#cart-item-qty').text() != 0)
				jQuery('#cart-hover-panel-table').fadeIn('slow');
		},
		mouseleave: function(){			
			fadeOut = true;			
			jQuery('#cart-hover-panel-table').on({
				mouseenter: function(){
					fadeOut = false;					
				},
				mouseleave: function(){
					jQuery('#cart-hover-panel-table').fadeOut('slow');
				}
			});	
			setTimeout(function(){			
				if(fadeOut){
					jQuery('#cart-hover-panel-table').fadeOut('slow');
				}		
			},700);													
		}		
	});	
}); 

window.get_miniCartItem_JSON = function(){
	jQuery.ajax({
		url:'https://www.acdd.com/checkout/cart/collectCartInfo/',
		method:'POST',		 
		cache: false
		 		
	}).done(function(response){			 
		 var res = JSON.parse(response);	
		 var key_arr = Object.keys(res);
		 var table_html = '<table id="cart-hover-panel-table"><thead><tr><th>#</th><th>Product Name</th><th>Qty</th></tr></thead><tbody>';
		 for(var i=0;i<key_arr.length;i++){
		 	table_html += '<tr><td>'+(i+1)+'</td><td>'+res[key_arr[i]].name+'</td><td>'+res[key_arr[i]].qty+'</td></tr>';

			var qty_total = res[key_arr[i]].qty,
				itemId = jQuery('#cart-qty-msg-reloaded').attr('data-item');
				
				
			//If product in cart is the same as currently viewing, display amount
			if(key_arr[i] === itemId) {
				//Hide the non-ajax counter
				jQuery('#cart-qty-msg').hide();
				//Show the new counter of cart qty specific to item
				jQuery('#cart-qty-msg-reloaded').html('').append('You have '+qty_total+' in your shopping cart');

				if(!jQuery('#product-view-initial-load').size())
				{
					var currentTotalAlloc = parseInt(jQuery('#allocation-limit-initial').val());
					var newTotalAlloc = currentTotalAlloc - window.qty_added
					jQuery('#allocation-limit-count').html(newTotalAlloc);
					jQuery('#allocation-limit-initial').val(newTotalAlloc)
				}
			}
		 }
		if(jQuery('#product-view-initial-load').size())
		{
			jQuery('#product-view-initial-load').remove();
		}
		 table_html += '</tbody></table>';
		 jQuery('#cart-hover-panel').html(table_html);
		 jQuery('#loginbox ul.links #cart-item-qty').html(key_arr.length);

	}); 
};
</script>
