A.mediarules.getCartVariables = function () {
	var cart = A.context.cart;
	var results = {
		products: [],
		quantities: []
	};
	if (cart != null && cart.entries != null) {
		cart.entries.forEach(function (cartEntry) {
			var product = cartEntry.product || cartEntry.offer.product;
			results.products.push(product.code);
			results.quantities.push(cartEntry.quantity);
		});
		if (cart.totalPrice != null && cart.totalPrice.value != null) {
			results.bS = (cart.totalPrice.value).toString();
		} else {
			results.bS = '0';
		}
	}
	return results;
};
A.mediarules.handleResponse = function (response) {
	var _event = {};
	_event.event = "mediarule_loaded";
	var _dataLayer = {};
	_dataLayer.ecommerce = {};
	_dataLayer.ecommerce.promoView = {};
	_dataLayer.ecommerce.promoView.promotions = [];
	if (response != null) {
		var spaces = response.spaces;
		if (spaces != null) {
			var keys = Object.keys(response.spaces);
			for (var i = 0; i < keys.length; i++) {
				var containerEl = A.utils.qs("#" + keys[i]);
				if (containerEl != null) {
					var trackingPosition = containerEl.dataset.position || '';
					var tmpEl = A.utils.createElement("tmp", null, spaces[keys[i]]);
					for (var j = 0; j < tmpEl.children.length; j++) {
						tmpEl.children[j].classList.add("mediarules--item");
						var promotion = {
							'id': tmpEl.children[j].dataset.trackingid || '',
							'name': tmpEl.children[j].dataset.trackingname || '',
							'creative': tmpEl.children[j].dataset.trackingcreative || '',
							'position': trackingPosition
						};
						_dataLayer.ecommerce.promoView.promotions.push(promotion);
						tmpEl.children[j].setAttribute("data-space-id", keys[i]);
						tmpEl.children[j].setAttribute("onclick", "dataLayer.push({'event': 'promotionClick','ecommerce': {'promoClick': {'promotions': [" + JSON.stringify(promotion) + "]}}});");
						containerEl.parentElement.insertBefore(tmpEl.children[j], containerEl);
					}
					containerEl.parentElement.removeChild(containerEl);
				}
			}
			A.mediarules.removeLoader();
		}
		if (_dataLayer.ecommerce.promoView.promotions != null && promotion != null && dataLayer != null) {
			dataLayer.push(_dataLayer);
		}
		if (_event != null && dataLayer != null) {
			dataLayer.push(_event);
		}
	}
};
