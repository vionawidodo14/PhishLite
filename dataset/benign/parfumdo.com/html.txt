        function keydownHandler(event) {
            // permet de savoir si un traitement à été réalisé ou si la touche n'est pas concernée
            var handled = false;
            // liste des elements présents à l'instant T dans la liste
            var lis = $('#list-o-marque').find('.em-elem');
            // si on monte ou descend
            if (event.keyCode==keycodes.FLECHE_BAS || event.keyCode==keycodes.FLECHE_HAUT) {
                // on désactive les li
                lis.removeClass('active');
                // on maximise et minimise l'index de l'item actif avec le nombre d'items disponibles
                active_index = event.keyCode==keycodes.FLECHE_BAS ? Math.min(++active_index,lis.length-1) : Math.max(--active_index,0);

                lis.eq(active_index).addClass('active');
                handled = true;
                // si on appuye sur entree et qu'on à un item selectionné
            } else if (event.keyCode==keycodes.ENTREE && active_index >= 0) {
                // récuperation du lien, et redirection de l'utilisateur

            else if (event.keyCode==keycodes.ECHAP && active_index >= 0) {

                active_index = -1;
                handled = true;

            // si l'event à déclanché un traitement, on stop la propagation pour éviter le scroll à l'écran (touche haut / bas)

                e.preventDefault();
                e.stopPropagation();
                return false;
