$(document).ready(function ($) {

window.__symudol && $(window).resize(onPageLayoutChange);

function onPageLayoutChange() {
  if($(window).width() >= 320 && $(window).width() < 768) {
    if($('#compact_slides').length == 0) {
      var slides = "<div id='compact_slides'>";
      var i = 0;
      
      $('.homepage-banner-rhs-ch h2').each(function() {
        if((i%3) == 0) {
          if(i > 0) {
            slides += "</div>";
          }
          slides += "<div class='tldonsale-slide'"+(i > 0 ? " style='display:none;'" : "")+">";
        }
        
        slides += "<span class='tldonsale'><span class='tldnameval'>"+$(this).text().toLowerCase()+"</span><span class='price'>"+$(this).parent().find('strong').text()+"</span></span>";
        i++;
      });
      
      slides += "</div></div>";
      $('.homepage-banner-lhs').append(slides);
    } else {
      $('#compact_slides').show();
    }
    
    var interval_id = $('#compact_slides').data('interval_id');
    
    if(!interval_id) {
      interval_id = setInterval(function() {
          $('#compact_slides > div:first').fadeOut(1000).next().fadeIn(1000).end().appendTo('#compact_slides');
        }, 5000);
        
      $('#compact_slides').data('interval_id', interval_id);
    }
  } else if($('#compact_slides').length == 1) {
    $('#compact_slides').hide();
    
    var interval_id = $('#compact_slides').data('interval_id');
    
    if(interval_id) {
      clearInterval(interval_id);
      $('#compact_slides').removeData('interval_id');
    }
  }
}

function check_domain_name() {
  var query = $.trim($("#field_domains-input").val()).toLowerCase();
  var arr = query.match(/^http(s?):\/\/([^:\/]+)/);
  
  if(arr) {
    query = arr[2];
  }

  if(query.indexOf("www.") == 0) {
    query = query.substring(4);
  }
  
  var pos = query.indexOf("/");
  
  if(pos > 1) {
    query = query.substring(0, pos);
  }
  
  query = $.trim(query);
  
  if(query.length == 0 || query.search(/[^a-z0-9-.\s]/) != -1) {
    $('form[name="form_domain_avail"]').addClass("validation-error");
    $("#error_msg").show();
    return false;
  }
  
  $("#field_domains-input").val(query);
  return true;
}

function remove_error_msg(){
  $('form[name="form_domain_avail"]').removeClass("validation-error");
  $("#error_msg").hide();
}
