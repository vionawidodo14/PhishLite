            ga('Client.send', 'event', 'Feedback text', currentPath, msg);
            
            ga('Waas.send', 'event', 'Feedback text', currentPath, msg);
            
        }
        else if ((t = c.closest('a[href*="."]')).length) {
            if ((h = (t.attr('href') || '')).indexOf('http') === 0 && h.indexOf(currentDomain) === -1) {
                
                ga('Client.send', 'event', 'Exit click on page', currentPath, h);
                
                ga('Waas.send', 'event', 'Exit click on page', currentPath, h);
                
            } else if (downloadFileTypes.test(h)) {
                
                ga('Client.send', 'event', 'Exit click on page', currentPath, h.replace(/^https?:\/\/[^\/]*/i, ''));
                
                ga('Waas.send', 'event', 'Exit click on page', currentPath, h.replace(/^https?:\/\/[^\/]*/i, ''));
                
            } else if (/^mailto:/i.test(h)) {
                
                ga('Client.send', 'event', 'Email click on page', currentPath, $.trim(t.text()).replace(/\s+/g, ' '));
                
                ga('Waas.send', 'event', 'Email click on page', currentPath, $.trim(t.text()).replace(/\s+/g, ' '));
                
            }
        }
    });
})();
