<script type="text/javascript">
  function disp_full_title(id){
    opacity_hover_fix(id, true);
    if ( ! is_shortened(id) ) { return; }
    copy_style(id);
    var ele = document.getElementById('title_' + id);
    ele.style.display = "block";
  }
  function hide_full_title(id){
    opacity_hover_fix(id, false);
    var ele = document.getElementById('title_' + id);
    ele.style.display = "none";
  }
  function is_shortened(id){
    return has_ellipsis( $("#short_title_" + id) );
  }

  function has_ellipsis(obj){
    if( obj.css("text-overflow") == "ellipsis" )
      return true;
    var children = obj.children();
    if(! children){ return false; }
    for(var i=0; i!=children.length; i++){
      var child = children.eq(i);
      if( has_ellipsis(child) ){
        return true;
      }
    }
    return false;
  }

  function copy_style(id){
    var $short = $('#short_title_' + id);
    var $full = $('#title_' + id);
    $full.css("font-size", $short.css("font-size"));
    $full.css("line-height", $short.css("line-height"));
    //$full.css("color", $short.css("color"));
    $full.css("font-family", $short.css("font-family"));
  }

  function opacity_hover_fix(id, hover){
    var $short = $('#short_title_' + id);
    if(hover){
      $short.css("opacity", "1");
      $short.css("color", "#8F8F8F");
      return;
    }
    $short.css("color", "#222");
  }
</script>
