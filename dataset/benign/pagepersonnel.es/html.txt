function detectLanguageSwitcher() {
	// detect if the page has a language switcher for the homepage banner
	if ( $('.language-switcher-block').length && $(location).attr('pathname').length <= 3 ) {

		if ($debug) {
			console.log("Detected language switcher on this page");
		}

		// do we have one single extra language or multiple
		if ( $('.language-switcher-block .single').length ) {
			$languagePathData = $('.language-switcher-locale-url .first.last a').attr('href');
			if ($debug) {
				console.log("Only a single extra language detected");
			}
		} else {
			// get the active language and the href attribute # = no path suffix, all other cases would be to include path
			$languagePathData = $('.language-switcher-locale-url .active a').attr('href');
			if ($debug) {
				console.log("Multiple languages deteceted");
			}
		}

		if ($debug) {
			console.log("Language path data is " + $languagePathData);
		}
		if ( $languagePathData === '#' ) {
			// if the path is a # set the path to just a slash
			$languagePath = '/';
			if ($debug) {
				console.log("Language path active is # therefore no changes needed");
			}
		} else {
			// if the data is not a hash, it will be valid path suffix
			$languagePath = $languagePathData;
			if ($debug) {
				console.log("Language path needs to be modified, new value is " + $languagePath);
			}
		}

		if ($debug) {
			console.log("The language path has been set to " + $languagePath);
		}
		return $languagePath;
	}
}

function getCookie($cookieName) {
  var value = "; " + document.cookie;
