<script>
	var escPulse = 0;
	function hideMainMenu() {
		if (escPulse == 0 && $(".sen-login-mobile") != undefined && $(".sen-navbar-mobile") != undefined) {
			if ($(".sen-login-mobile").hasClass("show")) {
				$(".sen-navbar-mobile").children().show();
			} else {
				$(".sen-navbar-mobile").children().hide();
				$(".sen-navbar-mobile").children().last().show();
			}
		} else {
			escPulse = 0;
		}
	}
	function showAllMainMenu() {
		$(".sen-navbar-mobile").children().show();
	}
	document.onkeydown = function(evt) {
	    evt = evt || window.event;
	    if (evt.key === "Escape") {
	    	escPulse = 1;
	    	showAllMainMenu();
	    }
	}
