    function valForm() {
        //alert($("#contact").val());
        if ($("#contact").val() == "" || $("#email").val()=="" || $("#countryID").val() == -1)  {
            $("#agreeMsg").text('Name, Email and Country are required');
            $("#valMsg").show();
            //alert($("#countryID").val());
        }
        else {
            if ($("#agree").prop('checked')) {
                //clear message if visibile
                $("#valMsg").hide();
                //alert("send form");
                sendForm();
            }
            else {
                $("#agreeMsg").text('You must agree to our Privacy Policy');
                $("#valMsg").show();
            }

        }
    }


    function sendForm() {
        var email = $("#email").val();
        var contact = $("#contact").val();
        var phone = $("#phone").val();
        var company = $("#company").val();
        var country = $("#countryID option:selected").val()
       var responses = $.ajax({
        type: "POST",
        url: '/AjaxCalls.aspx/EnewsSignUp',
        contentType: "application/json; charset=utf-8",
        data: '{_email: "' + email + '", _name:"' + contact + '", _phone:"' + phone + '",_company:  "' + company + '",  _countryID:"' + country +  '"}',
        dataType: "json",
        async: true,
        cache: false,
        crossDomain: true,

        success: function (msg) {
            if (msg.d) {
                //reload modal
                $("#divSuccess").show();
                $("#btnEnews").hide();
            }
        },
        error: function (error) {
            alert(error.statusText);
        }
    });
    }
