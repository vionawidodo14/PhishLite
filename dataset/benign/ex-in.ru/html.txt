function loadCommonModal( obj, e )
{
     var signup_form_name = 'signupemailforminmodal';
    
    e.preventDefault();
    $('#registrationInBlogModal').modal('hide');
    
    var modal_header_text = obj.attr( 'data-modal_header' );
    var modal_button_text = obj.attr( 'data-modal_button' );
    
    if (modal_header_text != undefined && modal_header_text != '') {
        $( '#' + signup_form_name + 'regFormText' ).html( modal_header_text );

    if (modal_button_text != undefined && modal_button_text != '') {
        $( '#' + signup_form_name + 'regFormSubmit' ).html( modal_button_text );
    }
}
