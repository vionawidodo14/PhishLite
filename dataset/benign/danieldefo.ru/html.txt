<script>
/*
 * Collapse plugin for jQuery
 * --
 * source: http://github.com/danielstocks/jQuery-Collapse/
 * site: http://webcloud.se/jQuery-Collapse
 *
 * @author Daniel Stocks (http://webcloud.se)
 * Copyright 2013, Daniel Stocks
 * Released under the MIT, BSD, and GPL Licenses.
 */

(function($, exports) {

  // Constructor
  function Collapse (el, options) {
    options = options || {};
    var _this = this,
      query = options.query || "> :even";

    $.extend(_this, {
      $el: el,
      options : options,
      sections: [],
      isAccordion : options.accordion || false,
      db : options.persist ? jQueryCollapseStorage(el.get(0).id) : false
    });

    // Figure out what sections are open if storage is used
    _this.states = _this.db ? _this.db.read() : [];

    // For every pair of elements in given
    // element, create a section
    _this.$el.find(query).each(function() {
      new jQueryCollapseSection($(this), _this);
    });

    // Capute ALL the clicks!
    (function(scope) {
      _this.$el.on("click", "[data-collapse-summary] " + (scope.options.clickQuery || ""),
        $.proxy(_this.handleClick, scope));

      _this.$el.bind("toggle close open",
        $.proxy(_this.handleEvent, scope));

    }(_this));
  }

  Collapse.prototype = {
    handleClick: function(e, state) {
      e.preventDefault();
      state = state || "toggle";
      var sections = this.sections,
        l = sections.length;
      while(l--) {
        if($.contains(sections[l].$summary[0], e.target)) {
          sections[l][state]();
          break;
        }
      }
    },
    handleEvent: function(e) {
      if(e.target == this.$el.get(0)) return this[e.type]();
      this.handleClick(e, e.type);
    },
    open: function(eq) {
      this._change("open", eq);
    },
    close: function(eq) {
      this._change("close", eq);
    },
    toggle: function(eq) {
      this._change("toggle", eq);
    },
    _change: function(action, eq) {
      if(isFinite(eq)) return this.sections[eq][action]();
      $.each(this.sections, function(i, section) {
        section[action]();
      });
    }
  };

  // Section constructor
  function Section($el, parent) {

    if(!parent.options.clickQuery) $el.wrapInner('<a href="#"/>');

    $.extend(this, {
      isOpen : false,
      $summary : $el.attr("data-collapse-summary",""),
      $details : $el.next(),
      options: parent.options,
      parent: parent
    });
    parent.sections.push(this);

    // Check current state of section
    var state = parent.states[this._index()];

    if(state === 0) {
      this.close(true);
    }
    else if(this.$summary.is(".open") || state === 1) {
      this.open(true);
    } else {
      this.close(true);
    }
  }

  Section.prototype = {
    toggle : function() {
      this.isOpen ? this.close() : this.open();
    },
    close: function(bypass) {
      this._changeState("close", bypass);
    },
    open: function(bypass) {
      var _this = this;
      if(_this.options.accordion && !bypass) {
        $.each(_this.parent.sections, function(i, section) {
          section.close();
        });
      }
      _this._changeState("open", bypass);
    },
    _index: function() {
      return $.inArray(this, this.parent.sections);
    },
    _changeState: function(state, bypass) {

      var _this = this;
      _this.isOpen = state == "open";
      if($.isFunction(_this.options[state]) && !bypass) {
        _this.options[state].apply(_this.$details);
      } else {
        _this.$details[_this.isOpen ? "show" : "hide"]();
      }

      _this.$summary.toggleClass("open", state !== "close");
      _this.$details.attr("aria-hidden", state === "close");
      _this.$summary.attr("aria-expanded", state === "open");
      _this.$summary.trigger(state === "open" ? "opened" : "closed", _this);
      if(_this.parent.db) {
        _this.parent.db.write(_this._index(), _this.isOpen);
      }
    }
  };

  // Expose in jQuery API
  $.fn.extend({
    collapse: function(options, scan) {
      var nodes = (scan) ? $("body").find("[data-collapse]") : $(this);
      return nodes.each(function() {
        var settings = (scan) ? {} : options,
          values = $(this).attr("data-collapse") || "";
        $.each(values.split(" "), function(i,v) {
          if(v) settings[v] = true;
        });
        new Collapse($(this), settings);
      });
    }
  });

  // Expose constructor to
  // global namespace
  exports.jQueryCollapse = Collapse;
  exports.jQueryCollapseSection = Section;
  
  //jQuery DOM Ready
  $(function() {
    $.fn.collapse(false, true);
  });

})(window.jQuery, window);
</script>

/*
 * Storage for jQuery Collapse
 * --
 * source: http://github.com/danielstocks/jQuery-Collapse/
 * site: http://webcloud.se/jQuery-Collapse
 *
 * @author Daniel Stocks (http://webcloud.se)
 * Copyright 2013, Daniel Stocks
 * Released under the MIT, BSD, and GPL Licenses.
 */

(function($) {

  var STORAGE_KEY = "jQuery-Collapse";

  function Storage(id) {
    var DB;
    try {
      DB = window.localStorage || $.fn.collapse.cookieStorage;
    } catch(e) {
      DB = false;
    }
    return DB ? new _Storage(id, DB) : false;
  }
  function _Storage(id, DB) {
    this.id = id;
    this.db = DB;
    this.data = [];
  }
  _Storage.prototype = {
    write: function(position, state) {
      var _this = this;
      _this.data[position] = state ? 1 : 0;
      // Pad out data array with zero values
      $.each(_this.data, function(i) {
        if(typeof _this.data[i] == 'undefined') {
          _this.data[i] = 0;
        }
      });
      var obj = this._getDataObject();
      obj[this.id] = this.data;
      this.db.setItem(STORAGE_KEY, JSON.stringify(obj));
    },
    read: function() {
      var obj = this._getDataObject();
      return obj[this.id] || [];
    },
    _getDataObject: function() {
      var string = this.db.getItem(STORAGE_KEY);
      return string ? JSON.parse(string) : {};
    }
  };

  jQueryCollapseStorage = Storage;

})(jQuery);
</script>

<div class="subHeading">
<a href="threads/rossija-ce-evropa.10795/">Россия це Европа</a>
</div>
<div class="primaryContent">
<div class="baseHtml excerptContent">
<blockquote class="coolqoute">Нет никаких оснований считать, что население в Европе "более качественное", чем на других континентах. Люди есть люди. Они слабы, грешны и несовершенны на любом материке. Каждый, кто хоть в какой-то степени причастен к христианской традиции, должен это понимать. Говорят, и саму Церковь организовал человек, трижды отрекшийся от Господа. Лучших не нашлось.<br/>
<br/>
Превосходство европейской цивилизации как раз в том, что, располагая крайне несовершенным человеческим материалом, она смогла на порядок превзойти все остальные и материальным благополучием, и уровнем культуры, и человечностью взаимоотношений внутри социума.<br/>
<br/>
Все знают сентенцию, восходящую, если не ошибаюсь, к наполеоновским войнам. "Один мамелюк в отдельной схватке побеждает одного француза, пять мамелюков бьются на равных с пятью французами, тридцать французов могут обратить в бегство сотню мамелюков". Это высказывание существует в двух десятках вариантов, но суть всегда одна. Европейцы сами по себе не качественнее. Качественнее европейская организация дел.<br/>
<br/>
Русским пришлось на себе испытать оба подхода - европейский и азиатский. В двух мировых войнах они сражались против одного и того же противника. Русский солдат ничуть не изменился. Но в первую войну у русских был европейский офицерский корпус, а во второй войне ими командовали какие-то мамелюки (главнокомандующим и вовсе был один абрек). B первой войне противник был остановлен на линии Рига-Львов (при этом ход войны считался не слишком удачным, так что недовольством общества в определённый момент воспользовались революционеры). Во второй неприятель доходил до берегов Волги и предгорий Кавказа (при этом победа над ним празднуется до сих пор, как величaйшее достижение страны). Но главное - потери во второй войне были на порядок выше, чем в первой.<br/>
<br/>
Вот эта десятикратная (если не больше) разница в потерях и выражает разницу между европейской и азиатской цивилизацией. У нормального человека она вызывает ужас, как и все азиатские нормы жизни, поэтому идея европеизации обычно воспринимается без малейших сомнений всеми, у кого есть на неё хоть какой-то шанс. Более того, ей увлечены и те, у кого шансов нет.<br/>
<br/>
В ЖЖ модно хвалиться, что автор шесть лет не смотрел телевизор. Зря. Иногда полезно и посмотреть. Если Вы увидите на каком-нибудь европейском телеканале грузина, то грузин, скорее всего, будет говорить, стараясь ногой вытолкнуть из кадра пасущегося рядом барана : "Мы - часть Западной цивилизации, форпост Европы". Ну ладно грузины. Они христиане. Но турки - мусульмане и пришельцы из Великой Степи. Они рвутся в Европу так, как туда не рвался ещё никто. Буквально - их гонят в дверь, а они в окно.<br/>
<br/>
Прорвутся или нет, я не знаю. Скорее, нет. Мне как-то попалось на глаза интервью с французским министром иностранных дел (бывшим), где он прямо говорил: их проблема не в том, что они мусульмане, а в том, что они проамериканские. 80 миллионов друзей Америки европейцам не нужны. Для Турции вступление в ЕС - иллюзорная цель. Но ведь тут важно не само вступление, а процесс подготовки к нему. Вслух европейцы никогда не скажут, почему не хотят видеть турок за своим столом. Им приходится говорить, что Турция не соответствует таким-то и таким-то стандартaм. Там у турок плохо с правами человека, здесь они зажали свободу прессы, тут качество иx продукции не выдерживает критики, к тому же сто лет назад они обидели армян. Но если четвёртый пункт явно надуман и призван заблокировать дорогу турецкому государству, то первые три - это спасательный круг для турецкого социума. Властям не остаётся ничего другого, как приближать определённые сферы жизни к европейским стандартам, а турецкому народу именно это и нужно.<br/>
<br/>
Это же нужно и русским. <span class="textShine">Ведь смысл европеизации не в том, чтобы страна вступила в какое-то европейское объединение, а в том, чтобы в стране воцарились европейские нормы жизни</span>. И не в карикатурном виде, как этого добивается "либеральная" "оппозиция", нагло и подло обманывающая народ утверждениями, будто Европа - это гей-парады и мультикультурализм. Гей-парады - вещь в себе, не интересующая никого, кроме геев. <span class="textShine">А европейские нормы жизни - это подконтрольные обществу чиновники, умеренные цены, серьёзные инвестиции в здравоохранение, достойные пенсии, детские пособия, на которые может прожить даже потерявшая работу семья, и охрана памятников архитектуры, перед которой бессильны любые деньги потенциальных застройщиков.</span><br/>
<br/>
И последнее, на что я ещё могу ответить в пределах этого поста. Даже следователи не в состоянии установить, кто является владельцем крупнейшего аэропорта страны "Домодедово". Следы обрываются на управляющей компании, зарегистрированной в оффшоре на острове Мэн <img alt=":babalisa:" class="mceSmilieSprite mceSmilie35" src="styles/default/xenforo/clear.png" title="Баба Лиза    :babalisa:"/>. Но в РФ едва ли не всё, что действительно чего-то стоит, так или иначе принадлежит фирмам, зарегистрированным на Мэне, Джерси, Кипре и в других оффшорах (обычно - находящихся под контролем британской короны). Это стандартная ситуация для Второго и Третьего миров. Запад работает на себя, остальное человечество работает на Запад.<br/>
<br/>
Прилагательное "европейская" неотделимо от слова Россия. Но в одну эпоху её полным определением было выражение "европейская страна" (1721-1917), а в другую - "европейская колония" (1917-?).</blockquote>
<a class="cqauth" href="https://bohemicus.livejournal.com/50626.html">bohemicus</a><br/>
<br/>
Французский президент Макрон не хуцпы ради назвал Россию "очень глубоко европейской страной". Во Франции до сих пор можно увидеть артефакты того, уничтоженного у нас, Русского-Европейского вектора. Это правда, в том числе потому дети нынешней рашкованской элитки так стремятся получить гражданства Европы, живут там и бегают с принтами наших царей.<br/>
<br/>
Внутренние проблемы искалеченной территории нашего отечества в том, что по сути русское (европейское) население пытаются затолкать в азиатское гетто, через сверхнизкий уровень жизни и надругательство над культурой, замещая её фантомом советского урода или рассказывая, что никаких русских нет, никакой русской истории нет, а сотворение мира началось с некой "Победы", где почти тридцать миллионов человек (половину населения Франции) трупами обменяли на небольшой город немецкой провинции.<br/>
<br/>
Русская культура в ассоциативном ряде её носителей (т.е. народа) сведена до примитивного фольклора - кокошник, матрешка, балалайка, ауе и 666 нацменов, наших "друзей", которым стоит отдать все посты в государстве и все деньги. Чтобы вот их дети жили в Европе.<br/>
<br/>
Сто лет работают, без выходных.<br/>

<div class="subHeading">
<a href="threads/to-o-chem-ne-zadumyvalsja.10775/">То, о чем не задумывался</a>
</div>
<div class="primaryContent">
<div class="baseHtml excerptContent">
<img alt="image.jpg" class="bbCodeImage LbImage" src="https://danieldefo.ru/attachments/image-jpg.30654/" style=""/>
<br/>
Тяжелее жизни в России - старость в России. Не говоря уже об инвалидности.<br/>
<br/>
Был вынужден свернуть с маршрута, отправившись снова туда, где плохо и нуждаются в поддержке, но уже давно ни на кого не надеются и ничего не ждут. Кроме смерти.<br/>
<br/>
На фото стол с попыткой продлить бабушке ежемесячную оплату коммунальных платежей. Единственное на что хватает 9 000 рублей пенсии.<br/>
<br/>
Безусловно тщетно из-за запущенности, но я бы очень хотел. Неделю изучаю особенности платной медицины 14 века. Но и на неё нет денег. На территории рекордных темпов убыли населения нет былых заслуг, стажа или накоплений. Все бабки, кто дожил до нищенской пенсии, подобны фараонам, но думают не о загробном, а о маленьком зиккурате.<br/>
<br/>
77 лет, из которых лет 7 подготовка к своим и дедовым похоронам, не повод лишать меня фирменных блинов. Неказистых как всё остальное вокруг, но хотя бы с маслом. Когда то даже не пальмовым!<br/>
<br/>
Любая продолжительность жизни на территории без медицины, но чудовищной экологии заключается в базовой прочности организма.<br/>
Если она есть, можно дожить живым трупом и до 90. Ни разу не обратившись в местную ветлечебницу за уколом физраствора на проточной воде или зеленкой.<br/>
К сожалению не быстрому отмиранию тела сопутствуют психологические и психические расстройства.<br/>
<br/>
Как не начать сходить с ума уже после тридцати не до конца ясно. Дальше деменция и резиновые свиньи.<br/>
Все больше понимаю мудрость сильных волей мужей, что пустили себе пулю под занавес. - Пора и честь знать.<br/>
<br/>
А так, живут люди уже с середины срока "сломанными" болванчиками.<br/>
<br/>
Бабка моя - стальная была женщина, двужильная, но сердце и сосуды надорвала. Жизнь её была тяжела, ребенок воины, тянула за троих.<br/>
Все мы на ней выросли как ракушки и коральчики на панцире старой черепахи.<br/>
В сознании и интеллекте до сих пор (отвечает на вопросы викторины), но уже нет зрения для книг, а от федеральных каналов портится характер и настроение.<br/>
Такой Бильбо Беггинс, в которого иногда вселяется жадность до кольца и просрочки. Желудок как у гиены или киборга. Остатки былой сборки.<br/>
<br/>
<img alt="image.jpg" class="bbCodeImage LbImage" src="https://danieldefo.ru/attachments/image-jpg.30656/" style=""/>
<br/>
<br/>
Высушены вены, а весь мозг в рубцах микроинсультов. Запередьное давление, мерцательная аритмия и едва фурычущее сердце. Переживающее за каждого из нас, живого или ушедшего.<br/>
Никакого лечения, молчала и ела мел, приписанный во время визита в<br/>
Петербург полутрезвым фельдшером скорой. Он ей и проткнул вены в процессе инсульта.<br/>
<br/>
Будем капаться на дому, накатались. Зарядил десять капельниц и добыл гору препаратов, купил редкий прибор, что не сбоит при прерывистом пульсе. Обнимаю и целую, потому что говорит, что не хочет жить, устала так. Но вижу, что радуется каждому прикосновению как ребенок.<br/>
<br/>
<img alt="image.jpg" class="bbCodeImage LbImage" src="https://danieldefo.ru/attachments/image-jpg.30655/" style=""/>
<br/>
<br/>
Мое счастливое детство еще теплится, тяжело вздыхает по ночам, наровит накинуть мне шаль на плечи и ноги.<br/>
Я не против, отопление дадут, Бог даст, к ноябрю.

<br/>
Теоретически можно представить, что после гигантского социального катаклизма у власти окажутся выходцы из низов и в течение нескольких десятилетий образуют новую элиту, образовав династии, в которых будут передаваться навыки управления, детям будет дано очень хорошее образование и т.д. Но ничего подобного не произошло. По меньшей мере вплоть до Ельцина у руля неизменно оказывались выдвиженцы из самых низов. Люди практически случайные, не имевшие ни достаточного образования, ни интеллекта, ни воспитания, вообще ничего.<br/>
<br/>
Конечно, теоретически можно представить и существование тайной элиты, полностью ушедшей в тень и прикрывшейся беспомощными марионетками. Есть даже попытки найти в СССР такую криптоэлиту. Но кроме нескольких имён в ранний период (братья Бонч-Бруевичи, Джунковский и т.п.) обнаружить невозможно ничего. То немногое, что было, кончилось очень быстро.<br/>
<br/>
Это значит, что советское руководство на международной арене оказывалось в роли деревенских дурачков в мегаполисе, полном прожжёных профессиональных мошенников.<br/>
<br/>
<div style="text-align: center">
<img alt="image.jpg" class="bbCodeImage LbImage" src="https://danieldefo.ru/attachments/image-jpg.30528/" style=""/>
		
​</div>Разумеется, таких с ходу берут в оборот и используют, как хотят. Страна без элиты суверенной быть не может, в ней отсутствуют носители суверенитета. Поэтому СССР был криптоколонией, кондоминиумом великих держав, спарринг-партнёром, големом с дистанционным управлением - чем угодно, но не равноценным игроком. Игроки во ВМВ боролись за то, где будут эмиссионные центры, чья будет мировая резервная валюта и т.п. СССР положил в 50-100 раз (!!!) больше жизней, чем любой из его союзников, и получил в награду Кёнигсберг.</blockquote>
<a class="cqauth" href="">Богемик</a><br/>
<br/>
Советским с рождения внушали, что страной может управлять и кухарка, командовать армией семинарист недоучка (он и командовал, отсюда такая "Победа").<br/>
В международных визитах ядерную дипломатию с мировым гегемоном доверили безграмотному селюку, не умеющему писать, монобровому цыгану - заниматься экономикой, а пятнистому комбайнеру - реформами 1/6 суши.<br/>
После известного результата, на качели посадили ссущего в штаны и колесо президентского самолета алкоголика.<br/>
Ну а хули? И Гагарин на колесо ссал.<br/>
<br/>
Эпоха несостоявшегося таксиста длится дольше брежневского застоя и не видно ей конца, западная медицина неплохо лечит африканских вождей (Мугабе до 95 дижил).<br/>
Поле выжжено.

<div class="subHeading">
<a href="threads/otxody-ot-obogaschenija-urana-iz-germanii-zaxoronjat-v-rossii.10812/">Отходы от обогащения урана из Германии захоронят в России</a>
</div>
<div class="primaryContent">
<div class="baseHtml excerptContent">
<blockquote class="coolqoute">Западноевропейская компания Urenco в 2019 году возобновила транспортировку в Россию урановых хвостов с немецкой уранообогатительной фабрики в городе Гронау.<br/>
<br/>
В документе властей немецкой земли Северный Рейн-Вестфалия сказано, что с 2019 по 2022 годы в Россию из Германии планируется отправить 12 тысяч тонн урановых хвостов (отходов) — это более тысячи контейнеров и примерно 20 железнодорожных составов.<br/>
<br/>
Кроме того, с мая по октябрь 2019 года из Гронау на АО «Уральский электрохимический комбинат» в Новоуральске в Свердловской области уже было отправлено шесть железнодорожных составов по 600 тонн, то есть около 3,6 тысяч тонн урановых хвостов.<br/>
<br/>
По данным Greenpeace, в России уже накоплено около 1 млн тонн урановых хвостов, которые не только радиоактивны, но и токсичны.<br/>
Ежегодные <a class="externalLink ProxyLink" data-proxy-href="proxy.php?link=http%3A%2F%2Fwww.gosnadzor.ru%2Fpublic%2Fannual_reports%2F&amp;hash=232704402486609de59ec8146f22a1a9" href="http://www.gosnadzor.ru/public/annual_reports/" rel="nofollow" target="_blank">отчёты о деятельности Ростехнадзора</a> (за 2004, 2006-2011 годы) упоминают о рисках разгерметизации контейнеров при хранении урановых «хвостов» под открытым небом на предприятиях в Ангарске, Северске, Новоуральске и Зеленогорске.</blockquote>
<a class="cqauth" href="https://greenpeace.ru/news/2019/10/23/v-rossiju-snova-nachali-vvozit-othody-obogashhenija-urana-iz-germanii/">greenpeace.ru</a><br/>
Наглядно про предназначение этой территории-свалки.
