<script type="text/javascript">
    Vue.component('http-www-in-domain', {
        template: '#dm-http-www-in-domain-template',
        props: {
            domain: String,
            selector: {
                type: String,
                default: ''
            },
            check3w: {
                type: Boolean,
                default: false
            }
        },
        computed: {
            url: function () {
                if (this.selector) {
                    return dmarcianTools.addParamsToCurrentUrl({domain: this.domainName, selector: this.selector});
                } else {
                    return dmarcianTools.addParamsToCurrentUrl({domain: this.domainName});
                }
            },
            texts: function () {
                return dmarcianTools.extractTexts('dm-http-www-in-domain-texts');
            },
            domainName: function () {
                var www = 'www.';
                var httpPattern = /^((http|https):\/\/)/;
                var domName = '';
                if (this.check3w == true) {
                //remove "http(s)://www." if any
                    if (this.domain.startsWith(www) || httpPattern.test(this.domain)) {
                        domName = this.domain.replace(/^(?:https?:\/\/)?(?:www\.)?/i, "").split('/')[0];
                        if (dmarcianTools.isValidDomain(domName)) {
                            return domName;
                        }
                    }
                }
                //remove "http(s)://" if any
                if ((this.check3w == false) &amp;&amp; (httpPattern.test(this.domain))) {    
                    domName = this.domain.replace(/^http(s?):\/\//i, "");
                    if (dmarcianTools.isValidDomain(domName)) {
                        return domName;
                    }
                }
                return domName;
            }
        }
    });
