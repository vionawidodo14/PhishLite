	jQuery( document ).ready(function( $ ) {

				e.preventDefault();
				// Validate email field prior to opening modal

					var msg = '';

			});

				event.preventDefault();
				element = $(this);

				// Validate select field prior to submitting signup
				if ( ! $('#mc-location-select2').val() ) {
					$('.mailchimp-modal').find("#response").hide().html('Please select your go-to By Chloe. location.').fadeIn('slow')
				} else {
					$('.mailchimp-modal').find("#response").hide().html('Adding email address...').fadeIn('slow');
					var listid = ( $('#mc-location-select2').find(':selected').data('listid'));
					var apikey = ( $('#mc-location-select2').find(':selected').data('apikey'));

						success: function(msg) {
// 							console.log('here: ' + msg);
