function doOnSubmitSearch(){
	switch($("#search-mode").val()){
		case 'quick':
			clearFormElements('#searchForm','mode,ad_type,description,my_ads');
			disableFormElements('#searchForm','mode,ad_type,description,my_ads');
			break;
		case 'detail':
			$("input[name='description']").val('');
			break;
	}
	
}



$("#searchForm").submit(function () {
    // prevent the default GET form behavior (name-value pairs)
    //event.preventDefault();

    doOnSubmitSearch(); 
    var searchIdType=$('#searchForm #detail_search_id_type');
    if(searchIdType){
			var surfaceSearchHeader=document.getElementById('surface_search_header');
			var landSurfaceSearchHeader=document.getElementById('land_surface_search_header');
			var minLandSurface=document.getElementById('min_land_surface');
			var maxLandSurface=document.getElementById('max_land_surface');
			var minSurface=document.getElementById('min_surface');
			var maxSurface=document.getElementById('max_surface');
			if(surfaceSearchHeader&&landSurfaceSearchHeader){
			switch (searchIdType.val()){
				case "5":
						surfaceSearchHeader.style.display="none";
						landSurfaceSearchHeader.style.display="block";
						minSurface.value='';
						maxSurface.value='';
						break;
						default:
						surfaceSearchHeader.style.display="block";
						landSurfaceSearchHeader.style.display="none";
						minLandSurface.value='';
						// minLandSurface.removeAttribute("value");
						// console.log(minLandSurface.value);
						maxLandSurface.value='';
						break;
				}
			}

    var strHoodPartValue = '';
    var strUserTypeValue = '';

    if (arrHoodValues != null){
 		strHoodValue = 'id_hood=' + arrHoodValues.join(',');
    }
    
 	var arrHoodPartValues = $("#detail_search_id_hood_part").val();

 		strHoodPartValue = 'id_hood_part=' + arrHoodPartValues.join(',');

    $("#detail_search_id_hood_part").attr('disabled',true);

    var searchUrl = action + '?' + strvalues + '&' + strHoodValue + '&' + strHoodPartValue + '&' + strUserTypeValue;

    window.location.href = searchUrl;
    
    return false;
});

<div class="blockText"><h1>Nekretnine Srbija</h1>
<div class="smallText">
<p><strong>NadjiDom.com</strong> je jedan od vodećih portala za oglašavanje nekretnina u Srbiji. Sajt promoviše i ubrzava prodaju i izdavanje nekretnina u gradovima kao što su Beograd, Novi Sad, Niš, Kragujevac, ali i ostalim gradovima širom Srbije. Oglasi sa slikama dostupni su Vam putem brze i efikasne pretrage.</p>
<p> </p>
<div class="three-columns">
<div style="padding: 0 1% 0 0;">
<h2>Prodaja nekretnina</h2>
<p><span>Oglase za nekretnine koje su na prodaju možete pronaći na strani </span><a href="http://www.nadjidom.com/sr/nekretnine/prodaja/">nekretnine prodaja</a>. Tu se nalaze stanovi, kuće, poslovni prostori, lokali, ali i vikendice, placevi, garaže i sl. Unošenjem ključnih reči u našu pretragu, kao što su: stanovi beograd, kuće beograd, stanovi novi sad, stanovi niš i sl. uz korišćenje filtera iz detaljne pretrage, dobićete vrlo precizne rezultate. </p>
</div>
<div style="padding: 0 1% 0 0; width: 31%;">
<h2>Izdavanje nekretnina</h2>
<p>Kada je u pitanju izdavanje, na našem portalu možete pronaći dve vrste oglasa tj. dugoročni i kratkoročni najam. Da bi korišćenje bilo što jednostavnije, dugoročni najam nalazi se u sekciji <a href="http://www.nadjidom.com/sr/nekretnine/izdavanje/">nekretnine izdavanje</a>, dok smo za kratkoročni najam napravili posebnu sekciju i nazvali je <a href="http://www.nadjidom.com/sr/nekretnine/smestaj/">smeštaj</a>.</p>
</div>
<div>
<h2>Oglasi</h2>
<p>Naši administratori proveravaju sadržaj svakog oglasa pre njegovog objavljivanja na portalu. Kvalitet oglasa je naš primarni cilj i ni pod kojim uslovima nećemo objaviti oglase za koje smatramo da ne zadovoljavaju određene kriterijume. </p>
<p> </p>
</div>
</div>
<h2>Region</h2>
<p>Od marta 2015. godine na našem sajtu mogu se oglašavati nekretnine iz čitavog regiona. Oglase agencija i fizičkih lica iz regiona možete pogledati na stranicama: <a href="http://www.nadjidom.com/hrvatska/">Nekretnine Hrvatska</a>, <a href="http://www.nadjidom.com/crna-gora/">Nekretnine Crna Gora</a> i <a href="http://www.nadjidom.com/bosna-i-hercegovina/">Nekretnine BiH</a></p>
