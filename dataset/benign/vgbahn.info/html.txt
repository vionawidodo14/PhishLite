document.addEventListener("DOMContentLoaded", function(event) { 
	$('.add_container a[href="#add"]').click(function() {
		doAddEmptyRow();
		return false;
	});
	
	function doAddEmptyRow() {
		var $doItem = $('.do_item').first().clone();
		$doItem.find('input[name="do_ordernbr[]"]').attr('value', '');
		$doItem.find('input[name="do_amount[]"]').attr('value', 1);
		$('.directorder_container .add_container').before($doItem);
		return $doItem;
	}
	// remove item
	$(document).on('click', '.do_trash a[href="#remove"]', function() { 
// 	$('.do_trash a[href="#remove"]').on('click', function() {
		// count .do_item - if there is only one, do not remove!
		var cnt = $('.do_item').length;
		if(cnt != 1) {
			$(this).parents('.do_item').remove();
		} else {
			$(this).parents('form')[0].reset();
		}
		return false;
	});
	// check form
	$('.directorder_container form').submit(function() {
		var filled = false;
		$('input[name="do_ordernbr[]"]').each(function() {
			if($(this).val() != '') {
				filled = true;
			}
		});
		if(filled == false) {
			alert('Bitte geben Sie eine Bestellnummer ein.');
			return false;
		}
	});
	
	function doAddOrderNum(orderNum) {
		// check if order number is already in input field
		var foundDouble = false;
		$('input[name="do_ordernbr[]"]').each(function(index, el) {
			if ($(this).attr('value') == orderNum) {
				foundDouble = true;
			}
		});
		
		if (!foundDouble) {
			var inserted = false;
			$('input[name="do_ordernbr[]"]').each(function(index, el) {
				if (!inserted && $(this).attr('value') == "") {
					$(this).attr('value', orderNum);
					inserted = true;
				}
			});
			if (!inserted) {
				var newRow = doAddEmptyRow();
				newRow.find('input[name="do_ordernbr[]"]').attr('value', orderNum);
			}
		}
	}
	
	// quickorder
	$('a.quickorder').click(function() {
		doAddOrderNum($(this).attr('data-ordernum'));
		return false;
	});
});
