        function submitLogin() {
            var button = document.getElementById('submitButton');
            button.disabled = true;
            var data = new FormData();
            data.append("email", document.getElementById('email').value);
            data.append("password", document.getElementById('password').value);
            fetch("/login/verifyLogin", {
                method: 'POST',
                credentials: 'same-origin',
                body: data
            }).then(function (response) {
                return response.text();
            }).then(function (login) {
                if (login == 1) {
                    alert("It looks like there was a problem with your email or password");
                    document.getElementById('email').value = "";
                    document.getElementById('email').focus();
                    button.disabled = false;
                } else if (login == 2) {
                    alert("Your account has been deactivated - please contact help@teach-now.com with any questions");
                    document.getElementById('password').value = "";
                    document.getElementById('email').focus();
                    button.disabled = false;
                } else if (login == 4) {
                    alert("That email is not recognized as a TEACH-NOW account");
                    document.getElementById('password').value = "";
                    document.getElementById('email').focus();
                    button.disabled = false;
                } else if (login == 5) {
                    alert("Your password is not correct");
                    document.getElementById('password').value = "";
                    document.getElementById('email').focus();
                    button.disabled = false;
                } else if (login == 7) {
                    alert("Please enter a valid email address to login");
                    document.getElementById('email').focus();
                    button.disabled = false;
                } else if (login == 8) {
                    alert("Please enter a password in order to login");
                    document.getElementById('password').focus();
                    button.disabled = false;
                } else if (login == 6) {
                    alert("There appears to be an environment misconfiguration");
                    button.disabled = false;
                } else if (login == 9) {                   
                    
                    var temp = document.createElement("div"),
                    message = 'The Teach-Now platform site will be down for 1 hour from 12:30 am EST to 1:30 am EST, on Thursday, September 5, 2019 for maintenance. Please contact the Help Desk at <a href="mailto:help@teach-now.edu">help@teach-now.edu</a> if you have any questions.';
                    
                    temp.innerHTML = message;
                    alert(temp.textContent || temp.innerText || "");
                    
                    button.disabled = false;
                } else if (login == 3) {
                    window.location = window.location.origin + "/home";
                } else {
                    alert("It looks like there's a technical problem - please contact help@teach-now.com");
                    button.disabled = false;
                }
            });
        }

        function checkSubmit(e, el) {
            var code = (e.keyCode ? e.keyCode : e.which);
            if (code === 13) { //Enter keycode
                submitLogin();
            }
        }
        function checkSubmitHelp(e, el) {
            var code = (e.keyCode ? e.keyCode : e.which);
            if (code === 13) { //Enter keycode
                submitHelp();
            }
        }

        function togglePopup(event,showPopup) {
            var popupElement = document.getElementById('help-popup');
            if(showPopup) {
                popupElement.style.display = 'block';
                $('#popupElement').show();
                $('.body-overlay').show();
                //$('#popupElement textarea').empty();
                $('.help-popup .input').val('');
            }else {
                $('.alertMsg').html('');
                $('#submitHelpButton').prop('disabled', false);
                $('#successMsg').html('');
                $('.body-overlay').hide();
                popupElement.style.display = 'none';
            }
            event.stopPropagation();
        }
        function submitHelp() {
            var email = document.getElementById('emailHelp').value,
                description = document.getElementById('help-description').value,
                userData = {
                'email' : email,
                'description': description 
                },
                atPosition= email.indexOf("@"),
                dotPosition= email.lastIndexOf(".");

                
            if(email=='' || description==''){
                if(email== '') {
                    $('.alertMsg').first().html('Please enter a valid email address');
                }else {
                    if (atPosition<1 || dotPosition<atPosition+2 || dotPosition+2>=email.length) {
                        $('.alertMsg').first().html('Please enter a valid email address');
                    }else {
                        $('.alertMsg').first().html('');
                    }
                }
                if(description=='') {
                    $('.alertMsg').eq(1).html('Please enter description of your problem');
                }else {
                    $('.alertMsg').eq(1).html('');
                }
                //return false;
            }
            else if (atPosition<1 || dotPosition<atPosition+2 || dotPosition+2>=email.length) {
                $('.alertMsg').html('');
                $('.alertMsg').first().html('Please enter a valid email address');
            }else {

                $('.alertMsg').html('');
                $.ajax({
                    method: "POST",
                    url: "/login/reportProblem",
                    data: userData,
                    success : function( data ) {
                        if(data.statusCode==0) {
                            $('.alertMsg').html('');
                            $('#submitHelpButton').prop('disabled', true);
                            $('#successMsg').html(data.response.msg);
                            $('.help-popup .inpul').val('');
                            $('#popupElement').hide();
                            
                        }else {
                            $('.alertMsg').first().html(data.response.msg);
                        }
                    },
                    error : function(data){
                        
                    }           
                 });
               }
        }
