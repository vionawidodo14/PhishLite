        function ValidateEmailMobile() {
            document.getElementById("v_SignupEmail").innerHTML = "";
            if ($("#SignupEmail").val() == "") {
                document.getElementById("v_SignupEmail").innerHTML = "Required";
                $("#SignupEmail").focus();
                return false;
            }
            if ($("#SignupEmail").val().length < 10 || $("#SignupEmail").val().length > 50) {
                document.getElementById("v_SignupEmail").innerHTML = "Email/Mobile should be atleast 10 & max 50 characters";
                $("#SignupEmail").focus();
                return false;
            }

            if (ck_mob.test($("#SignupEmail").val())) {
                if ($("#SignupEmail").val().length > 10) {
                    document.getElementById("v_SignupEmail").innerHTML = "Invalid mobile number";
                    $("#SignupEmail").focus();

                    return false;

        function validateEmail(email) {
            var re = /^([\w-]+(?:\.[\w-]+)*)@((?:[\w-]+\.)*\w[\w-]{0,66})\.([a-z]{2,6}(?:\.[a-z]{2})?)$/i;
            return re.test(email);
        }

                document.getElementById("v_SigninPassword").innerHTML = "Required";

                return false;

                $("#SigninPassword").focus();
                return false;

            if ($('#SigninRemember').is(":checked"))
                checked = 1;

            if ($("#CaptchaLogin").val() == "") {
                document.getElementById("v_SigninCaptcha").innerHTML = "Required";
                $("#CaptchaLogin").focus();
                return false;

                url: "/Ahmedabad/Ajax-YP/Signin",
                data: Param,
                success: function (result) {
                    $("#SigninSubmitFail").hide();

                        $("#SigninSubmitFail").show();
                        $("#SigninSubmitFail").html("Invalid Captcha");

                    if (result == "success")

        function getproduct() {
            var newcat = "";
            var oldcat = "";
            var a = "";

            var url = "";
            var item = "";
            var html = "";
            var div = "";
            var Param = "__RequestVerificationToken=" + $('input[name="__RequestVerificationToken"]').val();
            $.ajax({
                type: "POST",
                dataType: 'json',
                //contentType: 'application/json',
                beforeSend: function (xhr) {
                    xhr.setRequestHeader("Content-Encoding", "gzip");
                },
                url: "/Ahmedabad/Ajax-YP/GetHomeCategories",
                data: Param,
                success: function (data) {
                    var htmlstr = "";
                    jsonObj = jQuery.parseJSON(data);
                    for (var i = 0; i < jsonObj.cats.length; i++) {

                        url = "";
                        item = "";

                        url = "/Ahmedabad/" + jsonObj.cats[i].caturl + "/" + jsonObj.cats[i].catid + "?searchtext=" + jsonObj.cats[i].product.replace("&", "^") + "&FApplied=2:" + jsonObj.cats[i].prodid;
                        item = '<li><a href="' + url + '">' + titleCase(jsonObj.cats[i].disprod) + '</a></li>';
                        newcat = jsonObj.cats[i].dispcat;

                        if (newcat != oldcat) {
                            if (html != "") {
                                div += html.replace("[data]", a);
                                html = "";
                                a = "";
