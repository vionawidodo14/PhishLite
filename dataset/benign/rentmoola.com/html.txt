                        if(date._d.getDate() == 1) {

                            var lastCalendarDayOfMonth = new Date(currentDate.getFullYear(), currentDate.getMonth() + 1, 0).getDate();
                            var lastDateOfMonth = new Date(currentDate.getFullYear(), currentDate.getMonth() + 1, 0);
                            var cutoffDays = 1;

                            // If the last calendar day of the month falls on a Saturday, there are 2 days worth of cutoff time
                            if(lastDateOfMonth.getDay() == 6) {
                                cutoffDays = 2;
                            }

                            // If the last calendar day of the month falls on a Sunday, there are 3 days worth of cutoff time
                            if(lastDateOfMonth.getDay() == 0) {

                                cutoffDays = 3;
                            }

                            // First of the month dates that fall within 2 business days of the current date are also inactive

                            }
