                    $window.scroll(function(event) {

                        var sidebarHeight = $sidebar.outerHeight();

                        var scrollTop = $window.scrollTop();
                        var scrollBottom = scrollTop + windowHeight;

                        var sidebarTop = $sidebar.offset().top;
                        var sidebarBottom = sidebarTop + sidebarHeight;

                        var heightDelta = Math.abs(windowHeight - sidebarHeight);
                        var scrollDelta = lastScrollTop - scrollTop;

                        var isScrollingDown = (scrollTop > lastScrollTop);
                        var isWindowLarger = (windowHeight > sidebarHeight);

                        if ((isWindowLarger && scrollTop > initialSidebarTop) || (!isWindowLarger && scrollTop > initialSidebarTop + heightDelta)) {
                            $sidebar.addClass('fixed');
                            $sidebar.css('width', sidebarwidth);
                        } else if (!isScrollingDown && scrollTop <= initialSidebarTop) {
                            $sidebar.removeClass('fixed');
                        }

                        var dragBottomDown = (sidebarBottom <= scrollBottom && isScrollingDown);
                        var dragTopUp = (sidebarTop >= scrollTop && !isScrollingDown);

                        if (dragBottomDown) {
                            if (isWindowLarger) {
                                $sidebar.css('top', 0);
                            } else {
                                $sidebar.css('top', -heightDelta);
                            }
                        } else if (dragTopUp) {
                            $sidebar.css('top', 0);
                        } else if ($sidebar.hasClass('fixed')) {
                            var currentTop = parseInt($sidebar.css('top'), 10);
                            
                            var minTop = -heightDelta;
                            var scrolledTop = currentTop + scrollDelta;

                            var newTop = (isPageAtBottom) ? minTop : scrolledTop;

<script type="text/javascript">

        $(document).ready(function(){
            getWishListProducts();
            getCountWishListProducts();
        });

        function getCountWishListProducts()
        {
            products = $.cookie('wishlist_products');
            products = JSON.parse(products);
            if (products)
            {
                $('[id*=wish-counter]').html(products.length);
            }
        }

        function getWishListProducts()
        {
            products = $.cookie('wishlist_products');

            products = JSON.parse(products);

            if (products)
            {
                setWishProducts(products);
            }
        }

        function setWishProducts(productsIDs)
        {
            if (productsIDs.length)
            {
                productsIDs.forEach(function (productID){
                    $('[data-product-id="' + productID + '"]').addClass('selected');
                    //$(".add-to-wishlist.p-view.selected").html("<i></i> Scoate din favorite");
                });
            }
        }

        function addProductInWishList(el)
        {
            productID = $(el).data('productId');

            expiresDate = new Date();
            expiresDate.setHours(expiresDate.getHours() + 24 * 365);

            if ($(el).hasClass('selected'))
            {
                $(el).removeClass('selected');



                products = $.cookie('wishlist_products');
                products = JSON.parse(products);

                products.forEach(function (element, index)
                {
                    if (element == productID)
                    {
                        products.splice(index, 1);
                    }
                });

                var url = _url + '?module=Frontend&action=ajax_removeProductFromWishlist&product_id=' + productID;

            }
            else
            {
                $(el).addClass('selected');



                products = $.cookie('wishlist_products');
                if (products)
                {
                    products = JSON.parse(products);
                }
                else {
                    products = new Array();
                }
                products.push(productID);

                var url = _url + '?module=Frontend&action=ajax_addProductInWishlist&product_id=' + productID;
            }

            $.getJSON(url, function(data){
                if (data.success)
                {
                    $.cookie('wishlist_products', JSON.stringify(products), { expires: expiresDate, path: '/' });

                    getCountWishListProducts();
                }

            });
        }

    </script>
