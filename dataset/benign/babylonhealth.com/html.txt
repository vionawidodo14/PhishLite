    }

        header = document.querySelector('.header'),
        menuIsOpen = false;

    function toggleMenu() {
      if(!menuIsOpen) {
        header.classList.add('open-menu');
        menuIsOpen = true;
      } else {
        header.classList.remove('open-menu');
        menuIsOpen = false;
      }
    }

    hamburgerIcon.forEach(function(icon){
      icon.addEventListener('click', toggleMenu);
    });
