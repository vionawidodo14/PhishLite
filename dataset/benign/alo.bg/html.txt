function load_more(container_id,type)
	{
	height_step = 492;
		
	skip_ad_ids = new Array();
	$('#'+container_id).find(".ad_block_normal").each(function( index ) {
	  skip_ad_ids[skip_ad_ids.length] = $(this).attr('id').replace("adrows_", "");
	});

	jQuery.ajax({
	  url: "index.php",
	  type: "POST",
	  cache: false,
	  data: "action=load_more&type="+type+"&skip_ad_ids="+skip_ad_ids,
	  dataType: "text",
	  
	  success: function(text)
		{
		//current height
		newheight = $('#'+container_id).height();
		newheight += height_step;
		//alert(newheight);
		$('#'+container_id).append(text);
		$('#'+container_id).animate({'max-height':newheight+'px'},500);
		},

	error: function(req,stat,err)
		{
		show_message(stat);
		}
	});
	
	}
