            function UpdateCartTime() {
                if (window.cartTimerSyncTime == null || window.ItemsInCart == 0 || !window.cartTimerCanExpire) {
                    $(".cartExpireSpan").text("");
                    $("#nav-cart-button").removeClass("has-cart-timer");

                $("#nav-cart-button").addClass("has-cart-timer");
                var secondsSinceSync = Math.floor(((new Date() - window.cartTimerSyncTime)) / 1000);
                var syncedSecondsLeft = window.cartTimerSecondsLeftAtSync - secondsSinceSync;
