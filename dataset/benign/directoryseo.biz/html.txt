function decrease_height() {
   height = parseInt(document.getElementById('slide').style.marginTop);
   document.getElementById('panel').style.zIndex = -1;

   }
}

   height = parseInt(document.getElementById('slide').style.marginTop);

      increase_height();
   }

}
