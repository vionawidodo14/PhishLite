    $scope.toggleTags = function(tags) {

        $scope.tags = tags.slice(0, tags.length - 1);
      } else if ($scope.tags.length >= tags.length) {
        $scope.tags = tags.slice(0, tags.length - 1);
      } else {
        $scope.tags = tags;
      };
    };

    // Menu item should be shown if it is part of the currently selected menu

    };

      return $scope.tags.length != 0;
    };
    $scope.resetTags();
