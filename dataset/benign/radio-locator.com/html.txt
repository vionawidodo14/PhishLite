document.addEventListener('DOMContentLoaded', function(){
 if (navigator.geolocation) {
  var gd = document.getElementById("geodiv");
  gd.style.visibility = "visible";  
  gd.style.display = "block";
  var gl = document.querySelectorAll('.geolink');
  for (i = 0; i < gl.length; i++) {
    gl[i].addEventListener('click', geoLocate);
  }
}});
function resetGeoLink() {
  document.getElementById("geotext").innerHTML = 'find U.S. radio by your location';
}
function geoCallback(position) {
  resetGeoLink();
  window.location.assign('/cgi-bin/locate?select=lonlat&geo=y&latd=' + position.coords.latitude + '&lond=' + position.coords.longitude + '');
}
function geoFailback(error) {
  alert('Your location cannot be determined: ' + error.message);
  resetGeoLink();
}
function geoLocate() {
  navigator.geolocation.getCurrentPosition(geoCallback,geoFailback,{enableHighAccuracy:false,timeout:20000,maximumAge:120000});
  document.getElementById("geotext").innerHTML = 'getting your location';
}
