// Based on:  https://deanhume.github.io/lazy-observer-load/
var lazy_loader = (function() {

  "use strict";

  // If no ES6 support then exit here, or add transpiler with polyfill
  if (!supportsES6) {return false;}

  const dataAttr = "data-LL-src";
  const elementLoaded = "-js-lazyLoaded";
  const elementFadeClass = "-js-fadeIn";
  const elementInViewClass = "-js-inView";

  // Get elements that are marked to lazy load
  const lateLoadObjs = document.querySelectorAll("[" + dataAttr + "]");


  // Externalise this:
  const config = {
    // rootMargin format: Top Right Bottom Left (as per CSS margins)
    rootMargin: "-50px 50px", // The image gets 50px into viewport before loading, so you can see it kick in
    threshold: 0.01
  };


  let lateLoadObjsCount = lateLoadObjs.length;
  let observer;


  const _supportsIntersectionObserver = () => { // boolean
    const isSupported = !!('IntersectionObserver' in window);
    console.log(isSupported ? "IntersectionObserver supported" : "No IntersectionObserver support");
    return isSupported;
  };


  const _hasNativeSrc = (elem) => { // boolean
    if (!elem) {return false;}

    var nativeSrcElements = ["img", "script", "iframe", "audio", "video", "embed", "input", "source", "track"];
    return nativeSrcElements.includes(elem.tagName.toLowerCase());
  };



  // Apply the change to the element
  function applyElementChanges(elem, src) {

    // Prevent this from being lazy loaded a second time.
    elem.classList.add(elementLoaded);

    // No src value means just add the in-view class
    if (src !== "") {

      if (_hasNativeSrc(elem)) {
        elem.src = src;
      } else {
        elem.style.backgroundImage = "url(" + src + ")";
      }

    } else {

      // No src value so just add the in-viewport class
      elem.classList.add(elementInViewClass);

    }

    elem.classList.add(elementFadeClass);
  }



  // Fetch image for given URL -- Needs to fetch the relevant content type, not just Image()
  function fetchImage(url) {
    return new Promise((resolve, reject) => {
      // const image = new Image();
      const image = new Image();
      image.src = url;
      image.onload = resolve;
      image.onerror = reject;
    });
  }



  // Preloads the image
  function preloadElement(elem) {
    const src = elem.getAttribute(dataAttr);

    if (src === "") {
      return applyElementChanges(elem, src);
    }

    if (!src) {
      return;
    }

    return fetchImage(src).then(() => {
      applyElementChanges(elem, src);
    });
  }



  // On intersection of element in viewport
  function onIntersection(entries) {

    // Disconnect if we've already loaded all of the elements
    if (lateLoadObjsCount === 0) {
      observer.disconnect();
    }

    // Loop through the entries
    entries.forEach(entry => {
      // Are we in viewport?
      if (entry.intersectionRatio > 0) {
        lateLoadObjsCount--;

        // Stop watching and load the element
        observer.unobserve(entry.target);
        preloadElement(entry.target);
      }
    });
  }



  (function initialise() {

    if (_supportsIntersectionObserver()) {

      // Is supported, load the elements into the observer
      observer = new IntersectionObserver(onIntersection, config);
      lateLoadObjs.forEach(obj => {
        if (obj.classList.contains(elementLoaded)) {
          return;
        }

        observer.observe(obj);
      });

    } else {

      // Not supported, get elements immediately
      Array.from(lateLoadObjs).forEach(obj => preloadElement(obj));

    }
  }());

})();
