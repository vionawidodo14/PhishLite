	function showMore(tab)
	{
		if ($('.active .hidden-listing').css('display') == "none")
		{
			$('.active .hidden-listing').slideDown(1200);

			return false;
		}
		else
		{
			return true;
		}
	}

	$('a[data-toggle="tab"]').on('shown.bs.tab', function (e)
	{
		var target = $(e.target).attr("href"); // activated tab

		hideAllButtons();
		showButton(target);

		console.log(target);
	});

	function hideAllButtons()
	{
		$('.promotional-btn').hide();
		$('.recent-btn').hide();
		$('.our-best-dealers-btn').hide();
		$('.popular-btn').hide();
		$('.most-sold-btn').hide();
	}

	function showButton(target)
	{
		if (target === '#promotional')
			$('.promotional-btn').show();
		else if (target === '#recent')
			$('.recent-btn').show();
		else if (target === '#ourbestdealers')
			$('.our-best-dealers-btn').show();
		else if (target === '#mostpopular')
			$('.popular-btn').show();
		else if (target === '#mostsold')
			$('.most-sold-btn').show();
	}
</script>

	function SubmitTicket()
	{
		$.ajax({
			type: 'POST',
			url: '/ticket/createnew',
			data: '{ "ticketName": "' + encodeURIComponent($('#ticketName').val()) + '", "ticketEmail": "' + encodeURIComponent($('#ticketEmail').val()) + '", "ticketSubject": "' + encodeURIComponent($('#ticketSubject').val()) + '", "ticketOrderID": "' + $('#ticketOrderID').val() + '", "ticketMessage": "' + encodeURIComponent($('#ticketMessage').val()) + '" }',
			contentType: 'application/json; charset=utf-8',
			success: function (json)
			{
				if (!json.ErrorMessage)
				{
				$('#divOpenTicket').slideUp('slow', function ()
				{
					$('#formSubmitTicket')[0].reset();
					//showMessage('Ticket de suporte criado com sucesso, um operador irá responder em até 24 horas úteis.', 'success');
					window.location.href = "/ticket/" + json;
				});
				}
				else if (json.ErrorMessage)
				{
					showMessage(json.ErrorMessage, 'error');
				}
			}
		});
	}

	function ToggleTicketPanel()

		}
		else
		{
			$('#divOpenTicket').slideToggle('slow');
		}
		
		//return false;
	}

<script type="text/javascript">
    var pages = [];
    var titles = [];
    var index = 0;

    function PushPage() {
        pages.push($('#InformationalModal #div-inside-body').html());
        titles.push($('#InformationalModal .modal-title').html());

        index++;

        $('#spanNavBack').css('color', 'black');

        $('#InformationalModalNav').show();
    }

    function NavBack()
    {
    	if (index <= 0)
    	{
    		return;
    	}

        if (index >= pages.length) {
            pages.push($('#InformationalModal #div-inside-body').html());
            titles.push($('#InformationalModal .modal-title').html());
        }

        index--;

        $('#InformationalModal #div-inside-body').html(pages[index]);
        $('#InformationalModal .modal-title').html(titles[index]);

        BackForwardSetColor();
    }

    function NavForward()
    {
    	if (index >= pages.length - 1)
    	{
    		return;
    	}

        index++;

        $('#InformationalModal #div-inside-body').html(pages[index]);
        $('#InformationalModal .modal-title').html(titles[index]);

        BackForwardSetColor();
    }

    function BackForwardSetColor()
    {
		// Cor do botão de voltar
    	if (index <= 0)
    	{
    		$('#spanNavBack').css('color', 'gray');
    	}
    	else
    	{
    		$('#spanNavBack').css('color', 'black');
    	}

    	// Cor do botão de ir para frente
    	if (index < pages.length - 1)
    	{
    		$('#spanNavForward').css('color', 'black');
    	}
    	else
    	{
    		$('#spanNavForward').css('color', 'gray');
    	}

    	//console.log(index + ' ' + pages.length);
    }

	function informationalModal(body, title)
    {
	    if ($('#InformationalModal').is(':visible')) {
	        PushPage();
	    }
	    else {
	        pages = [];
	        titles = [];
	        index = 0;
	    }

		$('#InformationalModal #div-inside-body').html(body);

		if (title)
		{
			$('#InformationalModal .modal-title').html(title);
		}

		// Remove a barra de pesquisa do topo, pois ela fica atrapalhando o modal. E quando o usuário
		// dá scroll, a barra já volta automaticamente
		$(".header-scroll").hide();

		// Chama o modal, de fato.
		$("#InformationalModal").bsModal();

		ActivateElements();
	}

	function informationalModalURL(url, title)
	{
	    if ($('#InformationalModal').is(':visible')) {
	        PushPage();
	    }
	    else {
	        pages = [];
	        titles = [];
	        index = 0;
	    }

		$("#InformationalModal #div-inside-body").load(url, function ()
		{
			if (title)
			{
				$('#InformationalModal .modal-title').html(title);
			}

			// Remove a barra de pesquisa do topo, pois ela fica atrapalhando o modal. E quando o usuário
			// dá scroll, a barra já volta automaticamente
			$(".header-scroll").hide();

			// Chama o modal, de fato.
			$("#InformationalModal").bsModal();

			ActivateElements();
		});
	}
</script>
