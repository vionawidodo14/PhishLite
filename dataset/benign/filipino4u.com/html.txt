function DropDown(el) {
	this.wrapper = el;
	this.placeholder = this.wrapper.children('span');
	this.opts = this.wrapper.find('.dropdown span');
	this.val = '';
	this.index = -1;
	this.initEvents();
	this.$input = this.wrapper.find('input');
}

DropDown.prototype = {
	initEvents : function() {
		var obj = this;

		obj.wrapper.on('click', function(event){
			$(this).toggleClass('active');
			return false;
		});

		});
	},
	getValue : function() {
		return this.val;
	},
	getIndex : function() {
		return this.index;
	}
