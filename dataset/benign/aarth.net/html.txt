function validateFormOnSubmit(theForm) {
  var reason = "";
  document.getElementById('register-button').style.display = 'block';
  document.getElementById('res').innerHTML = '';
  reason += validateFname(theForm.fname);
  reason += validateLname(theForm.lname);
  reason += validateCname(theForm.cname);
  reason += validateEmail(theForm.email);
  reason += validatePhone(theForm.phone);
  
  if (reason != "") {
    return false;
  }
  else
  {
  	 document.getElementById('signup_txt').style.display = 'none';
     document.getElementById('signupform').style.display = 'none';
	 document.getElementById('res').innerHTML = 'Thank for registering! We will get back to you shortly';
	 return true;
  }
}

function validateCname(fld) {
    var error = "";
  document.getElementById('cn_error').innerHTML = '';

function validatePhone(fld) {
    var error = "";
     var numbers = /^[0-9 ]+$/;   
   document.getElementById('phone_error').innerHTML = '';
   if (trim(fld.value) == "") {
        error = "You didn't enter a phone number.\n";
        fld.style.background = 'Yellow';
		document.getElementById('phone_error').innerHTML = error;
    }  else if (trim(fld.value).length != 10) {
        error = "The phone number is the wrong length.\n";
        fld.style.background = 'Yellow';
		document.getElementById('phone_error').innerHTML = error;
    }
	else if(trim(fld.value).match(numbers)) {
	    error = "";
        fld.style.background = 'White';
		document.getElementById('phone_error').innerHTML = error;
        
    } 
	else
	{
	   error = "The phone number contains illegal characters.\n";
        fld.style.background = 'Yellow';
		document.getElementById('phone_error').innerHTML = error;
	}
    return error;
}


function trim(s)
{
  return s.replace(/^\s+|\s+$/, '');
} 

function validateEmail(fld) {
    var error="";
    var tfld = trim(fld.value);                        
    var emailFilter = /^[^@]+@[^@.]+\.[^@]*\w\w$/ ;
    var illegalChars= /[\(\)\<\>\,\;\:\\\"\[\]]/ ;
    document.getElementById('email_error').innerHTML = '';
    if (tfld == "") {
        fld.style.background = 'Yellow';
        error = "You didn't enter an email address.\n";
		document.getElementById('email_error').innerHTML = error;
    } else if (!emailFilter.test(tfld)) {              
        fld.style.background = 'Yellow';
        error = "Please enter a valid email address.\n";
		document.getElementById('email_error').innerHTML = error;
    } else if (tfld.match(illegalChars)) {
        fld.style.background = 'Yellow';
        error = "The email address contains illegal characters.\n";
		document.getElementById('email_error').innerHTML = error;
    } else {
        fld.style.background = 'White';
		document.getElementById('email_error').innerHTML = '';
    }
    return error;
}
