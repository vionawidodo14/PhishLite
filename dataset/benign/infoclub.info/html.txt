	bindRelativeItems : function(){
		var fields_names = [];
		var $form = $(this.options.formSelector);
		$form.find('.relativeItem').each(function(){
			var field_name = $(this).data('relative_field');
			var exist = false;
			for (var i in fields_names){
				if (field_name == fields_names[i]){
					exist = true;
					break;
				}
			}

			if (!exist){
				fields_names.push(field_name);
			}
		});

		var selector = '';
		for (var i in fields_names){
			selector += (selector!='' ? ',' : '') + '[name="'+fields_names[i]+'"]';
		}

        $form.on('change', selector, function(){
			var field_name = $(this).attr('name');
			var value = $(this).val();

			$form.find('.relativeItem[data-relative_field="'+field_name+'"]').each(function(){
				var visible_values = $(this).data('relative_values').split(',');
				var visible = false;

				for (var i in visible_values){
					if (value == visible_values[i]){
						visible = true;
						break;
					}
				}

				if (visible){
					$(this).addClass('visible');
					$(this).find('input[type="text"]').focus();
				}else{
					$(this).removeClass('visible');
				}
			});
        });

			if ($(this).prop('checked')){
				if ($(this).siblings('.relative_block').size()>0){

				}
			}
		});

	processError	: function(type, obj, subtype, errorText, position_method){
		
		if (typeof(subtype) == "undefined"){
			subtype = "";
		}
		if (typeof(position_method) == "undefined"){
			position_method = this.options.position_method;
		}
		obj.removeClass('successField').addClass('errorField');
		obj.parent().find('.inputSuccessBlock').remove();
		if (this.errorSender == this.errorSenderType.FORM_SUBMIT){
			obj.focus();
		}

		if (obj.parent().find('.errorType_'+type+'_'+subtype).size()>0){
			if (this.errorSender == this.errorSenderType.LIVE_CHECKER){
				return;
			}
		}else{
			obj.parent().find('[rel="errorBlock"]').fadeOut(function(){$(this).remove()});
		}

		if (typeof(errorText) == "undefined"){
			errorText = '';
		}
		
		switch(type){
			case this.errorType.EMPTY_FIELD:
				if (errorText == ''){
					errorText = this.getEmptyErrorForField(obj.attr('name'));
				}
				obj.addClass('errorEmptyField');
				break;
			case this.errorType.SPECIFIC_FIELD:
				if (errorText == ''){
					errorText = this.getSpecificErrorText(obj.attr('name'), subtype);
				}
				obj.addClass('errorSpecificField');
				break;
		}

		var errorClass = $(this.options.formSelector).data('error_class') ? $(this.options.formSelector).data('error_class') : 'inputErrorBlock';
		var $errorBlock = $('<div></div>').attr({"rel": "errorBlock", "class" : errorClass + ' errorType_'+type+'_'+subtype}).html(errorText).css('display', 'none');
		obj.parent().append($errorBlock);

        $errorBlock = obj.parent().find('[rel="errorBlock"]');

        if (position_method==1){
            $errorBlock.css({top: (obj.parent().height()-$errorBlock.outerHeight()-(obj.is('textarea') ? 3 : 0 ))/2 + parseInt(obj.parent().css('padding-top')) + 'px'}).fadeIn();
        }else{
            $errorBlock.css({top: (obj.parents('.textFieldWrapper, .textareaFieldWrapper').height()-$errorBlock.outerHeight()-(obj.is('textarea') ? 3 : 0 ))/2 + parseInt(obj.parents('.textFieldWrapper, .textareaFieldWrapper').css('padding-top')) + 'px'}).fadeIn();
        }

		if (subtype != 'send'){
			obj.parents(this.options.formSelector).find('button[type="submit"]').addClass("disabled");
		}

		if (this.options['onErrorCallback']){
			this.options['onErrorCallback'](obj, errorText);
		}

		if ($(this.options.formSelector).data('error_autohide')){
			window.setTimeout(function(){
				$errorBlock.fadeOut('fast', function(){
					$(this).remove();
				});
			}, 3000);
		}
	},

	validate:function($form, $rules, params){
		var defaults={
			afterValidate:false
		};

		var options = $.extend(defaults, params);
		
		$.each($rules, function(fname, frules){
			var $field=$form.find(':input[name='+fname+']');
			if($field.length==0) return;
			
			$field.tmchange({
				onenter:function(){
					$field.removeClass('errorField');
					$field.parent().find('[rel=errorBlock]').fadeOut('fast', function(){
						$(this).remove();
					})
				}
			})
		});
		
		$form.submit(function(){
			var focused=false;
			var has_errors=false;
			
			$.each($rules, function(fname, frules){
				var $field=$form.find(':input[name='+fname+']');
				if($field.length==0) return;
				
				$field.removeClass('errorField');
				$field.parent().find('[rel=errorBlock]').remove();
				
				var value=$field.val().trim();
				var error=false;
				
				if(frules.required && value=='') error='required';
				else if(frules.pattern && !frules.pattern.test(value)) error='pattern';
				else if(frules.callback){
					var test=frules.callback.call($field, value, frules);
					if(test!='') error=['callback', test];
				}
				
				if(error){
					has_errors=true;
					if(!focused){
						$field.focus();
						focused=true;
					}
					
					var errormsg=(typeof(error)=='string') ? (frules[error][1] ? frules[error][1] : frules.msg) : error[1];
					$field.addClass('errorField');
					
					var $errlabel=$('<div rel="errorBlock" class="inputErrorBlock"></div>').css({display:'none'}).html(errormsg);
					$errlabel.appendTo($field.parent()).fadeIn('fast');
				}
			});
			
			if(!has_errors){
				if(typeof(options.afterValidate)=='function') return options.afterValidate.call($form);
				else return true;
			} else {
				return false;
			}
		});

                            }
                            setState();
                        }
                    });

                    setState();
                }

                    window[params['onInit']]($carouselApi);
