	function updateSpotlight()
	{
		var what = $('div.indexSpotlight table.menu1 a.selected').attr('class').split(' ')[0];
		var type = $('input[name=serialType]:checked').val();
		$('div#spotlight').html('<div class="loader"><img src="/design/common/pic/ajax-loader.gif" /></div>');
		$.ajax({
			url: '/ajax/updateSpotlight?what=' + what + '&type=' + type
		}).done(function(result) {
			$('div#spotlight').html('');
			$('div#spotlight').append(result);
			makeTime( $('div#spotlight') );
		});
	}
	
	function updateFeatured(what)
	{
		if(!what || $('div#indexFeatured a.featured.' + what).hasClass("open"))
		{
			var scrollTo = $(document).scrollTop() - $('div#featured').outerHeight();
			if(scrollTo < 0) { scrollTo = 0; }
			$('div#indexFeatured a.featured').removeClass("open");
			$('div#featured').slideUp();
			$("html, body").animate({ scrollTop: scrollTo });
			$('div#indexFeatured').removeClass('open');
		}
		else
		{
			$('div#indexFeatured a.featured').removeClass("open");
			$('div#indexFeatured a.featured.' + what).addClass("open");
			$('div#indexFeatured').addClass('open');
			$('div#featured').slideUp();
			$('div#indexFeatured div.loader').html('<img src="/design/common/pic/ajax-loader.gif" />').slideDown();
			$.ajax({
				url: '/ajax/updateFeatured?what=' + what
			}).done(function(result) {
				$('div#indexFeatured div.loader').slideUp();
				$('div#featured').html('' + result).slideDown();
			});
		}
		return;
	}
</script>
