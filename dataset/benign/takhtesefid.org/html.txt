function format_duration(a)
{
    hh = Math.floor(a/3600)
    mm = Math.floor(a/60)-60*hh
    ss = a - Math.floor(a/60)*60
    s = ''
    s += hh ? hh+':' : ''
    s += (mm<10 & hh)? '0' : ''
    s += mm + ':'
    s += (ss<10 )? '0'+ss : ss
    return s        
}
