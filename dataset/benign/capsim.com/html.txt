<script>
	var mixer = mixitup('.product-container', {
		load: {
			filter: '.mix:not(.desc)'
		},
		selectors: {
			control: '[data-mixitup-control]'
		},
		classNames: {
			block: '',
			elementFilter: ''
		}
	});

	$(document).ready(function() {
		$('.tab-link').on('click', function(event) {
			// Prevent url change
			event.preventDefault();

			// figuring out which tab has been clicked, then applying that class to the solutions list
			var addressValue = $(this).attr("href");
			var addressTrim = addressValue.substring(1, addressValue.length);
			$( ".solutions-hold" ).removeClass( "hero-home hero-students hero-profs hero-execs" ).addClass(addressTrim);

			// `this` is the clicked <a> tag
			var target = $('[data-toggle="tab"][href="' + this.hash + '"]');

			// opening tab
			target.trigger('click');

			// reset filter on tabs to show all, unless we are in the exec tab
			if (addressTrim == 'hero-execs') {
					mixer.filter('.corporate');
			} else {
					mixer.filter('.mix:not(.desc)');
			}

		});

	})
</script>
