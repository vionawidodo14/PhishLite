                no_error = true;
            }
        }

        if (no_error) {
            ukryjBlad(input_name);
        }

        return no_error;
    }

    function checkData(day, month, year) {
        var d = new Date(year, month, day);
        return (d && (d.getMonth()) === month && d.getDate() === Number(day));
    }

    function sprawdzWiek(day, month, year) {
        var birthDate = new Date(year, month, day);
        var today = new Date();

        var age = today.getFullYear() - birthDate.getFullYear();
        var m = today.getMonth() - birthDate.getMonth();

        if (m < 0 || (m === 0 && today.getDate() < birthDate.getDate())) {
            age--;
        }

        return (age >= 18);

    }

    // Sprawdzanie daty urodzenia
    function sprawdzDaty() {
        var no_error = true;
        var input_name = 'day';
        var inputs_data = $('#form-main-register #data select');

        var input_day = $('[name="day"]').val();
        var input_month = $('[name="month"]').val();
        var input_year = $('[name="year"]').val();


        inputs_data.each(function () {
            input_name = $(this).attr('name');
            if ($(this).val() == null) {
                wyswietlEtykieteBledu(input_name, "Nie wybrano daty urodzenia");
                no_error = false;
            }
        });

        if (no_error) {
            if (!checkData(input_day, input_month - 1, input_year)) {
                wyswietlEtykieteBledu(input_name, "Nieprawidłowa data urodzenia");
                no_error = false;
            }
        }

        if (no_error) {
            if (!sprawdzWiek(input_day, input_month - 1, input_year)) {
                wyswietlEtykieteBledu(input_name, "Aby założyć konto musisz mieć ukończone 18 lat.");
                no_error = false;
            }
        }


        if (no_error) {
            ukryjBlad(input_name);
        }

        return no_error;
    }

    function walidacja(prev, active) {
        switch (prev) {
//            case 'login_or_email':
//                sprawdzPrzypomnienieHasla();
//                break;
//            case 'login':
//                sprawdzLoginLogowanie();
//                break;
            case 'plec':
                sprawdzCzyWybranoPlec();
                break;
            case 'registration_login':
                sprawdzLogin();
                break;
            case 'password':
                $('#form-login input[name="password"]').is(':visible') ? '' : sprawdzHaslo();
                break;
//            case 'new_password_reminder':
//                sprawdzHasloPrzypomnienie();
//                break;
            case 'email':
                sprwadzMail();
                break;
            case 'plec_para':
                sprawdzPlecPara();
                break;
            case 'day':
            case 'month':
            case 'year':
                var select_date = (['day', 'month', 'year']);
                if (!($.inArray(active, select_date) != -1 && $.inArray(prev, select_date) != -1)) {
                    sprawdzDaty();
                }
                break;
            case 'plec_para_partner':
                sprawdzPlecParaPartner();
                break;
            case 'day_para':
            case 'month_para':
            case 'year_para':
                var select_date = (['day_para', 'month_para', 'year_para']);
                if (!($.inArray(active, select_date) != -1 && $.inArray(prev, select_date) != -1)) {
                    sprawdzDatyParner();
                }
                break;
            case 'adress':
            case 'adress_show':
            case 'kraj':
            case 'wojewodztwo':
            case 'miasto':
                var select_date = (['adress', 'adress_show', 'kraj', 'wojewodztwo', 'miasto']);
                if (!($.inArray(active, select_date) != -1 && $.inArray(prev, select_date) != -1)) {
                    sprwadzMiejscowosc();
                }
                break;
        }
    }

    // Przechwytywanie kliknięć dla walidacji
    $(document).on('click', function (event) {
        if ($(event.target).is(':not([type="submit"])')) {
            if (active_selected_input != null) {
                prev_selected_input = active_selected_input;
            }
            active_selected_input = $(event.target).attr('name') || null;
            if (active_selected_input == null) {
                active_selected_input = $(event.target).find('input').attr('name') || null;
            }

            if ($(event.target).is('option')) {
                active_selected_input = $(event.target).parents('select').attr('name') || null;
            }

            if (prev_selected_input != null && prev_selected_input != active_selected_input) {
                walidacja(prev_selected_input, active_selected_input);
            }
        }
    });
