    function cutGeoNames(name) {
        var indexOfObl = name.indexOf('область');
        var indexOfResp = name.indexOf('Республика');
        if (~indexOfObl) {
            name = name.substr(0, indexOfObl) + ((indexOfObl > 18) ? '<br>' : '') + 'обл.';
        }
        if (~indexOfResp) {
            if (indexOfResp) {
                name = name.substr(0, indexOfResp) + ((indexOfObl > 18) ? '<br>' : '') + 'респ.';
            } else {
                name = 'респ. ' + name.slice(indexOfResp + 11);
            }
        }
        if (~name.indexOf('Респ.')) {
            name = (name.split(' ')).join('<br>');
        }
        return name;
    }

        RegionSelector.prototype.renderSearchResult = function (search) {
            var self = this;
            var searchResult = this.searchData;

            if (!!search) {
                searchResult = _.filter(searchResult, function (item) {
                    return (item.city.toLowerCase()).indexOf(search.toLowerCase()) !== -1;
                });
            } else {
                this.controls.search.results.empty();
                return;
            }

            var strResult = _.reduce(searchResult, function (acc, item) {
                return acc + self.linkTemplate(item);
            }, '');

            this.controls.search.results
                .empty()
                .append( this.searchResultTemplate({ items: strResult, col: searchResult.length }) );
        };

        RegionSelector.prototype.prepareSearchData = function () {
            var res = [];
            var self = this;
            var getDataForSearch = function(data) {
                return{
                    id          : data.id,
                    city        : data.city,
                    en          : data.en,
                    isCurrent   : data.id == self.currentArea
                }
            };

            _.forOwn(this.data.areas, function (area, area_id) {
                if (typeof area.cities !== 'undefined') {
                    _.forOwn(area.cities, function (city, city_id) {
                        if (city_id != window.COUNTRY_ID) {
                            res.push(getDataForSearch(city));
                        }
                    })
                } else {
                    if (area_id != window.COUNTRY_ID) {
                        res.push(getDataForSearch(area.region));
                    }
                }
            });

            return _.sortBy(res, ['city']);
        };

<script>
    (function() {
  function getScrollTop() {
    return document.documentElement.scrollTop || document.body.scrollTop;
  }

  function getEventPath(e) {
    var path = [];
    var root = document.querySelector('body');
    var currentEl = e.target;

    while (currentEl && currentEl !== root) {
      path.push(currentEl);
      currentEl = currentEl.parentNode;
    }
    return path;
  }

  function findEventTargetByDataset(e, dataKey, value) {
    var el;
    var path = getEventPath(e);
    for (var i = 0; i < path.length; i += 1) {
      if (
        path[i].dataset &&
        path[i].dataset[dataKey] !== undefined &&
        (value === undefined || value + '' === path[i].dataset[dataKey])
      ) {
        el = path[i];
        break;
      }
    }
    return el;
  }

  var bodyTag = document.querySelector('body');
  var hintBlock = document.querySelector('.xhint');
  var hintContent = document.querySelector('.xhint__content');
  var hintArrow = document.querySelector('.xhint__arrow');
  if (!hintBlock) return;

  var hintCss = window.getComputedStyle(hintBlock);
  var currentElemWithHintMinWidth =
    hintArrow.offsetWidth + parseInt(hintCss.paddingLeft, 10) + parseInt(hintCss.paddingRight, 10);

  var currentElemWithHint;

  var closeTimeoutId;
  var cursorOverHint = false;
  var cursorOverElemWithHint = false;

  var onHoverDataSetAttr = 'hintOnHover';

  hintBlock.addEventListener('mouseenter', function() {
    cursorOverHint = true;
  });
  hintBlock.addEventListener('mouseleave', function() {
    cursorOverHint = false;
    mouseOutHandler();
  });

  function hintCursorHandler(e) {
    var el = findEventTargetByDataset(e, onHoverDataSetAttr);
    if (!el) return;

    cursorOverElemWithHint = true;
    openHint(el, onHoverDataSetAttr);
    el.addEventListener('mouseleave', mouseLeaveFromHintElHandler);
  }

  function mouseLeaveFromHintElHandler() {
    currentElemWithHint.removeEventListener('mouseleave', mouseLeaveFromHintElHandler);
    cursorOverElemWithHint = false;
    mouseOutHandler();
  }

  function mouseOutHandler() {
    closeTimeoutId = setTimeout(function() {
      if (
        currentElemWithHint &&
        currentElemWithHint.dataset[onHoverDataSetAttr] !== undefined &&
        !cursorOverHint &&
        !cursorOverElemWithHint
      )
        closeCurrentHint();
    }, 300);
  }

  function setHintBgColor(color = '#fff') {
    hintContent.style.backgroundColor = color;
    hintArrow.style.color = color;
  }

  function openHint(el, contentAttr) {
    clearTimeout(closeTimeoutId);

    if (el && el !== currentElemWithHint) {
      hintContent.innerHTML =
        el.dataset.hintNext && el.nextElementSibling ? el.nextElementSibling.outerHTML : el.dataset[contentAttr];
      currentElemWithHint = el;

      if (!el.dataset.hintBgFrom) {
        var bgElem = document.querySelector(el.dataset.hintBgFrom);
        if (bgElem) {
          setHintBgColor(getComputedStyle(bgElem).backgroundColor);
        }
      } else {
        setHintBgColor();
      }
    }

    var hintMaxWidth = el.dataset.hintMaxWidthShow ? +el.dataset.hintMaxWidthShow : 0;
    var hintMaxWidthShow = (window.innerWidth >= hintMaxWidth);

    if (currentElemWithHint && hintMaxWidthShow) {
      positionHint();
      hintBlock.classList.add('shown');
      currentElemWithHint.dataset.hintOpened = 1;
    }
  }

  function positionHint() {
    if (!currentElemWithHint) return;

    var elCoords = currentElemWithHint.getBoundingClientRect();
    var elBottom = elCoords.top + elCoords.height;

    var newLeft = elCoords.left + elCoords.width / 2 - hintBlock.offsetWidth / 2;
    if (currentElemWithHint.dataset.hintAlign === 'right') {
      newLeft = elCoords.left + Math.max(elCoords.width, currentElemWithHintMinWidth) - hintBlock.offsetWidth;
    }

    if (bodyTag.offsetWidth < hintBlock.offsetWidth) {
      setHintCoords({ left: 0, right: 0 });
    } else if (newLeft < 0) {
      setHintCoords({ left: 0 });
    } else if (bodyTag.offsetWidth < hintBlock.offsetWidth + newLeft) {
      setHintCoords({ left: bodyTag.offsetWidth - hintBlock.offsetWidth - 1 });
    } else {
      setHintCoords({ left: newLeft });
    }

    hintArrow.style.left = elCoords.left + elCoords.width / 2 - parseInt(hintBlock.style.left, 10) + 'px';

    var yPosition;
    if (elBottom + hintBlock.offsetHeight <= window.innerHeight) {
      yPosition = 'bottom';
    } else if (elCoords.top >= hintBlock.offsetHeight) {
      yPosition = 'top';
    } else if (elCoords.top + getScrollTop() < hintBlock.offsetHeight) {
      yPosition = 'bottom';
    } else {
      yPosition = elCoords.top > window.innerHeight - elCoords.top - elCoords.height ? 'top' : 'bottom';
    }

    hintBlock.dataset.position = yPosition;

    var optimalTopValue = (yPosition === 'top' ? elCoords.top - hintBlock.offsetHeight : elBottom) + getScrollTop();
    setHintCoords({ top: optimalTopValue });
  }

  function setHintCoords({ top, left, right }) {
    if (left !== undefined) hintBlock.style.left = left + 'px';
    if (right !== undefined) hintBlock.style.right = right + 'px';
    if (top !== undefined) hintBlock.style.top = top + 'px';
  }

  function closeCurrentHint() {
    if (!currentElemWithHint) return;

    hintBlock.classList.remove('shown');
    currentElemWithHint.dataset.hintOpened = 0;
    currentElemWithHint = null;
  }

  window.addEventListener('mouseover', hintCursorHandler);
  window.addEventListener('resize', closeCurrentHint);
})();
