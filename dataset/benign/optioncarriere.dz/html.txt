var ancestors = getAncestors(el);
var selector_0 = selector.charAt(0);
var selector_type;
if( selector_0 == '#' ){
selector_type = 'id';
}
else if( selector_0 == '.' ){
selector_type = 'className';

else{
selector_type = 'nodeName';
}
var ancestor;

return ancestor;

}
else if( selector_type == 'nodeName' && ancestor.nodeName.toLowerCase() === selector ){
return ancestor;
}
}
return null;
};
