        if(!Date.now) Date.now = function() { return new Date(); }
        Date.time = function() { return Date.now().getUnixTime(); }

        var time = new Date().getUnixTime();
        var cookieSettings = {expires:365};
        $('#gdprsticky a.close, #gdprsticky a.accept').click(function(e){
            e.preventDefault();
            Cookies.set('gdpr_clicked', time,cookieSettings);
            Cookies.set('gdpr_accepted', time,cookieSettings);
            slideGDPRDown();
        });

	if(typeof Cookies.get('gdpr_clicked') === 'undefined'){
            $('#gdprsticky').show();
        }

        $('#gdprsticky a.moreInfo').click(function(e){
            e.preventDefault();
            Cookies.set('gdpr_clicked', time,cookieSettings);
            $.colorbox({
                html:$('#gdprLightboxContainer').html(),
                maxWidth:500,
                onComplete:function(){
                    $('.gdprLightbox .accept a').click(function(e){
                        e.preventDefault();
                        if($(this).hasClass('turnOff')) {
                            Cookies.set('gdpr_notrack', time,cookieSettings);
                            Cookies.remove('gdpr_accepted');
                            Cookies.remove('__ar_v4');
                            Cookies.remove('__qca');
                            Cookies.remove('_ga');
                            Cookies.remove('_gaexp');
                            Cookies.remove('_gid');
                            Cookies.remove('_gat');
                        }
                        slideGDPRDown();
                        $.colorbox.close();
                    });
                }
            });
        });

        function slideGDPRDown(){
            $('#gdprsticky').animate({'bottom':-$('#gdprsticky').height()-50},{
                done:function(){ $(this).remove(); window.location.reload(false); }
            });
        }
    });
</script>
