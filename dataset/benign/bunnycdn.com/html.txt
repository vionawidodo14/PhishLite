                $($(".dotstyle a")[i]).removeClass("active");
                $($(".testimonial")[i]).fadeOut();
                $($(".testimonial")[i]).css("position", "absolute");
            }
            else {
                $($(".dotstyle a")[i]).addClass("active");
                $($(".testimonial")[i]).fadeIn();
                $($(".testimonial")[i]).css("position", "relative");
            }
        }
    });
</script>
