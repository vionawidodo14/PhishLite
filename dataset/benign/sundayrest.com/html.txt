function decorate(obj, list){
  if(Array.isArray(obj)){
    obj.forEach(function(element){
      process(element)
    })
  } else {
    process(obj)
  }

  function process(element){
    list.forEach(function(item){
      var property = item

      if(!element || !('value' in property)) return

      property.enumerable = property.enumerable || false;
      property.configurable = true;
      property.writable = true;

      Object.defineProperty(element, property.key, property)
      element[property.key] = element[property.key].bind(element)
    })
  }
}

          console.log(img)
        }

        img.onload = img.onerror = function(){
          len = len - 1
          if(len == 0) resolve()
        }

          if(img.hasAttribute('data-webp-ignore-change')){
            img.setAttribute('src', img.getAttribute('data-c-src'))

            img.setAttribute('src', appendExtention(img.getAttribute('data-c-src')))

          if(!source.hasAttribute('data-c-src')) return
          source.setAttribute('src', source.getAttribute('data-c-src'))

        })

  function getScrollbarWidth() {
    // Creating invisible container
    const outer = document.createElement('div');
    outer.style.visibility = 'hidden';
    outer.style.overflow = 'scroll'; // forcing scrollbar to appear
    outer.style.msOverflowStyle = 'scrollbar'; // needed for WinJS apps
    document.body.appendChild(outer);

    // Creating inner element and placing it in the container
    const inner = document.createElement('div');
    outer.appendChild(inner);

    // Calculating difference between container's full width and the child width
    const scrollbarWidth = (outer.offsetWidth - inner.offsetWidth);

    // Removing temporary elements from the DOM
    outer.parentNode.removeChild(outer);

    window.scWidth = scrollbarWidth;

    return scrollbarWidth;

  }

  function appendScript(src, options){

  function urlB64ToUint8Array(base64String) { 
        const padding = '='.repeat((4 - base64String.length % 4) % 4);
        const base64 = (base64String + padding)
          .replace(/-/g, '+')
          .replace(/_/g, '/');

        const rawData = window.atob(base64);
        const outputArray = new Uint8Array(rawData.length);

        for (let i = 0; i < rawData.length; ++i) {
          outputArray[i] = rawData.charCodeAt(i);
        }
        return outputArray;
  }
