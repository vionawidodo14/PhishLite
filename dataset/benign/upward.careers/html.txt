function validateSearchForm() {
var errors = [];
	if($('#jobInput').val() == '') {
        $('#jobInput').val('All Jobs');
	}

	var zipCodeDigitPattern = /^\d{5}$/;
	 var letters = /^[A-Za-z]+$/;

	 if("" == $('#locationInput').val())
	 {
		 errors += " *Zip Code(5 digits)/Min 2 Characters\n";
		 $("#locationInput").css('background','#FDD');
	 }
	 else if("" != $('#locationInput').val() && !$('#locationInput').val().charAt(0).match(letters) && !zipCodeDigitPattern.test($('#locationInput').val()))
	 {
		 errors += " *Zip Code(5 digits)\n";
		 $("#locationInput").css('background','#FDD');
	 }
	else if("" != $('#locationInput').val() && $('#locationInput').val().match(letters))
      {
			if($('#locationInput').val().trim().length < 2)
			{
				errors += "  * Enter min 2 Characters\n";
				$("#locationInput").css('background','#FDD');
			}
	  }

	if (errors.length > 0) {
			alert("Please correct the following:\n\n"+errors);
		return false;
	}
	return true;
}
