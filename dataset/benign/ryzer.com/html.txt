    var canvas, stage, exportRoot, anim_container, dom_overlay_container, fnStartAnimation;
    function init() {
        canvas = document.getElementById("canvas");
        anim_container = document.getElementById("animation_container");
        dom_overlay_container = document.getElementById("dom_overlay_container");
        var comp=AdobeAn.getComposition("BB99354AE40FEE4090D96C93F583B751");
        var lib=comp.getLibrary();
        var loader = new createjs.LoadQueue(false);
        loader.addEventListener("fileload", function(evt){handleFileLoad(evt,comp)});
        loader.addEventListener("complete", function(evt){handleComplete(evt,comp)});
        var lib=comp.getLibrary();
        loader.loadManifest(lib.properties.manifest);
    }
    function handleFileLoad(evt, comp) {
        var images=comp.getImages();	
        if (evt && (evt.item.type == "image")) { images[evt.item.id] = evt.result; }	
    }
    function handleComplete(evt,comp) {
        //This function is always called, irrespective of the content. You can use the variable "stage" after it is created in token create_stage.
        var lib=comp.getLibrary();
        var ss=comp.getSpriteSheet();
        var queue = evt.target;
        var ssMetadata = lib.ssMetadata;
        for(i=0; i<ssMetadata.length; i++) {
            ss[ssMetadata[i].name] = new createjs.SpriteSheet( {"images": [queue.getResult(ssMetadata[i].name)], "frames": ssMetadata[i].frames} )
        }
        exportRoot = new lib.ryzerhomepage_HTML5Canvas_smaller();
        stage = new lib.Stage(canvas);	
        //Registers the "tick" event listener.
        fnStartAnimation = function() {
            stage.addChild(exportRoot);
            createjs.Ticker.setFPS(lib.properties.fps);
            createjs.Ticker.addEventListener("tick", stage);
        }	    
        //Code to support hidpi screens and responsive scaling.
        AdobeAn.makeResponsive(true,'both',true,1,[canvas,anim_container,dom_overlay_container]);	
        AdobeAn.compositionLoaded(lib.properties.id);
        fnStartAnimation();
    }

		var windowWidth = $(window).width();
		$(document).ready(function(){		
			var	w = $(window).height(),
				d = $('#animation_container').outerHeight(true);
				
			if (w > d || $(window).scrollTop() > 100)	{
				$('.scrollDown').hide();
			}

			$(document).on('click', 'a[href^="#"]', function (event) {
				event.preventDefault();
			
				$('html, body').animate({
					scrollTop: $($.attr(this, 'href')).offset().top
				}, 750);
			});
			
			$(document).on('click','.scrollDown', function() {
				$('.scrollDown').fadeOut(500);
				$('html, body').animate({
					scrollTop: $("#RyzerEvents").offset().top - 50
				}, 1000);
			});
			
			$(window).scroll(function() {
				$('.scrollDown').fadeOut(500);
			});
			
			$("#infoJoin").validate({ 
						errorPlacement: function(error, element) {
			var errorWrapper = "<div class='textError'><i class='fa fa-exclamation-triangle'></i></div>"
			$(element).parent("div").append(errorWrapper);
			$(element).parent("div").find(".textError").append(error);
			$(element).parent("div").addClass("errorBox");
			$(element).data("powertiptarget",error.attr("id"));
		},
		highlight: function(element,errorClass) {
			$(element).parent("div").find(".textError").show();
			$(element).addClass(errorClass);
			$(element).parent("div").addClass("errorBox");
			$(element).powerTip("destroy");
			$(element).powerTip({manual:true});
			$(element).on("focus", function(e){$(this).powerTip("show");});
			$(element).on("keydown", function(e){$(this).powerTip("reposition");});
			$(element).on("focusout", function(e){$(this).powerTip("hide");});
		},
		unhighlight: function(element,errorClass) {
			if($(element).hasClass(errorClass))
			{
				$(element).parent("div").find(".textError").hide();
				$(element).removeClass(errorClass);
				$(element).parent("div").removeClass("errorBox");
				$(element).powerTip("hide");
				$(element).powerTip("destroy");
				$(element).off("focus");
				$(element).off("keydown");
				$(element).off("focusout");
			}
		},
		errorElement: "div"
	
			})
			
		});
	</script>
