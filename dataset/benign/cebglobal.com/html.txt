		function updateSelectBox(selectedValue, key, selectBoxToUpdate, clearSelectBoxes) {
			if (clearSelectBoxes) {
				
				clearSelectBoxes.each(function(index, element) {
					$(element).empty();
					$(element).append($('<option>', {
						value : '',
						text : 'Select'
					}));
					$(element).selectBox('refresh');
				});
			}
	
			if (selectedValue && document.getElementById(key) != null) {
				var seValue = document.getElementById(key).value
				var subIndustryList = seValue.split('#');
				if (subIndustryList) {
					
					subIndustryList.map(function(val) {
						selectBoxToUpdate.append($('<option>', {
							value : val.trim(),
							text : val.trim()
						}));
					});
				}
				selectBoxToUpdate.selectBox('refresh');
			}
		}

		function validateAndPopulateFunctionRoleDetails() {
			
			var industry = $('#Function-Role-Form').find('#uIndustry option:selected').val();
			var subIndustry = $('#Function-Role-Form').find('#subindustry1 option:selected').val();
			
			if(!industry) {
				industry='Not Selected';
			}
			
			if(!subIndustry) {
				subIndustry = 'Not Selected';
			} 
			
			var json = '{"industry":"'+industry+'","subIndustry": "'+subIndustry+'","functionRoleMap": [], "kiTags" : []}';
			
			var functionList = [];
			var roleList = [];
			var isDuplicate = false;
			var partialRecFunctionValue='Not Selected';
			var isPartialRecUpdated = false;
			
			$('#Function-Role-Form select#ufunction').each(function(index,value) {
				
				var functionValue = $(value).val();
				var roleValue = $(value).closest('#functionRoleGroup').find('#urole option:selected').val();
				$(value).closest('#functionRoleGroup').find('#roleLabel').text('Role');
				$(value).closest('#functionRoleGroup').find('#roleLabel').css('color', 'black');
				$(value).closest('#functionRoleGroup').find('#roleLabel').css('font-size', '14px');
				$(value).closest('#functionRoleGroup').find('#roleLabel').remove('span');
				
				if(!functionValue) {
					functionValue = 'Not Selected';
				}
				
				if(!roleValue) {
					roleValue = 'Not Selected';
				}
				
				if(functionValue != 'Not Selected' && roleValue == 'Not Selected' && !isPartialRecUpdated) {
					partialRecFunctionValue = functionValue;
					isPartialRecUpdated = true;
				}
				
				if(roleList.indexOf(roleValue) > -1) {
					var isFunExist = false;
					for(var i=0;i<roleList.length;i++) {
						if(roleList[i]==roleValue && functionList[i]== functionValue) {
							isFunExist = true;
						}
					}
					if(!isFunExist && roleValue != 'Not Selected') {
						functionList.push(functionValue);
						roleList.push(roleValue);
					} else {
						isDuplicate = true;
						$(value).closest('#functionRoleGroup').find('#roleLabel').text('Role');
						$(value).closest('#functionRoleGroup').find('#roleLabel').css('font-size', '13px');
						$(value).closest('#functionRoleGroup').find('#roleLabel').append('<span style="font-size:11px"> (Please make another selection) </span>');
						$(value).closest('#functionRoleGroup').find('#roleLabel').css('color', 'red');
						
					}
				} else if(roleValue != 'Not Selected'){
					functionList.push(functionValue);
					roleList.push(roleValue);
				}
				
				if(json && roleValue != 'Not Selected') {
					var obj = JSON.parse(json);
					obj['functionRoleMap'].push({'function':functionValue,"role":roleValue});	
					json = JSON.stringify(obj);
				}
				
				
			});
			
			if(!isDuplicate) {
				if(functionList.length == 0) {
					var obj = JSON.parse(json);
					obj['functionRoleMap'].push({'function':partialRecFunctionValue,"role":'Not Selected'});	
					json = JSON.stringify(obj);
				}
				return json;
			}
		}
