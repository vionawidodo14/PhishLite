jQuery(document).ready(function(){
   jQuery('#hpHero-carousel').carousel({
      interval: 3500,
      keyboard: true,
      wrap: true
   });

   jQuery('#hpHero-carousel').on("touchstart", function(event) {
      var xClick = event.originalEvent.touches[0].pageX;
      jQuery(this).one("touchmove", function(event) {
         var xMove = event.originalEvent.touches[0].pageX;
         if( Math.floor(xClick - xMove) > 5 ){
            jQuery(this).carousel('next');
         }
         else if( Math.floor(xClick - xMove) < -5 ){
            jQuery(this).carousel('prev');
         }
      });
      jQuery('#hpHero-carousel').on("touchend", function(){
         jQuery(this).off("touchmove");
      });
   });

   jQuery('body').on('click','.play-pause button',function() {
      if(jQuery(this).children('.fa').hasClass('fa-pause')) {
         jQuery('#hpHero-carousel').carousel('pause');
      }
      if(jQuery(this).children('.fa').hasClass('fa-play')) {
         jQuery('#hpHero-carousel').carousel('cycle');
      }
      jQuery(this).children('.fa').toggleClass('fa-pause fa-play'); //Add or remove play/pause symbols
   });
});
