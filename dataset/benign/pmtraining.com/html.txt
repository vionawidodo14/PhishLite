    var timeRemaining;
    var ShowDemoDialog = function (dialogClass, offset) {
        // fix the background so it won't scroll
        if ($(document).height() > $(window).height()) {
            $('html').css("height", "100%");
            $('body').css("height", "100%").css('overflow', 'hidden');

            // move the window slightly to make the scrollbar disappear
            window.scrollBy(0, 1);
            window.scrollBy(0, -1);
        }

        // adjust height of too-tall dialogs
        console.log($(window).height());
        if ($(dialogClass).height() > $(window).height() - offset) {
            $(dialogClass).css('height', $(window).height() * .95 - offset);
            $(dialogClass + ' .modal-content').css('height', $(window).height() * .95 - offset);
            $(dialogClass + ' .scroll-area').css('height', $(window).height() * .95 - offset - 125).css('overflow', 'auto');
        }

       // $(".modal-dialog, .modal-content").css("height", "98%").css("width", "98%").css("top", 0).css("left", 0).css("right", 0).css("bottom", 0);

        // start the timer
        if ($('#ctl00_body_demoTest_trainingModeHiddenField').val() === "False") {
            if ($.isFunction($(document).everyTime)) {
                $(document).stopTime();
            }

            if (timeRemaining === undefined || Number($('#ctl00_body_demoTest_timeRemainingHiddenField').val() > timeRemaining)) {
                timeRemaining = Number($('#ctl00_body_demoTest_timeRemainingHiddenField').val());
            }

            $(document).everyTime(1000,
                function() {
                    if (timeRemaining <= 0) {
                        $(document).stopTime();
                        if ($('#ctl00_body_demoTest_modalDialog').is(":visible")) {
                            $('#ctl00_body_demoTest_hiddenSubmitButton').click();
                        }
                        return;
                    }

                    timeRemaining -= 1;
                    var minutes = Math.floor(timeRemaining / 60),
                        seconds = Math.floor(timeRemaining % 60);
                    $('#ctl00_body_demoTest_timeRemaining').text((minutes < 10 ? '0' + minutes : minutes) +
                        ':' +
                        (seconds < 10 ? '0' + seconds : seconds));
                    $('#ctl00_body_demoTest_timeRemainingHiddenField').val(timeRemaining);
                });
        } else {
            $(document).stopTime();
        }

        // add strike-through class to question responses if they are marked
        $("table.radio span[data-marked='true'] label").addClass("strike-through");

        // when the radio button label is right-clicked
        $(".test-question-container .radio label").contextmenu(function (e) {
            e.preventDefault();

            // show the strike through on the label
            $(this).toggleClass("strike-through");
            // set the corresponding checkbox checked status
            $("input[type='checkbox'][value='" + $(this).siblings("input").val() + "']").prop("checked", $(this).hasClass("strike-through"));
        });

        // when the mouse is released check to see if we need to select part of the question
        $(".test-demo-dialog").mouseup(function(e) {

            // if the highlight icon is visible
            if ($("#highlightIcon").is(":visible")) {

                // close the highlight icon
                $("#highlightIcon").hide();

                // if the click was outside of the highlight icon, remove the highlight
                if (e.srcElement.id !== "highlightIcon") {
                    $("#hiddenHighlightedText").val($("#hiddenQuestionText").val());
                    $("#questionText").html($("#hiddenQuestionText").val());
                }

                return;
            }

            const selection = window.getSelection(),
                offset = $("#questionContainer").offset();

            // if the selection is not a range, or if the string length is too small, return
            if (selection.type !== "Range" || selection.toString().trim().length < 2) {
                selection.removeAllRanges();
                return;
            }

            // get the selected text, and then deselect it
            const text = selection.toString().trim();
            selection.removeAllRanges();

            // if the selected text is not found in the question, return
            if ($("#hiddenQuestionText").val().indexOf(text) === -1) {
                return;
            }

            // replace all instances of the selected text with the same text wrapped in a 
            // span with a selected class in order to highlight it.
            $("#hiddenHighlightedText").val(
                $("#hiddenQuestionText").val()
                .split(text)
                .join("<span class='selected'>" + text + "</span>"));
            $("#questionText").html($("#hiddenHighlightedText").val());
            $("#highlightIcon").css({ "top": e.pageY - offset.top + 55, "left": e.pageX - offset.left + 22 }).show();

        });
    };
