    var mainBgWrapper = qs('.v-main-lp-top-side-bg');
    var imageToParallax = qs('.v-main-lp-top-side-img');

    function qs(el, getAll) {
        if (getAll) {
            return document.querySelectorAll(el);
        }
        return document.querySelector(el);
    }

    function perc(cursorParam, divParam) {
        return cursorParam * 100 / divParam;
    }

    function moveByMouse () {
        imageToParallax.style.transition = '.5s ease-out';
        imageToParallax.style.transformOrigin = '50% 50%';
        imageToParallax.style.transform = 'scale(1.03)';

        mainBgWrapper.addEventListener('mouseleave', function (e) {
            imageToParallax.style.setProperty('transform-origin', `50% 50%`);
        });

        mainBgWrapper.addEventListener('mousemove', function (e) {
            const r = mainBgWrapper.getBoundingClientRect();
            const {pageX, pageY} = e;
            const {offsetTop} = mainBgWrapper;
            const {width, height} = r;
            let cursorY = pageY - offsetTop;
            let [originX, originY] = [perc(pageX, width), perc(cursorY, height)];
            imageToParallax.style.setProperty('transform-origin', `${originX}% ${originY}%`);
        })
    }

    moveByMouse();


    countUp(.3);

    function countUp(sec) { // sec - time for animating in seconds
        var counters = qs('[data-counter-number]', true);
        [].forEach.call(counters, function(el) {
            countUpOne(el, sec);
        });
        function countUpOne(sel, sec) {
            let currNum = Number(sel.getAttribute('data-counter-number').replace(/\D/, ''));
            let counter = 7;
            counter = currNum/(sec*1000/5);
            function increase(number) {
                setTimeout(function(){
                    let sum = number + counter;
                    if (sum < currNum) {
                        sel.textContent = String(Math.round(sum)).replace(/\d{1,3}(?=(\d{3})+(?!\d))/g, '$& ');
                        increase(sum);
                    } else {
                        sel.textContent = String(currNum).replace(/\d{1,3}(?=(\d{3})+(?!\d))/g, '$& ');
                    }
                }, 5)
            }
            increase(0);
        }
    }
