        var paymentValue = $("#MainContent_ucFooter_BalancePayment_5_MakePayment_TBox_PaymentVal").val();
        $("#MainContent_ucFooter_BalancePayment_5_MakePayment_Pnl_QuickPayError").hide();
        $("#MainContent_ucFooter_BalancePayment_5_MakePayment_Pnl_CheckAmount").hide();


        if ($("#MainContent_ucFooter_BalancePayment_5_MakePayment_TBox_PaymentVal").val().length == 0) {
            $("#MainContent_ucFooter_BalancePayment_5_MakePayment_Lbl_QuickPayError").text('Please enter the amount you would like to pay.');
            $("#MainContent_ucFooter_BalancePayment_5_MakePayment_Lbl_QuickPayError").show();
            $("#MainContent_ucFooter_BalancePayment_5_MakePayment_Pnl_QuickPayError").show();
        }
        else if (paymentValue <= 0) {
            $("#MainContent_ucFooter_BalancePayment_5_MakePayment_Lbl_QuickPayError").text('Please enter a valid amount.');
            $("#MainContent_ucFooter_BalancePayment_5_MakePayment_Lbl_QuickPayError").show();
            $("#MainContent_ucFooter_BalancePayment_5_MakePayment_Pnl_QuickPayError").show();

        }
        else if (paymentValue > 0 && paymentValue < 100) {
            $("#MainContent_ucFooter_BalancePayment_5_MakePayment_Btn_Yes").click();
        }

        else if (paymentValue >= 100 && paymentValue < 1000) {
            var renameMe1 = 'Are you sure you want to pay &#163;' + paymentValue;

            $("#MainContent_ucFooter_BalancePayment_5_MakePayment_Lbl_CheckAmount").html(renameMe1);
            $("#MainContent_ucFooter_BalancePayment_5_MakePayment_Lbl_CheckAmount").show();
            $("#MainContent_ucFooter_BalancePayment_5_MakePayment_Pnl_CheckAmount").show();
        }
        else if (paymentValue >= 1000) {
            if (paymentValue.indexOf('.') != -1) {
                var renameMe1 = 'Are you sure you want to pay &#163;' + paymentValue;
                $("#MainContent_ucFooter_BalancePayment_5_MakePayment_Lbl_CheckAmount").html(renameMe1);
                $("#MainContent_ucFooter_BalancePayment_5_MakePayment_Lbl_CheckAmount").show();
                $("#MainContent_ucFooter_BalancePayment_5_MakePayment_Pnl_CheckAmount").show();
            }
            else {
                var length = paymentValue.length - 2;
                var first2 = paymentValue.substring(0, length);
                var last2 = paymentValue.substring(length, paymentValue.length);
                var renameMe2 = 'Are you sure you want to pay &#163;' + paymentValue + ' and you didn\'t mean &#163;' + first2 + '.' + last2 + '?';

                $("#MainContent_ucFooter_BalancePayment_5_MakePayment_Lbl_CheckAmount").html(renameMe2);
                $("#MainContent_ucFooter_BalancePayment_5_MakePayment_Lbl_CheckAmount").show();
                $("#MainContent_ucFooter_BalancePayment_5_MakePayment_Pnl_CheckAmount").show();
            }
        }

        return false;
    });
