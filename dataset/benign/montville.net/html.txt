	function loadVideo(guid, playlist) {
		videojs(guid, playlist, function () {
			var origWidth = jQuery('#' + guid).parent().data('width');
			var origHeight = jQuery('#' + guid).parent().data('height');

			var actualWidth = jQuery('#' + guid).parent().width();
			if (origWidth > actualWidth) {
				this.height(origHeight * (actualWidth / origWidth));
				this.width(actualWidth);

				//Retroactively fixes an issue with resizing videos
				//the style attribute shouldn't be on the parent
				jQuery('#' + guid).parent().removeAttr('style');

			} else {
				this.height(origHeight);
				this.width(origWidth);
			}
			if (playlist != '{}') {
				this.on('ended', function () {
					this.playlist.next();
				});
			}

			//fixes issue in IE when going full screen
			//video would be behind other content on the page
			this.on('enterFullWindow', function () {
				$(this.el).parents().not('html,body').each(function () {
					$(this)
						.data('oldzindex', $(this).css('z-index'))
						.css('z-index', 99999)
						.addClass('forcedzindex');
				});
			});
			this.on('exitFullWindow', function () {
				$('.forcedzindex').each(function () {
					$(this)
						.css('z-index', $(this).data('oldzindex'))
						.removeClass('forcedzindex')
						.removeData('oldzindex');
				});

			});
		});
	}

	if (!Array.prototype.indexOf)
		Array.prototype.indexOf = function (searchValue, index) {
			// In non-strict-mode, if the `this` variable is null
			// or undefined, then it is set the the window object.
			// Else, `this` is automaticly converted to an object.
			var len = this.length >>> 0; // convert ot number or 0

			index |= 0;              // rounds and NaN-checks
			if (index < 0)             // check if negative start
				index = Math.max(len - index, 0);
			else if (index >= len) // else, check if too big
				return -1;

			if (searchValue === undefined)
				// Because searchValue is undefined, keys that
				// don't exist will have the same value as the
				// searchValue, and thus do need to be checked.
				do {
					if (index in this && this[index] === undefined)
						return index;
				} while (++index !== len)
			else
				// Because searchValue is not undefined, there's no
				// need to check if the current key is in the array
				// because if the key isn't in the array, then it's
				// undefined which is not equal to the searchValue.
				do {
					if (this[index] === searchValue)
						return index;
				} while (++index !== len)

			// if nothing was found, then simply return -1
			return -1;
		};

	// https://stackoverflow.com/a/51789311/5629100
	var YoutubeVideos = (function () {
		var playersYoutube = [],	// players array (to control players individually)
			queueYoutube = [];		// videos queue (once api is ready, transform this into YT player)

		// Constructor
		function YoutubeVideos() { }

		// Add elements to queue
		YoutubeVideos.prototype.add = function ($video) {
			queueYoutube.push($video);
		};

		// Finds player based on the id of the iframe element
		YoutubeVideos.prototype.findPlayer = function (id) {
			var playerObj = null;

			playersYoutube.forEach(function (item) {
				if (item.playerId == id) {
					playerObj = item.player;
				}
			});

			return playerObj;
		};

		// Load Youtube API
		YoutubeVideos.prototype.loadApi = function () {
			$.getScript("//www.youtube.com/iframe_api", function () {
				window.onYouTubeIframeAPIReady = function () {
					queueYoutube.forEach(function ($video) {
						var loopBit = function () {
							if ($video.data('loop-video') == true) {
								return 1;
							}
							return 0;
						};

						var player = new YT.Player($video.get(0), {
							width: $video.data('frame-width'),
							height: $video.data('frame-height'),
							videoId: $video.data('video-id'),
							events: {
								'onStateChange': onPlayerStateChange,
								'onReady': onPlayerReady
							},
							playerVars: {
								autoplay: 0,
								rel: 0,
								enablejsapi: 1,
								loop: loopBit
							}
						});

						playersYoutube.push({
							playerId: $video.attr('id'),
							player: player
						});
					});
				};
			});
		};

		return YoutubeVideos;
	})();

	var VimeoVideos = (function () {
		var queueVimeo = [],		// queue of elements which will become players.
			playersVimeo = [];		// players array (to control players individually)

		// Constructor
		function VimeoVideos() { }

		// Add elements to queue
		VimeoVideos.prototype.add = function ($video) {
			queueVimeo.push($video);
		};

		// Converts queue to players.
		VimeoVideos.prototype.loadPlayers = function () {
			queueVimeo.forEach(function (item, index) {
				var vimeoOptions = {
					id: $(item).data('video-id'),
					autoplay: false,
					muted: ($(item).data('autoplay') == true) ? true : false
				};

				playersVimeo[index] = {
					playerId: $(item).attr('id'),
					player: new Vimeo.Player(item, vimeoOptions)
				};
			});
		};

		// Finds player based on the id of the iframe element
		VimeoVideos.prototype.findPlayer = function (id) {
			var playerObj = null;

			playersVimeo.forEach(function (item) {
				if (item.playerId == id) {
					playerObj = item.player;
				}
			});

			return playerObj;
		};

		// Adds listeners to the players that have been created now so they interact
		// correctly with the header galleries.
		VimeoVideos.prototype.bindListeners = function () {
			playersVimeo.forEach(function (item) {
				item.player.on('play', function () {
					var $wrapSlideshow = $(this.element.parentNode).parents('.wrap-slideshow');
					$wrapSlideshow.find('.cycle-pager').fadeOut();
					$wrapSlideshow.find('.playPause').fadeOut();
					$wrapSlideshow.find('.headerGalleryV2').cycle('pause');
				});
			});

			playersVimeo.forEach(function (item) {
				item.player.on('pause', function () {
					var $wrapSlideshow = $(this.element.parentNode).parents('.wrap-slideshow');

					$(this.element.parentNode).data('paused', true);
					$wrapSlideshow.find('.cycle-pager').fadeIn();
					$wrapSlideshow.find('.playPause').fadeIn();
					$wrapSlideshow.find('.headerGalleryV2').cycle('resume');
				});
			});

			playersVimeo.forEach(function (item) {
				item.player.on('ended', function () {
					if ($(this.element.parentNode).data('loop-video') == true) {
						this.play();
					} else {
						var $wrapSlideshow = $(this.element.parentNode).parents('.wrap-slideshow');

						$(this.element.parentNode).data('paused', true);
						$wrapSlideshow.find('.cycle-pager').fadeIn();
						$wrapSlideshow.find('.playPause').fadeIn();
						$wrapSlideshow.find('.headerGalleryV2').cycle('resume');
					}
				});
			});

			playersVimeo.forEach(function (item) {
				item.player.ready().then(function () {
					if ($(this).parents('.cycle-slide-active').length > 0 && $(this).data('autoplay') == true) {
						this.play();
					}
				});
			});
		};

		return VimeoVideos;
	})();

	// When the Youtube video is paused, ends, or begins, we need to either
	// show or hide the slideshow controls and either pause or resume the
	// slideshow. We also handle looping the Youtube videos here because the
	// Youtube API is not 100% reliable for that function.
	function onPlayerStateChange(event) {
		var $videoElem = $(event.target.a);
		var $wrapSlideshow = $videoElem.parents('.wrap-slideshow').first();
		var videoState = event.data;

		if (videoState == YT.PlayerState.PAUSED) {
            $videoElem.data('paused', true);
			$wrapSlideshow.find('.cycle-pager').fadeIn();
			$wrapSlideshow.find('.playPause').fadeIn();
			$wrapSlideshow.find('.headerGalleryV2').cycle('resume');
		}

		if (videoState == YT.PlayerState.ENDED) {
			if ($videoElem.data('loop-video') == true) {
				playYoutubePlayerById($videoElem.attr('id'));
            } else {
                $videoElem.siblings('.youtubeVideoThumbnail').fadeIn();
                $videoElem.siblings('.youtubeVideoThumbnailPlay').fadeIn();
				$wrapSlideshow.find('.cycle-pager').fadeIn();
				$wrapSlideshow.find('.playPause').fadeIn();
				$wrapSlideshow.find('.headerGalleryV2').cycle('resume');
			}
		}

		if (videoState == YT.PlayerState.PLAYING) {
            $videoElem.data('paused', false);
            $videoElem.siblings('.youtubeVideoThumbnail').fadeOut();
            $videoElem.siblings('.youtubeVideoThumbnailPlay').fadeOut();
			$wrapSlideshow.find('.headerGalleryV2').cycle('pause');
			$wrapSlideshow.find('.cycle-pager').fadeOut();
			$wrapSlideshow.find('.playPause').fadeOut();
		}
	}

	function onPlayerReady(e) {
		e.target.mute();

		if ($(e.target.a).parent('.cycle-slide-active').length > 0 && $(e.target.a).data('autoplay') == true) {
			forcePlayYoutubePlayer(e.target);
		}
	}

	function pauseYoutubePlayerById(id) {
		var player = youtubes.findPlayer(id);
		player.pauseVideo();

		if (player.getPlayerState() == 1) {
			forcePauseYoutubePlayer(player);
		}
	}

	function forcePauseYoutubePlayer(player) {
		player.pauseVideo();

		if (player.getPlayerState() == 1) {
			setTimeout(function () {
				forcePauseYoutubePlayer(player);
			}, 100);
		}
	}

	function playYoutubePlayerById(id) {
		var player = youtubes.findPlayer(id);
		forcePlayYoutubePlayer(player);
	}

	function forcePlayYoutubePlayer(player) {
		player.playVideo();

		if (player.getPlayerState() == 1) {
			setTimeout(function () {
				forcePlayYoutubePlayer(player);
			}, 100);
		}
	}

	function pauseVimeoPlayerById(id) {
		var player = vimeos.findPlayer(id);

		if (player != null) {
			player.pause();
		}
	}

	function playVimeoPlayerById(id) {
		var player = vimeos.findPlayer(id);

		if (player != null) {
			player.play();
		}
	}

	function addYouTubeStyles(videoId, containerWidth, containerHeight, videoWidth, videoHeight, videoZoom) {
		var containerRatio = containerWidth / containerHeight;
		var videoRatio = videoWidth / videoHeight;
		var trueSize = $('#' + videoId).parents('.headerGalleryV2').innerWidth() / containerWidth;
		var styles = '#' + videoId + ' {';

		if (containerRatio === videoRatio) { // Video already fits the size of the container.
			styles += 'top: 0;';
			styles += 'left: 0;';
			styles += 'width: 100%;';
			styles += 'height: 100%;';
		} else if (containerRatio > videoRatio) { // Container wider than video, needs to cut off top and bottom of video to scale.
			var scale = (containerWidth / videoWidth);

			styles += 'position: absolute;';
			styles += 'left: 0;';
			styles += 'width: 100%;';

			if (videoZoom === 'nozoom') {
				styles += 'top: 0;';
				styles += 'height: 100%;';
			} else if (videoZoom === 'zoomtop') {
				styles += 'top: 0;';
				styles += 'height: ' + (Math.floor((videoHeight * scale) * trueSize)) + 'px;';
			} else if (videoZoom === 'zoombottom') {
				styles += 'bottom: 0;';
				styles += 'height: ' + (Math.floor((videoHeight * scale) * trueSize)) + 'px;';
			} else if (videoZoom === 'zoommiddle') {
				styles += 'top: 0;';
				styles += 'height: ' + (Math.floor((videoHeight * scale) * trueSize)) + 'px;';
				styles += 'transform: translateY(-' + (((videoHeight * scale) - containerHeight) / (videoHeight * scale) * 50) + '%);'
			}
		} else { // Container taller than video, needs to cut off left and right of video to scale.
			var scale = (containerHeight / videoHeight);

			styles += 'position: absolute;';
			styles += 'left: 0;';
			styles += 'height: 100%;';

			if (videoZoom === 'nozoom') {
				styles += 'top: 0;';
				styles += 'height: 100%;';
			} else if (videoZoom === 'zoomtop') {
				styles += 'top: 0;';
				styles += 'width: ' + (Math.floor((videoWidth * scale) * trueSize)) + 'px;';
			} else if (videoZoom === 'zoombottom') {
				styles += 'bottom: 0;';
				styles += 'width: ' + (Math.floor((videoWidth * scale) * trueSize)) + 'px;';
			} else if (videoZoom === 'zoommiddle') {
				styles += 'top: 0;';
				styles += 'width: ' + (Math.floor((videoWidth * scale) * trueSize)) + 'px;';
				styles += 'transform: translateX(-' + (((videoWidth * scale) - containerWidth) / (videoWidth * scale) * 50) + '%);'
			}
		}

		styles += '}';
		$('#youtubeStyles').append(styles);
	}

<h1 class="feature-content-title title">
        Mission and Values
    </h1>
<div class="feature-content-body content">
<div class="feature-content-content">
<p style="text-align: center;"><strong>Mission Statement</strong></p>
<p>Montville Township Public Schools strive to develop well-rounded, invested, and inspired students by engaging them in an experience of collaboration with educators, parents, and community that fosters a culture of innovation, character and inclusivity, in a shared spirit of pride.</p>
<p style="text-align: center;"><strong>Values Statement</strong></p>
<p>Montville Township Public Schools believes that all students possess the tools necessary for achievement, and it is our role to guide them in crafting a life-long passion and curiosity in fulfilling their potential. The platform on which our students stand, learn, and grow is supported by three pillars:</p>
<p><strong>Innovation</strong></p>
<p>Montville Township Public Schools believes in being a leader in education by creating authentic active learning opportunities in the classroom and beyond.</p>
<p><strong>Character </strong></p>
<p>Montville Township Public Schools values service, engagement, and kindness in connecting our students to one another and the broader community. Through a wide assortment of experiences, course offerings, extracurricular activities and athletic opportunities, we afford our students the means to explore their world and participate in a personalized educational journey designed to help them discover their passions and achieve their goals.  We believe that all students should be ethical, self-motivated, caring and contributing members of the global community.</p>
<p><strong>Community</strong></p>
<p>Montville Township Public Schools seeks to cultivate a safe and inclusive environment that champions a sense of spirit and a respect for tradition. Our students are encouraged to break down barriers and bridge divides to create and participate in opportunities for service within their school, the district and the community at large.</p>
</div>
