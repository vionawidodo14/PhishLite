      function getPosts() {
        var infinityScroll = jQuery('.infinity-scroll');
        var isIndex = 0;
        // Disable infinity scroll until load is complete
        if(infinityScroll.length > 0) {
          infinityScroll.removeClass('infinity-scroll');
          // Parameters
                    var id = false;
          isIndex = 1;
          var type = 'all';
          var page = parseInt(infinityScroll.attr('data-page'), 10) + 1;
          var follows = infinityScroll.hasClass('infinity-scroll-follows');
          var modelId = parseInt(infinityScroll.attr('data-model'), 10);
          var isPrivate = parseInt(infinityScroll.attr('data-is-private'), 10);
          var isPremium = parseInt(infinityScroll.attr('data-is-premium'), 10);
          if (isIndex > 0) {
            jQuery.ajax({
                url: '/api/models?page=' + page,
                type: "post",
                timeout: 45000,
                async: true,
                cache: false,
                crossDomain: true,
                processData: false,
                success: function(data) {
                  if(data != '') {
                    // Add content
                    infinityScroll.attr('data-page', page);
                    infinityScroll.addClass('infinity-scroll');
                    infinityScroll.append(data);
                    lazyload();
                    twttr.widgets.load();
                  } else {
                    // No more content
                    jQuery('.infinity-scroll-footer').remove();
                  }
                }
              });
          } else {
            if(!isNaN(modelId)) {
              id = modelId;
            }
            var sort = infinityScroll.attr('data-sort');
            var sort_by = infinityScroll.attr('data-sort-by');
              jQuery.ajax({
                url: '/api/posts?id=' + id + '&page=' + page + '&follows=' + follows+ '&sort_by=' + sort_by + '&sort=' + sort + '&is_private=' + isPrivate+ '&is_premium=' + isPremium,
                type: "post",
                timeout: 45000,
                async: true,
                cache: false,
                crossDomain: true,
                processData: false,
                success: function(data) {
                  if(data != '') {
                    // Add content
                    infinityScroll.attr('data-page', page);
                    infinityScroll.addClass('infinity-scroll');
                    infinityScroll.append(data);
                    // Setup
                    setup_clipboard();
                    setup_comments();

                    setup_remove_comments();

                    setup_sharing();
                    setup_timestamps();

                    if (typeof twttr !== 'undefined') {
                      twttr.widgets.load();
                    }

                    // No more content
                    jQuery('.infinity-scroll-footer').remove();
                  }
                },

                  console.log(xhr, err);
                }

      // Update Messages
      function setup_messages(old_count) {
        if(jQuery("body").hasClass("account-conversation")) {
          jQuery.ajax({
                          url: "/api/messages/?id=false",
                        type: "GET",
            timeout: 3000,
            success: function(data) {
              var json = JSON.parse(data);
              var json_count = json.count;
              var json_messages = json.messages;
              var new_count = json_count - old_count;
              var messages = [];
              jQuery.each(json_messages, function( k, v ) {
                messages[k] = [v[0],v[1],v[2],v[3]];
              });
              if(new_count > 0) {
                new_messages = messages.splice(old_count,new_count);
                jQuery.each(new_messages, function( k, v ) {
                                      jQuery("#messages").append("<div class='message model-message "+v[3]+"-"+v[2]+"'>"+v[0]+"</div><p class='time-ago time-ago-"+v[3]+"-"+v[2]+"' data-time='"+v[1]+"'><span></span></p>");
                                  });
                setup_timestamps();
                setup_messages_scrollbar();
                setup_videos();
              }
              setTimeout(function() {setup_messages(json_count);},3000);
            },
            error: function(data) {
              setTimeout(function() {setup_messages(old_count);},3000);
            }
          });
        }
      }
