	/* common functions */
	function CheckAmt(fieldID) {
		var AmountPass = true;
		var lamount = $("#" + fieldID).val().replace(",", "");
		if (lamount == "") { AmountPass = false; }
		else { if (parseInt(lamount) < 100 || parseInt(lamount) > 50000) { AmountPass = false; } }
		if (AmountPass) {
			$("#" + fieldID).removeClass("is-invalid");
			$(".invalid-feedback").hide(); // manually hide because of input group
		}
		else {
			$("#" + fieldID).addClass("is-invalid");
			$(".invalid-feedback").show(); // manually show because of input group
		}
		return AmountPass;
	}
	function CheckSelect(field, nextid) 	{
		var label = $('label[for="' + field.id + '"]');
		if ($(field).val() != "0")
		{
			$(field).removeClass("is-invalid");
			$(field).addClass("field-completed");
			if (label) { label.addClass("showlabel"); }
			if (nextid) { GoToNext(nextid); }
		}
		else
		{
			$(field).removeClass("field-completed");
			$(field).addClass("is-invalid");
			if (label) { label.removeClass("showlabel"); }
		}
	}
	function CheckValue(field) {
		var label = $('label[for="' + field.id + '"]');
		if (label) {
			if ($(field).val() !== "") { label.addClass("showlabel"); }
			else { label.removeClass("showlabel"); }
		}
	}
	function CheckEnter(e, currentfield, nextid) {
		if (e.which == 13) {
			e.preventDefault();
			if (nextid) { GoToNext(nextid); }
		}
	}
	function GoToNext(nextid) {
		$('#' + nextid).focus();
	}
	function BindFx() {
		// done here so it can be called after update panel partial postback
		$('input').on("keyup", function () { CheckValue(this); });
		$('input,select').on("focus", function () { $('label[for="' + this.id + '"]').addClass("currentfocus"); });
		$('input,select').on("blur", function () { $('label[for="' + this.id + '"]').removeClass("currentfocus"); });
	}
	$(document).ready(function () { BindFx(); });
