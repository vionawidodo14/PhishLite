<script>
  $(function() {
  
    var on_cookie_settings_page = false;
    function setCookieSettings(name, value) {
      Cookies.set(name, value, { expires: 730 });
    }
    function are_analytical_cookies_accepted(){
      if (on_cookie_settings_page) {
        return $('#analytical').prop("checked");
      } else {
        return true;
      }
    }
    var analytical_cookies_value = Cookies.get('analytical');
    if ((analytical_cookies_value=='true') || (analytical_cookies_value === undefined)) {
      $('#analytical').prop('checked', true);
      $('.analytical').addClass('text-success');
    }
    if (Cookies.get('functional')=='true') {
      $('#functional').prop('checked', true);
      $('.functional').addClass('text-success');
    }
    if (Cookies.get('advertisers')=='true') {
      $('#advertisers').prop('checked', true);
      $('.advertisers').addClass('text-success');
    }
    $('.cookies_').on('click', function(e) {
      setCookieSettings($(this).attr('id'), $(this).is(":checked"));
      if($(this).is(":checked")) {
        $('.'+$(this).attr('id')).addClass('text-success');
      }
      else {
        $('.'+$(this).attr('id')).removeClass('text-success');
      }
    });
    $('#save-cookie-settings, #accept-cookies, .cookieinfo-close').unbind();
    $(document).on("click",'#save-cookie-settings, #accept-cookies, .cookieinfo-close', function(e) {
      e.preventDefault();
      setCookieSettings('analytical', are_analytical_cookies_accepted());
      setCookieSettings('bauhaus_cookies', 1);
      setCookieSettings('cb-enabled', 'accepted');
      
      window.location.reload(true);
    });
      
  });
</script>
