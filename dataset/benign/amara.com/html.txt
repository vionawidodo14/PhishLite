    amaraReady(function () {
        var $link = $('#footer-survey-link');
        var $modal = $('#footer-survey-modal');
        var $iframe = $('#footer-survey-modal-iframe');

        var windowWidth = Math.max(document.documentElement.clientWidth, window.innerWidth || 0);
        if (windowWidth < 768) {
            var windowHeight = Math.max(document.documentElement.clientHeight, window.innerHeight || 0);
            $('.survey-modal-dialog .modal-content').css('minHeight', windowHeight + 'px');
        }

        $link.click(function () {
            if('undefined' !== typeof window.amaraGa ) {
                amaraGa.sendEvent({
                    eventCategory: 'Buttons',
                    eventAction: 'SurveyStart',
                    eventLabel: 'footerFeedback'
                });
            }

            $modal.modal('show');
        });

        $modal.on('shown.bs.modal', function () {
            if (!$iframe.attr('src')) {
                $iframe.attr('src', "https://amaraliving.eu.qualtrics.com/jfe/form/SV_b4m6fBIBGEdxpeB?" + getIframeQueryString());

                window.addEventListener('message', function (ev) {
                    try {
                        // Qualtrics surveys are sending messages we don't want to interact with to our site.
                        // They are of type object so we can filter them out here
                        if (typeof ev.data === "object") {
                            return;
                        }

                        var data = JSON.parse(ev.data);
                        // Add 60px for fixed position bottom link
                        var newHeight = data.height + 60;

                        if (newHeight) {
                            $iframe.attr('height', newHeight + 'px');
                        }
                    } catch (e) {}
                });
            }
        });

        function getIframeQueryString() {
            var extraData = document.getElementById('js-survey-data');

            var queryParams = [
                'site_id=' + encodeURIComponent(extraData.dataset.siteId),
                'device_breakpoint=' + encodeURIComponent(BoostrapViewportHelper.getCurrentViewPort()),
                'page_url=' + encodeURIComponent(window.location.href)
            ];

            if (extraData.dataset.contactId) {
                queryParams.push('contact_id=' + encodeURIComponent(extraData.dataset.contactId))
            }

            return queryParams.join('&');
        }
    });
