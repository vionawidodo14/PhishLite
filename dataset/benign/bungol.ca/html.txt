			var autocompleteAddressField = document.getElementById('autocompleteAddressField');
var autocompleteItems = document.getElementById('autocompleteItems');
var autocompleteInputTimeout;
var autocompleteCurrentFocus = -1;

function addActive(items, keyCode) {
    if (!items) {
        return false;
    }
    removeActive(items);
    if (autocompleteCurrentFocus >= items.length) {
        autocompleteCurrentFocus = 0;
    }
    if (autocompleteCurrentFocus < 0) {
        autocompleteCurrentFocus = items.length - 1;
    }
    while (items[autocompleteCurrentFocus].classList.contains('noKeydown')) {
        if (autocompleteCurrentFocus == 0) {
            if (keyCode == 40) {
                autocompleteCurrentFocus++;
            } else if (keyCode==38) {
                autocompleteCurrentFocus = items.length - 1;
            }
            
        } else if (autocompleteCurrentFocus < items.length) {
            if (keyCode == 40) {
                autocompleteCurrentFocus++;
            } else if (keyCode==38) {
                autocompleteCurrentFocus--;
            }
            
        }
    }
    items[autocompleteCurrentFocus].classList.add('autocompleteActive');
}

autocompleteAddressField.addEventListener('keydown', function(e) {
    let items = autocompleteItems.getElementsByClassName('item');
    if (e.keyCode == 40) {
        autocompleteCurrentFocus++;
        addActive(items, e.keyCode);
    } else if (e.keyCode == 38) {
        autocompleteCurrentFocus--;
        addActive(items, e.keyCode);
    } else if (e.keyCode == 13) {
        //e.preventDefault();
        if (autocompleteCurrentFocus > -1) {
            if (items) {
                e.preventDefault();
                items[autocompleteCurrentFocus].click();
            }
        }
    } else if (e.keyCode == 27) {
        this.value='';
        clearACItems();
    }
})

function removeActive(items) {
    for (let i=0; i<items.length;i++) {
        items[i].classList.remove('autocompleteActive');
    }
}

autocompleteAddressField.addEventListener('input', function(e) {
    autocompleteCurrentFocus = -1;
    if (this.value.length > 2) {
        let value = this.value;
        clearTimeout(autocompleteInputTimeout);
        autocompleteInputTimeout = setTimeout(function() {
            addressAutocomplete(value);
        }, 150)
    } else {
        autocompleteItems.setAttribute('hidden','');
        clearACItems();
    }
})

autocompleteAddressField.addEventListener('click', function(e) {
    e.stopPropagation();
    if (this.value.length > 2 && (!autocompleteItems.firstChild)) {
        let value = this.value;
        addressAutocomplete(value);
    } else if (this.value.length == 0 && localStorage.indexSelectedAddressCount) {
        clearACItems(); 
        createRememberedAddresses();
    }
})

function addressAutocomplete(value) {
    let xhr = new XMLHttpRequest();
    xhr.open('GET', '/api/property-search-autocomplete/?value=' + encodeURIComponent(value));
    xhr.setRequestHeader("Content-type", "application/x-www-form-urlencoded");
    xhr.setRequestHeader('X-Requested-With', 'XMLHttpRequest');
    xhr.onload = function() {
        if (xhr.status == 200 && autocompleteAddressField.value.length > 2) {
            clearACItems();
            autocompleteItems.removeAttribute('hidden');
            let response = JSON.parse(xhr.responseText);
            let addresses = response.addresses;
            let listings = response.listings;
            let cities = response.cities;
            let subways = response.subways;
            let intersections = response.intersections;
            let schools = response.schools;
            if (cities) {
                if (cities.length > 0) {
                    for (let i=0; i<cities.length; i++) {
                        let c = cities[i];
                        let district = '';
                        let text = '';
                        if (c.cd_type == 'RM') {
                            district = 'Region';
                        } else if (c.cd_type == 'CTY') {
                            district = 'County';
                        }
                        if (c.csd_name == c.cd_name) {
                            text = c.cd_name;
                        } else {
                            text = c.csd_name + ', ' + c.cd_name;
                        }
                        if (district != '') {
                            let addedDistrict = ' ' + district
                            text += addedDistrict;
                        }
                        createLocationAutocompleteItem(text, c.la, c.lo, c.zoom, 'city');
                    }
                }
            }
            if (subways) {
                if (subways.length > 0) {
                    for (let i=0; i<subways.length; i++) {
                        let s = subways[i];
                        createLocationAutocompleteItem(s.display_name, s.la, s.lo, 16, 'subway');
                    }
                }
            }
            if (intersections) {
                if (intersections.length > 0) {
                    for (let i=0; i<intersections.length; i++) {
                        let s = intersections[i];
                        let text = s.street1 + ' / ' + s.street2 + ', ' + s.city;
                        createLocationAutocompleteItem(text, s.la, s.lo, 16, 'intersection');
                    }
                }
            }
            if (addresses) {
                if (addresses.length > 0) {
                    /*
                    if (listings.length >0) {
                        let addressHeader = document.createElement('div');
                        addressHeader.innerHTML = 'Locations';
                        addressHeader.classList.add('header');
                        addressHeader.onclick = function(e) {
                            e.stopPropagation();
                        }
                        autocompleteItems.appendChild(addressHeader);
                    }*/
                    for (let i=0; i<addresses.length; i++) {
                        let municipality;
                        if (addresses[i].district) {
                            municipality = addresses[i].district;
                        } else {
                            municipality = addresses[i].city;
                        }
                        createLocationAutocompleteItem(addresses[i].street + ', ' + municipality, addresses[i].la, addresses[i].lo, 17, 'address');
                    }
                }
            }
            if (listings) {
                if (listings.length > 0) {
                    /*
                    let listingsHeader = document.createElement('div');
                    listingsHeader.innerHTML = 'Listings';
                    listingsHeader.classList.add('header');
                    listingsHeader.onclick = function(e) {
                        e.stopPropagation();
                    }
                    autocompleteItems.appendChild(listingsHeader);*/
                    createAutocompleteListings(listings);
                }
            }
            if (schools) {
                if (schools.length > 0) {
                    for (let i=0;i<schools.length;i++) {
                        let name = schools[i].name + ' (' + schools[i].board_name + ')';
                        createLocationAutocompleteItem(name, schools[i].lat, schools[i].lng, 17, 'school');
                    }
                }
            }
        }
    }
    xhr.send();
}

function clearACItems() {
    while (autocompleteItems.firstChild) {
        autocompleteItems.removeChild(autocompleteItems.firstChild);
    }
    autocompleteItems.setAttribute('hidden','');
}

function createAutocompleteListings(listings) {
    for (let i=0; i<listings.length; i++) {
        let l = listings[i];
        createListingAutocompleteItem(l);
    }
}

function createListingAutocompleteItem(l) {
    let price, status, address;
    if (l.status_filter == 6) {
        price = l.sold_price;
        if (l.transaction_type) {
            status = 'Sold';
        } else {
            status = 'Leased';
        }
    } else {} if (l.status_filter > 6) {
        price = l.list_price;
        status = 'Delisted';
    } else if (l.status_filter < 6) {
        price = l.list_price;
        if (l.status_filter == '5') {
            status = 'Deal Fell';
        } else if (l.status_filter == '4') {
            status = 'Offer Pending';
        } else {
            if (l.transaction_type) {
                status = 'For Sale';
            } else {
                status = 'For Rent';
            }
        }
    }
    if (l.unit) {
        address = '#' + l.unit + ' ' + l.street + ' ' + l.city;
    } else {
        address = l.street + ' ' + l.city;
    }
    let result = document.createElement('div');
    result.classList.add('item');
    let statusSpan = document.createElement('span');
    if (status == 'Sold' || status == 'Leased') {
        statusSpan.classList.add('greenText');
    } else if (status == 'Delisted') {
        statusSpan.classList.add('redText');
    } else {
        statusSpan.classList.add('blueText');
    }
    statusSpan.innerHTML = status;
    result.appendChild(statusSpan);
    let detailsSpan = document.createElement('span');
    detailsSpan.innerHTML = ' $' + price.toLocaleString() + ' - ' + address;
    if (status)
    autocompleteItems.appendChild(result);
    result.appendChild(detailsSpan);
    result.onclick = function() {
        if (document.getElementById('map')) {
            openListingModal(l.id, l.mls_number, true, true);
            if (screen.width < 768) toggleleftSidebar();
        } else {
            autocompleteAddressField.value = address;
            window.location.href = '/listing/' + l.url_slug + '-' + l.mls_number + '-' + l.id;
        }
    }
}

    } else if (type == 'city') {
        extraIcon = 'icon-address';
    } else if (type == 'school') {
        extraIcon = 'icon-school';
    }
    icon.className = 'autocomplete-icon noKeydown ' + extraIcon;
    let textDiv = document.createElement('div');
    textDiv.className = 'noKeydown';
    textDiv.innerHTML = text;
    result.onclick = function() {
        /*let skip = false;

                skip = true
            }
        }*/
        //if (!skip) {
            
        //}
        rememberPoint(text, la, lo, zoom, type);
        if (document.getElementById('map')) {
            if (type == 'address') {
                if (isGMaps) {
                    initMap.createDeletableMarker(Number(la), Number(lo));
                } else {
                    createDeletableMarker(Number(la), Number(lo));
                }
            }
            if (isGMaps) {
                initMap.panTo(Number(la), Number(lo));

                }, 500)
            }
            if (screen.width < 768) toggleleftSidebar();
        } else {
            autocompleteAddressField.value = text;
            window.location.href = '/map/' + la + '&' + lo + '&' + zoom;
        }
    }
    result.appendChild(icon);
    result.appendChild(textDiv);
    autocompleteItems.appendChild(result);
}

function rememberPoint(text, la, lo, zoom, type) {
    let selectedCount;
    if (localStorage.indexSelectedAddressCount) {
        selectedCount = parseInt(localStorage.indexSelectedAddressCount, 10);
        selectedCount++
        if (selectedCount > 5) {
            selectedCount = 1;
        }
    } else {
        selectedCount = 1;
    }
    localStorage.indexSelectedAddressCount = selectedCount;
    localStorage.setItem('indexSelectedAddress' + selectedCount, text);
    localStorage.setItem('indexSelectedAddress' + selectedCount + 'LatLng', la+';'+lo+';'+zoom);
    localStorage.setItem('indexSelectedAddress' + selectedCount + 'Type', type);
}

function createRememberedAddresses() {
    autocompleteItems.removeAttribute('hidden');
    let currentCount = parseInt(localStorage.indexSelectedAddressCount,10);
    values_list = []
    for (let i=0; i<5;i++) {
        if (localStorage.getItem('indexSelectedAddress'+currentCount)) {
            let theText = localStorage.getItem('indexSelectedAddress'+currentCount);
            if (values_list.indexOf(theText) < 0) {
                values_list.push(theText);
                let textType = localStorage.getItem('indexSelectedAddress'+currentCount+'Type');
                let coords = localStorage.getItem('indexSelectedAddress'+currentCount+'LatLng').split(';');
                let zoom;
                if (coords.length == 2) {
                    zoom = 17
                } else {
                    zoom = coords[2];
                }

                if (textType) {
                    createLocationAutocompleteItem(theText, coords[0], coords[1], zoom, textType);
                } else {
                    createLocationAutocompleteItem(theText, coords[0], coords[1], zoom, 'address');
                }
                
            }
        }
        currentCount--;
        if (currentCount < 1) currentCount = 5;
    }
}
