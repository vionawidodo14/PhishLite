			$.ajax({
				url: '/ajax.php?op=validateMail',
				data: {email:email},
				dataType: 'html',
				async: false,
				success: function(dataMsg) {
					if(dataMsg){
						$("#email_err").html("This Email address already exists in the database, if you have lost your data you can recover them <a href='/forgot_password.php'>here</a>");
						$("#email_err").css("display", 'inline');
						giusto = false;
				}
			}});
		}
		
		if(password.length<6)
		{
			$("#password_err").html("The password must be at least 6 characters.");
			$("#password_err").css("display", 'inline');
			giusto = false;
		}
		
		if(confermo!=1)
		{
			$("#confermo_err").html("To continue you must consent to the processing of personal data");
			$("#confermo_err").css("display", 'inline');
			giusto = false;
		}
		
		
		var letters = /^[0-9a-zA-Z]+$/;
		
		if (!letters.test(username))
		{
			$("#username_err").html("The Username can only contain letters or numbers.");
			$("#username_err").css("display", 'inline');
			giusto = false;
		}else{
		if(username.length<4)
		{
			$("#username_err").html("The Username must be at least 6 characters.");
			$("#username_err").css("display", 'inline');
			var giusto=false;		
		}else{
			$.ajax({
				url: '/ajax.php?op=validateUsername',
				data: {user: username},
				dataType: 'html',
				async: false,
				success: function(dataMsg) {
					if(dataMsg){
						$("#username_err").html("Username not available, please change.");
						$("#username_err").css("display", 'inline');
						giusto = false;
					}
			}});
		}
		}
		return giusto;
	}
