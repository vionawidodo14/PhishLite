function cartSave(success) {
  //Iterate through cart items and save them to cookie
  cartGet(function (data) {
    if (data && data.items) {
      var saveData = [];
      for (var i=0; i<data.items.length; i++) {
        var item = data.items[i];
        console.log('saving item', item);
        //Compress data for cookie
        saveData.push({
          'v': item['variant_id'], //Variant
          'q': item['quantity'],   //Quantity
        })
      }
      setCookie(cartRestoreCookieName, JSON.stringify(saveData), 1800000, '/');
      if (success) {
        success();
      }
    }
  });
}

function cartRestore(restored) {
  //Find cart-restore cookie
  var cartRestoreData = getCookie(cartRestoreCookieName);

    //Clear cookie data data before beginning
    //setCookie(cartRestoreCookieName,'',-100000, '/', 'sipsby.com');

        var cartItems = JSON.parse(cartRestoreData);
        console.log('cartItems', cartItems)
      }
      catch(err) {
        console.error('unable to restore cart',err);
        return false;

      $(document.body).append(

      );

      //Start adding items
      var itemCount = 0;
      addNextItem = function() {
        if (cartItems.length == 0) {
          //Hide "Updating cart" div
          $('#sipsby-cart-restore').hide();
          
          $('.sb-cart-item-count').text(itemCount);
          $('.sb-cart-item-count-parens').text('('+itemCount+')');

          //Done. Only redirect to cart if there were actual items restored
          if (restored && itemCount > 0) {
            restored();
          }
        }
        else {
          //Pop an item
          var item = cartItems.shift();
          itemCount += item['q'];
          console.log('  addItems', item);
          try {
            cartAdd({
              'id': item['v'],
              'quantity': item['q']
            }, addNextItem);
          }
          catch (err) {
            console.error('unable to re-add item')
          }
        }

      }

      //Add first item
      addNextItem();
      
    });

    return true; //Working on restoring
  }
  return false; //Not restoring
}
