        var player;    
        var idContainer = 'player-0';
        var currentItem = 0;

        var timeout;
        var owl = $('.main-slider').owlCarousel({
            autoplay: false,
            autoplayTimeout: 5000,
            nav: true,
            dots: true,        
            navText: [],
            mouseDrag: true,
            touchDrag: true,
            items : 1,
            loop: false,    
            singleItem:true,                    
            dotsContainer: '#carousel-custom-dots.video_dots'                
        }); 

        owl.on('changed.owl.carousel', function(event) {

            var currentItem = event.item.index;
            var countItem = event.item.count;           
            var idContainer = $('.main-slider .owl-item').find('#player-' +currentItem);
            var idVideo =  $('.main-slider .owl-item').find('#player-' +currentItem).data("id"); 
        
            if (undefined !== idVideo && idVideo.length){ 
             
                $('.video_dots .owl-dot.video_play:not(.active) .owl-bgcolor').removeAttr('style');

                $('.main-slider .owl-item').each(function () {
                 
                    if($(this).find('.video div#player-'+currentItem).length > 0){                       
                                 
                        onYouTubePlayerAPIReady('player-'+currentItem, idVideo);

                    }else if($(this).find('.video iframe#player-' +currentItem).length > 0){  
                       
                        $('iframe#player-' +currentItem, this)[0].contentWindow.postMessage('{"event":"command","func":"playVideo","args":""}', '*');
                     
                    }else if($(this).find('.video iframe:not(#player-' +currentItem+')').length > 0){  
                     
                        $('iframe:not(#player-' +currentItem+')', this)[0].contentWindow.postMessage('{"event":"command","func":"pauseVideo","args":""}', '*');

                    }
                });                
                     
            }else{               
                $('.video_dots .owl-dot.video_play:not(.active) .owl-bgcolor').removeAttr('style');

                $('.main-slider .owl-item').each(function () {
                    if($(this).find('.video iframe').length > 0){
                        $('iframe', this)[0].contentWindow.postMessage('{"event":"command","func":"pauseVideo","args":""}', '*');
                    }
                });
            }        
           
            
            $('.main-slider .owl-item iframe').removeClass('stoped');
            
            CheckActive(currentItem, countItem);
            
        });      

       
        function CheckActive(currentItem, countItem) {   
            clearTimeout(timeout);              
            if(!(currentItem == (countItem - 1))){                
                if(!($('.main-slider .owl-item').eq(currentItem).find('iframe').length) > 0){ 
                    timeout = setTimeout(function(){               
                        $('.main-slider').trigger('to.owl.carousel', [(currentItem + 1), 300]); 
                    }, 5000); 
                }
            }else if(currentItem == (countItem - 1)){           
                if(!($('.main-slider .owl-item').eq(currentItem).find('iframe').length) > 0){
                    timeout = setTimeout(function(){ 
                        $('.main-slider').trigger('to.owl.carousel', [0, 300]);                 
                    }, 5000); 
                }
            }                   
        }    

        $('.main-slider .owl-next').click(function () {                
            var currentItem = $(".main-slider .owl-item.active");
            var currentIndex = $(".main-slider .owl-item").index(currentItem) + 1;
            
            $('.main-slider .owl-item iframe').removeClass('stoped'); 

            if(currentIndex == 4){            
                if($(this).hasClass('disabled')){                 
                    $(this).removeClass('disabled');
                }else{                   
                    $('.main-slider').trigger('to.owl.carousel', [0, 150]);
                    $('.owl-prev').removeClass('disabled');
                }
            }   
        });

        $('.main-slider .owl-prev').removeClass('disabled');
        
        $('.main-slider .owl-prev').click(function () {

            $('.main-slider .owl-item iframe').removeClass('stoped'); 

            if(currentIndex == 1){

        });

        $('.sound_video').click(function () {   
            var player = YT.get($(this).data("id"));            
            if(player.isMuted()){
                player.unMute();       
                $(this).addClass('active');
            }else{
                player.mute();
                $(this).removeClass('active');
            }                    
        });

        function onYouTubePlayerAPIReady(idContainer, idVideo) { 
            if(!idContainer){
                idContainer = 'player-0';
                idVideo = $('.main-slider .owl-item').find('#player-0').data("id");
            } 

            player = new YT.Player(idContainer, {
                playerVars: { 'autoplay': 1, 'modestbranding': 0,'iv_load_policy': 3, 'rel': 0, 'controls': 0,'autohide':1,'showinfo': 0, 'wmode':'opaque' },
                videoId: idVideo,
                events: {
                    'onReady' : initialize,
                    'onStateChange': onPlayerStateChange                    
                }        
            });  
        }    

        function onPlayerStateChange(event) {  
            if($('.main-slider .owl-item.active iframe').length > 0){
                var playerID = $('.main-slider .owl-item.active iframe').attr('id');
                var player = YT.get(playerID);
                if(event.data == 0){
                    player.playVideo();  
                }            
            }             

            var currentItem = $(".main-slider .owl-item.active");
            var currentIndex = $(".main-slider .owl-item").index(currentItem) + 1;

            if(event.data == 0  && 4 == currentIndex) { 
                $('.main-slider').trigger('to.owl.carousel', [0, 300]); 
                $('.video_dots .owl-dot.video_play:not(.active) .owl-bgcolor').removeAttr('style');                     
            }else if(event.data == 0 && 4 !== currentIndex){
                $('.main-slider').trigger('next.owl.carousel', [300]); 
                $('.video_dots .owl-dot.video_play:not(.active) .owl-bgcolor').removeAttr('style');
            }else if(event.data == 0){
                player.playVideo();                   
            }
        }

        function initialize(event){   
             if($('.main-slider .owl-item.active iframe').length > 0){
                var playerID = $('.main-slider .owl-item.active iframe').attr('id');
                var player = YT.get(playerID);

                player.playVideo(); 
            }     

            player.mute();  
           
            var time_update_interval;

            clearInterval(time_update_interval);            
         
            time_update_interval = setInterval(function () {
                updateTimerDisplay();               
            }, 1000)
          
        }
        
        function updateTimerDisplay(){            
            if($('.main-slider .owl-item.active iframe').length > 0){ 
                var playerID = $('.main-slider .owl-item.active iframe').attr('id');
                var player = YT.get(playerID);

                var videoPos = !player.getCurrentTime ? 0.0 : player.getCurrentTime(); 
                var speed = (videoPos * 100)/ player.getDuration();  
                
                $('.video_dots .owl-dot.active.video_play .owl-bgcolor').attr('style', 'width:' + speed + '%');  
             
            }
        }   
        
        $('.main-slider .touch-block').on("click", function () {
            if($('.main-slider .owl-item.active iframe').length > 0){                
                var playerID = $(this).parents('.owl-item').find('iframe').attr('id');
                var player = YT.get(playerID); 

                if($(this).parents('.owl-item').find('iframe').hasClass('stoped')){
                    $(this).parents('.owl-item').find('iframe').removeClass('stoped'); 
                    player.playVideo();                    
                }else{
                    $(this).parents('.owl-item').find('iframe').addClass('stoped');                
                    player.pauseVideo();
                } 
            }
        });


        CheckActive(currentItem);
