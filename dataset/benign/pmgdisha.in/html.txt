<script>
	var appNo;
	var typeOfApp;
	var ajaxUrl = "https://www.pmgdisha.in/wp-admin/admin-ajax.php";
	
	function checkApplicationStatus(){		
		var $ = jQuery;
		appNo = $("#applicationNo").val().trim();
		typeOfApp = "trackTpApp";		
		$('#review').hide();
		$('#reject').hide();
		$('#approve').hide();
		$('#invalid').hide();
		$("#hold").hide();
		$('#reasonDiv').hide();
		$("#under-review").hide();
		if(appNo){
			jQuery.ajax({
				type:"POST",
				url: ajaxUrl,
				data: {
					action: "my_action",
					appNo: appNo,
					type: typeOfApp
				},
				beforeSend: function(){
					$(".custom_loader").show();
				},
				success:function(data){
					//console.log(data);
					if(data){
						var obj = JSON.parse(data);
						var statusVal = obj.registrationStatus;
						var comments = obj.comments;

						if (statusVal == 'DUPLICATE_EMAIL' || statusVal == 'APPLICATION_UPDATED') {
							$("#under-review").show();
							if(comments){
								showReason(comments);
							}							  
						} else if (statusVal == 'APPROVED') {
							$('#approve').show();
						} else if (statusVal == 'REJECTED') {
							$('#reject').show();
							showReason(comments);
						} else if (statusVal == 'APPLICATION_HOLD'){
							$('#hold').show();
							showReason(comments);
						} else if (statusVal == 'UN_APPROVED'){
							$("#review").show();
							if (comments){
								$("#review").hide();
								$('#under-review').show();
								showReason(comments);	
							}					
						} else {
							$('#invalid').show();
						}
					} else {
						$('#invalid').show();
					}
				},
				complete: function(){
					$(".custom_loader").hide();
				},
				error: function(errorThrown){
					alert(errorThrown);
				}
			});			
		} else {
			$('#invalid').show();
		}
	}
	function showReason(comments){
		var $ = jQuery;
		$("#reasonDiv").find("b").html(comments);
		$("#reasonDiv").show();
	}
	
	jQuery("#statusTpModal").on("hidden.bs.modal", function(){
		jQuery('#review').hide();
		jQuery('#reject').hide();
		jQuery('#approve').hide();
		jQuery('#invalid').hide();
		jQuery("#hold").hide();
		jQuery('#reasonDiv').hide();
		jQuery("#under-review").hide();
		jQuery("#applicationNo").val("");
	});
	
	function _tracksocialAuditStatus(){
		var isValid = true;
		var emailVal = jQuery("#_emailForSocial").val().trim();
		typeOfApp = "socialAuditApp";
		
		if(!emailVal){
			jQuery(".error-email-social").show();
			isValid = false;
		} else {
			if(!(disha.validator.validateEmail(emailVal))){
				 jQuery(".error-email-social").show();
				 isValid = false;
			 } else {
				 jQuery(".error-email-social").hide();	 
			 }
		}
		
		if(isValid){
			jQuery.ajax({				 
			 url:	ajaxUrl,
			 type:	'POST',
			 data: {
				 action: "my_action",
				 email: emailVal,
				 type: typeOfApp
			 },
			 beforeSend: function(){
				 jQuery(".custom_loader").show();
			 },
			 success: function(data, success){
				var status;
				var latestComment;
				var obj;
				if(data){
					if(data == "application not found"){						 
						status = data;						
						jQuery(".appStatus").html("Status:" + " " + status).addClass("text-danger").removeClass("text-success");
						jQuery(".latestComment").html("");
					} else if(data == "Some error Occured"){
						jQuery(".appStatus").html("Some error Occured.").addClass("text-danger").removeClass("text-success");
						jQuery(".latestComment").html("");
					} else {
						obj = JSON.parse(data);
						status = obj.status;
						latestComment = obj.reason;						
						if(status == "UNAPPROVED"){
							jQuery(".latestComment").html("");	
						} else {
							jQuery(".latestComment").html("Reason:" + " " + Disha.Utils.HtmlEncode(latestComment)).addClass("text-success");
						}
						jQuery(".appStatus").html("Status:" + " " + status).removeClass("text-danger").addClass("text-success");
					}
				} else {
					alert("something went wrong");
					jQuery(".appStatus").html("");
					jQuery(".latestComment").html("");
				}				
			 },				 
			 error: function(){
				alert("something went wrong");
			 },
			 complete: function(){
				 jQuery(".custom_loader").hide();
			 }
		 });
		}		
		return isValid;
	}
	jQuery("#trackSocialModel").on("hidden.bs.modal", function(){
		jQuery("#_emailForSocial").val("");
		jQuery(".appStatus").html("");
		jQuery(".latestComment").html("");
	});
</script>
