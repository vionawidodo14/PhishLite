        };
     
     

        //  function to validate E-Newsletter form
        function modalSubmit(){
            localStorage.setItem('submitChk', 1); //changes to 1 after submit button pressed
            
            if (localStorage.getItem('submitChk') == 1 && localStorage.getItem('processedChk') != 1){
                //reset test var to 0 to avoid multiple submissions
                submitChk = 0;
                
                // check to see if @ symbol is present in emails
                if (emailaddress.value.indexOf("@") == -1 || emailaddress.value.indexOf(' ') != -1){
                    modalInvalidEmail.classList.remove('hidden');
                } 
                // check for letters in first and last name (no blank spaces)
                else if(!letters.test(firstname.value.charAt(0)) || !letters.test(firstname.value.slice(-1))){
                     modalInvalidFirstName.classList.remove('hidden');
                }
                else if(!letters.test(lastname.value.charAt(0)) || !letters.test(lastname.value.slice(-1))){
                     modalInvalidLastName.classList.remove('hidden');  
                }
                else if(modalBranchSelector.value == '0'){
                     modalInvalidBranchSelector.classList.remove('hidden');  
                }
                // all seems well so proceed to modalSubmit page
                else{
                    if (thankYou.classList.contains('hidden')){
                        thankYou.classList.remove('hidden');
                        modalTable.classList.add('hidden');
                        localStorage.setItem('processedChk', 1);  //changes to 1 when all validations are good and prior to processing
                        //call the function to submit if all validations pass
                        modalSubmitted(firstname.value,lastname.value,emailaddress.value);
                        modalButton.classList.add('hidden');
                    }
                    else{
                        thankYou.classList.add('hidden');
                        modalTable.classList.remove('hidden');
                    };
                }
            }
            else{
                //do nothing, form alredy submitted
            }
        };
     
     
        //submits the form upon clicking the submit button
        function modalSubmitted(a,b,c){  
           $.post("/temp/nl/goRobly.cfm",{firstname: a, lastname: b, emailaddress: c},
                function(result){
                    //Do Nothing for now
                });
        };
        
     
        // function to close modal, and unblur body
        function modalConfirmSub(){
            var newLoc = window.location.href = '/about/become_a_member.cfm'; //var in function scope to avoid a redirect loop
            
            if (rblyFrmSb == null){
                //if there is no sidebar
                localStorage.setItem('nlSignUpChk', 1); //local variable used to determine if member came from the nlSignUp, used on become_a_member.cfm
                modal.classList.add('hidden');
                modalBackground.classList.add('hidden');
                mainBody.classList.remove('blurredBody');
                rblyFrm.classList.add('hidden');
                nlFrmTxt.classList.add('hidden');
                invalidEmail.classList.add('hidden');
                modalButton.classList.remove('hidden');
                newLoc;
            }
            else{
                //if there is a sidebar
                localStorage.setItem('nlSignUpChk', 1); //local variable used to determine if member came from the nlSignUp, used on become_a_member.cfm
                modal.classList.add('hidden');
                modalBackground.classList.add('hidden');

            };
