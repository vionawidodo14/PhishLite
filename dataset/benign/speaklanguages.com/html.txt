<script>function initialise_sticky_ads() {
    var sticky = document.getElementById('sticky_ad');

    if (sticky != null) {
        // Get sticky ad distance from top
        var rect = sticky.getBoundingClientRect();
        var initial_scroll_y = document.body.scrollTop + document.documentElement.scrollTop;
        var sticky_top = rect.top + initial_scroll_y;

        // Get sticky ad height
        var sticky_height = sticky.offsetHeight;

        // Get sticky ad bottom margin (in pixels as an integer)
        var sticky_style = sticky.currentStyle || window.getComputedStyle(sticky);
        var sticky_margin_bottom = parseInt(sticky_style.marginBottom, 10);

        // Calculate bottom of outer column
        var outer_col = document.getElementById('outer');
        var outer_col_bottom = outer_col.offsetTop + outer_col.offsetHeight;

        // Apply appropriate classes
        apply_sticky_classes();

        // Handle scroll
        window.onscroll = function() {
            apply_sticky_classes();
        };

        // Handle window resizing
        window.onresize = function() {
            // Window resized, so reinitialise code to handle sticky ad
            initialise_sticky_ads();
        };

        // Apply appropriate classes to sticky ad depending on scroll position
        function apply_sticky_classes() {
            // This logic necessary to support different JavaScript implementations
            var scroll_y = document.body.scrollTop + document.documentElement.scrollTop;
            if (scroll_y > outer_col_bottom - sticky_height - (2*sticky_margin_bottom)) {
                // Scrolling has hit bottom of column
                sticky.className = "ad sticky_end";
            } else if (scroll_y > sticky_top - sticky_margin_bottom) {
                // Scrolling has got to point where ad should stick
                sticky.className = "ad sticky";
            } else {
                // Scrolling not yet to point where ad should stick
                sticky.className = "ad";
            }
        }
    }
}</script>
