<script type="text/javascript">
    $(document).ready(function () {
        outputResize();
        $('#cityStateInput').on('keyup keydown', function (e) {
            if (e.type == "keyup") {
                var key = e.keyCode
                if (key != 40 && key != 38 && key != 9) {
                    if ($('.placesOutput').width() < 100) {
                        outputResize();
                    }
                    if ($('#cityStateInput').val().length > 2) {
                        var inputVal = $('#cityStateInput').val();
                        url = SiteSettings.SiteRoot + "location/getlocations/";
                        $.post(url, { input: inputVal },
                            function (result) {
                                $('.placesOutput').empty();
                                $.each(result, function (key, value) {
                                    $('.placesOutput').append('<div class=\"location-result\" style=\"width:100%;\" onclick=\"selectionToggle(this)\">' + value + '</div>');
                                });
                            }
                        );
                    }
                    else {
                        $('.placesOutput').empty();
                    }
                }
            }
            else if (e.type == "keydown") {
                var $listItems = $('.placesOutput').children();
                var key = e.keyCode,
                    $selected = $listItems.filter('.selected'),
                    $current;
                if (key != 40 && key != 38 && key != 9) {
                    return;
                }
                $listItems.removeClass('selected');
                if (key == 40) //down
                {
                    if (!$selected.length || $selected.is(':last-child')) {
                        $current = $listItems.eq(0);
                    }
                    else {
                        $current = $selected.next();
                    }
                }
                else if (key == 38) //up
                {
                    if (!$selected.length || $selected.is(':first-child')) {
                        $current = $listItems.last();
                    }
                    else {
                        $current = $selected.prev();
                    }
                }
                else if (key == 9) //tab
                {
                    $('.placesOutput').empty();
                }
                $('#cityStateInput').val($current.text());
                $current.addClass('selected');
            }
        });
        $(window).resize(function () {
            outputResize();
        });
        $(document).on('click', function (e) {
            if (!($(e.target).is("#placesOutput") || $(e.target).is("#cityStateInput") || $(e.target).is(".location-result"))) {
                $('.placesOutput').empty();
            }
        });
    });
    function outputResize()
    {
        $('.placesOutput').width($('#cityStateInput').width());
    }
    function selectionToggle(caller) {
        var x = $(caller).text();
        $('#cityStateInput').val(x);
        $('.placesOutput').empty();
    }

</script>
