        /**
         * Handles events on the email field
         */
        x.emailHandler = function(evt) {
            // Keypress handling
            if (evt.type == 'keypress') {
                if (evt.keyCode == 13) {
                    evt.stopPropagation();
                } else {
                    return;
                }
            }

            var $field = $('#login-email'),
                email = $field.val();

            // Focus handling
            if (evt.type == 'focus') {
                if (!email) {
                    $field.popover('show');
                }
                return;
            }

            // We only care when the text value changes
            if ($field.data('oldval') == email) {
                return;
            }
            $field.data('oldval', email);

            // Check if we have an invalid entry
            if (!x.validateEmailField(email)) {
                if (x.isBrandedLogin()) {
                    return
                }
                x.resetOrg();
                return;
            }

            // Indicate that the email looks good
            x.showFieldFeedback($field, x.SUCCESS);

            // Indicate that we're looking up the company
            if (x.validateEmail(email)) {
                x.detectOrg(email);
            } else {
                x.resetOrg();
            }
        };

         * Handles events on the email field

                password = field.val();

            // We only care when the text value changes
            if (field.data('oldval') == password) {
                return;
            }
            field.data('oldval', password);

            // Check if we have an invalid entry
            if (!x.validatePasswordField(password)) {
                return;
            }

            // Indicate that the email looks good
            x.showFieldFeedback(field, x.SUCCESS);
        };

        /**

        x.validateForm = function() {

            x.hideFormFeedback();

            var emailField = $('#login-form [name="login"]');
            if (!x.validateEmailField(emailField.val())) {
                x.showFieldFeedback(emailField, x.ERROR, "Please enter your email/username.");
                isValid = isValid && false;

            var passwordField = $('#login-form [name="password"]');
            if (!x.validatePasswordField(passwordField.val())) {
                x.showFieldFeedback(passwordField, x.ERROR, "Don't forget to enter your password.");
                isValid = isValid && false;
            }

            // Nothing has stopped us yet
            return isValid;

        /**
         * For form fields, wraps them up with some helpful inline messaging
         * @param Element element The textbox element to wrap with feedback
         * @param String type The type of feedback {x.SUCCESS, x.WARN, x.ERROR, x.INFO}
         * @param String message
         */
        x.showFieldFeedback = function(element, type, message) {
            // Create the new feedback box
            var cssClass,
                feedbackBox,
                formGroupElement,
                elementId = $(element).attr('id');

            feedbackBox = $('<div>', {
                'class': 'feedback textbox-feedback',
                'for': elementId
            });

            // Style the box
            switch (type) {
                case x.SUCCESS:
                    cssClass = 'success';
                    break;
                case x.INFO:
                    cssClass = 'info';
                    break;
                case x.WARN:
                    cssClass = 'warning';
                    break;
                case x.ERROR:
                    cssClass = 'error';
                    break;
                default:
                    break;
            }

            // Update & style the message
            feedbackBox.html(message);
            feedbackBox.addClass('bg-' + cssClass);

            // Style the form field
            formGroupElement = element.closest('.form-group');
            formGroupElement.removeClass('has-success has-info has-warning has-error');
            formGroupElement.addClass('has-' + cssClass);

            // Remove any previous messages
            element.next('div.textbox-feedback[for="' + elementId + '"]').remove();

            // Insert the feedbackBox (only if there's a message)
            if (message) {
                element.after(feedbackBox);
            }
        };

                        org.logopath,
                        org.iconpath,

        x.setOrg = function(

                $('header.site-header').css('border-bottom-color', '');
