function loadExtImages(host = false) {
	try {
		var allowedImgSources = host ? null : JSON.parse(localStorage.getItem("allowedImgSources") || "[]");
		var extImages = document.querySelectorAll("[extsrc]");
		for (var i = 0; i < extImages.length; i++) {
			var item = extImages[i];
			if (host) {
				if (item.getAttribute("extsrc").indexOf(host) > -1) {
					loadExtImage(item);
				}
			} else {
				for (var ii = 0; ii < allowedImgSources.length; ii++) {
					host = allowedImgSources[ii];
					if (item.getAttribute("extsrc").indexOf(host) > -1) {
						loadExtImage(item);
					}
				}
			}
		}
	} catch (e) {
		console.log('Failed to load external images', e);
	}
}

function loadIntImages() {
	try {
		var extImages = document.querySelectorAll("[extsrc]");
		for (var i = 0; i < extImages.length; i++) {
			var item = extImages[i];

				loadExtImage(item);
			}
		}
	} catch (e) {
		console.log('Failed to load internal images', e);
	}
}
