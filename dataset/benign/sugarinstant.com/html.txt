				setTimeout(wait, 5);

	function cancelEvent(e) {
		var ev = (e||window.event);
		if(ev) {
			ev.preventDefault();
			ev.stopPropagation();
		}
		return ev
	}

		// Go to physical checkout if there physical goods and only physical goods in the cart, otherwise show digital cart

		if(0 <= 0 || (store && store.data && store.data.cart.Items && store.data.cart.Items.length > 0)) {
