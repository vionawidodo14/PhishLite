        var videoSegmenter = {
            init: function (player, segments) {
                this.player = player;
                this.segments = segments;
                this.index = 0;
                this.languages = ["en"];
                this.waiting = false;
                this.sleeping = true;
                this.timing = true;

                this.player.addEventListener('timeupdate', this.wakeup.bind(this), false);
                window.setTimeout(this.switch.bind(this, 1), 1000.0 * this.segments[0].until);
            },
            wakeup: function () {
                this.sleeping = false;
                this.update();
            },
            timeout: function () {
                this.waiting = this.sleeping;
                this.update();
                this.waiting = false;
                this.sleeping = true;
            },
            switch: function (index) {
                if (this.player.paused) {
                    window.setTimeout(this.switch.bind(this, (index + 1) % this.segments.length), 5000);
                    this.found(index);
                    this.timing = true;
                }
                else this.timing = false;
            },
            update: function () {
                if (this.player.paused) return;
                for (var i = 0; i < this.segments.length; i++) {
                    // var start = i > 0 ? this.segments[i - 1].until : 0.0;
                    if (this.player.currentTime < this.segments[i].until) {
                        this.found(i);
                        return;
                    }
                }
                this.found(0);
            },
            found: function (index) {
                var next = (index + 1) % this.segments.length;
                if (this.index !== index) {
                    this.index = index;
                    this.segments[next].fetching = false;
                    this.render();
                }
                if (!this.waiting) {
                    this.waiting = true;
                    window.setTimeout(
                        this.timeout.bind(this),
                        1000.0 * (this.segments[index].until - this.player.currentTime)
                    );
                }
                if (!this.segments[next].fetching && !this.segments[next].hasOwnProperty('data'))
                {
                    this.segments[next].fetching = true;
                    window.setTimeout(
                        this.fetch.bind(this, next), 400
                    );
                }
            },
            fetch: function (index) {
                $.get('/templates/render-template/', {
                    template_code: this.segments[index].template,
                    language_code: this.languages[index % this.languages.length],
                    subdomain: this.segments[index].subdomain,
                }, this.ready.bind(this, index));
                $('.invoice-preload.video-wrapper')
                    .attr('class', 'invoice-preload video-wrapper ' + this.segments[index].template);
            },
            ready: function (index, data) {
                var segment = this.segments[index];
                segment.data = data;
                segment.fetching = false;
                segment.invoice = $(data.html);
                $('#intro-invoice-preload').empty().append(segment.invoice);
                if (window.Typekit && !window.WebFont) window.WebFont = {};
                $('head').append($('<style>').text(data.css)).append(data.fonts);
            },
            render: function () {
                var segment = this.segments[this.index];
                if (segment.hasOwnProperty('invoice'))
                    $('#intro-invoice').empty().append(segment.invoice);
                $('.intro.intro-top.intro-invoice')
                    .attr('class', 'intro intro-top intro-invoice ' + segment.template);
            }
        };

        videoSegmenter.init($('#player')[0], videoSegments);
