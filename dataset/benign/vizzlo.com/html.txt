    try{
        var currentUserName = null;
        document.addEventListener("DOMContentLoaded", function () {
            (function ($) {
                if (currentUserName) {
                    var ratingsDisabled = false;
                    var lastVizzard = null;
                    var ratedVizzard = [];
                    var lastChangedVizzardKey = currentUserName+ '.LAST_CHANGED_VIZZARD';
                    var ratedVizzardKey = currentUserName + '.RATED_VIZZARDS';
                    var disabledKey = currentUserName + '.RATINGS_DISABLED';

                    try {
                        ratingsDisabled = JSON.parse(window.localStorage.getItem(disabledKey)) === true;
                    } catch (e) {}

                    try {
                        lastVizzard = JSON.parse(window.localStorage.getItem(lastChangedVizzardKey));
                    } catch (e) {

                    }
                    try {
                        ratedVizzard = JSON.parse(window.localStorage.getItem(ratedVizzardKey)) || [];
                    } catch (e) {

                    }

                    var disableDialog = function () {
                        try {
                            window.localStorage.removeItem(lastChangedVizzardKey);
                            window.localStorage.setItem(disabledKey, true);
                        } catch (e) {}
                        $('#ratingDialog').removeClass('show');
                    };

                    var ignoreDialog = function () {
                        try {
                            window.localStorage.removeItem(lastChangedVizzardKey);
                        } catch (e) {}
                        $('#ratingDialog').removeClass('show');
                    };

                    window.addEventListener('beforeunload', ignoreDialog);

                    if (lastVizzard && ratedVizzard.indexOf(lastVizzard.id) === -1 && !ratingsDisabled) {
                        var ratingForm = document.forms.ratingForm;
                        var closeTimeout = setTimeout(ignoreDialog, 5000);
                        $('#ratingDialog').bind('mouseover', function () {
                            closeTimeout && clearTimeout(closeTimeout);
                        });
                        $(ratingForm).bind('submit', function (e) {
                            e.preventDefault();
                            try {
                                ratedVizzard.push(lastVizzard.id);
                                window.localStorage.setItem(ratedVizzardKey, JSON.stringify(ratedVizzard));
                                window.localStorage.removeItem(lastChangedVizzardKey);
                            } catch (e) {

                            }
                            fetch("/api/v1/vizzards/" + lastVizzard.id + "/reviews", {
                                method: 'POST',
                                include: 'credentials',
                                body: JSON.stringify({
                                    rating: parseInt(ratingForm.elements.rating.value) || 0,
                                    feedback: ratingForm.elements.feedback.value
                                })
                            });
                            $('#ratingContent').removeClass('step-1', 'step-3').addClass('step-2');
                            setTimeout(function () {
                                $('#ratingDialog').removeClass('show');
                            }, 2000);
                        });
                        $('#ratingDialog input[type="radio"]').bind('change', function(){
                            $('#ratingDialog button[type="submit"]').removeAttr("disabled");
                        });
                        $('#ratingDialog *[data-dismiss="modal"]').click(disableDialog);
                        $('.rating-label-vizzard').html(lastVizzard.title);
                        $('#ratingDialog').addClass('show');
                        $('.rating > label').bind('click mouseover', function () {
                            $('#ratingLabel').html($(this).attr('title'));
                        });
                        $('.rating').bind('mouseleave', function () {
                            var value = ratingForm.elements.rating.value;
                            $('#ratingLabel').html(value ? $("label[for='star" + value + "']").attr('title') : "Select a rating");
                        });
                    }
                }
            })(jQuery);
        });

    function debounce(func, wait, immediate) {
        var timeout;
        return function() {
            var context = this, args = arguments;
            var later = function() {
                timeout = null;
                if (!immediate) func.apply(context, args);
            };
            var callNow = immediate && !timeout;
            clearTimeout(timeout);
            timeout = setTimeout(later, wait);
            if (callNow) func.apply(context, args);
        };
    }

    var substringMatcher = function (strs) {
        return function findMatches(q, cb) {
            var matches, substrRegex;

            
            matches = [];

            
            substrRegex = new RegExp(q, 'i');

            
            
            $.each(strs, function (i, str) {
                if (substrRegex.test(str)) {
                    matches.push(str);
                }
            });

            cb(matches);
        };
    };
