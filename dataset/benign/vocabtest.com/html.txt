var mostMissedWait = 5000;

function showNextMostMissed() {
	clearTimeout(mostMissedTO);
	
	var tmpID = (mostMissedID + 1) % mostMissedArr.length;
	mostMissedShow(tmpID, false);
	
	mostMissedTO = setTimeout('showNextMostMissed()', mostMissedWait);
}
function mostMissedShow(id, stopTimeout) {
	// stop looping if we need to
	if (stopTimeout)
		clearTimeout(mostMissedTO);
	
	// ignore same selection
	if (id == mostMissedID)
		return;
		
	// update buttons
	document.getElementById('mostMissedButton' + mostMissedID).className = 'inactive';
	document.getElementById('mostMissedButton' + id).className = 'active';
	// move items
	var tmpLeft = id * -100;
	document.getElementById('mostMissedContent').style.left = tmpLeft + '%';
	
	// set new id
	mostMissedID = id;
}

mostMissedTO = setTimeout('showNextMostMissed()', mostMissedWait);
