    heroCarousel = function() {
        var el = document.getElementById('homeHeroCarousel');
        var navId = 'homeHeroNav';
        //id for navigation circles
        var swiping = new Hammer(el);
        var items = $('#homeHeroCarousel .c-hero-carousel__item').length;
        //get how many items are in the carousel
        var count = 0;
        // set the active item number to 0
        var heroTimer;
        var heroTimerPaused = false;
        var navHTML = '<div class="c-hero-carousel__nav" id="' + navId + '">';
        for (var i = 0; i < items; i++) {
            var itemNum = i + 1;
            navHTML += '<a href="#homeHero' + itemNum + '"' + (i < 1 ? 'class="is-active">' : '>') + '<span class="is-vishidden">Carousel Item ' + itemNum + '</span></a>'
        }
        navHTML += '</div>';
        $(el).parent().append(navHTML);
        $(el).find('.c-hero-carousel__item').each(function(idx) {
            $(this).attr('data-carousel-item', idx);
        })
        $(el).addClass('is-init');
        heroTimer = window.setInterval(function() {
            if (!heroTimerPaused) {
                count < items - 1 ? count++ : count = 0;
                changeItem(count, "next");
            }
        }, 6000);
        $(el).on('mouseover', '.c-button', function() {
            heroTimerPaused = true;
        });
        $(el).on('mouseout', '.c-button', function() {
                heroTimerPaused = false;
            })
            /*$(window).on('resize', function(e) {
waitForFinalEvent(function() {
changeItem(count, count, "specific");
}, 100, 'homeCarouselResize');
});*/
        $('#homeHeroNav a').each(function(idx) {
            //select each item inidividually so we can use the index of each item for assigning click handlers
            $(this).on('click', function(e) {
                e.preventDefault();
                changeItem(idx, "specific");
                count = idx;
                clearInterval(heroTimer);
            });
        });

        function changeItem(newer, dir) {
            if (!(newer <= items - 1)) {
                // if the count is greater than the number of items, we need to reset
                newer = 0;
            }
            if (newer < 0) {
                // if the count is less than zero, we need to go to the other end of the carousel items
                newer = items - 1;
            }
            count = newer;
            if (dir === "next") {
                // scroll
                $(el).animate({
                        'scrollLeft': $(el).scrollLeft() + $(el).outerWidth()
                    },
                    500, //scroll the carousel to the left using the itemWidth as how many pixels to scroll it
                    function() {
                        // move first item
                        $(el).find('.c-hero-carousel__item:first').appendTo(el);
                        //find the first item and move it to the end of the items
                        // set scroll left to 0
                        $(el).scrollLeft(0);
                    }
                );
            } else if (dir === "prev") {
                //move last item
                $(el).find('.c-hero-carousel__item:last').prependTo(el);
                //find the last item and move it to the beginning of the items
                //set scroll left to outerwidth
                $(el).scrollLeft($(el).outerWidth());
                //scroll
                $(el).animate({
                    'scrollLeft': 0
                }, 500);
                //scroll the carousel to the left using the itemWidth as how many pixels to scroll it
            } else {
                var pos = $('[data-carousel-item=' + newer + ']').position().left + $(el).scrollLeft();
                console.log(pos);
                $(el).animate({
                        'scrollLeft': pos
                    },
                    500,
                    function() {
                        $($('[data-carousel-item=' + newer + ']').prevAll().get().reverse()).appendTo(el);
                        $(el).scrollLeft(0);
                    }
                );
                //scroll the carousel to the left using the itemWidth as how many pixels to scroll it
            }
            $('#homeHeroNav a').removeClass('is-active').eq(newer).addClass('is-active');
            // add the active class to the nav coresponding to the active item
            $(el).find('use').each(function() {
                if (this.href && this.href.baseVal) {
                    this.href.baseVal = this.href.baseVal;
                    // trigger fixing of href
                }
            });
        }
        swiping.on('swipe', function(e) {
            clearInterval(heroTimer);
            if (e.direction === 2) {
                //if swiped left (direction 2), then add 1 to the count
                count++;
                changeItem(count, "next");
            } else {
                // if swiped right, remove one from the count
                count--;
                changeItem(count, "prev");
            }
        });
    }
    heroCarousel();
</script>
