    function authenticate(email, password, rememberMe,loginAction,loginData) {
        if ($(email).val() == "" || $(password).val() == "") {
            $('#formLogin').submit();
            return;
        }
        var urlToAuthenticate = 'https://property24ph.property24.com.ph/api/Authentication/Authenticate';
        urlToAuthenticate += '?username=' + encodeURIComponent($(email).val());
        urlToAuthenticate += '&password=' + encodeURIComponent($(password).val());
        urlToAuthenticate += '&rememberMe=' + encodeURIComponent($(rememberMe).val());
        if(loginAction!=undefined && loginAction.val()!='')
        {
            urlToAuthenticate += '&postLoginAction=' + encodeURIComponent($(loginAction).val());

        }
        if(loginData!=undefined && loginData.val()!='')
        {
            urlToAuthenticate += '&postLoginData=' + encodeURIComponent($(loginData).val());

        }
        $.ajax({
            url: urlToAuthenticate,
            type: 'GET',
            dataType: 'jsonp',
            success: function (data) {
                if (data != undefined) {
                    if (data == "Invalid" || data == "NotFound") {
                        $("#formLogin span.errorText").show();

                    } else {
                        document.cookie = data;
                        location.reload();
                    }
                }
            },

                alert('error : ' + errorThrown);

        });
    }
