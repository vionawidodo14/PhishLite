<script>
    $(function() {
        function handleToggle(e) {
            var div = $('.browse-talent-container');
            var h6 = div.find('h6');
            var ul = div.find('ul');

            if ($(window).width() <= 479) {
                div.addClass('toggle-control');

                if(!ul.hasClass('collapse')) {
                    ul.addClass('expand').height(0);
                }

                if(!h6.hasClass('collapse')) {
                    h6.addClass('expand');
                }

                h6.off('click').on('click', function(e) {
                    var target = $(e.target);

                    if(target.is('h6')) {
                        var childUl = target.parent().find('ul');

                        switch(target.hasClass('expand')) {
                            case true:
                                var height = 0;
                                target.parent().find('li').each(function(index, el) {
                                    height += $(el).outerHeight();
                                });

                                target.removeClass('expand')
                                    .addClass('collapse');
                                childUl.removeClass('expand')
                                    .addClass('collapse');
                                childUl.animate({ height: height });

                                break;

                            case false:
                                target.removeClass('collapse')
                                    .addClass('expand');
                                childUl.removeClass('collapse')
                                    .addClass('expand');
                                childUl.animate({ height: 0 });
                                break;
                        }
                    }
                });
            } else {
                div.removeClass('toggle-control');
                ul.removeClass('expand').removeClass('collapse').height('auto');
                h6.removeClass('expand').removeClass('collapse');
            }
        }

        setTimeout(handleToggle, 500);
        $(window).on('resize', handleToggle);
    });
</script>
