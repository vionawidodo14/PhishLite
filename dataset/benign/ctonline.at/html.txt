    $(function () {
        $('#nav-read').click(function (e) {
            try {
                e.preventDefault();
                $.removeCookie('readFilter', {path: '/'});
                $.removeCookie('readTopicId', {path: '/'});
                $.removeCookie('readScroll', {path: '/'});
                window.location.href = $(this).attr('href');
            } catch (e) {
                console.log(e);
            }
        });
        $('#nav-learn').click(function (e) {
            try {
                e.preventDefault();
                $.removeCookie('learnFilter', {path: '/'});
                $.removeCookie('learnTopicId', {path: '/'});
                $.removeCookie('learnScroll', {path: '/'});
                window.location.href = $(this).attr('href');
            } catch (e) {
                console.log(e);
            }
        });
        $('#nav-exam').click(function () {
            $.removeCookie('examTestmodelIds', {path: '/'});
        });
        $('#nav-result').click(function () {
            $.removeCookie('resultTrainingId', {path: '/'});
            $.removeCookie('resultScroll', {path: '/'});
        });

        // bugfix, second click always follows link. we want to prevent that
        $('li.dropdown-submenu > a').each(function () {
            var $this = $(this);
            $this.attr('href', 'javascript:void(0);');
        });
        $('li.dropdown-submenu > a').on('click', function (e) {
            e.stopPropagation();
            e.preventDefault();

            var $this = $(this),
                isOpen = $this.hasClass('open');

            $('li.dropdown-submenu').find('ul').hide();
            $('li.dropdown-submenu > a').removeClass('open');

            if (!isOpen) {
                $this.parent().find('ul').show();
                $this.blur();
                $this.addClass('open');
            }
            return false;
        });
    });
</script>
