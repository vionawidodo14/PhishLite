function startAutoComplete() {
	autoComplete(formatDatumFrontPage);
	setACSuccession('from','to');
	setACSuccession('to',openRoute);
	setACSuccession('overnight_start',openRecommendations);

function loadHeroImage() {
	var d=new Date; var img='hero'+(Math.floor(d.getMinutes()/20)+1);
	byId('fp_header_image').className=img;
	byId('fp_header_image').style.backgroundImage="url(/site_images/fp2/"+img+"_1000px.jpg)";

function openRoute() { 
	checkAutoCompleteValues('from'); checkAutoCompleteValues('to');
	var l      =lastAutoCompleteValues;
	var from   =l.from ? l.from.name : $('#from').typeahead('val');
	var to     =l.to   ? l.to.name   : $('#to'  ).typeahead('val');
	var fromLL =l.from ? (l.from.lat+','+l.from.lon) : '';
	var toLL   =l.to   ? (l.to.lat  +','+l.to.lon  ) : '';
	var explLL =l.from ? "lat="+l.from.lat+"&lon="+l.from.lon : '';
	var url;
	if (!from) { byId('from').style.border='2px solid red'; return; }
	if      (byId('rtd').checked) { url='/explore/'  +encodeURIComponent(from)+'?'+explLL; }	// pub/cafe ride
	else if (byId('rtr').checked) { url = '/map?from=%&fromLL=%&suggest=1'.ip(encodeURIComponent(from),fromLL); } // circular ride
	else if (byId('rth').checked) { url='/overnight/'+encodeURIComponent(from)+'?show=hotels&'+explLL; } // hotel ride
	else if (byId('rtc').checked) { url='/overnight/'+encodeURIComponent(from)+'?show=campsites&'+explLL; } // camping ride
	else {  // A-B
		if (!to) { byId('to').style.border='2px solid red'; return; }
		url = '/map?from=%&to=%&fromLL=%&toLL=%'.ip(encodeURIComponent(from),encodeURIComponent(to),fromLL,toLL); 
	}
	window.location.href = url;
}

function focusedFrom(event) {
	focusRouteInput(event);
}
function focusedTo(event) {
	byId('rtd').checked=byId('rtr').checked=byId('rth').checked=byId('rtc').checked=false;
	focusRouteInput(event);
}
function focusRouteInput(event) {
	if (screen.availHeight<700) { document.body.scrollTop += event.target.getBoundingClientRect().top-25; }
	byId('route_expander').style.display='block';
	byId('route_input_box').style.backgroundColor='rgba(47,145,91,1.0)';
	byId('fp_dimmer').style.display='block';
	byId('fp_headline1').style.opacity=0.5;
	byId('fp_headline2').style.opacity=0.5;
}
function unfocusRouteInput(event) {
	if (hasParentNode(event.target,byId('route_input_box'))) return;
	if (hasParentOfClass(event.target,'tt-suggestions')) return;
	byId('route_expander').style.display='none';
	byId('route_input_box').style.backgroundColor='rgba(47,145,91,0.8)';
	byId('fp_dimmer').style.display='none';
	byId('fp_headline1').style.opacity=1;
	byId('fp_headline2').style.opacity=1;
}
function loadRecommendations(a) {
	$.ajax({ url: '/explore/recommendations/'+homeCountry, dataType: 'json', success: function(data,textStatus,jqXHR) {
		var html='';
		for (var i=0; i<data.recommendations.length; i++) {
			html+=renderRecommendation(data.recommendations[i]);
		}
		byId('explore_recs').innerHTML = html;
	}});
	// make cachable
}
function renderRecommendation(a) {
	var link="/map?from=%,%&to=%,%&circular=yes&listing=%".ip(a.start_lat,a.start_lon,a.end_lat,a.end_lon,a.id);
	var html="<div><h3><a href='%'>%</a></h3>".ip(link,a.title);
	html+="<p class='starting'><span>%</span> start · Ride % there</p>".ip(a.start_town,commonReadableDistance(a.distance,a.region));
	html+="<div class='image'>"
	html+="<div class='lh'>%</div>".ip(a.subtype);
	if (a.local_price && a.local_price>0) {
		html+="<div class='rh'>Stay from %</div>".ip(formatPrice(a.local_price,a.currency,a.region));
	}
	html+="<a href='%'><img src='%' ></a>".ip(link, a.image);
	html+="</div><p>%, %.</p>".ip(a.name,a.end_town);
	html+="</div>"; return html;
}

	if (l.overnight_start) url+="&lat=%&lon=%".ip(l.overnight_start.lat,l.overnight_start.lon);
	window.location.href=url;
}
