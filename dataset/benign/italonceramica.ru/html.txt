				function totalericercaavanzata(filtri,fsuccess,args)
				{
					if(typeof args == "undefined") args = {}

					$.ajax({
						url : '/ajax/risultatiricerca.php',
						data: filtri,
						cache: false,
						dataType : 'html',
						beforeSend : function () {
							// $('#attesaricercaavanzata').css('visibility','visible');
						},
						success : function (html) {
							fsuccess(html,args);
						},
						error : function (richiesta,stato,errori) {
							console.log(errori);
						}
					});
				}
				function get_totali_ricercaavanzata($form)
				{
					on_start_get_totali_ricercaavanzata();

					var totselezionati = 0;
					var filtri = {};
					for (var index=0;index<elencoinputs.length;index++)
					{
						var name = elencoinputs[index];
						var $select = $form.find('select[name='+name+']');

						if($select.length)
						{
							var value = $select.val();
							filtri[name] = value;

							if(value) totselezionati++;
						}
					}
					var args = {
						totselezionati: totselezionati
					}
					// console.log(filtri);
					//calcola il nr di risultati
					totalericercaavanzata(filtri,function(html,args){

						// console.log(html,args);

						var almenounfiltro = args.totselezionati;

						//abbiamo calcolato i risultati, disabilitiamo tutti input che non portano risultati
						if(almenounfiltro) //almeno un filtro selezionato
						{
							var tot_richieste_da_eseguire = filtri.length;
							var richieste_eseguite = 0;
							var $gruppidifiltri = [];

							for (var index=0;index<elencoinputs.length;index++)
							{
								var name = elencoinputs[index];
								var $select = $form.find('select[name='+name+']');

								if($select.length)
								{
									$option = $select.find('option');
									$gruppidifiltri.push($option);
								}
							}

							//per ogni filtro disponibile, ciclo le sue opzioni
							for(var index in $gruppidifiltri)
							{
								$opzionifiltro=$gruppidifiltri[index];

								//per tutte le opzioni verifico se portano a risultati
								$opzionifiltro.each(function(){

									var name = $(this).closest('select').attr('name');
									var value = $(this).attr('value');

									//filtri attuale + questa opzione in particolare
									var filtri_con_questa_opzione = jQuery.extend({}, filtri); //copia di un oggetto
									filtri_con_questa_opzione[name]=value;

									var variabilidebug = {
										name: name,
										value: value,
										filtri_con_questa_opzione: filtri_con_questa_opzione
									}

									totalericercaavanzata(filtri_con_questa_opzione,function(html,args){
										// console.log(name,html);
										// console.log(args.$thischeckbox);
										if(parseInt(html)<=0)
										{
											// console.log('disable');
											args.$thischeckbox.attr("disabled", "disabled");
										}
										else
										{
											// console.log('enable');
											args.$thischeckbox.removeAttr("disabled");
										}

										// if(args.variabilidebug.name=="din51097") console.log(html);

										richieste_eseguite++;
										var tot_richieste_da_eseguire = args.tot_richieste_da_eseguire;
										// console.log(richieste_eseguite,"<",tot_richieste_da_eseguire);
										if(richieste_eseguite==tot_richieste_da_eseguire)
										{
											on_complete_get_totali_ricercaavanzata();
										}

									},{$thischeckbox: $(this), variabilidebug:variabilidebug, tot_richieste_da_eseguire:tot_richieste_da_eseguire});
									// console.log(filtri_con_questa_opzione);
								});
							}
						}
						else
						{
							//tutti gli input sono abilitati
							on_complete_get_totali_ricercaavanzata();
						}
					},args);
				}
			});
