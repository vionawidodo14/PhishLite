    function urlfriendly(text){
        var newtext = text.trim();
        
        newtext = text.replace(/-/g," ");
        newtext = newtext.replace(/_/g," ");
        
        newtext = newtext.replace(/[^A-Za-z0-9\s]/g,'');

        newtext = newtext.replace(/[\t\n]/g,'');
