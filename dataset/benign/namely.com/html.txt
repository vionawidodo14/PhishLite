        function switchNav(evt) {
            evt.preventDefault();

            if (evt.target.nodeName == 'A') {
                sliderLi.forEach(function(item, i, sliderLi) {
                    item.classList.remove('active');
                });
                evt.target.parentElement.classList.add('active');

                currentIndex = evt.target.getAttribute('data-index');
                switchImages(currentIndex);
                switchDescs(currentIndex);
            }
        }

        function switchImages(index) {
            sliderImages.forEach(function(item, i, sliderImages) {
                item.classList.remove('active');

                if (item.getAttribute('data-img') == index) {
                    item.classList.add('active');
                }
            });
        }

        function switchDescs(index) {
            sliderDescs.forEach(function(item, i, sliderDescs) {

                if (item.getAttribute('data-desc') == index) {
                    item.classList.add('active');
                }
