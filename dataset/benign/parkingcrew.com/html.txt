function sendFeedback() {
	var urlPara = (document.location.href),
			reportType = $('#reporttype').val(),
			comment = $('#comment').val(),
			subject = $('#subject').val(),
			name = $('#name').val(),
			email = $('#email').val(),
			copy2user = $('#copy2sender').is(':checked') ? 1 : 0,
			newClass = $("#newClass").val(),
			domain = $('#domain').val();

	if ((reportType == "bug" || reportType == "feedback" || reportType == "feature" || reportType == "inquiry") && comment != "") {
		$.ajax({
			type: 'POST',
			url: 'usercp.php',
			data: {
				'mode': 'bugreport',
				'urlPara': urlPara,
				'type': reportType,
				'comment': comment,
				'subject': subject,
				'name': name,
				'email': email,
				'copy2user': copy2user
			},
			success: function (responsetext) {
				if (responsetext == "failed") {
					$('#feedback_messages').html('<span style="color:#CC7070;">Error, could not send Report!</span>');
				} else {
					$('#subbutton').remove();
					$('#feedback_messages').html('<span style="color:#70CC70;">' + responsetext + '</span>');
				}
			},
			error: function () {
				alert('Sorry, an error occured.');
			}
		});
	} else if (reportType === "classification") {
		if (newClass == '') {
			betterAlert('Please enter a reclassification for this domain.');
		} else if (domain === '') {
			betterAlert('Please enter a domain.');
		} else {
			var domains = domain.split("\n");
			if (domains.length > 50) {
				betterAlert('You exceeded the maximum of 50 domains');
			} else {
				$.ajax({
					type: 'POST',
					url: 'usercp.php',
					data: {
						'mode': 'ajax_saveFeedbackDomainReclassification',
						'newClass': newClass,
						'domain': domains
					},
					dataType: 'json',
					success: function (response) {
						if (response.success == false) {
							$('#feedback_messages').html('<span style="color:#CC7070;">' + response.message + '</span>');
						} else {
							$('#subbutton').remove();
							$('#feedback_messages').html('<span style="color:#70CC70;">' + response.message + '</span>');
						}
					},
					error: function () {
						alert('Sorry, an error occured.');
					}
				});
			}
		}
	} else if (comment == "") {
		alert('Please enter a comment.');
	} else {
		alert('Please choose a report Type.');
	}
}

var Urly = (function () {
	function oUrly() {
	}

	oUrly.prototype = {
		url: '',

		getBaseUrl: function (url) {
			var baseurl = this.url.split('?')[0];//url without params
			return baseurl;
		},
		getParamsAsString: function (url) {
			var params = url.split('?')[1];//params of url
			return params;
		},
		getParamsAsArray: function () {
			var params = this.url.split('?')[1];// params
			var url2 = params.split('&');
			return url2;
		},
		getParamsAsObject: function () {
			var paramObj = {}
			var paramStr = this.url.split('?')[1];// params
			var paramAr = paramStr.split('&');
			var parts;
			for (var i = 0; i < paramAr.length; i++) {
				parts = paramAr[i].split('=');
				paramObj[parts[0]] = parts[1];
			}
			return paramObj;
		},
		addParams: function (parobj) {
			var url2 = this.getParamsAsArray();
			var newurl = this.url.split('?')[0];//baseurl
			newurl = this._joinParams(url2, newurl, parobj);
			return newurl;
		},
		_joinParams: function (url2, newurl, parobj) {
			var flaglist = {};
			var prop;
			for (prop in parobj) {
				flaglist[prop] = 0;
			}
			var found;
			var cntparams = 0;
			var parsep = '?';
			// check if existing parameter are among the new ones to be added
			for (var i = 0; i < url2.length; i++) {
				var found = false;
				for (prop in parobj) {
					var parts = url2[i].split('=');
					if (parts[0] == prop) {
						//parameter to be added already exists, so add and flag it
						found = true;
						flaglist[prop] = 1;
						if (cntparams > 0) parsep = '&';
						newurl = newurl.concat(parsep + prop) + '=' + parobj[prop];
						cntparams++;
					}
				}
				// if parameter is not a new one , then it already exists and has to be added
				if (!found) {
					if (cntparams > 0) parsep = '&';
					newurl = newurl.concat(parsep + url2[i]);
					cntparams++;
				}
			}
			//add all new parameters which have not been already added
			for (var prop in parobj) {
				if (flaglist[prop] == 0) {
					if (cntparams > 0) parsep = '&';
					newurl = newurl.concat(parsep + prop + '=' + parobj[prop]);
				}
			}
			return newurl;
		},
		removeParams: function (rmAr) {
			var url2 = this.getParamsAsArray();
			var newurl = this.url.split('?')[0];//baseurl
			var flaglist = {};
			for (var i = 0; i < url2.length; i++) {
				var paramName = this._getParamName(url2[i]);
				flaglist[paramName] = 0;
			}
			for (var i = 0; i < url2.length; i++) {
				paramName = this._getParamName(url2[i]);
				for (var j = 0; j < rmAr.length; j++) {
					if (paramName == rmAr[j]) {
						flaglist[paramName] = 1;  //to be removed
					}
				}
			}
			var sep = '?';
			countParams = 0;
			for (var i = 0; i < url2.length; i++) {
				var paramName = this._getParamName(url2[i]);
				if (flaglist[paramName] == 0) {
					if (countParams > 0) sep = '&';
					newurl = newurl.concat(sep + url2[i]);
					countParams++;
				}
			}
			return newurl;
		},
		_getParamName: function (paramAndValue) {
			var parts = paramAndValue.split('=');
			return parts[0];
		}
	};
	return new oUrly;
})();
