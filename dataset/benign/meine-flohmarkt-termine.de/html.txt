                $('.newsletter .terms-missing').fadeIn();
                hasErrors = true;
                e.preventDefault();
            }

            //validate email
            if($('.newsletter input[name=email]').val() == "") {
                $('.newsletter input[name=email]').fadeOut().fadeIn();
                $('.newsletter .email-missing').fadeIn();
                hasErrors = true;
                e.preventDefault();

            //validate plz selectbox

                $('.newsletter select[name="u_plz"]').fadeOut().fadeIn();
                $('.newsletter .plz-missing').fadeIn();
                hasErrors = true;
                e.preventDefault();
            }

            if(hasErrors) {
                $('.newsletter .alert-warning').fadeIn();
            }

        });
    });
