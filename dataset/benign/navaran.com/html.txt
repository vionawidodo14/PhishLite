    var globalToday = '1398/08/02';
    $(document).ready(function () {
        var dateTo;
        $('#frmRent #deliveryDate').datepicker({
            dateFormat: 'yy/mm/dd',
            isRTL: true,
            minDate: '1398/08/02',
            numberOfMonths: ($(window).width() > 640) ? 2 : 1,
            onSelect: function (selectedDate) {
                dateTo.datepicker('option', 'minDate', addDay(selectedDate));
            }
        });
        dateTo = $('#frmRent #returnDate').datepicker({
            dateFormat: 'yy/mm/dd',
            isRTL: true,
            minDate: '1398/08/02',
            numberOfMonths: ($(window).width() > 640) ? 2 : 1
        });
        $('#deliveryCity').html('');
        $('#returnCity').html('');
        for (var i = 0; i < cityLocationArray.length; i++) {
            $('#deliveryCity').append('<option value="' + cityLocationArray[i].id + '">' + cityLocationArray[i].name + '</option>');
            $('#returnCity').append('<option value="' + cityLocationArray[i].id + '">' + cityLocationArray[i].name + '</option>');
        }
        $('#deliveryCity').change(function () {
            setLocation(get('#deliveryCity'), 'd');
        });
        $('#returnCity').change(function () {
            setLocation(get('#returnCity'), 'l');
        });
        $('#deliveryCity').val(18);
        $('#returnCity').val(18);
        $('#deliveryCity').change();
        $('#returnCity').change();
        $('#frmRent #btnRent').click(function () {
            var deliveryDate = get('#deliveryDate');
            var deliveryLocation = parseInt(get('#deliveryLocation option:selected'));
            var returnLocation = parseInt(get('#returnLocation option:selected'));
            var onway = $('#check-dropoff').is(':checked');
            if (deliveryLocation == 0) {
                toastr.error('لطفا ایستگاه تحویل را انتخاب نمایید');
                return false;
            }
            if (onway == true && returnLocation == 0) {
                toastr.error('لطفا ایستگاه بازگشت را انتخاب نمایید');
                return false;
            }
            if (deliveryDate == globalToday) {
                var today = new Date();
                var h = parseInt(today.getHours());
                if(h<10){
                    h='0'+h;
                }
                var m = parseInt(today.getMinutes());
                if(m<10){
                    m='0'+m;
                }
                today=h+':'+m;
                var time = get('#deliveryTime');
                time = time.split(':');
                time[0] = parseInt(time[0]);
                if(time[0]<10){
                    time[0]='0'+time[0];
                }
                time[1] = parseInt(time[1]);
                if(time[1]<10){
                    time[1]='0'+time[1];
                }
                if(time<today) {
                    toastr.error('ساعت تحویل را بعد از ساعت فعلی انتخاب نمایید');
                    return false;
                }
            }
            var action = '';
            action += 'deliveryLocation=' + get('#frmRent #deliveryLocation option:selected');
            if ($('#check-dropoff').is(':checked')) {
                action += '&returnLocation=' + get('#frmRent #returnLocation option:selected');
            }
            else {
                action += '&returnLocation=' + get('#frmRent #deliveryLocation option:selected');
            }
            action += '&deliveryDate=' + get('#frmRent #deliveryDate').replace(/\//g, '-');
            action += '&deliveryTime=' + get('#frmRent #deliveryTime option:selected');
            action += '&returnDate=' + get('#frmRent #returnDate').replace(/\//g, '-');
            action += '&returnTime=' + get('#frmRent #returnTime option:selected');
            action += '&referer='+get('#frmRent #referer');
            window.location.href = '/rent?' + action;
        });
    });

    function setLocation(cityId, type) {
        var obj = $('#deliveryLocation');
        if (type == 'l') {
            obj = $('#returnLocation');
        }
        for (var i = 0; i < cityLocationArray.length; i++) {
            if (cityLocationArray[i].id == cityId) {
                var cursor = cityLocationArray[i];
                $(obj).html('');
                for (var j = 0; j < cursor['location'].length; j++) {
                    $(obj).append('<option value="' + cursor['location'][j].id + '">' + cursor['location'][j].name + '</option>');
                }
            }
        }
    }
