    //CONTROLLI FORM RICERCA CORSI HOME
    $j('#parola_chiave, #select_area').keypress(function(e) {
      if(e.which == 13) {
        e.preventDefault();
        invia_form_ricerca_corsi();
      }
    });

    $j('#invia_ricerca_corsi').click(function(e) {
      e.preventDefault();
      invia_form_ricerca_corsi();
    });

    function invia_form_ricerca_corsi() {
      //FILTRO PAROLA CHIAVE
      parola_chiave = $j('#parola_chiave').val();

      if ((parola_chiave.length > 0) && (parola_chiave.length < 3)) {
        alert('La parola chiave deve essere almeno di 3 caratteri');
      } else {
        $j('form#cerca_corsi_home').submit();
      }

    }
    //FINE CONTROLLI FORM RICERCA CORSI HOME


    //CONTROLLI FORM RICERCA LAVORO HOME
    $j('#ricerca_luogo, #ricerca_testo').keypress(function(e) {
      if(e.which == 13) {
        e.preventDefault();
        invia_form_ricerca_lavoro();
      }
    });

    $j('#invia_ricerca_lavoro').click(function(e) {
      e.preventDefault();
      invia_form_ricerca_lavoro();
    });


    function invia_form_ricerca_lavoro() {
      //FILTRO CHE LAVORO STAI CERCANDO
      ricerca_testo = $j('#ricerca_testo').val();

      //FILTRO LOCALITA'
      ricerca_luogo = $j('#ricerca_luogo option:selected').attr('ref');
      ricerca_luogo_sel = $j('#ricerca_luogo option:selected').val();

      if (ricerca_luogo != "") {
        ricerca_luogo = "&" + ricerca_luogo;
      }

      if ((ricerca_testo.length > 0) && (ricerca_testo.length < 3)) {
        alert('Il lavoro cercato deve essere almeno di 3 caratteri');
      } else {
        window.location.href = "/offerte/?testo=" + ricerca_testo + ricerca_luogo + "&luogo_sel=" + ricerca_luogo_sel + "&contratto=All&titolo=All";
      }

    }
    //FINE CONTROLLI FORM RICERCA LAVORO HOME

  });
