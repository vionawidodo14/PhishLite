<script type="text/javascript">
        $(function () {
            setInterval(function () { changePicture(); }, 2000);
        });

        var usePicture = 1;
        function changePicture() {
            var picture;
            if (usePicture == 1) {
                usePicture = 2;
                picture = ".picture1";
            } else if (usePicture == 2) {
                usePicture = 3;
                picture = ".picture2";
            } else {
                usePicture = 1;
                picture = ".picture3";
            }

            $(picture + ' img').removeClass('showY').removeClass('showX');
            showNewPicture(picture);
        }

        function showNewPicture(picture) {
            var randomNumber = Math.floor((Math.random() * 4) + 1);

            switch (randomNumber) {
                case 1:
                    $(picture + ' .imageYL').addClass('showY');
                    break;
                case 2:
                    $(picture + ' .imageYR').addClass('showY');
                    break;
                case 3:
                    $(picture + ' .imageXB').addClass('showX');
                    break;
                default:
                    $(picture + ' .imageXT').addClass('showX');
            }
        }

        function setLowerAndRemoveDots(val) {
            var lowerCase = val.value.toString().toLowerCase();
            val.value = lowerCase.replace(/å/g, 'a').replace(/ä/g, 'a').replace(/ö/g, 'o').replace(/ü/g, 'u').replace(/ÿ/g, 'y');
        }

        function SetActionUrl(){

            var dataValue = JSON.stringify({
                "userName": $("#username").val(), "actionTarget": document.getElementById('login-form').action
            });

            $.ajax({
                type: "POST",
                url: "NewLoginPage.aspx/GetActionIdpTarget",
                data: dataValue,
                async: false,
                contentType: 'application/json; charset=utf-8',
                dataType: 'json',
                success: function (data) {
                    document.getElementById('login-form').action = data.d;
                }
            });

            return true;
        }
    </script>
