        function ConvertAmount() {

            var link = '/Site/ExchangeCalculator';
            $('#ConvertCurrenciesResults').html("");
            $('#convertCurrenciesError').html("");

            var amount = $('#txtAmountValue').val();
            var fromCurrency = $('#ddlFromCurrency').val();
            var toCurrency = $('#ddlToCurrency').val();

            if (amount != "" && $.isNumeric(amount)){
                if (fromCurrency != "0" && toCurrency != "0") {
                    showLoader();
                    $.ajax({
                        url: link,
                        type: "post",
                        data: {
                            amount: amount,
                            from_currency: fromCurrency,
                            to_currency: toCurrency,
                            __RequestVerificationToken: $(':input[name="__RequestVerificationToken"]').val()
                        },
                        success: function (response) {
                            if (response.status == "ok") {

                                $('#ConvertCurrenciesResults').slideDown();
                            }
                            hideLoader();
                        },
                        error: function (jqXHR, textStatus, errorThrown) {
                            hideLoader();
                        }
                    });

                    $('#convertCurrenciesError').html("الرجاء اخيار عملات الصرف");
                }
            } else {
                $('#convertCurrenciesError').html("الرجاء ادخال مبلغ صحيح");
