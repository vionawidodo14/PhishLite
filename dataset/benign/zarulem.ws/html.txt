function winH(){
	return $(window).height();
}

function matchSize(el){
	el = $('.pagewrap ' + el);
	var wH = winH();
	el.height('auto');
	if ( wH > el.height() ){
		el.height( wH );
		return wH;
	} else {
		return el.height();
	}
}

function sizeLayer(){
	var menuH = matchSize('.menu-layer');
	var pW = $('.pagewrap');
	if( pW.hasClass('mm-open') ){
		pW.height(menuH);
	} else {
		pW.height('auto');
	}
}

$(window).load(function(){
	sizeLayer();
});

$(window).resize(function(){
	sizeLayer();
});
