  // Set TOC containers' height to be the same as that of the content, so that
  // TOC's `position: sticky` can work as user scrolls the whole page.
  //
  // Also apply a polyfill for older browsers.
  function setupTOC() {
    var tocContainer = document.getElementById('toc-container');

    if (!tocContainer) { return; }

    var content = document.getElementById('content');
    tocContainer.style.height = content.offsetHeight + 'px';

    if (Stickyfill.stickies.length) {
      Stickyfill.refreshAll();
    }
    else {
      var toc = document.getElementById('toc') || document.getElementById('lefttoc');
      Stickyfill.addOne(toc);
    }
  }

  document.addEventListener('DOMContentLoaded', function () {
    // LOCALNAV
    (function() {
      var currentUrl = window.location.href.split("#")[0];
      var localNavs = [].slice.call(document.querySelectorAll('ul.lnav a'));
      var currentLocalNav = localNavs.filter(function (nav) { return nav.href === currentUrl })[0];
      if (currentLocalNav) { currentLocalNav.classList.add('current'); }
    }());

    BLQ.closeUp('a.fb');

    // KEYBOARD SHORTCUTS
    var os = BLQ.getOS();
    if (os == "Mac" || os == "iOS") {
      [].slice.call(document.querySelectorAll('.ctrl')).forEach(function (node) { node.innerHTML = '⌘' });
    }

    // SVG PNG FALLBACK
    if (!Modernizr.svg) {
      [].slice.call(document.querySelectorAll('img[src$=".svg"]')).forEach(function (node) {
        node.setAttribute('src', node.getAttribute('src').replace('.svg', '.png'));
      });
    }

    // KONAMI
    (function() {
      var a = 0;
      var kkeys = '';
      var konami = '38,38,40,40,37,39,37,39,66,65';
      var nom = '78,79,77,13';
      var jedi = '74,69,68,73,13';
      window.addEventListener('keydown', function (e) {
        kkeys += ',' + e.keyCode;
        if(kkeys.indexOf(konami)>=0 && a<=1) {
          a += 1;
          // DO KONAMI THING
          if(a==1) {
            document.getElementById('page-wrapper').classList.add('comicsans');
            kkeys = '';
          }
          else {
            document.getElementById('page-wrapper').classList.add('starwars');
          }
        }
        if(kkeys.indexOf(nom)>=0) {
          window.location = '/dinner';
        }
        if(kkeys.indexOf(jedi)>=0) {
          document.getElementById('page-wrapper').classList.add('starwars');
        }
      });
    }());

    anchors.options = {
      icon: '#',
      placement: 'left'
    };

    anchors.add('h3:not(.no-anchor), h4:not(.no-anchor), h5:not(.no-anchor)');

    [].slice.call(document.querySelectorAll('img[alt="gif"]')).forEach(function (node) {
      BLQ.gifPlayer(node);
    });

    [].slice.call(document.querySelectorAll('img[alt="os"]')).forEach(function (node) {
      BLQ.getOSimg(node);
    });

    // clear the search field text when looses focus
    document.getElementById('s').addEventListener('blur', function (e) {
      e.target.value = '';
    });

    // Sticky TOC
    setupTOC();
    window.addEventListener('load', setupTOC); // redo after all images are loaded
  });
</script>
