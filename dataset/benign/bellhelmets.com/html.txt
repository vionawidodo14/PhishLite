	function _getCategories(list) {
		var categories = [];

		for(var i = 0; i < list.length; i++) {
			if(!inArray(list[i].category, categories)) {
				categories.push(list[i].category);
			}
		}

		return categories;

	function _getProducts(category, list) {
		var products = [];

		for(var i = 0; i < list.length; i++) {
			if(list[i].category == category && !inArray(list[i].product, products)) {
				products.push(list[i].product);
			}
		}

		return products;
	}

	function _getManuals(product, category, list) {

				list[i].product == product &&
				!inArray(list[i].name, manuals)
			){
				manuals.push(list[i]);
			}
		}

		return manuals;
	}

	function camelCase(str) {
		return str.replace(/(?:^\w|[A-Z]|\b\w|\s+)/g, function(match, index) {
			if (/\s+/.test(match)) return "";
			return index == 0 ? match.toLowerCase() : match.toUpperCase();
		});
	}
