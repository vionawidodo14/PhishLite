<script>

    $(document).ready(function () {
        handleCartCount();
    });

    $('.cart-toggle').click(function () {
        showCart();
    });

    function showCart() {
        $('.cart-box').fadeToggle(400);
        getCartBox('Fetching your cart. Please Wait...');
    }

    $(document).on("click", ".remove-cart-item", function () {
        removeCartItem($(this));
    });

    $(document).on("change", ".update-cart-item-qty", function () {
        let memberId = $('#site').data('member-id');
        let qty = $(this).val();
        let sku = $(this).data('sku');
        let full = $(this).data('full');

        updateCartItem(memberId, qty, sku, full);      
    });

    function removeCartItem(el) {
        let sku = $(el).data('sku');
        let id = $(el).data('id');
        let full = $(el).data('full');
        let membership = $(el).data('membership');
        let memberId = $("#site").data("member-id");

        let node = $(el).closest('.cart-item-title-grid');

        $('#cart-loading').show();
        $('#cart-loading-message').html('Updating your cart. Please Wait...');

        let url = "";

        if (full) {
            url = "/umbraco/Surface/StoreSurface/RemoveCartItemReturnView?id=" + id + "&full=" + full;
        }
        else {
            url = "/umbraco/Surface/StoreSurface/RemoveCartItem?id=" + id;
        }

        if (membership) {
            url = "/umbraco/Surface/StoreSurface/RemoveMembershipReturnView?memberid=" + memberId;
        }


        $.ajax({
            type: "POST",
            url: url,
            success: function (data) {

                if (membership) {
                    $('.cart-box').hide();

                    $.ajax({
                        type: "GET",
                        url: "/umbraco/Surface/StoreSurface/GetCartCount?memberid=" + memberId,
                        success: function (newCount) {
                            setCartCookie(newCount);
                            handleCartVisibility(newCount);

                            Cookies.set('mic', "false", { expires: 365 });

                            if (!full && newCount > 0) {
                                showCart();
                            }

                            if (full && newCount > 0) {
                                $('.cart-results').html(data);  // Full Cart page, return view.
                            }
                            else if (full && newCount == 0) {
                                window.location = "\store";
                            }
                        }
                    });
                }
                else {
                    if (Cookies.get('cart-count')) {
                        count = Cookies.get('cart-count');
                        count = (count - 1);
                        setCartCookie(count);
                        handleCartVisibility(count);
                        if (count == 0) { //redirect to store (no items)
                            window.location = "\store";
                        }
                    }

                    if (full && count > 0) {
                        $('.cart-results').html(data);  // Full Cart page, return view.
                    }
                    else if (full && count == 0) {
                        window.location = "\store";
                    }
                    else {
                        $(node).remove(); // cart box just remove node.                             
                    }

                    $('#cart-loading').hide();
                }

                gaSendProductRemoveCart($(el), 1);
            },
            error: function () {
                $('#cart-loading').hide();
            }
        });
    }

    function updateCartItem(memberId, qty, sku, full) {
        $('#cart-loading').show();
        $('#cart-loading-message').html('Updating your cart. Please Wait...');

        $.ajax({
            type: "POST",
            url: "/umbraco/Surface/StoreSurface/AddToCartReturnView?sku=" + sku + "&qty=" + qty + "&memberid=" + memberId + "&full=" + full,
            success: function (data) {
                $('.cart-results').html(data);
                $('#cart-loading').hide(); 
            },
            error: function () {
                $('#cart-loading').hide();
            }
        });
    }

    function getCartBox(message) {
        var memberId = $('#site').data('member-id');

        $('#cart-loading').fadeIn();
        $('#cart-loading-message').html(message);

        $.ajax({
            type: "GET",
            url: "/umbraco/Surface/StoreSurface/GetCartBox?memberid=" + memberId,
            success: function (data) {
                $('.cart-results').html(data);
                $('#cart-loading').hide();
            },
            error: function () {
                $('#cart-loading').hide();
            }
        }); 
    }

    function handleCartCount() {
        var memberId = $('#site').data('member-id');

        // Member is logged in
        if (memberId > 0) {
            var count = 0;

            // get cookie cart count
            if (Cookies.get('cart-count')) {
                count = Cookies.get('cart-count');
                handleCartVisibility(count);
            }
            else {
                getCartCountFromServer(memberId);
            }
        }
        else {
            handleCartVisibility(0); // hide cart icon
        }
    }

    function getCartCountFromServer(memberId) {
        $.ajax({
            type: "GET",
            url: "/umbraco/Surface/StoreSurface/GetCartCount?memberid=" + memberId,
            success: function (count) {
                if (count > 0) {
                    setCartCookie(count);
                }
                handleCartVisibility(count);
            }
        });      
    }

    function handleCartVisibility(count) {
        if (count > 0) {
            $('.cart-icon').show();
            $(".cart-count").html(count);
            $('.hamberder-icon').css('grid-column', '2/3');
        }
        else {
            $('.cart-icon').hide();
            $('.hamberder-icon').css('grid-column', '1/3');
        }
    }

    function setCartCookie(count) {
        Cookies.set('cart-count', count, { expires: 30 });
    }

</script>
