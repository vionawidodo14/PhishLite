						arrayOfNames = objectReference.split('.');
					} else {
						throw new Error('getValue can only accept a string for object reference');
					}
				}

				var theValue = window;
				for (var i = 0; i < arrayOfNames.length; ++i) {
					var objName = arrayOfNames[i];
					if (theValue.hasOwnProperty(objName)) {
						theValue = theValue[objName];
					} else {
						return notFoundFn(arrayOfNames, i);
					}
				}
				return theValue;
			};

					}

	(function() {
		var heroId = 'hero-banner-image';
		var heroPreviewId = 'hero-banner-image-preview';
		var heroPreview = document.getElementById(heroPreviewId);

		if (!heroPreview) return;

		var heroPreviewSrc = getElBackground(heroPreview);
		if (heroPreviewSrc) {
			var link = document.createElement('link');
			link.rel = 'preload';
			link.as = 'image';
			link.href = heroPreviewSrc;
			document.head.appendChild(link);
		}

		var hero = document.getElementById(heroId);
		if (!hero) return hideHeroPreview();

		var heroSrc = getElBackground(hero);
		if (!heroSrc) return hideHeroPreview();

		var img = document.createElement('img');

		img.onload = hideHeroPreview;
		img.onerror = hideHeroPreview;
		img.src = heroSrc;
		if (img.complete) hideHeroPreview();

		function getElBackground(el) {
			var src;
			try {
				src = window.getComputedStyle(el)
					.backgroundImage
					.slice(4)
					.replace(/['"()]/g, '');
			} catch (err) {
				hideHeroPreview();
				throw new Error('[HeroLoader] failed to extract bg img src: ' + el.outerHTML);
			}
			return src;
		}

		function hideHeroPreview() {
			setTimeout(function() {
				heroPreview.className = heroPreview.className.replace('is-active', '');
			}, 10);
			setTimeout(function() {
				if (heroPreview && heroPreview.parentNode) heroPreview.parentNode.removeChild(heroPreview);
			}, 600);
		}
	})();
</script>
