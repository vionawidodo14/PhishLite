function adpostop()
{
	var scrollTop=$(window).scrollTop();
	if(scrollTop<headerHeight)
	{
		$('.floatholder').css('top',headerHeight-scrollTop);
	}
	else
		$('.floatholder').css('top',2);

function adposbottom()
{
	var winHeight=$(window).height();
	var scrollTop=$(window).scrollTop();
	var headremain=headerHeight-scrollTop;

	//left banner to bottom
	var topLeft=winHeight-leftHeight;
	if(topLeft<2)
		topLeft=2;	
	if(headremain>0 && winHeight-headremain < leftHeight)
		topLeft=headremain;
	$('#floatleft').css('top',topLeft);

	//right banner to bottom
	var topRight=winHeight-rightHeight;
	if(topRight<2)
		topRight=2;

	if(headremain>0 && winHeight-headremain < rightHeight)
		topRight=headremain;
	$('#floatright').css('top',topRight);
}
function ad_pos()
{
	//hpos
	if(sticktoborder)
	{
		var winWidth=$(window).width();

		var space=Math.floor((winWidth-1001)/2);

		if(space<rightWidth)

		else
			$('#floatright').css('right',-space);
	}
	//vpos
	if(adfloattop)
		adpostop();
	else
