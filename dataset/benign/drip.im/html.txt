        function openMenu() {
            let menuList = document.getElementById("menuList").getElementsByTagName("li")[0];
            menuList.style.display = 'block';
            menuList.className = 'menu-item-1 menuIn';
            let item = menuList.getElementsByTagName("a")[0];
            let item2 = menuList.getElementsByTagName("span")[0];

                if (menuList.className == 'menu-item-1 menuIn') {
                    item2.style.display = 'inline-block';
                    item.style.width = '7rem';
                }
            }, false);
        }
        //关闭菜单
        function closeMenu() {

                }
