    function cart_checking(qty, max){
        qty = parseInt(qty);
        max = parseInt(max);

        var warn = "";
        if(qty > 0 && qty <= max)
        {
            warn = "";
        }
        else if(qty > 0 && qty > max)
        {
            warn = "Unavailable of Stock";
            warn = warn.replace("[QUANTITY]", max);
        }
        else
        {
            warn = "Incorrect Quantity";
        }

        return warn;
    }
