<script>
  (function (document, window) {
    var currentPage = 1;

    // DOM nodes
    var body = document.querySelector('.eugdpr-modal__body');
    var bodyTitle = body.querySelector('.eugdpr-modal__body-title');
    var bodyContents = body.querySelector('.eugdpr-modal__body-contents');
    var declinedTitle = body.querySelector('.declined-title');
    var declinedContents = body.querySelector('.declined-contents');
    var footer = document.querySelector('.eugdpr-modal__footer');
    var footerQuestion = footer.querySelector('.eugdpr-modal__footer-question');
    var footerButtonContainer = footer.querySelector('.eugdpr-modal__footer-button-container');
    var footerAcceptButton = footer.querySelector('.eugdpr-modal__footer-button');
    var footerDenyButton = footer.querySelector('.eugdpr-modal__footer-button--hollow');

    // Page 1 constants
    var REVIEW_PAGE_BODY_TITLE = bodyTitle.innerHTML;
    var REVIEW_PAGE_BODY_CONTENTS = bodyContents.innerHTML;
    var REVIEW_PAGE_FOOTER_QUESTION = 'Do you consent to our use of cookies and privacy policy?';
    var REVIEW_PAGE_FOOTER_ACCEPT_BUTTON = 'Accept';
    var REVIEW_PAGE_FOOTER_DENY_BUTTON = 'Deny';

    // Page 2 constants
    var DECLINED_PAGE_BODY_TITLE = declinedTitle.innerHTML;
    var DECLINED_PAGE_BODY_CONTENTS = declinedContents.innerHTML;
    var DECLINED_PAGE_FOOTER_QUESTION = 'Want to review our privacy policy again?';
    var DECLINED_PAGE_FOOTER_ACCEPT_BUTTON = 'Back to policy review';
    var DECLINED_PAGE_FOOTER_DENY_BUTTON = '';

    function startTransition() {
      body.classList.add('transition');
      footer.classList.add('transition');
    }

    function endTransition() {
      body.classList.remove('transition');
      footer.classList.remove('transition');
    }

    function setupReviewPage() {
      startTransition();

      // We are using a timeout here for the opacity transition
      setTimeout(function () {
        bodyTitle.innerHTML = REVIEW_PAGE_BODY_TITLE;
        bodyContents.innerHTML = REVIEW_PAGE_BODY_CONTENTS;
        footerQuestion.innerHTML = REVIEW_PAGE_FOOTER_QUESTION;
        footerAcceptButton.innerHTML = REVIEW_PAGE_FOOTER_ACCEPT_BUTTON;
        footerDenyButton.innerHTML = REVIEW_PAGE_FOOTER_DENY_BUTTON;
        footerButtonContainer.classList.remove('declined');

        endTransition();
        currentPage = 1;
      }, 500);
    }

    function setupDeclinedPage() {
      startTransition();

      // We are using a timeout here for the opacity transition
      setTimeout(function () {
        bodyTitle.innerHTML = DECLINED_PAGE_BODY_TITLE;
        bodyContents.innerHTML = DECLINED_PAGE_BODY_CONTENTS;
        footerQuestion.innerHTML = DECLINED_PAGE_FOOTER_QUESTION;
        footerAcceptButton.innerHTML = DECLINED_PAGE_FOOTER_ACCEPT_BUTTON;
        footerDenyButton.innerHTML = DECLINED_PAGE_FOOTER_DENY_BUTTON;
        footerButtonContainer.classList.add('declined');

        endTransition();
        currentPage = 2;
      }, 500);
    }

    footerAcceptButton.addEventListener('click', function (e) {
      // If someone was on the deny page and wants to review the terms again
      if (currentPage === 2) {
        setupReviewPage();
        return;
      }

      window.compliance.consentToEugdprRegulations(2);
    }, false);

    footerDenyButton.addEventListener('click', function (e) {
      // If current page is review page, go to page 2 (declined page)
      if (currentPage === 1) {
        setupDeclinedPage();
      }
    }, false);
  })(document, window);
</script>
