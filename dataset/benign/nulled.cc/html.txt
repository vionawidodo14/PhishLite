function EstFE_replaceText (callback) {

    if (document.selection) {
        // --- IE4+
        var s = document.selection.createRange(); 
        if (s.text) {
            eval('s.text = ' + callback + '(s.text);');
            s.select();
            return true;
        }
    } else if (typeof (obj.selectionStart) == 'number') {
        // --- Mozilla/Gecko
        var start = obj.selectionStart;
        var end = obj.selectionEnd;
        var __selection = obj.value.substr(start, end-start).toString();
        
        eval('var rs = ' + callback + '(obj.value.substr(start, end-start));');
        obj.value = obj.value.substr(0, start) + rs + obj.value.substr(end);
        
        if (rs.length > __selection.length) {
            end += rs.length - __selection.length;
        }
        
        obj.setSelectionRange(end, end);
        
        return true;
    }

    return false;
}

                var __dialog = '';
                switch (code) {
                    case 'font':
                        __dialog = 'Font family:';
                        break;
                        
                    case 'size':
                        __dialog = 'Font size:';
                        break;
                        
                    case 'color':
                        __dialog = 'Font color:';
                        break;
                }
                
                var __data_val = prompt (__dialog, '');
                
                if (__data_val == null) {
                    return false;
                }
                
                if (__data_val == '') {
                    replacement = '[' + code + '=]' + selected_text + '[/' + code + ']';
                } else {
                    replacement = '[' + code + '=' + __data_val + ']' + selected_text + '[/' + code + ']';
                }
                
                break;
                
            /**
             * Вывод диалога ввода ссылки медиа-объекта.
             */
            case 'media':
                replacement = '[media=][/media]';
                break;
                
            /**
             * Очистка контента от BB кодов.
             */
            case 'clear':
                if (!selected_text) {
                    return false;
                }
                
                replacement = selected_text.replace(/\[img\]([^\]\[]+|)\[\/img\]/ig, '');
                
                replacement = replacement.replace(/\[(media|size|font|color)=([a-zA-Z0-9]+|)\]([a-zA-Z0-9\s]+|)\[\/(media|size|font|color)\]/ig, '');
                replacement = replacement.replace(/\[(\/|)(media|size|font|color)(=|)([a-zA-Z0-9]+|)\]/ig, '');
                
                replacement = replacement.replace(/\[\/?(b|i|u|s|left|center|right|url|img|media|quote|spoil|hide|html|code|php|size|font|color)\]/ig, '');
                replacement = replacement.replace(/\[url=[^\]]+\]/ig, '');
                
                break;
            
            case 'fast':
                replacement = code;
                break;
                
            /**
             * Обычные вставки. Наличие выделенного текста - не обязательно. 
             */
            case 'regular':
            default:
                replacement = '[' + code + ']' + selected_text + '[/' + code + ']';
        }
        
        if (is_IE) {
            s.text = replacement;
            s.select();
        } else {
            var __start = obj.selectionStart,
                __end = obj.selectionEnd;
            obj.value = obj.value.substr(0, __start) + replacement + obj.value.substr(__end);

            obj.setSelectionRange(__start + replacement.length, __start + replacement.length);
        }
        
        return false;
    });
});
</script>

function scrollToTop ( ) {
