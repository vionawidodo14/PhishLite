function correctHeight(objId)
{
    var $obj = $(objId);
    if ($obj.length > 0) {
        if($obj.innerHeight() > window.innerHeight) {
            $obj.innerHeight(window.innerHeight - 55);
            $obj.css('overflow-y', 'scroll');
        } else {
            $obj.innerHeight('unset');
            $obj.css('overflow-y', 'unset');
        }
    }
}
