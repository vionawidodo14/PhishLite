				showCloseHandle: true,
				content: '/intro-video.php'
			});
			
			var w = 1020;
			$('.modal').css({ width: w, marginLeft: (w / 2) * -1 });
			$('.modal').css({ height: 620});
			$('.modal').find('.modal-header').hide();
			$('.modal').find('.modal-body').css({ 'max-height': 620});
			$('.modal').find('.modal-body').css({ height: 620});

		if ('') {
			$('.tabbox ul li a.').click();
		} else {
			$('.tabbox ul li a:first').click();
		}
		
		$('.tab button.btn').click(function() {
			$('#tab-form').submit();
		});

		$('.tab input[type=text]').each(function() {
			var w = $(this).width();

		$('.tabbox ul .tools').click(function() {
			window.location = '/tools';
		});
		
		$('.tab input').keydown(function(e) {
			if (e.keyCode == 13 || e.keyCode == 10) {
				$('#tab-form').submit();
			} else if (e.keyCode == 9) {
				if (e.shiftKey) {
					if ($('.tabbox ul li.active a').hasClass('domain')) {
						$('.tabbox ul li a.name').trigger('click');
					} else {
						$('.tabbox ul li.active').prev().children('a').trigger('click');
					}
				} else {
					if ($('.tabbox ul li.active a').hasClass('name')) {
						$('.tabbox ul li:first a').trigger('click');
					} else {
						$('.tabbox ul li.active').next().children('a').trigger('click');
					}
				}
				return false;
			}
		});
		
		$(document).keyup(function(e) {
			if (e.altKey) {
				switch (e.keyCode) {
					case 49:  // 1
					case 112: // F1
						$('.tabbox ul li a.domain').click();
						return false;
						
					case 50:  // 2
					case 113: // F2
						$('.tabbox ul li a.email').click();
						return false;
						
					case 51:  // 3
					case 114: // F3
						$('.tabbox ul li a.name').click();
						return false;
				}
			}
		});

		var stat_posted = false;

		$('#tab-form').submit(function() {
			
			var param = $('.tabbox ul li.active a').attr('href').replace('#', '');
			
			var val = $('.tabbox .tab:visible input[type=text]').val();
			if (val.length == 0) return false;

			if (!stat_posted) {
				$.post('/ac', { auth: getAuth(), element: 'action_stat', action: 'hp_' + param, val: val });
				stat_posted = true;
			}
			
			switch (param) {
				case 'domain':
					if (val.indexOf(".") == -1) {
						show_tab_error("Please enter a valid domain name.");
					} else if (val.match(/\./g).length == 3) {
						window.location = '/ip?' + enc(val);
					} else {
						window.location = '/domain?' + enc(val);
					}
					break;
					
				case 'email':
					var re = /^(([^<>()[\]\\.,;:\s@\"]+(\.[^<>()[\]\\.,;:\s@\"]+)*)|(\".+\"))@((\[[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\.[0-9]{1,3}\])|(([a-zA-Z\-0-9]+\.)+[a-zA-Z]{2,}))$/;
					if (re.test(val) == false) {
						show_tab_error("Please enter a valid email address.");
					} else {
						window.location = '/email?' + enc(val, true);
					}
					break;
					
				case 'name':
					if (val.length < 4) {
						show_tab_error("Please enter a name.");
					} else {
						window.location = '/name?' + enc(val);
					}
					break;
					
				default:
					break;
			}
			
			return false;
			
		});

	});
