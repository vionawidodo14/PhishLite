		function loadShoppingCartItemsReceived(data) {
			var data = JSON.parse(data.d);
			var cartCount = 0;
			var cartContents = '';

			for (var key in data) {
				if (data.hasOwnProperty(key)) {
					cartContents += "<span class='cart-item'>" + data[key].Name + "</span>";
					cartCount += data[key].Quantity;
				}
			}
			jQuery('#rptCartItems').html(cartContents);

			if (cartCount > 0) {
				//jQuery('.shopping-cart').show();
				jQuery('.shopping-cart').addClass('has-items');
				jQuery('#aShoppingCart').html('<span id="spanShoppingCartCount" class="cart-counter">' + cartCount + '</span>').addClass('has-items');

				$('.cart-items').show();
