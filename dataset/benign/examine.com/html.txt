window.examine.ready(function() {
    $script.ready(
        ['algolia', 'lib-bundle'],
        function () {

            var ALGOLIA_INDEX = 'live_search'; // search index

            // algolia setup
            var algoliaClient = algoliasearch('GNDG2Z8078', '0bcefd73de9be8f77feec3dbe8c402a0');

            // set containers
            var $q = $('input.search-box.search-autocomplete');

            // initialize algolia indexes
            var suggestionTemplate = function() {
                var template = Handlebars.compile($('script#hit-template').html());

                return function(hit) {
                    return template(hit);
                };
            };

            var algoliaIndex = algoliaClient.initIndex(ALGOLIA_INDEX);
            var searchParams = {
                hitsPerPage: 10,
                attributesToRetrieve: '_id,page_ID,type,section,summary_quick,slug,visibility,objectTable,objectID,objectName',
                attributesToHighlight: 'objectName',
                distinct: true,
            };

            // custom typeahead adapter with delay
            var customTTAdapter = function(params) {
                var self = algoliaIndex;
                var debounceFunction = function(query, syncCb, asyncCb) {
                    var rootParent = this.$el.closest('.twitter-typeahead');
                    var cb;

                    if (typeof asyncCb === 'function') {
                        // typeahead 0.11
                        cb = asyncCb;
                    } else {
                        // pre typeahead 0.11
                        cb = syncCb;
                    }

                    // show the preloader
                    //rootParent.append('<span class="icon-spin4 animate-spin"></span>');
                    // use lodash debounce
                    return self.search(query, params, function(err, content) {
                        // hide the preloader
                        //rootParent.find('.animate-spin').remove();
                        if (query === 'testosterone') {
                            content.hits.unshift({
                                objectTable: 'data_stacks',
                                objectName: 'Increasing Testosterone',
                                slug: 'increasing-testosterone'
                            });
                        }
                        cb(content.hits);
                    });
                };

                return _.debounce(debounceFunction, 50);
              };

            var typeaheadParams = {
                //source: algoliaIndex.ttAdapter(searchParams),
                source: customTTAdapter(searchParams),
                displayKey: 'objectName',
                name: 'search',
                templates: {
                    suggestion: suggestionTemplate()
                }
            };

            // typeahead input
            $q.each(function () {
                var elem = $(this).typeahead(
                    {
                        hint: $(this).hasClass('search-hint'),
                        highlight: true,
                        minLength: 2
                    },
                    typeaheadParams
                );

                if (elem.hasClass('homepage-search')) {
                    elem.focus();
                }
            });

            // get url
            $q.on("typeahead:selected", function(e, data, dataset) {
                switch (data.objectTable) {
                    case 'data_nutrition':
                        if (data.type && data.type.toUpperCase() != 'DRAFT') {
                            var pageUrl = '/nutrition/' + encodeURIComponent(data.slug) + '/';
                        }
                        break;

                    case 'products':
                        if (data.visibility && data.visibility == 'visible') {
                            var pageUrl = '/' + encodeURIComponent(data.slug) + '/';
                        } else {
                            var pageUrl = '/products/';
                        }
                        break;
                    case 'data_stacks':
                        var pageUrl = '/stacks/' + encodeURIComponent(data.slug) + '/';
                        break;
                    case 'data_supplements':
                    default:
                        var pageUrl = '/' + (data.section == 'supplements' ? 'supplements' : 'topics') + '/' + encodeURIComponent(data.slug) + '/';
                }
                if (pageUrl) {
                    window.location.href = pageUrl;
                }
            });
    });
});
