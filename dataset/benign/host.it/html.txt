<script>
  var prodId = ""
  var isProductSelected = !!prodId
  var PRICE_THRESHOLD = 10

  
  function isPriceBelowThreshold(price) {
    return price <= PRICE_THRESHOLD
  }

  function priceToText(price) {
    if (!isProductSelected && price === 0) return "gratis"
    if (!isProductSelected) return localizePrice(price) + " €"
    
    if (isPriceBelowThreshold(price)) return "incluso"
    
    return "+ " + localizePrice(price - PRICE_THRESHOLD) + " €"
  }

  function localizePrice(price) {
    var formatNumberOptions = {
      maximumFractionDigits: 2
    }

    return price.toLocaleString('it-IT', formatNumberOptions)
  }

  function formatState(state) {
    if (!state.id) {
      return state.text;
    }

    var rawPrice = Number($(state.element).data('price'))
    var textPrice = priceToText(rawPrice)

    var includedClass = '';      
    if (textPrice == "incluso") includedClass="--incluso"

    return $(
      '<span class="mod-searchform-select__item"><span class="mod-searchform-select__item-label">' + state.text + '</span><span class="mod-searchform-select__item-price' + includedClass  + '">' + textPrice + '</span></span>'
    );
  }

  function memoizedFormatState(state) {
    var fn = formatState
    fn.cache = fn.cache || {}
    if (!state.id) return fn(state)
    return fn.cache[state.id] ? fn.cache[state.id] : (fn.cache[state.id] = fn(state))
  }

  function initSelect2($select) {
    $select.select2({
      placeholder: ".estensione",
      templateSelection: formatState,
      templateResult: memoizedFormatState,
      selectOnClose: true
    });

    
    $select.select2('open');
    $select.select2('close');
    
    $select.val(null).trigger('change')
    resetFocus()
  }

  function resetFocus() {
    var allFocusableElements = document.querySelectorAll('button, [href]:not(link), input, select, textarea, [tabindex]:not([tabindex="-1"])');

    // Find first visible element
    for (var i = 0; i < allFocusableElements.length; i++) {
      var currentEl = allFocusableElements[i]
      if (currentEl.offsetParent !== null) {
        currentEl.focus()
        return
      }
    }
  }

  
  function debounce(func, wait, immediate) {
    var timeout;
    return function() {
      var context = this, args = arguments;
      var later = function() {
        timeout = null;
        if (!immediate) func.apply(context, args);
      };
      var callNow = immediate && !timeout;
      clearTimeout(timeout);
      timeout = setTimeout(later, wait);
      if (callNow) func.apply(context, args);
    };
};

      }
      $select.trigger('change.select2');

<p class="com-typo">
									Il <strong>Dominio</strong> è il punto di partenza della tua presenza online e costituisce il vero e proprio “nome” sul web per il tuo progetto o la tua azienda. La registrazione di un dominio ha in genere un costo annuale molto contenuto che tuttavia può variare in base all’estensione scelta (esempio .it, .com, .net, .org, ecc.). Host è un Registrar accreditato ICANN e consente di registrare domini con oltre 300 estensioni. Puoi registrare un dominio per proteggere il tuo marchio, il nome della tua azienda o una tua idea. La registrazione di un dominio include la gestione delle zone DNS e la gestione dei reindirizzamenti.
								</p>
<p class="com-typo">
									Un <strong>Hosting</strong> è invece una soluzione completa che include, oltre alla registrazione di un nome a dominio, anche un certo quantitativo di spazio web e di caselle di posta elettronica. Hai bisogno di un hosting se intendi costruire un sito web utilizzando un CMS (come Joomla, WordPress, PrestaShop, ecc.) oppure la tua applicazione web personalizzata in HTML, PHP e MySQL.
