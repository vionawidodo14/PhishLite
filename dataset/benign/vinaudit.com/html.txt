function onVinInputFocus(focus) {
  var input = document.getElementById('va_header_vin_input');
  var vin = input.value.toUpperCase();
  var isValid = isValidVin(vin);
  input.style.backgroundColor = isValid ? '#E4FFD8' : (focus ? '#FFEBD8' : '#FFFFFF');
  input.style.borderColor = focus ? '#3E5260' : '#6D90A8';
}

function onVinInput() {
  var input = document.getElementById('va_header_vin_input');
  var locateBox = document.getElementById('va_header_vincheck_locate_box');
  if (!locateBox.origInnerHTML) {
     locateBox.origInnerHTML = locateBox.innerHTML;
  }
  var vin = input.value.toUpperCase();
  vin = vin.replace(/ /g, '');
  vin = vin.replace('I', '1');
  vin = vin.replace('O', '0');
  vin = vin.replace('Q', '0');
  if (input.value != vin) {
    input.value = vin;
  }
  var count = vin.length;
  var isValid = isValidVin(vin);
  if (count >= 17) {
     locateBox.innerHTML = isValid ? 'VALID VIN ('+count + '/17)' : 'INVALID VIN ('+count + '/17)';
  } else if (count > 0) {
     locateBox.innerHTML = '(' + count + '/17)';
   } else {
     locateBox.innerHTML = locateBox.origInnerHTML;
   }
  input.style.background = isValid ? '#E4FFD8' : '#FFEBD8';
}
