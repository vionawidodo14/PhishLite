function SpalteRechts( obj_Spalte_1, obj_Spalte_2 ) {
	// 
	
	// console.log( 'FN SpalteRechts : ' + obj_Spalte_1.offsetTop + ' : ' + obj_Spalte_2.offsetTop );// TEST
	
	var Return = false;
	var i_Diff_Max = 10;
	var i_Differenz = 0;
	var blSichtbar = true;
	
	i_Differenz = obj_Spalte_1.offsetTop - obj_Spalte_2.offsetTop;
	
	if ( i_Differenz < 0 ) {
		i_Differenz = i_Differenz * -1;
	}
	
	// console.log( 'FN SpalteRechts : ' + i_Differenz + ' : ' + i_Diff_Max );// TEST
	
	if ( typeof obj_Spalte_2.style.display != 'undefined' ) {
		if ( obj_Spalte_2.style.display == 'none' ) {
			blSichtbar = false;
		}
	}
	
	if ( i_Differenz < i_Diff_Max && blSichtbar == true ) {
		 Return = true;
