    var email =$("#newsletter_email").val();
    if(email!='')
    {
        if($("#agree_desk").prop('checked') == false)
        {
            $("#message_agree_desk").show(0).delay(4000).hide(0);
        }
        else
        {
            $.ajax({
            type: "POST",
            data: { newsletter_email: email},
            dataType:'json',
            async: false,
            cache: false,
            url: "https://internationalopenacademy.com/Newsletter/newsletterSignup",

                    $("#news_letter_sign_up_div").hide();
                    $('#newsletter-message-div_desk').html(data.msg);
                    $('#newsletter-message-div_desk').show();
                }
            },    
            });
        }
    }
});
