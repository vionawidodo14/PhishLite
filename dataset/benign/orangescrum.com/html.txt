function getTimeRemaining(endtime) {
  var t = Date.parse(endtime) - Date.parse(new Date());
  var seconds = Math.floor((t / 1000) % 60);
  var minutes = Math.floor((t / 1000 / 60) % 60);
  var hours = Math.floor((t / (1000 * 60 * 60)) % 24);
  var days = Math.floor(t / (1000 * 60 * 60 * 24));
  return {
    'total': t,
    'days': days,
    'hours': hours,
    'minutes': minutes,
    'seconds': seconds
  };
}

    function formatSplit(stringraw) {

        var out = '';
        for (var z = 0; z < stringraw.length; ++z) {
            var ch = stringraw.charAt(z);
            if (ch == '<') {
                while (ch != '>') {
                    out += ch;
                    ch = stringraw.charAt(++z);
                }
                out += ch;
                continue;
            }
            out += '<span>' + ch + '</span>';
        }

function validate_inpt(formid, id) {

				$("#"+formid+'_'+id).css({"border":"1px solid #FF0000"});

				$("#"+formid+'_'+id).css({"border":"1px solid #666"});
			}
		}
	}else {
		$("#"+formid+'_'+id).css({"border":"1px solid #FF0000"});
	}

	if(id == "seo_url") {
		var seo_url = $("#"+formid+'_'+id).val().trim();

		if(seo_url) {

				$("#"+formid+'_'+id).css({"border":"1px solid #FF0000"});
				$("#"+formid+'_'+id).html("Only letters and numbers are alloweded.");
			}

    function setSuptrackCookie(cname, cvalue, exdays) {
        var d = new Date();
        d.setTime(d.getTime() + (exdays * 24 * 60 * 60 * 1000));
        var expires = "expires=" + d.toUTCString();
        document.cookie = cname + "=" + cvalue + "; " + expires + "; path=/";
    }
    function getSuptrackCookie(cname) {
        var name = cname + "=";
        var ca = document.cookie.split(';');
        for (var i = 0; i < ca.length; i++) {
            var c = ca[i];
            while (c.charAt(0) == ' ')
                c = c.substring(1);
            if (c.indexOf(name) == 0)
                return c.substring(name.length, c.length);
        }
        return "";
    }
