        ],
        function (
            $,
            modal
        ) {
            $(document).ready(function () {
                $('#subscribe').removeAttr("disabled");
                //modal part
                var options = {
                    type: 'popup',
                    responsive: true,
                    innerScroll: true,
                    modalClass: 'custom-popup-subscribe'
                };
                var popup = modal(options, $('#custom-popup-subscribe'));

                var dataForm = $('#newsletter-validate-detail');
                dataForm.mage('validation', {});
                $(document).on('click', '#subscribe', function (e) {
                    e.preventDefault();
                    var status = dataForm.validation('isValid');
                    var email = $('#newsletter').val();
                    var url = dataForm.attr('action');
                    if (status) {
                        $.ajax({
                            showLoader: true,
                            url: url,
                            data: {email: email},
                            type: 'POST',
                            dataType: 'json'
                        }).done(function (data) {
                            if (data.status == 'OK') {
                                $('#error').hide();
                                $('#success').show();
                                $('#success .message').html('Your information completed the cube. </br>You are now subscribed.');
                                $('.modal-footer button span').html('Continue');
                            } else if (data.status == 'ERROR') {
                                $('#success').hide();
                                $('#error').show();
                                $('#error .message').html('Something is amiss. </br>Please re-enter your information.');
                                $('.modal-footer button').addClass('error');
                                $('.modal-footer button span').html('retry');
                            }

                            $('#custom-popup-subscribe').modal('openModal');
                            return true;
                        });
                    }
                });
