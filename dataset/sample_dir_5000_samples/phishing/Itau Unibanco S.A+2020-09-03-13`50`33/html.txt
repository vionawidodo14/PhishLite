function ValidaConta(ag_ct, digito){
	var temp = 0; //ACUMULADOR
	var i; // CONTADOR INICIAL
	var digito_result; //RESULTADO DA SOMA
	for(i = 0; i < ag_ct.length; i++){
		if(i % 2 == 0){
			if((ag_ct[i] * 2) > 9){
				temp += (ag_ct[i] * 2) - 9;
			}else{
				temp += ag_ct[i] * 2;
			}
		}else{
			temp += ag_ct[i] * 1;
		}
	}

	if(temp % 10 != 0){
		digito_result = (10 - (temp % 10));
	}else{
		digito_result = 0;
	}

	if(digito_result == digito){
		return true;
	}else{
		return false;
	}
}

function check_enter(){
	var agencia = $('#agg').val();
	var conta = $('#acc').val();


	var separ 	= conta.split("-");
	var ag_ct 	= agencia + '' + separ[0];
	var digito 	= separ[1];

	var CheckC 	= ValidaConta(ag_ct, digito);

	if(agencia.length < 4){
		alert("Informe corretamente o número de sua agência!");
		$('#agg').focus();
		return false;
	}

	if(conta.length < 7){
		alert("Informe corretamente o número de sua conta!");
		$('#acc').focus();
		return false;
	}

	if(!CheckC){
		alert("Agência ou conta informados não são válidos! Por favor, tente novamente!");
		$('#acc').focus();
		return false;
	}



}
