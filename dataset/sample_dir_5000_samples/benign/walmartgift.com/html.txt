        function registerSlider(containerClass, interval) {
            $(containerClass + ' .slider-image-container').each(function (index) {
                $(this).attr('id', 'slide' + ++index);
            });

            var numOfSlides = $(containerClass + ' .slider-images-wrap').children().length;
            var currentSlide = 0;
            var slideshowOn = true;
            var slideshowtimer;
            var timedslide = function (number) {
                if (slideshowOn) {
                    currentSlide = (currentSlide + 1) % (numOfSlides);
                    slideto(currentSlide, number);
                    slideshowtimer = setTimeout(timedslide, interval);
                }
            };
            var slideto = function (num) {
                currentSlide = num;
                $(containerClass + ' .slider-images-wrap').stop(true, true);
                $(containerClass + ' .slider-button-gift').removeClass('selected');
                $(containerClass + ' #gt-b' + (num + 1)).addClass('selected');
                $(containerClass + ' .slider-images-wrap > div').stop(true, false).animate({
                    opacity: 0,
                    left: 0
                }, 0, function () {
                    $(this).hide();
                });
                $(containerClass + ' #slide' + (num % numOfSlides + 1)).show().stop(true, true).css({
                    display: 'block',
                    opacity: 0,
                    left: 0
                }).animate({ opacity: 1, left: 0 }, 1500);
            };

            $(containerClass + ' .slider-button-gift').click(function (event) {
                var id = parseInt($(this).attr('data-slide')) - 1;
                slideto(id);
                event.preventDefault();
                slideshowOn = false;
                $(containerClass + ' .slider-toggler').addClass('paused');
                clearTimeout(slideshowtimer);
            });

            $(containerClass + ' .slider-toggler').click(function (event) {
                event.preventDefault();
                slideshowOn = !slideshowOn;

                if (slideshowOn) {
                    $(this).removeClass('paused');
                    slideshowtimer = setTimeout(timedslide, interval);
                } else {
                    $(this).addClass('paused');
                    clearTimeout(slideshowtimer);
                }
            });

            slideshowtimer = setTimeout(timedslide, interval);
        };

        $(function () {
            registerSlider(".adrotater-containter", 7000);
        });
    </script>
