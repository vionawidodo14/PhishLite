<script>
$(document).ready(function(){
	var html = $('#XF');
	
	
		// Advanced search menu
		$('.aqua-search-advanced').on('click', function(){ html.toggleClass('aqua-search-menu-active'); });
		$('.aqua-search-flex input').on('focus', function(){ html.addClass('aqua-search-menu-active'); });
		$(document).on('click', function() { html.removeClass('aqua-search-menu-active'); });
		$('.aqua-wrap-search').on('click', function(event){ event.stopPropagation(); });
	

	
		$("[data-macbb-editor]").hover(function() {
			html.addClass('aqua-picker--loaded');
		});
		// macbb editor: open and close with data-macbb-editor
		$("[data-macbb-editor]").on('click', function(event){
			html.toggleClass('aqua-editor-open').addClass('aqua-picker--loaded');
			event.preventDefault();
		});
		// ..and close by pressing ESC
		$(document).keyup(function(e) {
			if (e.keyCode === 27){
				html.removeClass('aqua-editor-open');
			}
		});
		
		$('.aqua-editor-panel').each(function(){
			// This value is used for cookie names and class names.
			var settingName = $(this).attr('data-setting');
			var settingDefault = $(this).attr('data-setting-default');
			var settingCookie = getCookie(settingName);
			if(settingCookie === 'on'){
				$(this).attr('data-setting-status','on');
				console.log(settingName + ': on (cookie)');
			} else if(settingCookie === 'off') {
				$(this).attr('data-setting-status','off');
				console.log(settingName + ': off (cookie)');
			} else {
				if(settingDefault === 'on'){
					$(this).attr('data-setting-status','on');
					console.log(settingName + ': on (cookie)');
				} else{
					$(this).attr('data-setting-status','off');
					console.log(settingName + ': off (cookie)');
				}
			}
		});
		// Change the class and cookie when the toggle is clicked
		$('.aqua-editor-panel').on('click', function(){
			var settingName = $(this).attr('data-setting');
			var settingStatus = $(this).attr('data-setting-status');
			if(settingStatus === 'on'){
				html.removeClass(settingName);
				setCookie(settingName, 'off', 365);
				$(this).attr('data-setting-status', 'off');
			} else {
				html.addClass(settingName);
				setCookie(settingName, 'on', 365);
				$(this).attr('data-setting-status', 'on');
			}
		});
	
});
