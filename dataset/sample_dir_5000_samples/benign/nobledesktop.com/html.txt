function scrollAbstractCodeLines(svgEl) {
  // select all the code line elements and convert the NodeList into an Array
  var codeLinesArr = Array.from(svgEl.querySelectorAll('rect')); // Array

  // take only the top half of the code lines (each code line is duplicated in the svg file so that it tiles)
  var codeLinesTopHalf = codeLinesArr.slice(0, codeLinesArr.length/2)

  // build nested array pairing corresponding elements from first and second halves:
  // [[0_a, 0_b], [1_a, 1_b], ... [N-1_a, N-2_b]]
  // where n_a.x === n_b.x, but n_b.y === n_a.y + svgEl.height/2
  var pairedCodeLines = codeLinesTopHalf.map(function(topEl) {
    var svgElHeight = Number(svgEl.getAttribute('height')),
             topElY = Number(topEl.getAttribute('y')),
             topElX = Number(topEl.getAttribute('x'))

    // grab the bottom element that has the same x value but whose y value is half the height of the svg more than the top element y value
    // elements that are at x=0 don't have an x attribute, so don't use x attribute selector
    if (topElX === 0) { var queryString = 'rect[y=\"' + (topElY + (svgElHeight / 2)) + '\"]:not([x])' }
    else { var queryString = 'rect[y=\"' + (topElY + (svgElHeight / 2)) + '\"][x=\"' + topElX + '\"]' }
    
    // select the corresponding bottom element
    var bottomEl = svgEl.querySelector(queryString);
    
    // return the element pair
    return [topEl, bottomEl];
  });

  // shuffle the nested array
  pairedCodeLines.sort(function () { return Math.random() - .5 });

  // animate: fade in first third of code lines, then scroll
  new TimelineMax({ repeat: 0 })
    .staggerTo(codeLinesArr.slice(0, codeLinesArr.length/3), .7, { opacity: 1, ease: Back.easeIn.config(2) }, .15) // fade in the first third of the lines (approx initially visible)
    .set(codeLinesArr, { opacity: 1 }) // set the rest to opacity: 1 instantly (these should not be visible when it happens)
    .staggerTo(pairedCodeLines, 2, { opacity: .2, ease: Back.easeInOut.config(2), repeat: -1, yoyo: true }, .1) // subtly pulsate codes lines
    // .to(svgEl, 60, { y: '-50%', ease: Power0.easeNone, repeat: -1 }, 0) // scroll linearly starting at the beginning
}
