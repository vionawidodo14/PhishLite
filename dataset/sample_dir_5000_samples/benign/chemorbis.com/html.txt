        , funcDoSearch: function (category) {

            var inputO = jQuery("#" + this.inputId);

                    var docs = new Array();

                        var resultUl = jQuery('#' + result_ul_id);
                        resultUl.show();
                        resultUl.append(thisO.funcResultItemCategoryHTML(COSearchCategory_Base[category].title
                                , searchText, numfoundSt * 1));
                        contentCat = thisO.funcResultItemContHTMLStart(COSearchCategory_Base[category].title
                                , searchText, numfoundSt * 1);
                        jQuery.each(docs, function () {
                            //Give the list element a rel with the data results ID incase we want to act on this later, like selecting from the list

                            if (COSearchCategory_Base[category].funcItemTitle)

                            if (COSearchCategory_Base[category].funcItemSubtitle)

                        var totalLink = null;

/**
 * data-img-hover kullanımı:
	 <p>
 <img id="id123" src="resim-1-URL" data-img-hover="yeni-resim-URL" />
 <i class="fa fa-play-circle" data-img-hover-continue="true" data-img-hover-id="id123"></i>
 	</p>
 * !!! img.id ile "data-img-hover-id" alanı aynı değeri taşımalı 	
 */
jQuery(document).ready(function(){
	var hoverTime=200;
	//
	jQuery("[data-img-hover]").each(function() {
		var jThis=jQuery(this);
		var hoverUrl=jThis.data("img-hover");
		var timerId=jThis[0].id;
		window[timerId]={timer:null,mouseOver:null,mouseOut:null};
		var hoverOutUrl=jThis.attr("src");		
		function mouseOver(){
        	// zamanlanmis kapama istegi varsa, onu durdurmak yeterli
	        if(window[timerId].timer!=null){
	        	clearTimeout(window[timerId].timer);
	        	window[timerId].timer=null;
	            return;
	        }
        	// eylem
	        jQuery("#"+timerId).attr("src", hoverUrl);
        }
		function mouseOut(sure){
        	// kapama istegini zamanla
        	window[timerId].timer=setTimeout(function(){
        		jQuery("#"+timerId).attr("src", hoverOutUrl);
        		window[timerId].timer=null;
        	},sure);
        }
		window[timerId].mouseOver=mouseOver;
		window[timerId].mouseOut=mouseOut;
		jQuery("#"+timerId).hover(
      					            function() {mouseOver();
               
            }
                    ,function() {
                mouseOut(hoverTime);
            }
        ); 
		
    });
	//										
	jQuery("[data-img-hover-continue]").each(function() {											
		var jThis=jQuery(this);
		var timerId=jThis.data("img-hover-id");
		jThis.hover(
      					            function() {window[timerId].mouseOver();
               
            }
                    ,function() {
                window[timerId].mouseOut(hoverTime);
            }
        ); 
	});
});
</script>

			    if(fireBaseToken!=currentToken){
				    removeTheCurrentBrowser();
			    }
			    
			    sendTokenToServer(currentToken);
			    
			  } else {
				    // Show permission request.
				    console.log('No Instance ID token available. Request permission to generate one.');//Need permission but may be canceled
				    // Show permission UI.
				    setTokenSentToServer(false);
				    showPoup=true;
				    startTimer();
			  }
			}).catch(function(err) {
			  console.log('An error occurred while retrieving token. ', err);//Block it
			  sendBrowserStatusToServer('An error occurred while retrieving token. '+ err.message,"BLOCK_BY_USER_OR_HAS_ERROR");
			  setTokenSentToServer(false);
			});
			
			// Callback fired if Instance ID token is updated.
			messaging.onTokenRefresh(function() {
			  messaging.getToken().then(function(refreshedToken) {
			    console.log('Token refreshed.');
			    // Indicate that the new Instance ID token has not yet been sent to the app server.
			    setTokenSentToServer(false);
			    
			    removeTheCurrentBrowser();
			    
			    // Send Instance ID token to app server.
			    sendTokenToServer(refreshedToken);
			    
			  }).catch(function(err) {
			    console.log('Unable to retrieve refreshed token ', err);
			    sendBrowserStatusToServer('Unable to retrieve refreshed token '+ err.message,"BLOCK_BY_USER_OR_HAS_ERROR");
			  });
			});
			
			messaging.onMessage(function(payload) {
				    console.log('Message received. ', payload);
				    // Update the notification bar to include the received message .
				    refreshNotificationBar();
				    showNotificationPopupWithDelay(payload);
			  });
	});

					sendBrowserStatusToServer('Unable to get permission to notify.'+ err.message,"DOES_NOT_GIVE_PERMISSION");
				});
	}
