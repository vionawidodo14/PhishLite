<script type="text/javascript">
var timerLength = 3500; //miliseconds
var maxRotations = 100; //limit to avoid endless loops
var imagePath = "/images/images-top/home-top/";


var imageFiles = {
    "findEvent" : [
	'Find_Event/1.jpg'
    ],

    "shopping" : [
	'Shopping/1.jpg'
    ],

    "connect" : [
	'Connect/1.jpg'
    ],

    "listEvent" : [
	'List_Event/1.jpg'
    ]
};

var cats = ['findEvent', 'shopping', 'connect', 'listEvent'];
var lastNum = [1,1,1,1];
var newNum = [-1,-1,-1,-1];
var rotateCount = 0;

function getRandomNums() {
    for(var i=0, l=cats.length; i<l; i++){
	var fileCount = imageFiles[cats[i]].length;
	do {
	    newNum[i] = Math.floor(Math.random() * fileCount);
	} while (newNum[i] == lastNum[i] && fileCount>1);

	var newEl = i<0 ? false : document.getElementById(cats[i]+"Image"+newNum[i]);
	if (!newEl) {
	    var lastEl = document.getElementById(cats[i]+"Image"+lastNum[i]);
	    newEl = lastEl.cloneNode(true);
	    newEl.id = cats[i]+'Image'+newNum[i];
	    newEl.style.display = 'none';
	    newEl.src = imagePath + imageFiles[cats[i]][newNum[i]];
	    if (imageFiles[cats[i]][newNum[i]] != '') lastEl.parentNode.appendChild(newEl);
	}
    }
}

function randomImages() {

    if (rotateCount >= maxRotations) {
	 clearInterval(homeShuffleInterval);
    } else {
	rotateCount++;
	for (var i=0, l=cats.length; i<l; i++) {
	    var lastEl = document.getElementById(cats[i]+"Image"+lastNum[i]);
	    var newEl = document.getElementById(cats[i]+"Image"+newNum[i]);
	    lastEl.style.display = 'none';
	    newEl.style.display = 'inline';
	    lastNum[i] = newNum[i];
	}

	getRandomNums();
    }
}

</script>
