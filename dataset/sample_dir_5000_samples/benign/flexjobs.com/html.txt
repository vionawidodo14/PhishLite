<script>
  jQuery.ajaxSetup({
    'beforeSend': function(xhr) {xhr.setRequestHeader("Accept", "text/javascript")}
  });

  $(document).ready(function(){
    var introOpen = false;
    var jobUpdatesPaused = false;
    var keywordTags = [];
    var locationTag = '';
    var tags = [];

    logAnalEvent('Shown Intro');
    checkLocalStorage();

    /* Functions */

    function addKeywordTag(tag) {
      tag = tag.trim();
      if (tag.length < 1) {
        return false;
      }

      var exists = false;
      jQuery.each(keywordTags, function (idx, eachTag) {
        if (eachTag.toUpperCase() === tag.toUpperCase()) {
          exists = true;
          return false;
        }
      });
      if (exists) {
        return false;
      }

      keywordTags.push(tag);      
      updateTags();
      return true;
    }

    function addLocationTag(tag) {
      locationTag = tag;
      updateTags();
      return true;
    }

    function addTag(tag) {
      tag = tag.trim();
      if (tag.length < 1) {
        return false;
      }

      tags.push(tag);
      updateTags();
      return true;
    }

    function checkLocalStorage() {
      var params = window.localStorage.getItem('introParams');
      if (params) {
        debugIntro('Found local storage params');
        params = JSON.parse(params);
        processParams(params);
        debugIntro(params);
      }
      else {
        debugIntro('Did not find local storage params');        
      }
    }

    function debugIntro(debugMessage) {
    }

    function getParams() {
      params = {};
      if (keywordTags.length > 0) {
        params['keywords'] = keywordTags;
      }

      if ($('#full_time').prop('checked')) {
        params['full_time'] = true;
      }

      if ($('#part_time').prop('checked')) {
        params['part_time'] = true;
      }

      if ($('#remote_work').prop('checked')) {
        params['remote_work'] = true;
      }

      if ($('#flexible_schedule').prop('checked')) {
        params['flexible_schedule'] = true;
      }

      if ($('#alternative_schedule').prop('checked')) {
        params['alternative_schedule'] = true;
      }

      if (locationTag.length > 0) {
        params['location'] = locationTag;
      }

      params['open'] = introOpen;
      debugIntro(params)
      return params;
    }

    function getSearchParamQueryString(countOnly) {
      var queryString = '';

      if (keywordTags.length > 0) {
        queryString += 'search=';
        queryString += keywordTags.map(encodeURIComponent).join('+');
        queryString += '&';
      }

      if ($('#full_time').prop('checked')) {
        queryString += encodeURIComponent('schedule[]') + '=' + encodeURIComponent('Full-Time') + '&';
      }

      if ($('#part_time').prop('checked')) {
        queryString += encodeURIComponent('schedule[]') + '=' + encodeURIComponent('Part-Time') + '&';
      }

      if ($('#remote_work').prop('checked')) {
        queryString += encodeURIComponent('tele_level[]') + '=' + encodeURIComponent('All Telecommuting') + '&';
        queryString += encodeURIComponent('tele_level[]') + '=' + encodeURIComponent('Mostly Telecommuting') + '&';
        queryString += encodeURIComponent('tele_level[]') + '=' + encodeURIComponent('Partial Telecommuting') + '&';
        queryString += encodeURIComponent('tele_level[]') + '=' + encodeURIComponent('Option for Telecommuting') + '&';
      }

      if ($('#flexible_schedule').prop('checked')) {
        queryString += encodeURIComponent('schedule[]') + '=' + encodeURIComponent('Flexible Schedule') + '&';
      }

      if ($('#alternative_schedule').prop('checked')) {
        queryString += encodeURIComponent('schedule[]') + '=' + encodeURIComponent('Alternative Schedule') + '&';
      }

      if (locationTag.length > 0) {
        queryString += 'location=' + encodeURIComponent(locationTag) + '&';
      }

      return queryString;
    }

    function hasParams() {
      return (keywordTags.length > 0 || tags.length > 0 || locationTag.length > 0);
    }

    function htmlentities(str) {
      return str.replace(/[\u00A0-\u9999<>\&]/gim, function(i) {
        return '&#'+i.charCodeAt(0)+';';
      });
    }

    function logAnalEvent(event, data) {
      if (event.length > 0) {
        // debugIntro('Event: ' + event);
        // debugIntro(data ? data : {});
      }
    }

    function processIntroCheckbox(element, tagName) {
      if (element.prop('checked')) {
        introOpen = true;
        element.parent().addClass('checked');
        addTag(tagName);
        logAnalEvent('Checkbox Toggle On', { param: tagName } );
      }
      else {
        element.parent().removeClass('checked');
        removeTag(tagName);
        logAnalEvent('Checkbox Toggle Off', { param: tagName } );
      }
      updateJobResults();
    }

    function processParams(params) {
      jobUpdatesPaused = true;

      if (params['keywords']) {
        jQuery.each(params['keywords'], function (idx, keyword) {
          addKeywordTag(keyword);
        });
      }

      if (params['full_time']) {
        $('#full_time').prop('checked', true);
        $('#full_time').parent().addClass('checked');
        addTag('Full-Time');
      }

      if (params['part_time']) {
        $('#part_time').prop('checked', true);
        $('#part_time').parent().addClass('checked');
        addTag('Part-Time');
      }

      if (params['remote_work']) {
        $('#remote_work').prop('checked', true);
        $('#remote_work').parent().addClass('checked');
        addTag('Remote Work');
      }

      if (params['flexible_schedule']) {
        $('#flexible_schedule').prop('checked', true);
        $('#flexible_schedule').parent().addClass('checked');
        addTag('Flexible Schedule');
      }

      if (params['alternative_schedule']) {
        $('#alternative_schedule').prop('checked', true);
        $('#alternative_schedule').parent().addClass('checked');
        addTag('Alternative Schedule');
      }

      if (params['location'] && params['location'].length > 0) {
        addLocationTag(params['location']);
        $('#location').val(params['location']);
      }

      introOpen = params['open'];
      jobUpdatesPaused = false;

      if (introOpen && hasParams()) {
        updateJobResults(true);
      }
      updateIntro();
    }

    function removeKeywordTag(tag) {
      tag = tag.trim();
      if (tag.length) {
        var newTags = [];
        jQuery.each(keywordTags, function (idx, eachTag) {
          if (tag.toUpperCase() !== eachTag.toUpperCase()) {
            newTags.push(eachTag);
          }
        });
        keywordTags = newTags;
        updateTags();
      }
    }

    function removeTag(tag) {
      tag = tag.trim();
      if (tag.length) {
        var newTags = [];
        jQuery.each(tags, function (idx, eachTag) {
          if (tag.toUpperCase() !== eachTag.toUpperCase()) {
            newTags.push(eachTag);
          }
        });
        tags = newTags;
        updateTags();
      }
    }

    function resetIntro() {
      tags = [];
      locationTag = '';
      keywordTags = [];
      introOpen = false;
      resetSteps();
      saveParams();
      updateIntro();
    }

    function resetSteps() {
      $('.intro2-step4').hide();
      $('.intro2-step3').hide();
      $('.intro2-step2').hide();
      $('.intro2-step1').fadeIn();;
      $('.intro2-steps input.checkbox').prop('checked', false);
      $('.intro2-steps label').removeClass('checked');
      $('#location').val('');
      $('#search').focus();
    }

    function resetTags() {
      updateTags();
    }

    function saveParams() {
      debugIntro('Saving Params');
      var params = getParams();
      window.localStorage.setItem('introParams', JSON.stringify(params));
    }

    function updateIntro() {

      // Only show close icon when user has at least one param
      if (hasParams()) {
        var closeIcon = $('.intro2-close-intro').children('i').first();
        if (introOpen) {
          closeIcon.removeClass('fa-chevron-down').addClass('fa-chevron-up');
        }
        else {
          closeIcon.removeClass('fa-chevron-up').addClass('fa-chevron-down');
        }
        $('.intro2-close-intro').fadeIn();;
      }
      else {
        $('.intro2-close-intro').hide();
      }

      // Show results & tags?
      if (introOpen && hasParams()) {
        $('.intro2-tags').fadeIn();;
        $('.intro2-results-container').slideDown();
      }
      else {
        $('.intro2-tags').hide();
        $('.intro2-results-container').slideUp();
      }
    }

    function updateJobResults(noSave) {
      debugIntro('Update Job Results');
      var url = '/introsearch.json?' + getSearchParamQueryString();

      if (hasParams()) {
        debugIntro('Performing search request');
        debugIntro(url);
        $.getJSON(url)
          .done(function(data) {
            debugIntro('Loaded ' + data.jobCount + ' results');
            if (data.jobCount < 1) {
              $('#intro2-jobs').html('<div class="alert alert-danger">We\'re sorry, there are no jobs matching your search criteria.</div>');
              $('.see-all-results').hide();
            }
            else {
              $('#intro2-jobs').html(data.list);
              $('.see-all-results').fadeIn();;
            }
          })
          .fail(function(jqxhr, textStatus, error) {
            debugIntro(textStats + ' ' + error);
          });
      }
      else {
        debugIntro('No params - no search request required');
      }

      if (! noSave) {
        saveParams();
      }
      updateIntro();
    }

    function updateTags() {
      debugIntro('Update Tags');
      var htmlTags = '';

      // Keyword Tags
      jQuery.each(keywordTags, function (idx, keywordTag) {
        htmlTags += '<button class="intro2-tag keyword-tag" data-tag="' + htmlentities(keywordTag) + '">' + htmlentities(keywordTag) + ' &nbsp; <b>X</b></button>';
      });

      // Checkbox Tags
      jQuery.each(tags, function (idx, tag) {
        htmlTags += '<button class="intro2-tag checkbox-tag" data-tag="' + htmlentities(tag) + '">' + htmlentities(tag) + ' &nbsp; <b>X</b></button>';
      });

      // Location Tag
      if (locationTag.length > 0) {
        htmlTags += '<button class="intro2-tag location-tag" data-tag="' + htmlentities(locationTag) + '">' + htmlentities(locationTag) + ' &nbsp; <b>X</b></button>';
      }

      $('.intro2-tags-buttons').html(htmlTags);
    }

    /* Events */

    $('.intro2-see-results').click(function() {
      introOpen = true;
      updateJobResults();     
      /*
      var location = $('#location').val().trim();
      if (location.length > 0) {
        addLocationTag(location);
        logAnalEvent('Add Location', { location: location } );
        introOpen = true;
        updateJobResults();
      }
      return false;
      */
    });

    $(document.body).on('click', '.intro2-tag-reset', function() {
      resetIntro();
      logAnalEvent('Reset')
      return false;

    $('#location').keyup(function(e){
      if (e.keyCode == 13) {

    });

    $(document.body).on('click', '.keyword-tag', function() {
      var tagName = $(this).attr('data-tag');

      logAnalEvent('Remove Tag', { tag: tagName} );

    $(document.body).on('click', '.keyword', function(e) {

    $('#full_time').click(function() {

    });

    $('#alternative_schedule').click(function() {
      processIntroCheckbox($(this), 'Alternative Schedule');
    });
