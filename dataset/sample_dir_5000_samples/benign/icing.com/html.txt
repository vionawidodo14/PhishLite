var minAge = 13;
var authenticated = false;
var subscribed = false;
var pageLoads = parseInt(localStorage.getItem('pageLoads'), 10);
var sitePath = window.location.pathname;

function validateCountry() {
    if ($('.crs-country').val() == '') {
        $('.crs-country').addClass('error');
    } else {
        $('.crs-country').removeClass('error');
    }
}

function validateRegion() {
    if ($('.crs-region').val() === '' || $('.crs-region').val() === null) {
        $('.crs-region').addClass('error');
    } else {
        $('.crs-region').removeClass('error');
    }
}

function validateCheckbox1() {
    if (!$('#newsletter-checkbox1').is(':checked')) {
        $("label[for='newsletter-checkbox1']").addClass('error');
    } else {
        $("label[for='newsletter-checkbox1']").removeClass('error');
    }
}

function validateCheckbox2() {
    if (!$('#newsletter-checkbox2').is(':checked')) {
        $("label[for='newsletter-checkbox2']").addClass('error');
        $('.checkbox2-link').css('color', '#e0142c !important');
        $('.checkbox2-link').css('font-weight', '700');

    } else {
        $("label[for='newsletter-checkbox2']").removeClass('error');
    }
}

function validateDob() {
    if ($('#s_date_of_birth').length && $('#s_optin_date').length) {
        var newsletterDob = $('#s_date_of_birth').val().split('-');
        var day = newsletterDob[2];
        var month = newsletterDob[1];
        var year = newsletterDob[0];
        var myDate = new Date();
        myDate.setFullYear(year, month - 1, day);
        var currDate = new Date();
        currDate.setFullYear(currDate.getFullYear() - minAge);
        $('#s_optin_date').val(date.toLocaleDateString('en-GB'));

        if (day != undefined && month != undefined && year != undefined) {
            if (currDate < myDate) {
                $('.day, .month, .year').addClass('error');
            } else {
                $('.day, .month, .year').removeClass('error');
            }
        }
    }
}

function validateForm() {
    validateDob();
    validateCountry();
    validateRegion();
    validateCheckbox1();
    validateCheckbox2();

    $('#s_optin_date').val(date.toLocaleDateString('en-GB'));

    $('#newsletter-general-error').css('display', '');
    $('#newsletter-dob-error').css('display', '');
    $('#newsletter-already-signedup-message').css('display', '');

    if ($("#newsletter-item input, .crs-country, .crs-region, label[for='newsletter-checkbox1'], label[for='newsletter-checkbox2']").hasClass('error')) {
        $('#newsletter-general-error').css('display', 'flex');
        return false;
    } else if ($('.day, .month, .year').hasClass('error')) {
        $('#newsletter-dob-error').css('display', 'flex');
        return false;
    } else {
        if (authenticated && !subscribed) {
            $.ajax({
                url: "https://www.icing.com/on/demandware.store/Sites-icingNA-Site/en_US/Account-NewsletterSubscribe",
                success: function () {
                    $('#newsletter-homepage-modal').css('display', 'none');
                    $('#newsletter-item').submit();
                }
            });
        } else if (authenticated && subscribed) {
            $('#newsletter-already-signedup-message').css('display', 'flex');
            return false;
        } else if (!authenticated && !localStorage.getItem('subscribed')) {
            localStorage.setItem('subscribed', 'true');
            $('#newsletter-item').submit();
        }
    }
}

function initNewsletterPopup() {	
	$(document).ready(function() {
	    // renders date dropdown
	    $('#s_date_of_birth').dateDropdowns({
	        displayFormat: 'mdy',
	        required: true,
	        daySuffixes: false,
	        defaultDateFormat: 'dd/mm/yyyy'
	    });

            // selects first option of day dropdown instead of showing up blank
            $('.day').prop('selectedIndex', 0);
	
	    // disables first option and divider of country dropdown
	    $('#s_countryiso option:eq(0)').attr('disabled','disabled');
	    $('#s_countryiso option:eq(4)').attr('disabled','disabled');
	
	    // disables first option of dob dropdowns
	    $('.day option:eq(0)').attr('disabled','disabled');
	    $('.month option:eq(0)').attr('disabled','disabled');
	    $('.year option:eq(0)').attr('disabled','disabled');
	
	    $('#newsletter-dob-info').click(function() {
	        $('#newsletter-dob-info-exapnd').toggle();
	    });
	
	    $('.crs-country').on('change', function() {
	        validateCountry();
	        $('#s_region option:eq(0)').attr('disabled','disabled');
	    });
	
	    $('.crs-region').on('change', function() {
	        validateRegion();
	    });
	
	    $('#newsletter-button').click(function() {
	        $('.newsletter-modal').css('display', 'flex');
	    });
	
	    $('#close-newsletter-form, #newsletter-no-thanks').click(function(e) {
	        e.preventDefault();
	        $('.newsletter-modal').css('display', 'none');
	        localStorage.setItem('resetNewsletterFlow', resetNewsletterFlowSetDate);
	    });
	
	    $('#newsletter-checkbox1').on('click', function() {
	        validateCheckbox1();
	    });
	
	    $('#newsletter-checkbox2').on('click', function() {
	        validateCheckbox2();
	    });
	
	    $('#footer-close-newsletter-form').click(function() {
	        $('.newsletter-modal').css('display', 'none');
	    });
	
	    if (!parseInt(localStorage.getItem('resetNewsletterFlow'), 10)) {
	        var pageLoadNum = pageLoads >= 0 ? pageLoads + 1 : 1;
	        localStorage.setItem('pageLoads', pageLoadNum);
	        // pageLoadsUntilPopup in newsletter-js content asset
	        if (pageLoadNum >= pageLoadsUntilPopup) {
	            showNewsletter();
	            localStorage.setItem('pageLoads', 0);
	        }
	    }
	
	    if (date.getDate() >= parseInt(localStorage.getItem('resetNewsletterFlow'), 10)) {
	    	localStorage.removeItem('resetNewsletterFlow');
	    }
	
	    $('.crs-region').on('change', function() {
	        validateDob();
	    });
	});
	
	(function isAuthenticated() {
	    $.ajax({
	        url: "https://www.icing.com/on/demandware.store/Sites-icingNA-Site/en_US/Account-IsAuthenticated",
	        success: function (response) {
	            authenticated = response.isAuthenticated;
	        }
	    });
	})();
	
	(function isSubscribed() {
	    $.ajax({
	        url: "https://www.icing.com/on/demandware.store/Sites-icingNA-Site/en_US/Account-IsNewsletterSubscribed",
	        success: function (response) {
	            subscribed = response.isSubscribed;
	        }
	    });
	})();
	
	(function getCountryCode() {
	    $.ajax({
	        url: "https://www.icing.com/on/demandware.store/Sites-icingNA-Site/en_US/OrgClaires-GetCountryCode",
	        success: function (response) {
	            var countryCode = response.countryCode;            
	            if (countryCode == 'DE') {
	                minAge = 16;
	            }
	            $('.crs-country').val(countryCode).trigger('change');
	            $('#s_region option:eq(0)').attr('disabled','disabled');
	        }
	    });
	})();
	
	function showNewsletter() {
	    // pagesToExclude defined in newsletter-js content asset
	    if ((authenticated && subscribed) || pagesToExclude.includes(sitePath.toLowerCase().replace(/\//g, '')) || localStorage.getItem('subscribed')) {
	        $('.newsletter-modal').css('display', 'none');
	    } else {
	        $('.newsletter-modal').css('display', 'flex');
	    }
	}
}

var initPopupInterval = setInterval(function () {
    if (window.jQuery) {
        clearInterval(initPopupInterval);
        initNewsletterPopup();
    }
}, 100);
