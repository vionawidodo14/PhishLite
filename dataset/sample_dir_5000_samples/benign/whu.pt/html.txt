	$('input[type="email"]').blur(function (e) {

		}
	});

	function CommonFormAJAX(form){
		// Disable submit button
		var control = $(form).find(".button-submit");
		var control_js = document.querySelectorAll(".button-submit")[0];
		var control_text = control.text();
		var listener = control[0].onclick;
		console.log(listener);
		control.attr("disabled",true);
		control.text("Connecting");
		$(form).find(".ajax-return-text").html("");
		// Send AJAX Request
		$.post(form.action,$(form).serialize(),function(data,status,xhr){
			try{
				data = jQuery.parseJSON(data);
				console.log(data);
			}catch(e){
				console.log("No JSON returned!");
				console.log(data);
			}
			if(data.status == undefined || data.status == "error"){
				if(typeof nc != undefined){
					//try{
						nc.reload()
						//gtobjects['#panel-login'].refresh();
					//}catch(e){
					//}
				}	
				$(form).find(".wrapper-captcha img").click();
				$(form).find(".wrapper-captcha input").val("");
				control.attr("disabled",false).text(control_text);
				if(data.status == undefined){
					$(form).find(".ajax-return-text").html("Server has reponded with unknown message.");
				}else{
					$(form).find(".ajax-return-text").html(data.message);
				}
				if(data.fail_login){
					var countdown = parseInt($(".attempt-left-counter")[0].innerHTML) - parseInt(data.fail_login);
					$(".attempt-left-counter").text(countdown > 0 ? countdown : 0);
					$("#login-failure-code").html(data.fail_code);
					$('#login-failure').modal();
				}
			}else if(data.status == "redirect"){
				control.unbind();
				window.location = data.redirect;
			}else if(data.status == "email_found"){
				control.attr("disabled",false).text(control_text);
				$("#input-login-email").val(data.email);
				UpdateCaptcha($("#panel-login .container-captcha img"));
				alert("Success! We find your email address and will take you to the login section.");
				$('#main-nav a[href="#panel-login"]').tab('show');
			}else if(data.status == "reset_sent"){
				control.unbind();
				$("#panel-recover-password").html('<h2>Reset Password</h2><div class="panel panel-success"><div class="panel-heading"><h3 class="panel-title">Success</h3></div><p>A confirmation email has been sent to '+data.email+'. Please allow a few minutes for the mail to arrive.</p></div>');
			}else if(data.status == "resend"){
				control.unbind();
				$("#panel-re-confirm").html('<h2>Confirm Email Address</h2><div class="panel panel-success"><div class="panel-heading"><h3 class="panel-title">Success</h3></div><p>A confirmation email has been sent to '+data.email+'. Please allow a few minutes for the mail to arrive.</p></div>');
			}else if(data.status == "email_confirm"){
				control.unbind();
				$("#panel-recruitment").html('<h2>注册成功，请去您的邮箱中查看验证邮件，并完成最后的验证。</h2><h3>在使用账号前你必须按邮件内容指示通过验证。如果你不这样做，注册的新账号会在较短时间后被删除。</br>如果你长时间未收到验证邮件，请用注册邮箱发送邮件到admin@whupt.net或者加客服群229655622（验证信息WHUPT），我们将手动帮你激活账号。</h3>');
			}else if(data.status == "automatic_confirmed"){
				control.unbind();
				$("#panel-recruitment").html('<h2>Success</h2>');
				$("#input-login-email").val(data.extra);
				alert("Success! No further authentication is needed. I will take you to the login section.");
				$('#main-nav a[href="#panel-login"]').tab('show');
			}else if(data.status == "manual_confirm_admin"){
				$("#panel-recruitment").html('<h2>Success</h2>But authentication will be taken by admin.');
			}else if(data.status == "manual_confirm_inviter"){
				$("#panel-recruitment").html('<h2>Success</h2>But authentication will be taken by admin or your inviter.');
			}
		});
	}

		function init_geetest(itarget){
			var target = itarget || '#login_verify'

				target = '#recruitment_verify'
			else if(target=='#panel-recover-password')
				target = '#recover-password_verify'		
			nc = new noCaptcha();
			var nc_appkey = 'FFFF0000000001779879';  // 应用标识,不可更改
			var nc_scene = 'login';  //场景,不可更改
			var nc_token = [nc_appkey, (new Date()).getTime(), Math.random()].join(':');
			var nc_option = {
				renderTo: target,//渲染到该DOM ID指定的Div位置
