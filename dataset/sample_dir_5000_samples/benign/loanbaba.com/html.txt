       function getEMITenure() {
           if (document.getElementById('tenureInYears').checked) {
               document.getElementById("emiLoanTenureYears").style.display = 'block';
               document.getElementById("emiLoanTenureMonths").style.display = 'none';
           } else if (document.getElementById('tenureInMonths').checked) {

       function getEMIChartData() {
           var loanAmt = document.getElementById("emiLoanAmount").value;
           if (isNaN(loanAmt)) {
               return false;
           }
           var interest = document.getElementById("emiInterestRate").value;

           if (isNaN(interest)) {
               return false;
           }
           var tenure = '';
           var tenureUnit = '';
           if (document.getElementById('tenureInYears').checked) {
               tenure = document.getElementById("emiLoanTenureYears").value;
               tenureUnit = "years";
           }
           else if (document.getElementById('tenureInMonths').checked) {
               tenure = document.getElementById("emiLoanTenureMonths").value;
               tenureUnit = "months";
           }
           if (isNaN(tenure)) {
               return false;
           }
           getEMIChart(loanAmt, interest, tenure, tenureUnit)
       }
       function getEMIChart(loanAmount, interestRate, tenure, tenureUnit) {
           $.ajax({
               type: "POST",
               url: Serviceurl + "/Appservice.asmx/GetEMICalcData",
               data: '{ "loanAmount" : "' + loanAmount + '","interestRate" : "' + interestRate + '", "tenure" : "' + tenure + '", "tenureUnit" : "' + tenureUnit + '"}',
               contentType: "application/json; charset=utf-8",
               dataType: "json",
               success: function (data) {
                   var emiDataInfo = $.parseJSON(data.d);
                   renderEMIChart(emiDataInfo);
               },
               failure: function (data) {
               }
           })
       };
