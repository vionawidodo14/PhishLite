// speichert eingegeben input daten und stellt nächstes Input Feld
function safeInfoShowNext() {
   var formelement = document.getElementById("signupnl");
   var mailInput = document.getElementById("mail").value; 
   
   // ueberpruefe, ob daten ueberhaupt valide sein können 
   if( (mailInput.length>3) && (mailInput.includes("@")) && (mailInput.includes(".")) ) {
      // Eingabedaten speichern
      document.getElementById("email").value = mailInput;
      
      removeChildren(formelement);
     
      // füge input 'Vorname' und 'Nachname' sowie Glyphicon span-Element für Absenden an
      // input
      var prenameInp = document.createElement("INPUT"); // mail
      prenameInp.setAttribute("type", "text");
      prenameInp.setAttribute("class", "nlinput");
      prenameInp.setAttribute("required", "true");
      prenameInp.setAttribute("placeholder", "Vorname *");
      prenameInp.setAttribute("id", "pre");
      prenameInp.setAttribute("autocomplete", "off");
      
      // input
      var lastnameInp = document.createElement("INPUT"); // mail
      lastnameInp.setAttribute("type", "text");
      lastnameInp.setAttribute("class", "nlinput");
      lastnameInp.setAttribute("required", "true");
      lastnameInp.setAttribute("placeholder", "Nachname *");
      lastnameInp.setAttribute("id", "last");
      lastnameInp.setAttribute("onkeyup", "colorSendGlyph()");
      lastnameInp.setAttribute("autocomplete", "off");
      
      // span Abschicken
      var spanEl = document.createElement("SPAN");
      spanEl.setAttribute("class", "glyphicon glyphicon-ok-sign");
      spanEl.setAttribute("placeholder", "Nachname *");
      spanEl.setAttribute("id", "glyphOK");
      spanEl.setAttribute("onclick", "confirmSignUp()");
      spanEl.setAttribute("type", "submit");
      // type="submit"
      
      // append 
      formelement.appendChild(prenameInp);
      formelement.appendChild(lastnameInp);
      formelement.appendChild(spanEl);
   } else {
      // Kleine Fehlermeldung, nur wenn element noch nicht existiert
      var elementExists = document.getElementById("fehlerText");
      if(!elementExists) {
         var fehlertext = document.createElement("div");
         fehlertext.innerHTML = "Bitte geben Sie eine <em>valide Mail-Adresse</em> ein, um sich für den Newsletter anzumelden.";
         fehlertext.setAttribute("class", "well");
         fehlertext.setAttribute("id", "fehlerText");
         formelement.appendChild(fehlertext);
      }
   }
} 

function confirmSignUp() {
   var formelement = document.getElementById("signupnl");
   var mailInput = document.getElementById("email").value;
   var prename = document.getElementById("pre").value;
   var lastname = document.getElementById("last").value
   
   // ueberpruefe, ob daten ueberhaupt valide sein können 
   if( (prename.length>2) && (lastname.length>2) ) {
      // Eingabedaten speichern
      document.getElementById("prename").value = prename;
      document.getElementById("lastname").value = lastname;
      
      // remove elements
      removeChildren(formelement);
      document.getElementById("nlinfo").remove();
      document.getElementById("oder").remove();
      
      // Success Text -> vielleicht erst nach Auswertung in handle_signup.php anzeigen?
      //var textEl = document.createElement("text");
      //textEl.innerHTML = "<em>Herlichen Glückwunsch</em>. Sie erhalten ab sofort den Newsletter.";
      //textEl.setAttribute("style", "");
      
      // gif animation des erfolgreichen Anmeldens
      var imgEl = document.createElement("IMG");
      imgEl.setAttribute("src", "/signupnewsletter/css/animate-paper-airplane-color.gif");
      imgEl.setAttribute("id", "paperplane");
      //imgEl.setAttribute("style", "");
      imgEl.setAttribute("title", "In den Newsletter eintragen");   
      
      var success = true; // fuer asynchorne datenverarbeitung
      
   } else {
      // Fehlermeldung Text
      var textEl = document.createElement("div");
      textEl.setAttribute("id", "fehlermeldung");
      textEl.setAttribute("class", "well");
      textEl.innerHTML = "Bitte füllen Sie <em>beide Fehler</em> aus, um den Newsletter zu erhalten.";
      
      // kein gif
      var imgEl = document.createElement("text");
      imgEl.innerHTML = "";
   }
   
   // append, entweder success-Nachricht oder Fehlermeldung
   // Wenn Fehlermeldung schon vorhande, kein append mehr
   if(document.getElementById("fehlermeldung") === null) {
      //formelement.appendChild(textEl);
      formelement.appendChild(imgEl);
   }
   
   // asynchroner Umgang mit den Form-Daten
   if(success) {
      // form asynchron abschicken
      var xhttp = new XMLHttpRequest();
      xhttp.onreadystatechange = function() {
         if (this.readyState == 4 && this.status == 200) {
            document.getElementById("catchoutput").setAttribute("style", "border: 1px solid black;");// HIER WEITER!!!
            document.getElementById("catchoutput").innerHTML = this.responseText;
         }
      };
      xhttp.open("GET", "/signupnewsletter/handle_signup.php?mail="+mailInput+"&prename="+prename+"&lastname="+lastname, true);
      xhttp.send();
   }
}

// entferne child felder
function removeChildren(formelement) { 
   var child = formelement.lastElementChild;
   while (child) { 
      formelement.removeChild(child); 
      child = formelement.lastElementChild;
   }
}

// Färbe glypicon schwarz, wenn nutzer tippt und input '@' enthält
function colorSendGlyph() {
   var preInput  = document.getElementById("pre").value;
   var lastInput = document.getElementById("last").value;

   if( (preInput.length>0) && (lastInput.length>0) ) {
      document.getElementById("glyphOK").style.color = "black";
      var event = window.event ? window.event : e;
      if(event.keyCode === 13) {
         // Nutzer drückt Enter
         confirmSignUp();
      }
   } else {
      document.getElementById("glyphOK").style.color = "grey";
   }
}

// Färbe glypicon schwarz, wenn nutzer bei Inputs ausgefüllt hat
function colorGlyph() {
   var mailInput = document.getElementById("mail").value; 

   if( (mailInput.length>3) && (mailInput.includes("@")) && (mailInput.includes(".")) ) {
      document.getElementById("glyphNext").style.color = "black";
      var event = window.event ? window.event : e;
      if(event.keyCode === 13) {
         // Nutzer drückt Enter
         safeInfoShowNext();
      }
   } else {
      document.getElementById("glyphNext").style.color = "grey";
   }
}
