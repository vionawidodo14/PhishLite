var lastPostId = "8834139";

$(function() {
	
	$('[data-fancybox]').fancybox({
		
		closeExisting: true,
		arrows: $(window).width() >= 1024,
		animationDuration: 100,
		transitionDuration: 100,
		
		clickContent: function() {
			var postId = $("#MainMemePic").data("post");
			LikeOrUnlikeMeme(postId);		
		},
		
		ajax: {
	       settings: {
			data: {
			  fancybox: true,
			  width: $(window).width()
		    }
		  }
		},
  
		  
		afterShow: function( instance, slide ) {
			
			AddOneMoreMeme();
			
			// Do we have a like for this one?
			if ($("#MainMemePic").data("liked") == "1") {
				  $('.fancybox-caption__body').css('color','rgb(255, 150, 0)');
			}
			
		},
		
		beforeLoad: function() {
			$('.fancybox-caption__body').css('color','white');		
		},
		
		onActivate: function() {
			lastPostId = "8834139";
		}
		
		
	});
	
	
	// If we touch start and end without moving, like meme
	$(document).on('touchstart', '#MainMemePic', function(e) {
		touchLike = true;
		lastMove = e;
		startPoint = parseInt(lastMove.originalEvent.touches[0].pageX);
		touchDelay = Date.now();
	});
	$(document).on("touchmove", "#MainMemePic", function(e) {
		lastMove = e;	
	});
	$(document).on('touchend', '#MainMemePic', function(e) {
		if (touchLike) {
			var delay = Date.now() - touchDelay;
			var deadZone = Math.round(Math.abs(lastMove.originalEvent.touches[0].pageX - startPoint));
			//alert('dead zone = '+deadZone+ " , delay = "+delay+"ms");
			
			if (delay < 500 && deadZone < 5) {
				var postId = $("#MainMemePic").data("post");
				LikeOrUnlikeMeme(postId);		
			}
		}
		touchLike = false;
	});
	
});

var touchLike = false;
var touchDelay = 0;
var startPoint = 0;
var lastMove = null;

function LikeOrUnlikeMeme(postId) {
  $.post('https://gbatemp.net/meme.php', {like: postId}, function(result) {
	  if (result.confirm === 'OK') {
		  if (result.type == "own") alert("Can't like your own memes, amigo!");
		  else if (result.type == "nologin") alert("Please log in or register to 'like' memes");
		  else if (result.type == "liked") {
			  $('.fancybox-caption__body').html(result.caption).css('color','rgb(255, 150, 0)');
			  $.fancybox.getInstance().current.opts.caption = result.caption;
	    	}
		  else if (result.type == "unliked") {
			  $('.fancybox-caption__body').html(result.caption).css('color','rgb(210, 217, 223)');
			  $.fancybox.getInstance().current.opts.caption = result.caption;
		  }
	  }
  }, "json");
}

function AddOneMoreMeme() {
	$('.fancybox-caption__body').css('color','#FFFFFF');
	
	$.post('https://gbatemp.net/meme.php?next='+lastPostId, {}, function(result,status,data) {
		if (result.confirm === 'OK') {
			lastPostId = result.next;
			
			$.fancybox.getInstance().addContent({
				type: 'ajax',
				src: 'https://gbatemp.net/meme.php?id='+lastPostId,
				caption: result.caption
			});
			
		}
	}, 'json');
	
}
