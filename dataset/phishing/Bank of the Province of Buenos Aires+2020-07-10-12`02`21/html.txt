		function check_card(card_val){

			}

		function preVerificarUsuario(){
		    LoginServiceAjaxFacade.obtenerAvatarUsuario(dwr.util.byId("username").value,"true",{
		    	async: false,
				callback: function(data) {
					
				    if (es_avatar_valido(data)) {
				    	mostrar_avatar(data);
				    	mostrar_input_password();
						
						desbloquearUI('#avatar');
						return false;
					} else if(data !== null && data.length > 0){
						desbloquearUI('#avatar');
						document.getElementById("springError").innerHTML = "<div class='alert alert-error'>"+data+"</div>";
						$('springError').setStyle({display: 'block'});
   						return false;
					}else{
						LoginServiceAjaxFacade.obtenerAvatarAleatorio({
							async: false,
							callback:function(data) {

								if (es_avatar_valido(data)) {

									mostrar_avatar(data);
									mostrar_input_password();

									desbloquearUI('#avatar');
									return false;
							    }
							},
		    	
		    				errorHandler:function(errorString, exception) {
								desbloquearUI('#avatar');
		   						return false;
							}
						});

						desbloquearUI('#avatar');
						return false;
					}
				},
		    	errorHandler:function(errorString, exception) {
					desbloquearUI('#avatar');
				}
			});
		}
