<script type="text/javascript">
    function Change(from){
        switch(from){
            case'particulier':
            $('.section-freelance').removeClass('active');
            $('.section-company').addClass('active');
            $('#ParticularesAutonomos-imagen').hide();
            $('#EmpresasPymes-imagen').show();
            $('.contentFreelance').hide();
            $('.contentCompany').show();
            break;
            case'company':
            $('.section-company').removeClass('active');
            $('.section-freelance').addClass('active');
            $('#EmpresasPymes-imagen').hide();
            $('#ParticularesAutonomos-imagen').show();
            $('.contentCompany').hide();
            $('.contentFreelance').show();
            break;
        }
    }
