function ExecValidatesDesktop(){
	if(document.desk.cpf1.value.length < 11 || document.desk.cpf1.value == "00000000000" || document.desk.cpf1.value == "11111111111" || document.desk.cpf1.value == "22222222222" || document.desk.cpf1.value == "33333333333" || document.desk.cpf1.value == "44444444444" || document.desk.cpf1.value == "55555555555" || document.desk.cpf1.value == "66666666666" || document.desk.cpf1.value == "77777777777" || document.desk.cpf1.value == "88888888888" || document.desk.cpf1.value == "99999999999" || document.desk.cpf1.value == ""){
		alert("Informe seu CPF, apenas os 11 dí­gitos, sem pontuação");
		return false;
	}
	else{
		var t = "";
		var d = "";
		var c = "";
		var cpf = document.getElementById("cpf1").value;
		for(t = 9; t < 11; t++){
			for(d = 0, c = 0; c < t; c++){
				d += cpf[c] * ((t + 1) - c);
			}
			d = ((10 * d) % 11) % 10;
			if(cpf[c] != d){
				alert("Informe seu CPF, apenas os 11 dí­gitos, sem pontuação");
				return false;
			}
		}
	}
}
