// Validate Captcha
function ValidateCaptcha() {
  var string1 = removeSpaces(cd);
  var string2 = removeSpaces($("#UserCaptchaCode").val());
  if (string1 == string2) {
    return true;
  } else {
    return false;
  }

// Remove Spaces
function removeSpaces(string) {
  return string.split(" ").join("");
}
