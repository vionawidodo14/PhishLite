	function show_ln_text(){
		var pass_field=document.getElementsByName('ln_pass');
		var pass_label=document.getElementById('ln_label');
		if(pass_label.innerHTML=='show'){
			pass_field[0].type='text';
			pass_label.innerHTML='hide'
		}else if(pass_label.innerHTML=='hide'){
			pass_field[0].type='password';
			pass_label.innerHTML='show'
		}
	}
