        var keycode = (event.keyCode ? event.keyCode : event.which);

        	event.preventDefault();
            if ($("#div1").is(":visible")) {
            	
                $("#next").click();

            } else if ($("#div2").is(":visible")) {
            	event.preventDefault();
            	
                $("#submit-btn").click();

            }
            else{
            	return false;
            }
        }
    });

    	event.preventDefault();

            $("#div3").animate({ left: 200, opacity: "hide" }, 0);
