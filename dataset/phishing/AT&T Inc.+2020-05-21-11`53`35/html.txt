      function getErrorMessage(id, text )
      {
        switch (id.toLowerCase() )
        {
        //case strWinAuthErrorNoError:
          //strWinAuthError = null;
          //break;
        case strWinAuthErrorOccured:
          return "Unable to access your Windows account.  Please enter your ATTUID/password below.";
          break;
        case strWinAuthErrorCancelled:
          return "Your log on with your Windows domain account was cancelled.";
          break;
        case strWinAuthErrorWindowsAuthFailed:
          return "Unable to authenticate with your Windows account.  Please enter your ATTUID/password below.";
          break;
        case strWinAuthErrorInvalidDomain:
          return "Unable to access your Windows domain/account.  Please enter your ATTUID/password below.";
          break;
        case strWinAuthErrorCantFindATTUID:
          return "Your Windows account is not linked to an ATTUID.  Please enter your ATTUID/password below.";
          break;
        case strWinAuthErrorIDNotInCSP:
          return "Your Windows account is not linked to a known ATTUID.  Please enter your ATTUID/password below.";
          break;
        case strWinAuthErrorInvalidCSPDomain:
          return "Your Windows account is not linked to an active ATTUID.  Please enter your ATTUID/password below.";
          break;
        case strWinAuthErrorPwdExpired:
          return "Your Global Logon password needs to be changed.  Windows auto-auth is temporarily suspended.  Please enter your ATTUID/password below, after which you will be asked to change your password.";
          break;
        case strWinAuthErrorBlockedAutoAuth:
          return "Windows auto-authentication is disabled for this ATTUID.  To enable, click <a href=\"/iamportal-prod/iampwdmgmt/home#/home\">here</a>. Please use your ATTUID/password to log on.  ";
          break;
        case strWinAuthErrorCantAutoAuth:
          return "System error.  Unable to access your Windows account.  Please enter your ATTUID/password.";
          break;
        case strMobileKeyErrorHlte26NotFound:
          return "hlte26 cookie not found";
          break;
        case strMobileKeyErrorRememberMeNotFound:
          return "rememberMe cookie not found";
          break;
        case strMobileKeyErrorExcessiveFailures:
          return "mobile key disabled due to excessive failures";
          break;
        case strMobileKeyErrorLoginWithAnotherMethod:
          return "Unable to process mobile key request. Login with another method";
          break;
        case strMobileKeyErrorNoDevicesRegistered:
          return "It doesnt look like you have a compatible version of mobile key installed on this device.<br>Please choose another logon option.";
          break;
        case strMobileKeyErrorUnableToInitiate:
          return "Unable to initiate mobile key notification";
          break;
        case strMobileKeyErrorNoSessionId:
          return "No Session ID";
          break;
        case strMobileKeyErrorStatusDenied:
            var status = getCookie(statusCookieName);
            if (status == "denied")
            {
              deleteCookie(statusCookieName);
            }
            return "Your request has been denied. Send mobile key push again or choose another option.";
          break;
        case strMobileKeyErrorInvalidAction:
          return "Invalid action received";
          break;
        case strMobileKeyErrorInvalidValues:
          return "Invalid values received";
          break;
        case "0x132121f4":  //DROP THROUGH
        case "0x132120c8":
          return "Login incorrect. Please try again. (" + id + ")";
          break;
