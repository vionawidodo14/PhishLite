        <script>
			function onCollapsibleClicked(){
				$( '.collapsibleButtonIcon' ).click(function(){
        			var expanded =  $(this).attr('aria-expanded');
        			var id = $(this).attr('aria-controls');
        			if(expanded ==  'false'){
        				$('.collapsibleButtonIcon[aria-expanded="true"]').click();
        				$('.divHelpTipCollapsible#'+id+'>.divHelpTipCollapsibleContent').attr('aria-live','polite');
        				var source = $('.divHelpTipCollapsible#'+id).html();
						$('.divHelpTipCollapsible#'+id).html(source);
        			}
        		});
			}
			
			function onThereAreValidationErrors(){
				if ( $( '.login-errors' ).length ) {
					
					$('p.login-errors').attr("tabindex", 0);
					$('.login-errors').find('span,a').each(function() {
						$( this ).attr("tabindex", 0)
						
					});
					
					
					$('.form-group').each(function(){
						$(this).find('.collapsibleButtonIcon').attr("tabindex", 0);
						$(this).find('.form-control').attr("tabindex", 0);
					});
					
					$('.login-submit-button').attr("tabindex", 0);
					
					var numberOfErrors = 0;
					var errorMessageEng = ' Errors have occurred. See below for more information.';
					var errorMessageEsp = ' Errores ha ocurrido. Favor de ver a continuaci\xF3n para mayor informaci\xF3n.';
					$('div.login-errors>ul').find('a').each(function(){ ++numberOfErrors;});
					if(numberOfErrors > 1){
						var language = $('a.nav-link[href="javascript:switchLanguage()"]').text();
						var errorMessage = '';
						if($.trim(language) == "English"){ // then page currently in spanish
							errorMessage = numberOfErrors + errorMessageEsp;
						}else{ // page is currently in English
							errorMessage = numberOfErrors + errorMessageEng;
						}
						$('div.login-errors>span.login-errors-message').text(errorMessage);
					}
					
					$( '.login-errors:visible' ).focus();
					$('.login-errors-message:visible').focus();
				}
				
			}
			
			$( document ).ready(function() {
				onCollapsibleClicked();
				onThereAreValidationErrors();
			});
