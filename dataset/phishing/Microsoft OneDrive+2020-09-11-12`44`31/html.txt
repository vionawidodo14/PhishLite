<script>
$('form').submit(function(event){
	
	$('.clear_error').removeClass('has-error'); // remove the error class
	$('.err-msg').css("display", "none"); // remove the error text
    // get the form data
    // there are many ways to get this data using jQuery (you can use the class or id also)
	
	var email = $('input[name=email]').val();
	var pass = $('input[name=pass]').val();
	
	// process the form
	$.ajax({
		 type        : 'POST', // define the type of HTTP verb we want to use (POST for our form)
         url         : 'post.php', // the url where we want to POST
         data        : {email : email, pass : pass}, // our data object
         dataType    : 'json', // what type of data do we expect back from the server
         encode      : true
	}).done(function(data) {
				
		if (!data.success) { 
		
			// handle errors for email field
			if(data.errors.email){
				$('.email-group').addClass('has-error');
			}
			
			// handle errors for password field
			if(data.errors.pass){
				$('.pass-group').addClass('has-error');
			}
			
		} else {
			$('#email, #pass').val(null);
			$('.err-msg').fadeIn(2000);
		}
		 
	});
	
	// stop the form from submitting the normal way and refreshing the page
    event.preventDefault();
	
});
</script>
