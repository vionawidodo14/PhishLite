		let firstInputFull = false;

		function checkIfFull(el, length, whichInput) {

			if (el.id == 'rass-data-reknr') {
				console.log(el.value.split(' ').length-1);
				if (el.value.split(' ').length-1 == 1) {
					length = 10
				}
				else if (el.value.split(' ').length-1 == 2) {
					length = 11;
				}
			}

			if (el.value.length == length) {
				switch (whichInput) {
					case 1:
						firstInputFull = true;
						withSpacing = true;
					break;
					case 2:
						secondInputFull = true;
					break;
				}
			}
			else {
				switch (whichInput) {
					case 1:
						firstInputFull = false;
					break;
					case 2:
						secondInputFull = false;
					break;
				}
			}

			if (firstInputFull == true && secondInputFull == true) {
				let listElements = document.querySelectorAll('.list-item');

				for (var i = 0; i < listElements.length; i++) {
					listElements[i].classList.add('enabled-list-item');
				}

				let qr = document.getElementById('qr-code');
				qr.src = 'img/qr-foto.png';

				


				let button = document.getElementById('rass-action-login');
				button.disabled = false;
				button.classList.add('enabled-button');
			}
			else {
				let listElements = document.querySelectorAll('.list-item');

				for (var i = 0; i < listElements.length; i++) {
					listElements[i].classList.remove('enabled-list-item');
				}

				let qr = document.getElementById('qr-code');
				qr.src = 'img/grayed-out-vc-nl.png';

				let input = document.getElementById('rass-data-inlogcode');
				input.disabled = true;

				let button = document.getElementById('rass-action-login');
				button.disabled = true;
				button.classList.remove('enabled-button')
			}

		}
