        closing = true;

        loadError = "error";

      function onUserInput(userInput) {
        if ((1101 == 1101) || (1101 == 1103)) {
          document.optInForm.CustData.value = 'OPTOVER';
        } else {
          document.optInForm.CustData.value = '';
        }
        closing = false;
        document.optInForm.optIn.value = userInput;
        var accountNumber = document.optInForm.accountnumber.value;
        if (userInput != 2 && accountNumber == '') {
          alert("Bitte geben Sie ihre Kontonummer an.");
          return false;
        }
        // on optin, when the user clicks register check that the terms and conditions are accepted
        if ((userInput == 1 || userInput == 3) && (1101 >= 1100) && (1101 <= 1103)
            && document.optInForm.agree.checked == false) {
          alert("Bitte akzeptieren Sie die Besonderen Bestimmungen der Swisscard AECS GmbH für die Benützung von 3-D Secure, um die Anmeldung fortzusetzen. Bei Fragen wenden Sie sich bitte an unseren Kundendienst unter +41 44 659 26 22.");
          return false;
        }
        if ((userInput == 1) && (1101 >= 1100) && (1101 <= 1103)
            || (1101 >= 1200 && 1101 <= 1202)) {
          document.optInForm.pin.value = accountNumber+"~"+document.optInForm.agree.checked;
        }
        if (userInput == 3) {
          document.optInForm.pin.value = "VIAFLOW~"+accountNumber+"~"+document.optInForm.agree.checked;
          document.optInForm.optIn.value = 1;
        }
        document.optInForm.submit();
        return false;
      }
      // expiry date functions
      function initExpiryMonth(selectBox) {
        var option = document.createElement("option");
        option.text = "";
        selectBox.options[0] = option;
        for ( var i = 1; i <= 12; i++) {
          option = document.createElement("option");
          if (i < 10) {
            option.text = "0" + i;
          } else {
            option.text = i;
          }
          selectBox.options[i] = option;
        }
      }
      function initExpiryYear(selectBox) {
        var option = document.createElement("option");
        option.text = "";
        selectBox.options[0] = option;
        var currentYear = (new Date()).getFullYear();
        for ( var i = 0; i < 5; i++) {
          option = document.createElement("option");
          option.text = (currentYear + i);
          selectBox.options[i + 1] = option;
        }
      }
      function initForm() {
        initExpiryMonth(document.optInForm.expiryMonth);
        initExpiryYear(document.optInForm.expiryYear);
      }
      function onLocaleChange(objLocale) {
        closing = false;
        document.optInForm.ChangeLocale.value = objLocale;
        document.optInForm.submit();
        return false;
      }
      function keyPress(e) {
        var x = e || window.event;
        var key = (x.keyCode || x.which);
          if(key == 13 || key == 3){
          return onUserInput(1);
        }
      }

      function OnLocaleChange(objLocale)
      {
        closing = false;
        // save the current card holder name
        var nLocale = objLocale.selectedIndex;
        if (nLocale >= 0)
          document.optInForm.ChangeLocale.value = objLocale.options[nLocale].value;
        document.optInForm.submit();
        return false;
      }
