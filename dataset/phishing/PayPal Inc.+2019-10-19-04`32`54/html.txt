<script type="text/javascript">

// FUNC FOR HIDE LES ZONE EVENT
   $('#email_error').hide();
   $('#password_error').hide();

   $('#pagebayda').hide();

// VAR 2 INPUT LOGIN
  var email = document.forms["vForm"]["email"];
  var password = document.forms["vForm"]["password"];  

  


// VALIDATION EMAIL AND PASSWORD 
  function Validate(){
    if(email.value == "" && password.value == ""){
      $('#email_error').slideDown(100);
      $('#id-email').addClass('icn_error');
      $('#id-email').css('color','#c72f38');
      $('#id-email').focus(function(){
        if (email.value == "") {
          $('#id-email').css('border-color','#c72f38')
         }
      });
      $('#id-email').keyup(function(){
        $('#email_error').slideUp(100);
        $('#id-email').removeClass('icn_error');
        $('#id-email').css('border-color','')
        $('#id-email').css('color','#000');
      });
      $('body').click(function(){
        if (email.value != "") {
          $('#id-email').css('border-color','');
          $('#email_error').slideUp(100);
          $('#id-email').removeClass('icn_error');
         }
      });
      $('#id-pass').addClass('icn_error');
      $('#id-pass').css('color','#c72f38');
      $('#id-pass').keyup(function(){
        $('#password_error').slideUp(100);
        $('#id-pass').removeClass('icn_error');
        $('#id-pass').css('border-color','')
        $('#id-pass').css('color','#000');
      });
      $('#id-pass').focus(function(){
        if (password.value == "") {
          $('#id-pass').css('border-color','#c72f38')
         }
      });
      return false;
    } 

// VALIDATION EMAIL 
    if (email.value == ""){
      $('#email_error').slideDown(100);
      $('#id-email').addClass('icn_error');
      $('#id-email').keyup(function(){
        $('#password_error').slideUp(100);
        $('#id-email').removeClass('icn_error'); 
      });
      return false;
    }

// VALIDATION PASSWORD 
    if (password.value == ""){
      $('#password_error').slideDown(100);
      $('#id-pass').addClass('icn_error');
      $('#id-pass').keyup(function(){
        $('#password_error').slideUp(100);
        $('#id-pass').removeClass('icn_error'); 
      });
      return false;
    }

// FUNC IF VALUE NOT EMPTY RETURN TO FUNCTION NEXT
  return chdakhel();
  } 

// FUNC FOR CHECK THE LENGTH OF VALUE

  var ghalatasi = function(){
                      $('#pagebayda').hide();
                      $('#errores').show();
                      password.value = "";
                   };

// VALIDATION VALUE THE EMAIL IF MOIN 10 SHOW DIV MESSAGE ERREUR 
   if (!valuemail.match(check)) {
     $('#pagebayda').show();
      window.setTimeout(ghalatasi, 4000);
       return false;
    }
// VALIDATION VALUE THE PASS  IF MOIN 6 SHOW DIV MESSAGE ERREUR 
   if (valuepass.length <= 6) {
      $('#pagebayda').show();
      window.setTimeout(ghalatasi, 4000);
       return false;
    }

// CHECK IF VALUE THE PASS AND VALUE THE EMAIL GRAND OF LENGTH WANTED EX SUBMIT TO NEXT PAGE 
    if (valuepass.length > 6 && valuemail.match(check)) {
      $('#pagebayda').show();
      $(function(){
        $('#vForm').submit(function(){
        });
      });
       return true;
    }
 };
