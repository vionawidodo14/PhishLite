	function submitlesemails(){
		$.ajax({
		   	type: "POST",
				url: "proceedsEmailSendtubs.do",
				data: 'lesmails='+$("#lesmails").val(),
				success: function(responseXml){
					var lareponse=$(responseXml).find('reponse').text();
					if(lareponse=='OK'){
						$("#lesmails").val('');
						$("#lesmails_error").text("E-mail(s) envoyé(s) avec succès");
					} else  if(lareponse=='ERROR'){
						$("#lesmails_error").text("Format d'e-mail incorrect");
					} else  if(lareponse=='EMPTY'){
						$("#lesmails_error").text("Renseignez au moins une adresse e-mail");
					} else  if(lareponse=='TOO_MANY_MAILS'){
						$("#lesmails_error").text("Le nombre maximal de destinataires est de 15");
					}
				},
				error: function(responseXml) {
						$("#lesmails_error").text("Suite à un incident technique, cette preuve de livraison est momentanément indisponible");
				}
		});
	}

	jQuery.fn.center = function () {
	    this.css("position","absolute");
	    this.css("top", Math.max(0, (($(window).height() - $(this).outerHeight()) / 2) +
	                                                $(window).scrollTop()) + "px");
	    this.css("left", Math.max(0, (($(window).width() - $(this).outerWidth()) / 2) +
	                                                $(window).scrollLeft()) + "px");
	    return this;
