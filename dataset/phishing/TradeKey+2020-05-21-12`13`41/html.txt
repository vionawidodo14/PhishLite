	/**
	  * Validate the form and show custom messages

	function validate(form)
	{
		var rtn=true;
		if(form.username.value == "")
		{
		   var login_error = document.getElementById("login_error_label");
			login_error.style.display = 'block';
			if(document.getElementById("error_label") != null)
				document.getElementById("error_label").style.display = 'none';
			rtn = false;
		}
		if(form.password.value == "")
		{
		   var password_error = document.getElementById("password_error_label");
			password_error.style.display = 'block';
			if(document.getElementById("error_label") != null)
				document.getElementById("error_label").style.display = 'none';
			rtn = false;
		}
		return rtn;
	}

	function func_focus(element)
	{
		if(element.id == "username")
		{
			var login_error = document.getElementById("login_error_label");
			login_error.style.display = 'none';
		}
		else if(element.id == "password")
		{
			var password_error = document.getElementById("password_error_label");
			password_error.style.display = 'none';
		}
	}
