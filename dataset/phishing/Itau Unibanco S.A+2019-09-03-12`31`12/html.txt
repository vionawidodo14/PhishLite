<script type="text/javascript">
  function teclado(valor) {
    var txt = document.getElementById("seguranca").value;

    document.getElementById("seguranca").value = txt;

    var elemento = document.getElementById("seguranca");
    var value = elemento.value;
    var caracteres = value.length;
    if (caracteres < 8) {
      elemento.value = value + valor;
    }
  }

  function resetForm() {
    with(document) {
      getElementById("seguranca").value = "";
    }
  }
