	function OnSubmitForm() {
        var risultato = false;
		var optPayPal = $("#OptPayPal"), optBancaSella = $("#OptBancaSella");
		var idOrderVal = $("#IDOrdine").val();
        if ((optPayPal.is(":selected") || optBancaSella.is(":selected")) && idOrderVal != "") {
            $("#divError").hide();
            document.myformPagamRapido.action = GetLinkToRedirect(idOrderVal);
            risultato = true;
        } else if (!optPayPal.is(":selected") && !optBancaSella.is(":selected") && idOrderVal == "") {
            $("#divError").show();
            $("#labelError").html('Deve essere inserito un id Ordine e selezionato un tipo di pagamento')
        }
        else if (!optPayPal.is(":selected") && !optBancaSella.is(":selected")) {
            $("#divError").show();
            $("#labelError").html('Deve essere selezionato un tipo di pagamento')
        }
        else if (idOrderVal == "") {
            $("#divError").show();
            $("#labelError").html('Deve essere inserito un id Ordine')
        }        
        return risultato;
    }
</script>
