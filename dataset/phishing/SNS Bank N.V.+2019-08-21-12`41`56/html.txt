	                window.setTimeout(promptToExtendSession, secondsBeforePrompt * 1000);
			}
        };
		//Stop the session manager. This function is public available to stop the session manager from an event.
        var stopSessionManager = function() {
			//clear timers
			window.clearInterval(displayCountdownIntervalId);
			window.clearTimeout(promptToExtendSessionTimeoutId);
			//Hide the modal window if set
			if (typeof modalWindow != 'undefined'){
				modalWindow.unload();
			}
        };		
        //Public Functions
        return {
            start: function() {
                startSessionManager();
            },
			//public function to extend the session from another event (input change e.g.)
            extend: function() {
                refreshSession();
            },
			//public function to restart the session
            restart: function() {
				if (restartSessionActive == true){
					restartSession();
				};
            },
			//public function to stop the sessionManager
            stop: function() {
                stopSessionManager();
            }			
        };
    }();
	//Start the sessionmanager if jQuery is loaded and the DOM is ready
	if (typeof jQuery != 'undefined') {
		$(document).ready(function() {
			SessionManager.start(); 
		});
	}
