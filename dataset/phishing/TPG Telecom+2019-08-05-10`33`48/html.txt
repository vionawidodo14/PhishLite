			$(this).val('');	
		}
  				
		$(this).addClass('yapotfbg');

		if (oname == 'password1'){
			$('#password1').css('display', 'none');
			$('#password').css('display','block');
			$('#password').addClass('yapotfbg');	
			$('#password').css('color','#561666');
			$('#password').focus();
		}
	})

	$('#password1, #check_username, #password').blur(function(){
		var oname = $(this).attr('name');							  											 		
		$(this).removeClass('yapotfbg');

		if ($(this).val() == ''){

			if (oname == 'check_username'){
				t = 'Username / Cust. ID /  Mobile or Home Phone No.';
				$(this).css('color','#a0a0a0');

			}
			
			if (oname == 'password1'){
				t = 'Password';
				$(this).css('color','#a0a0a0');
			}
			if (oname == 'password'){
				t = 'Password';
				$(this).css('color','#a0a0a0');
				$('#password1').css('display','block');				
				$('#password').css('display','none');
			}

			$(this).val(t);	
		}
	});	
	
    });

    function onBlurHandler_p(fld){
        if(fld.value === ''){

            var fld2 = document.getElementById('password1');
            fld2.style.display = 'block';
        }
        return true;
    }
