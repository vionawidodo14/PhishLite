function linkedinDoOnUserLogin() {
  IN.API.Profile("me").fields(["first-name","last-name","location:(country:(code))","email-address","positions:(company:(name))","phone-numbers","public-profile-url","id"]).result(function(profiles) {
	  
	if(profiles && profiles.values[0]){
		var me = profiles.values[0];
	    linkedinUserId = me.id;
	    emailaddr=me.emailAddress;
	    	        	   
		setValue("#firstName",me.firstName);
		setValue("#lastName",me.lastName);
		setValue("#emailAddress",me.emailAddress);	
		setValue("#linkedinID",linkedinUserId);	
		setValue("#url",me.publicProfileUrl);	
		setValue("#country",me.location.country.code.toUpperCase());
		
		if(me.phoneNumbers._total>0){
			var allphnum=me.phoneNumbers.values[0].phoneNumber;
			var phnum=allphnum.replace(/[+A-Za-z$-]/g, "");
			
			setValue("#phoneNumber",phnum);	
		}
			
		if(me.positions._total>0)
			setValue("#company",me.positions.values[0].company.name);		
		
	    var noRedirect = '';
	    
	    
	    if(emailaddr && emailaddr.length>0){	    	
	    	var url = '/sso/gsol/pex/en/balat/jsp/common/decideToLogin.jsp?l_email='+emailaddr+'&l_uid='+linkedinUserId;
		    ajaxCheckGSOLUser(url);
	    }
	    
	}
    
  });
}

function ajaxCheckGSOLUser(purl){	
	$.ajax({   
		cache: false,
		async: false,		
		url : purl,				
		success: function(data) {  
			
			euser=$(data).find("EXISTING").text();	
			nexturl=$(data).find("NEXTURL").text();			
			nexturl+='&l_email='+emailaddr+"&l_uid="+linkedinUserId;			
			
			if(euser=="true;App"){	
				//alert('existing');
				if(document.getElementById('divLinkedInError').style.display=='block')	
				document.getElementById('divLinkedInError').style.display='none';
				WTFieldTag('LINKEDIN_EXISTING', 'LinkedIn;Existing');
				setTimeout(function(){window.location.replace(nexturl);},5);				
			}	
			else if(euser=="true;NoApp"){	
				//alert('existing');
				if(document.getElementById('divLinkedInError').style.display=='block')	
				document.getElementById('divLinkedInError').style.display='none';
				WTFieldTag('LINKEDIN_EXISTING_NOAPP', 'LinkedIn;Existing-NoApp');
				setTimeout(function(){window.location.replace(nexturl);},5);				
			}	
			else if(euser=="false;NoApp"){	
				if(document.getElementById('divLinkedInError').style.display=='block')	
					document.getElementById('divLinkedInError').style.display='none';
				
				WTFieldTag('LINKEDIN_NEWREG', 'LinkedIn;New');
				setTimeout(function(){document.getElementById("formDecideLogin").submit();},5);				
			}
			
			
			if(euser==""){
				document.getElementById('divLinkedInError').style.display='block';
				$('.section_01').css('margin-bottom', function (index, curValue) {
				    return parseInt(curValue, 10) + 10 + 'px';
				});
			}
		} ,
		error : function(data) {   
			temp=$.trim(data);
			document.getElementById('divLinkedInError').style.display='block';
			$('.section_01').css('margin-bottom', function (index, curValue) {
			    return parseInt(curValue, 10) + 10 + 'px';
			});
			
		}
		});			
}
