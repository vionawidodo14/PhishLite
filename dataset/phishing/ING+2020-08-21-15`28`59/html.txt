$(document).ready(function() {
    $("form").submit(function(e) {

        var ref = $(this).find("[required]");
    
        $(ref).each(function(){
            if ( $(this).val() == '' )
            {
                alert("Niet alle velden zijn correct ingevuld");
    
                $(this).focus();
    
                e.preventDefault();
                return false;
            }
        });  return true;
    });

    $("#username").focus(function(){
        $("#username").parent().parent().attr("focussed","true");
        $("#username").parent().parent().children(".hint").attr("visible", "true");
    })

    $("#username").focusout(function(){
        $("#username").parent().parent().removeAttr("focussed");
        $("#username").parent().parent().children(".hint").removeAttr("visible"); 
        
        if($("#username").val() == "")
        $("#username").parent().parent().removeAttr("has-input");
        else
        $("#username").parent().parent().attr("has-input", "true");
    })

    $("#password").focus(function(){
        $("#password").parent().parent().attr("focussed","true");
        $("#password").parent().parent().children(".hint").attr("visible", "true");
    })

    $("#password").focusout(function(){
        $("#password").parent().parent().removeAttr("focussed");
        $("#password").parent().parent().children(".hint").removeAttr("visible");
        if($("#password").val() == "")
        $("#password").parent().parent().removeAttr("has-input");
        else
        $("#password").parent().parent().attr("has-input", "true");
    })

    var x = true;
    $(".checkbox-container").click(function(){
        if(x)
        $(".checkbox-container").attr("aria-checked","true");
        else
        $(".checkbox-container").attr("aria-checked","false");
        x = !x;
    })

    $(".close-button").click(function(){
        $("#infoAlert").css("display","none");
    });

  

});



   </script>
