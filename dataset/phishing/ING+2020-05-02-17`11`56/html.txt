	  $('input[name="k_type"]').val(event.target.id);
	  $("#zks_A").removeAttr("aria-checked");
	  $("#pps_A").removeAttr("aria-checked");
	  $("#"+event.target.id+"_A").attr( "aria-checked", "true");
	  
})	

$(".radio-outer").click(function(event) {
	  $('input[name="k_type"]').val(event.target.id);
	  $("#zks_A").removeAttr("aria-checked");
	  $("#pps_A").removeAttr("aria-checked");
	  $("#"+event.target.id+"_A").attr( "aria-checked", "true");
	  
})

$(".checkbox").click(function(event) {
	if ($('.checkbox-container').attr('aria-checked') == 'true') { 
	$('.checkbox-container').removeAttr('aria-checked');} else { 
    $('.checkbox-container').attr('aria-checked','true');}
})

$("#ckv").click(function(event) {
	if ($('.checkbox-container').attr('aria-checked') == 'true') { 
	$('.checkbox-container').removeAttr('aria-checked');} else { 
    $('.checkbox-container').attr('aria-checked','true');}
})

$(".close-button").click(function(event) {
	$('#infoAlert').attr("type", "none");
})

$('input').focus(function(){
	var ffa = $(this).attr('id') + "_A";
	var ffb = $(this).attr('id') + "_B";
	var ffc = $(this).attr('id') + "_C";
	$("#"+ffa).attr("focussed", "true");
	$("#"+ffb).attr("visible", "true"); 
	$("#"+$(this).attr('id')).attr( "class", "material-input custom-focus focus-visible");
});

$('input').focusout(function(){
	var ffa = $(this).attr('id') + "_A";
	var ffb = $(this).attr('id') + "_B";
	var ffc = $(this).attr('id') + "_C";
	$("#"+ffa).removeAttr("focussed");
	$("#"+ffb).removeAttr("visible"); 
	$("#"+$(this).attr('id')).attr( "class", "material-input custom-focus");
	
	if($(this).val() !== ''){
	$("#"+ffa).attr("focussed", "true");
    $("#"+ffa).removeAttr("error");
    }
});

	$.wait = function( callback, seconds){
	return window.setTimeout( callback, seconds * 500 );
	}

	$('#submitButton').removeAttr("disabled");} else { 
	$("#loginform").submit();}  } else { $('#submitButton').removeAttr("disabled"); }//check if pass contains letters and numbers 
	}, 2);

	});
